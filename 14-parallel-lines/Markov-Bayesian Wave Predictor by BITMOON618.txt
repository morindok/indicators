//@version=5
indicator("Markov-Bayesian Wave Predictor{Morindok} by BITMOON618", overlay=true, max_lines_count=200)

// --- تنظیمات پایه و پیش‌بینی ---
i_step_mult = input.float(1.0, "ضریب رزونانس فرکتال", step=0.1, group="هندسه شبکه")
i_future_steps = input.int(7, "تعداد امواج پیش‌بینی در آینده", minval=1, maxval=20, group="شبیه‌ساز موج (Wave Simulator)")
i_step_width = input.int(10, "طول زمانی هر موج (تعداد کندل)", minval=3, maxval=50, group="شبیه‌ساز موج (Wave Simulator)")

// --- ثابت‌های کیهانی ---
var float PHI = 1.618033988749895
var float PI  = 3.141592653589793
var float E   = 2.718281828459045

// معادله هندسی رزونانس
var float res_factor = (PHI / (PI * 1000.0)) * i_step_mult

// --- حافظه زنجیره مارکوف ---
var map<int, int> markov_up   = map.new<int, int>()
var map<int, int> markov_down = map.new<int, int>()
var int last_active_node      = na

// --- حافظه ایجنت‌های لایو ---
var map<int, float> agent_bull_score = map.new<int, float>()
var map<int, float> agent_bear_score = map.new<int, float>()

// متغیرهای ایمن کندل
float candle_rng = high - low
float atr_val    = ta.atr(14)

// --- تابع استخراج احتمال ترکیبی برای هر مدار هندسی ---
get_node_bull_prob(int node_idx) =>
    int m_up   = na(markov_up.get(node_idx)) ? 0 : markov_up.get(node_idx)
    int m_down = na(markov_down.get(node_idx)) ? 0 : markov_down.get(node_idx)
    int m_total = m_up + m_down
    float p_markov_bull = m_total > 0 ? (m_up / m_total) * 100.0 : 50.0

    float a_b = na(agent_bull_score.get(node_idx)) ? 0.0 : agent_bull_score.get(node_idx)
    float a_s = na(agent_bear_score.get(node_idx)) ? 0.0 : agent_bear_score.get(node_idx)
    float a_total = a_b + a_s
    float p_agent_bull = a_total > 0 ? (a_b / a_total) * 100.0 : 50.0

    // ادغام بیزی (شرطی)
    float final_bull = (p_markov_bull * 0.5) + (p_agent_bull * 0.5)
    final_bull

// --- موتور پردازش شبکه عصبی و مارکوف ---
bool is_ready = barstate.ishistory or barstate.isconfirmed

if is_ready and candle_rng > 0 and atr_val > 0
    int current_node = int(math.round(math.log(close) / res_factor))
    
    // آپدیت مارکوف
    if na(last_active_node)
        last_active_node := current_node
    else if current_node != last_active_node
        if current_node > last_active_node
            int c_up = na(markov_up.get(last_active_node)) ? 0 : markov_up.get(last_active_node)
            markov_up.put(last_active_node, c_up + 1)
        else
            int c_down = na(markov_down.get(last_active_node)) ? 0 : markov_down.get(last_active_node)
            markov_down.put(last_active_node, c_down + 1)
        last_active_node := current_node

    // آپدیت ایجنت‌ها
    int min_N = int(math.ceil(math.log(low) / res_factor))
    int max_N = int(math.floor(math.log(high) / res_factor))
    
    if min_N <= max_N
        float a1_b = ((close - low) / candle_rng)
        float a1_s = ((high - close) / candle_rng)
        
        float a2_b = math.max(0, (close - open) / atr_val)
        float a2_s = math.max(0, (open - close) / atr_val)
        float a2_sum = a2_b + a2_s
        a2_b := a2_sum > 0 ? a2_b / a2_sum : 0.5
        a2_s := a2_sum > 0 ? a2_s / a2_sum : 0.5

        for i = min_N to max_N
            float lvl = math.exp(i * res_factor)
            float a3_b = 0.5
            float a3_s = 0.5
            if low < lvl and close > lvl
                a3_b := (lvl - low) / candle_rng
                a3_s := 1.0 - a3_b
            else if high > lvl and close < lvl
                a3_s := (high - lvl) / candle_rng
                a3_b := 1.0 - a3_s

            float total_b = a1_b + a2_b + a3_b
            float total_s = a1_s + a2_s + a3_s
            
            float prev_b = na(agent_bull_score.get(i)) ? 0.0 : agent_bull_score.get(i)
            float prev_s = na(agent_bear_score.get(i)) ? 0.0 : agent_bear_score.get(i)
            
            agent_bull_score.put(i, prev_b + total_b)
            agent_bear_score.put(i, prev_s + total_s)

// --- رندرینگ لایو و شبیه‌سازی موج آینده ---
if barstate.islast
    int center_N = int(math.round(math.log(close) / res_factor))
    
    // رسم شبکه‌های افقی
    for i = center_N - 50 to center_N + 49
        float lvl = math.exp(i * res_factor)
        color line_col = color.new(#d4af37, 85)
        if i == center_N + 1
            line_col := color.new(color.red, 40)
        else if i == center_N - 1
            line_col := color.new(color.green, 40)
        line.new(bar_index, lvl, bar_index + (i_future_steps * i_step_width) + 10, lvl, color=line_col, width=1, extend=extend.none)

    // ==========================================
    // 🔮 موتور پیش‌بینی موج آینده (Future Wave Projection)
    // ==========================================
    int sim_node = center_N
    int sim_bar = bar_index
    float sim_price = close
    
    for k = 1 to i_future_steps
        // دریافت احتمال صعود در گره فعلیِ شبیه‌سازی شده
        float node_bull_prob = get_node_bull_prob(sim_node)
        
        // تصمیم‌گیری مسیر موج بر اساس احتمال قالب
        bool will_go_up = node_bull_prob > 50.0
        
        // حرکت به مدار بعدی
        int next_node = will_go_up ? sim_node + 1 : sim_node - 1
        float next_price = math.exp(next_node * res_factor)
        int next_bar = sim_bar + i_step_width
        
        // رسم خط زیگزاگ موج آینده
        color wave_col = will_go_up ? color.new(color.aqua, 10) : color.new(#ff00ff, 10) // فیروزه‌ای برای صعود، سرخابی برای نزول
        line.new(sim_bar, sim_price, next_bar, next_price, color=wave_col, width=3, style=line.style_arrow_right)
        
        // آماده‌سازی برای قدم بعدی حلقه
        sim_node := next_node
        sim_bar := next_bar
        sim_price := next_price

    // ==========================================
    // جدول مانیتورینگ لایو
    // ==========================================
    var table prob_table = table.new(position.top_right, 4, 12, border_width=1, border_color=color.rgb(30, 30, 30))
    
    table.cell(prob_table, 0, 0, "مدار کیهانی", bgcolor=color.rgb(10, 10, 12), text_color=color.rgb(220, 180, 60))
    table.cell(prob_table, 1, 0, "ماتریس مارکوف", bgcolor=color.rgb(10, 10, 12), text_color=color.rgb(180, 150, 255))
    table.cell(prob_table, 2, 0, "احتمال نهایی", bgcolor=color.rgb(30, 20, 5), text_color=color.white)
    table.cell(prob_table, 3, 0, "وضعیت موج", bgcolor=color.rgb(10, 10, 12), text_color=color.rgb(220, 180, 60))

    int row = 1
    for i = center_N + 5 to center_N - 5
        float lvl = math.exp(i * res_factor)
        float final_bull = get_node_bull_prob(i)
        
        int m_up = na(markov_up.get(i)) ? 0 : markov_up.get(i)
        int m_down = na(markov_down.get(i)) ? 0 : markov_down.get(i)
        
        bool is_touched = (high >= lvl) and (low <= lvl)
        string status_txt = is_touched ? "💥 درگیری لایو" : (i == center_N + 1 ? "مقاومت موجی" : (i == center_N - 1 ? "حمایت موجی" : "آماده"))
        color status_bg = is_touched ? color.rgb(180, 100, 10) : color.rgb(20, 20, 22)
        
        color final_bg = final_bull > 50 ? color.new(color.green, 70) : color.new(color.red, 70)
        string final_display = str.tostring(math.round(math.max(final_bull, 100 - final_bull), 1)) + "% " + (final_bull > 50 ? "صعود" : "نزول")

        table.cell(prob_table, 0, row, str.tostring(math.round(lvl, 5)), bgcolor=status_bg, text_color=color.white)
        table.cell(prob_table, 1, row, str.tostring(m_up) + " U / " + str.tostring(m_down) + " D", bgcolor=color.rgb(25, 15, 35), text_color=color.rgb(200, 150, 255))
        table.cell(prob_table, 2, row, final_display, bgcolor=final_bg, text_color=color.white)
        table.cell(prob_table, 3, row, status_txt, bgcolor=status_bg, text_color=color.white)
        
        row += 1
