//@version=5
indicator("Markov-Bayesian Harmonic Network{Morindok} by BITMOON618", overlay=true, max_lines_count=100)

// --- تنظیمات پایه ---
i_step_mult = input.float(1.0, "ضریب رزونانس فرکتال", step=0.1)

// --- ثابت‌های کیهانی ---
var float PHI = 1.618033988749895
var float PI  = 3.141592653589793
var float E   = 2.718281828459045

// معادله هندسی رزونانس (فضای توزیع نمایی)
var float res_factor = (PHI / (PI * 1000.0)) * i_step_mult

// --- حافظه زنجیره مارکوف (اتصال خطوط به یکدیگر) ---
// ثبت می‌کنیم که قیمت از مدار n به کدام مدار رفته است (n+1 یا n-1)
var map<int, int> markov_up   = map.new<int, int>()
var map<int, int> markov_down = map.new<int, int>()
var int last_active_node      = na

// --- حافظه ایجنت‌های لایو (Live Evidence) ---
var map<int, float> agent_bull_score = map.new<int, float>()
var map<int, float> agent_bear_score = map.new<int, float>()

// متغیرهای ایمن کندل
float candle_rng = high - low
float current_v  = na(volume) ? 1.0 : volume
float atr_val    = ta.atr(14)

// --- موتور پردازش شبکه عصبی و زنجیره مارکوف ---
bool is_ready = barstate.ishistory or barstate.isconfirmed

if is_ready and candle_rng > 0 and atr_val > 0
    // 1. محاسبه گره فعال فعلی (نزدیکترین خط هندسی به قیمت بسته شدن)
    int current_node = int(math.round(math.log(close) / res_factor))
    
    // 2. به‌روزرسانی اتصالات زنجیره‌ای (Markov Chain Update)
    if na(last_active_node)
        last_active_node := current_node
    else if current_node != last_active_node
        // انتقال بین گره‌ها رخ داده است!
        if current_node > last_active_node
            // قیمت از مدار قبلی به مدار بالاتر رفته است (انتقال زنجیره‌ای صعودی)
            int c_up = na(markov_up.get(last_active_node)) ? 0 : markov_up.get(last_active_node)
            markov_up.put(last_active_node, c_up + 1)
        else
            // انتقال زنجیره‌ای نزولی
            int c_down = na(markov_down.get(last_active_node)) ? 0 : markov_down.get(last_active_node)
            markov_down.put(last_active_node, c_down + 1)
        
        // آپدیت حافظه مارکوف
        last_active_node := current_node

    // 3. محاسبه ایجنت‌های در لحظه (Live Evidence) برای گره‌هایی که برخورد داشته‌اند
    int min_N = int(math.ceil(math.log(low) / res_factor))
    int max_N = int(math.floor(math.log(high) / res_factor))
    
    if min_N <= max_N
        // ایجنت 1: دلتای کندل
        float a1_b = ((close - low) / candle_rng)
        float a1_s = ((high - close) / candle_rng)
        
        // ایجنت 2: مومنتوم جنبشی نسبت به ATR
        float a2_b = math.max(0, (close - open) / atr_val)
        float a2_s = math.max(0, (open - close) / atr_val)
        float a2_sum = a2_b + a2_s
        a2_b := a2_sum > 0 ? a2_b / a2_sum : 0.5
        a2_s := a2_sum > 0 ? a2_s / a2_sum : 0.5

        for i = min_N to max_N
            float lvl = math.exp(i * res_factor)
            
            // ایجنت 3: جذب نقدینگی روی مدار دقیق
            float a3_b = 0.5
            float a3_s = 0.5
            if low < lvl and close > lvl
                a3_b := (lvl - low) / candle_rng
                a3_s := 1.0 - a3_b
            else if high > lvl and close < lvl
                a3_s := (high - lvl) / candle_rng
                a3_b := 1.0 - a3_s

            // تجمیع امتیاز ایجنت‌های کندلی برای این مدار خاص
            float total_b_score = a1_b + a2_b + a3_b
            float total_s_score = a1_s + a2_s + a3_s
            
            float prev_b = na(agent_bull_score.get(i)) ? 0.0 : agent_bull_score.get(i)
            float prev_s = na(agent_bear_score.get(i)) ? 0.0 : agent_bear_score.get(i)
            
            agent_bull_score.put(i, prev_b + total_b_score)
            agent_bear_score.put(i, prev_s + total_s_score)

// --- رندرینگ و محاسبه احتمال نهایی مبتنی بر بیز (Bayes) در لحظه ---
if barstate.islast
    int center_N = int(math.round(math.log(close) / res_factor))
    
    // رسم شبکه
    for i = center_N - 50 to center_N + 49
        float lvl = math.exp(i * res_factor)
        color line_col = color.new(#d4af37, 85)
        
        if i == center_N + 1
            line_col := color.new(color.red, 40)
        else if i == center_N - 1
            line_col := color.new(color.green, 40)
            
        line.new(bar_index, lvl, bar_index + 10, lvl, color=line_col, width=1, extend=extend.right)

    // UI و خروجی ماتریس
    var table prob_table = table.new(position.top_right, 5, 12, border_width=1, border_color=color.rgb(30, 30, 30))
    
    color bg_h = color.rgb(10, 10, 12)
    color txt_h = color.rgb(220, 180, 60)
    
    table.cell(prob_table, 0, 0, "مدار کیهانی", bgcolor=bg_h, text_color=txt_h)
    table.cell(prob_table, 1, 0, "احتمال زنجیره‌ای (مارکوف)", bgcolor=bg_h, text_color=color.rgb(180, 150, 255))
    table.cell(prob_table, 2, 0, "قدرت لایو (ایجنت‌ها)", bgcolor=bg_h, text_color=color.rgb(150, 255, 180))
    table.cell(prob_table, 3, 0, "وضعیت درگیری", bgcolor=bg_h, text_color=txt_h)
    table.cell(prob_table, 4, 0, "🔥 احتمال نهایی ترکیبی", bgcolor=color.rgb(30, 20, 5), text_color=color.white)

    int row = 1
    for i = center_N + 5 to center_N - 5
        float lvl = math.exp(i * res_factor)
        
        // 1. استخراج احتمال زنجیره‌ای (Markov Chain - Prior)
        int m_up   = na(markov_up.get(i)) ? 0 : markov_up.get(i)
        int m_down = na(markov_down.get(i)) ? 0 : markov_down.get(i)
        int m_total = m_up + m_down
        float chain_bull_prob = m_total > 0 ? (m_up / m_total) * 100.0 : 50.0
        
        // 2. استخراج احتمال لایو ایجنت‌ها (Evidence)
        float a_bull = na(agent_bull_score.get(i)) ? 0.0 : agent_bull_score.get(i)
        float a_bear = na(agent_bear_score.get(i)) ? 0.0 : agent_bear_score.get(i)
        float a_total = a_bull + a_bear
        float live_bull_prob = a_total > 0 ? (a_bull / a_total) * 100.0 : 50.0
        
        // 3. ترکیب شرطی (Bayesian Fusion)
        // اگر دیتا مارکوف قوی باشد، وزن آن بیشتر است. در غیر این صورت ایجنت‌ها تصمیم می‌گیرند.
        float final_bull_prob = (chain_bull_prob * 0.5) + (live_bull_prob * 0.5) // ادغام ۵۰/۵۰ برای تعادل ریاضی
        float final_bear_prob = 100.0 - final_bull_prob

        // منطق برخورد در لحظه (Dynamic at Touch)
        bool is_touched = (high >= lvl) and (low <= lvl)
        
        string status_txt = ""
        color status_bg = color.rgb(20, 20, 22)
        color status_txt_c = color.rgb(190, 190, 190)

        if is_touched
            status_txt := "💥 محاسبه لایو فعال"
            status_bg := color.rgb(180, 100, 10)
            status_txt_c := color.white
        else if i == center_N + 1
            status_txt := "ایستگاه بعدی (مقاومت)"
        else if i == center_N - 1
            status_txt := "ایستگاه بعدی (حمایت)"
        else
            status_txt := "آماده ارتباط"

        // استخراج خروجی ستون نهایی
        float winning_prob = math.max(final_bull_prob, final_bear_prob)
        string final_dir = final_bull_prob > 50 ? "صعودی" : "نزولی"
        color final_bg = final_bull_prob > 50 ? color.new(color.green, 70) : color.new(color.red, 70)
        string final_display = str.tostring(math.round(winning_prob, 1)) + "% " + final_dir

        // چاپ جدول
        table.cell(prob_table, 0, row, str.tostring(math.round(lvl, 5)), bgcolor=status_bg, text_color=status_txt_c)
        table.cell(prob_table, 1, row, str.tostring(math.round(chain_bull_prob, 1)) + "% صعود", bgcolor=color.rgb(25, 15, 35), text_color=color.rgb(200, 150, 255))
        table.cell(prob_table, 2, row, str.tostring(math.round(live_bull_prob, 1)) + "% صعود", bgcolor=color.rgb(15, 35, 25), text_color=color.rgb(150, 255, 180))
        table.cell(prob_table, 3, row, status_txt, bgcolor=status_bg, text_color=status_txt_c)
        table.cell(prob_table, 4, row, final_display, bgcolor=final_bg, text_color=color.white)
        
        row += 1
