//@version=5
indicator("Price Rotation Around Level", overlay=true)

// ÊäÙíãÇÊ æÑæÏí
useMa = input.bool(false, "Use Moving Average as Center")
maLength = input.int(20, "MA Length", minval=1)
maType = input.string("SMA", "MA Type", options=["SMA", "EMA"])
centerPrice = input.price(0, "Center Level", confirm=true)
pivotTime = input.time(timestamp("01 Jan 2024 00:00 +0000"), "Pivot Time", confirm=true)
slopePercent = input.float(0, "Slope (% per bar)", minval=-10, maxval=10, step=0.1)

// ãÍÇÓÈå ãíÇäíä ãÊÍÑ˜
ma = maType == "SMA" ? ta.sma(close, maLength) : ta.ema(close, maLength)

// ÇäÊÎÇÈ ÓØÍ ãÑ˜Òí
center = useMa ? ma : centerPrice

// ãÍÇÓÈå ÔíÈ
barsSincePivot = (time - pivotTime) / (time - time[1])
slopeOffset = (slopePercent / 100) * center * barsSincePivot

// ãÍÇÓÈå ÂİÓÊ ŞíãÊåÇí æÇŞÚí äÓÈÊ Èå ãÑ˜Ò
openOffset = open - center
highOffset = high - center
lowOffset = low - center
closeOffset = close - center

// ŞíãÊåÇí ÑÎÔ íÇİÊå
rotatedOpen = center + slopeOffset + openOffset
rotatedHigh = center + slopeOffset + highOffset
rotatedLow = center + slopeOffset + lowOffset
rotatedClose = center + slopeOffset + closeOffset

// ŞíãÊåÇí ãÚ˜æÓ
invertedOpen = center + slopeOffset - openOffset
invertedHigh = center + slopeOffset - highOffset
invertedLow = center + slopeOffset - lowOffset
invertedClose = center + slopeOffset - closeOffset

// ÑÓã ˜äÏáåÇí ÑÎÔ íÇİÊå
plotcandle(rotatedOpen, rotatedHigh, rotatedLow, rotatedClose, "Rotated Price", 
           color=rotatedClose >= rotatedOpen ? color.green : color.red, 
           wickcolor=rotatedClose >= rotatedOpen ? color.green : color.red, 
           bordercolor=rotatedClose >= rotatedOpen ? color.green : color.red)

// ÑÓã ˜äÏáåÇí ãÚ˜æÓ
plotcandle(invertedOpen, math.max(invertedHigh, invertedLow), math.min(invertedHigh, invertedLow), invertedClose, "Inverted Price", 
           color=invertedClose >= invertedOpen ? color.blue : color.orange, 
           wickcolor=invertedClose >= invertedOpen ? color.blue : color.orange, 
           bordercolor=invertedClose >= invertedOpen ? color.blue : color.orange)

// ÑÓã ÎØ ãÑ˜Òí ÔíÈÏÇÑ
plot(center + slopeOffset, "Sloped Center", color=color.gray, linewidth=1, style=plot.style_line)
