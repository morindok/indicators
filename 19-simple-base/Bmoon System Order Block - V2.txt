//@version=6
indicator("BMoon System Order Block{Morindok}", overlay=true, max_bars_back=4000)

n = input.int(50, title="Periods", minval=2)
lineMode = input.string("Unified Highs/Lows (Yellow/Green)", title="Line Mode", options=["Separated Individual (Yellow/Green)", "Individual Chronological (White)", "Individual Alternating (Orange)", "Unified Alternating (Red)", "Unified Highs/Lows (Yellow/Green)"])

// Point Selections on Chart
mirrorLevel1 = input.price(0, title="P1 Level / Range Point 1", confirm=true)
selectedTime1 = input.time(0, title="P1 Time / Range Point 1", confirm=true)
mirrorLevel2 = input.price(0, title="P2 Level / Range Point 2", confirm=true)
selectedTime2 = input.time(0, title="P2 Time / Range Point 2", confirm=true)

// Auto Fib Settings
useAutoFib = input.bool(true, title="Enable Auto Fib (Uses P1 & P2 as Range)", group="Auto Fib Settings")
useSmartEngine = input.bool(true, title="Enable Smart Dynamic Engine", tooltip="Adapts coefficients based on post-range breakout and velocity.", group="Auto Fib Settings")

timeCoefP1 = input.float(0.382, title="Time Coefficient P1", step=0.001, group="Auto Fib Settings")
levelCoefP1 = input.float(0.0, title="Level Coefficient P1", step=0.001, group="Auto Fib Settings")
timeCoefP2 = input.float(0.618, title="Time Coefficient P2", step=0.001, group="Auto Fib Settings")
levelCoefP2 = input.float(1, title="Level Coefficient P2", step=0.001, group="Auto Fib Settings")

// Determine Range Boundaries
int tStart = math.min(selectedTime1, selectedTime2)
int tEnd = math.max(selectedTime1, selectedTime2)

var float rangeHigh = na
var float rangeLow = na
if time >= tStart and time <= tEnd
    rangeHigh := na(rangeHigh) ? high : math.max(rangeHigh, high)
    rangeLow := na(rangeLow) ? low : math.min(rangeLow, low)

// --- Smart Dynamic Engine Logic ---
var float dynLevelCoefP1 = na
var float dynLevelCoefP2 = na
var float dynTimeCoefP1 = na
var float dynTimeCoefP2 = na
var bool engineLocked = false
var int barsAfterRange = 0

if useSmartEngine and time > tEnd and not engineLocked and not na(rangeHigh) and not na(rangeLow)
    barsAfterRange += 1
    float rangeDist = math.max((rangeHigh - rangeLow), syminfo.mintick)
    
    // Calculate breakout and velocity
    bool upBreakout = close > rangeHigh
    bool dnBreakout = close < rangeLow
    float velocity = ta.tr / rangeDist
    
    if upBreakout or dnBreakout or barsAfterRange > 20
        if upBreakout or (barsAfterRange > 20 and close > rangeHigh - (rangeDist*0.5))
            dynLevelCoefP1 := 0.618  
            dynLevelCoefP2 := 0.786  
        else
            dynLevelCoefP1 := 0.382  
            dynLevelCoefP2 := 0.214  
            
        if velocity > 0.5 
            dynTimeCoefP1 := 0.236
            dynTimeCoefP2 := 0.382
        else 
            dynTimeCoefP1 := 0.5
            dynTimeCoefP2 := 0.618
            
        engineLocked := true

float finalLevelCoefP1 = useSmartEngine and engineLocked ? dynLevelCoefP1 : levelCoefP1
float finalLevelCoefP2 = useSmartEngine and engineLocked ? dynLevelCoefP2 : levelCoefP2
float finalTimeCoefP1 = useSmartEngine and engineLocked ? dynTimeCoefP1 : timeCoefP1
float finalTimeCoefP2 = useSmartEngine and engineLocked ? dynTimeCoefP2 : timeCoefP2
// ----------------------------------

// Calculate Final Levels and Times
float mLevel1 = useAutoFib ? (rangeLow + (rangeHigh - rangeLow) * finalLevelCoefP1) : mirrorLevel1
float mLevel2 = useAutoFib ? (rangeLow + (rangeHigh - rangeLow) * finalLevelCoefP2) : mirrorLevel2
int sTime1 = useAutoFib ? int(tStart + (tEnd - tStart) * finalTimeCoefP1) : selectedTime1
int sTime2 = useAutoFib ? int(tStart + (tEnd - tStart) * finalTimeCoefP2) : selectedTime2

float mHigh1 = float(2.0 * mLevel1 - low)
float mLow1  = float(2.0 * mLevel1 - high)

float mHigh2 = float(2.0 * mLevel2 - low)
float mLow2  = float(2.0 * mLevel2 - high)

bool u_dn = true, bool u_u0 = true, bool u_u1 = true, bool u_u2 = true, bool u_u3 = true, bool u_u4 = true
bool d_dn = true, bool d_u0 = true, bool d_u1 = true, bool d_u2 = true, bool d_u3 = true, bool d_u4 = true

for i = 1 to n
    u_dn := u_dn and (high[n-i] < high[n])
    u_u0 := u_u0 and (high[n+i] < high[n])
    u_u1 := u_u1 and (high[n+1] <= high[n] and high[n+i+1] < high[n])
    u_u2 := u_u2 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+i+2] < high[n])
    u_u3 := u_u3 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+i+3] < high[n])
    u_u4 := u_u4 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+4] <= high[n] and high[n+i+4] < high[n])
    
    d_dn := d_dn and (low[n-i] > low[n])
    d_u0 := d_u0 and (low[n+i] > low[n])
    d_u1 := d_u1 and (low[n+1] >= low[n] and low[n+i+1] > low[n])
    d_u2 := d_u2 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+i+2] > low[n])
    d_u3 := d_u3 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+i+3] > low[n])
    d_u4 := d_u4 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+4] >= low[n] and low[n+i+4] > low[n])

bool flagUp = u_u0 or u_u1 or u_u2 or u_u3 or u_u4
bool flagDn = d_u0 or d_u1 or d_u2 or d_u3 or d_u4

bool upFractal1 = u_dn and flagUp and time[n] < sTime1
bool downFractal1 = d_dn and flagDn and time[n] < sTime1

bool upFractal2 = u_dn and flagUp and time[n] < sTime2
bool downFractal2 = d_dn and flagDn and time[n] < sTime2

var int refBarIndex1 = na
var int refBarIndex2 = na
if time <= sTime1
    refBarIndex1 := bar_index
if time <= sTime2
    refBarIndex2 := bar_index

var upFracIdx1 = array.new_int(0)
var dnFracIdx1 = array.new_int(0)
if upFractal1
    array.push(upFracIdx1, int(bar_index - n))
if downFractal1
    array.push(dnFracIdx1, int(bar_index - n))

var upFracIdx2 = array.new_int(0)
var dnFracIdx2 = array.new_int(0)
if upFractal2
    array.push(upFracIdx2, int(bar_index - n))
if downFractal2
    array.push(dnFracIdx2, int(bar_index - n))

bool p1Up = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(upFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p1Dn = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(dnFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p2Up = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(upFracIdx2, 2 * refBarIndex2 - bar_index) >= 0
bool p2Dn = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(dnFracIdx2, 2 * refBarIndex2 - bar_index) >= 0

var int yPrevX = na, var float yPrevY = na
var int yCurrX = na, var float yCurrY = na
var int gPrevX = na, var float gPrevY = na
var int gCurrX = na, var float gCurrY = na

var float ix1 = na, var float iy1 = na
var float ix2 = na, var float iy2 = na
var float ix3 = na, var float iy3 = na
var float ix4 = na, var float iy4 = na
var int intCount = 0

// --- 5-Wave Optimal Box Engine (Background Tracker) ---
var float[] pivots = array.new_float(0)
var int[] pivotDirs = array.new_int(0)

float ph = ta.pivothigh(high, 10, 10)
float pl = ta.pivotlow(low, 10, 10)

if not na(ph)
    array.unshift(pivots, ph)
    array.unshift(pivotDirs, 1)
if not na(pl)
    array.unshift(pivots, pl)
    array.unshift(pivotDirs, -1)

if array.size(pivots) > 6
    array.pop(pivots)
    array.pop(pivotDirs)
// --------------------------------------------------------

var bool finalBoxDrawn = false
bool triggerIntersectionCheck = false

if lineMode == "Unified Highs/Lows (Yellow/Green)"
    if p1Up
        yPrevX := yCurrX
        yPrevY := yCurrY
        yCurrX := bar_index
        yCurrY := mHigh1
        triggerIntersectionCheck := true
    if p2Up
        yPrevX := yCurrX
        yPrevY := yCurrY
        yCurrX := bar_index
        yCurrY := mHigh2
        triggerIntersectionCheck := true

    if p1Dn
        gPrevX := gCurrX
        gPrevY := gCurrY
        gCurrX := bar_index
        gCurrY := mLow1
        triggerIntersectionCheck := true
    if p2Dn
        gPrevX := gCurrX
        gPrevY := gCurrY
        gCurrX := bar_index
        gCurrY := mLow2
        triggerIntersectionCheck := true

    if triggerIntersectionCheck and not na(yPrevX) and not na(yCurrX) and not na(gPrevX) and not na(gCurrX)
        if (yCurrX != yPrevX) and (gCurrX != gPrevX)
            float m1 = (yCurrY - yPrevY) / (yCurrX - yPrevX)
            float m2 = (gCurrY - gPrevY) / (gCurrX - gPrevX)
            if m1 != m2
                float ix = (m1 * yPrevX - m2 * gPrevX + gPrevY - yPrevY) / (m1 - m2)
                if ix >= math.max(yPrevX, gPrevX) and ix <= math.min(yCurrX, gCurrX)
                    float iy = m1 * (ix - yPrevX) + yPrevY
                    if intCount == 0
                        ix1 := ix
                        iy1 := iy
                        intCount := 1
                    else if intCount == 1
                        ix2 := ix
                        iy2 := iy
                        intCount := 2
                    else if intCount == 2
                        ix3 := ix
                        iy3 := iy
                        intCount := 3
                    else if intCount == 3
                        ix4 := ix
                        iy4 := iy
                        intCount := 4

    // --- Box Drawing with Optimal Reversal Zone Check ---
    if intCount == 4 and not finalBoxDrawn and bar_index >= math.round(ix4)
        int offset4 = math.max(0, int(bar_index - math.round(ix4)))
        
        if offset4 < 4000
            float optTop = high[offset4]
            float optBottom = low[offset4]
            
            // Apply 5-Wave logic to optimize the box coordinates
            if array.size(pivots) >= 5
                float p0 = array.get(pivots, 4)
                float p1 = array.get(pivots, 3)
                float p2 = array.get(pivots, 2)
                float p3 = array.get(pivots, 1)
                float p4 = array.get(pivots, 0)
                
                bool isBullWave = array.get(pivotDirs, 4) == -1 and p1 > p0 and p2 > p0 and p3 > p1 and p4 > p2
                bool isBearWave = array.get(pivotDirs, 4) == 1 and p1 < p0 and p2 < p0 and p3 < p1 and p4 < p2
                
                if isBullWave or isBearWave
                    float w3Length = math.abs(p3 - p2)
                    float target1 = isBullWave ? p4 + (w3Length * 1.272) : p4 - (w3Length * 1.272)
                    float target2 = isBullWave ? p4 + (w3Length * 1.618) : p4 - (w3Length * 1.618)
                    
                    optTop := math.max(target1, target2)
                    optBottom := math.min(target1, target2)
            
            box.new(left=math.round(ix4), top=optTop, right=math.round(ix4)+1, bottom=optBottom, border_color=color.orange, border_width=2, bgcolor=color.new(color.orange, 80), extend=extend.right)
            finalBoxDrawn := true
