//@version=5
indicator("Live Order Flow Predictor - History Log", overlay=true, max_labels_count=500)

// --- ãÊÛíÑåÇí ĞÎíÑå ÓÇÒí áÇíæ (varip) ---
varip float lastPrice = na
varip float lastVol   = na

varip int   upTicks   = 0
varip int   downTicks = 0

varip float upVol     = 0.0
varip float downVol   = 0.0

varip float upDist    = 0.0
varip float downDist  = 0.0

// ÂÑÇíå ÈÑÇí ãÍÇİÙÊ ÏÇÆãí ÇÒ áíÈáåÇí ˜äÏáåÇí ÈÓÊå ÔÏå
var label[] history_labels = array.new_label()

// ÇíÌÇÏ ÌÏæá ÈÑÇí äãÇíÔ ÂãÇÑ æ íÔÈíäí
var table statsTable = table.new(position.top_right, 2, 6, border_width = 1, border_color = color.new(color.gray, 50))

// ÑíÓÊ ˜ÑÏä ÔãÇÑäÏååÇ ÏÑ Çæáíä Êí˜ åÑ ˜äÏá ÌÏíÏ
if barstate.isnew
    upTicks   := 0
    downTicks := 0
    upVol     := 0.0
    downVol   := 0.0
    upDist    := 0.0
    downDist  := 0.0
    lastPrice := open
    lastVol   := 0.0

// İŞØ ÏÑ áÇíæ ãÇÑ˜Ê ãÍÇÓÈå ÔæÏ
if barstate.isrealtime
    
    // ãÍÇÓÈå ÊÛííÑÇÊ ÇÒ Êí˜ ŞÈáí
    if not na(lastPrice)
        float priceChange = close - lastPrice
        float volChange   = volume - nz(lastVol, 0)
        volChange := volChange < 0 ? 0 : volChange
        
        // ÊÔÎíÕ Êí˜ ÕÚæÏí íÇ äÒæáí æ ÊÎÕíÕ ÏíÊÇ
        if priceChange > 0
            upTicks += 1
            upVol   += volChange
            upDist  += priceChange
        else if priceChange < 0
            downTicks += 1
            downVol   += volChange
            downDist  += math.abs(priceChange)
            
    // ÂÏíÊ ŞíãÊ æ ÍÌã ÈÑÇí Êí˜ ÈÚÏí
    lastPrice := close
    lastVol   := volume

// --- ÓíÓÊã ÂãÇÑí æ ãÍÇÓÈå İÔÇÑ (Pressure Score) ---
float totalVol  = upVol + downVol
float totalDist = upDist + downDist

float bullScore = (upVol / math.max(totalVol, 1)) * 0.5 + (upDist / math.max(totalDist, 1)) * 0.5
float bearScore = (downVol / math.max(totalVol, 1)) * 0.5 + (downDist / math.max(totalDist, 1)) * 0.5

// íÔÈíäí Ñä ˜äÏá ÈÚÏí ÈÑÇí ÌÏæá
string predictionText  = "Neutral"
color  predictionColor = color.gray

if bullScore > bearScore
    predictionText  := "Green (Bullish)"
    predictionColor := color.new(color.green, 20)
else if bearScore > bullScore
    predictionText  := "Red (Bearish)"
    predictionColor := color.new(color.red, 20)

// --- ÈÑæÒÑÓÇäí ÌÏæá (İŞØ ÏÑ áÇíæ ãÇÑ˜Ê) ---
if barstate.isrealtime
    table.cell(statsTable, 0, 0, "Metric", text_color=color.white, bgcolor=color.new(color.black, 20))
    table.cell(statsTable, 1, 0, "Live Value", text_color=color.white, bgcolor=color.new(color.black, 20))
    
    table.cell(statsTable, 0, 1, "Up / Down Ticks", text_color=color.white, bgcolor=color.new(color.blue, 80))
    table.cell(statsTable, 1, 1, str.tostring(upTicks) + " / " + str.tostring(downTicks), text_color=color.white, bgcolor=color.new(color.black, 40))
    
    table.cell(statsTable, 0, 2, "Up / Down Vol", text_color=color.white, bgcolor=color.new(color.blue, 80))
    table.cell(statsTable, 1, 2, str.tostring(upVol, "#.##") + " / " + str.tostring(downVol, "#.##"), text_color=color.white, bgcolor=color.new(color.black, 40))
    
    table.cell(statsTable, 0, 3, "Up / Down Dist", text_color=color.white, bgcolor=color.new(color.blue, 80))
    table.cell(statsTable, 1, 3, str.tostring(upDist, "#.####") + " / " + str.tostring(downDist, "#.####"), text_color=color.white, bgcolor=color.new(color.black, 40))
    
    table.cell(statsTable, 0, 4, "Next Candle Prediction", text_color=color.white, bgcolor=color.new(color.black, 0))
    table.cell(statsTable, 1, 4, predictionText, text_color=color.white, bgcolor=predictionColor)

// --- ?? ËÈÊ ŞØÚí áíÈáåÇ ÏÑ ÇíÇä ˜äÏá Èå åãÑÇå äÊíÌå íÔÈíäí ?? ---
if barstate.isconfirmed
    // ÊÔÎíÕ äæÚ íÔÈíäí ÏÑ ÂÎÑíä Êí˜ ˜äÏá ÌÇÑí
    string histPred = bullScore > bearScore ? "g" : (bearScore > bullScore ? "r" : "n")
    
    // ÊäÙíã Ñä áíÈá ÈÑ ÇÓÇÓ íÔÈíäí (ÓÈÒ ÈÑÇí g¡ ŞÑãÒ ÈÑÇí r¡ ÎÇ˜ÓÊÑí ÈÑÇí ÎäËí)
    color lblColor  = histPred == "g" ? color.new(color.green, 30) : (histPred == "r" ? color.new(color.red, 30) : color.new(color.gray, 50))
    
    // ÓÇÎÊ ãÊä áíÈá ÊÇÑíÎí (ÔÇãá ÍÑİ íÔÈíäí æ ÇØáÇÚÇÊ ÍÌã)
    string historyText = "Pred: " + histPred + "\nVol: " + str.tostring(upVol, "#.#") + "/" + str.tostring(downVol, "#.#")
    
    saved_lbl = label.new(bar_index, high, historyText, color=lblColor, textcolor=color.white, style=label.style_label_down, size=size.small)
    
    // ĞÎíÑå ÏÑ ÂÑÇíå ÈÑÇí ÌáæíÑí ÇÒ Ç˜ ÔÏä ÊæÓØ áÊİÑã
    array.push(history_labels, saved_lbl)
    
    // ãÏíÑíÊ ÍÇİÙå
    if array.size(history_labels) > 490
        label.delete(array.shift(history_labels))
