//@version=5
indicator("Quantum Tick-Volume Collapse{Morindok} by @BITMOON618 V3", overlay=true, max_labels_count=500)

// =====================================================================
// QUANTUM TIMELINE COLLAPSE SYSTEM
// Two probability "timelines" (UP-branch / DOWN-branch) are derived from
// the same combined tick+volume metric (f_probUp). While the live bar is
// still forming, both timelines coexist (superposition) and the real
// candle is painted with a translucent blend of both. The instant the
// bar closes (barstate.isconfirmed = the observation event), whichever
// timeline carries more weight collapses into the bar's REAL, solid
// color. Historical bars have no live tick data, so they are left
// untouched (native chart color).
// =====================================================================

// --- Inputs: Tick/Volume Engine ---
int   lookback = input.int(50, "Lookback Bars (History)", minval=1, maxval=480, group="Tick/Volume Engine")
float wTick    = input.float(0.4, "Tick Count Weight", minval=0.0, maxval=1.0, step=0.05, group="Tick/Volume Engine")
float wVol     = input.float(0.6, "Volume Weight",     minval=0.0, maxval=1.0, step=0.05, group="Tick/Volume Engine")

// --- Inputs: Quantum Timeline Collapse ---
color upColorInput       = input.color(color.green, "Up Timeline Color (Collapsed)",   group="Quantum Timeline Collapse")
color downColorInput     = input.color(color.red,   "Down Timeline Color (Collapsed)", group="Quantum Timeline Collapse")
color neutralColorInput  = input.color(color.gray,  "Neutral / Tie Color",             group="Quantum Timeline Collapse")
float neutralZone        = input.float(5.0, "Neutral Zone Width (± around 50%)", minval=0.0, maxval=45.0, step=0.5, group="Quantum Timeline Collapse")
bool  showSuperposition  = input.bool(true, "Show Superposition Blend While Forming", group="Quantum Timeline Collapse")
bool  showCollapseMarker = input.bool(true, "Show Collapse Marker On Close", group="Quantum Timeline Collapse")

// --- Live (per-tick) variables ---
varip float lastPrice = na
varip float lastVol   = 0.0
varip int   upTicks   = 0
varip int   downTicks = 0
varip float upVol     = 0.0
varip float downVol   = 0.0

// --- Persistent history ---
var label[] history_labels = array.new_label()
var float[] hUpTicks = array.new_float()
var float[] hDnTicks = array.new_float()
var float[] hUpVol   = array.new_float()
var float[] hDnVol   = array.new_float()

// Moving label for the live bar
var label live_label = label.new(na, na, "", color=color.new(color.blue, 70), textcolor=color.white, style=label.style_label_down)

// Quantum-derived candle color (na = untouched / native chart color)
var color quantumColor = na

// --- Cumulative probability (history + current live bar) ---
f_probUp() =>
    float sUpT = float(upTicks)
    float sDnT = float(downTicks)
    float sUpV = upVol
    float sDnV = downVol
    int sz = array.size(hUpTicks)
    int nStart = math.max(0, sz - lookback)
    if sz > 0
        for i = nStart to sz - 1
            sUpT += array.get(hUpTicks, i)
            sDnT += array.get(hDnTicks, i)
            sUpV += array.get(hUpVol, i)
            sDnV += array.get(hDnVol, i)
    float totT = sUpT + sDnT
    float totV = sUpV + sDnV
    float rT = totT > 0 ? sUpT / totT : 0.5
    float rV = totV > 0 ? sUpV / totV : 0.5
    float wSum = wTick + wVol
    wSum > 0 ? ((wTick * rT + wVol * rV) / wSum) * 100.0 : 50.0

// --- Reset on the first tick of a new bar ---
if barstate.isnew
    upTicks   := 0
    downTicks := 0
    upVol     := 0.0
    downVol   := 0.0
    lastPrice := open
    lastVol   := 0.0

// --- Live market only ---
if barstate.isrealtime
    // Volume delivered by this tick
    float dVol = math.max(volume - lastVol, 0.0)

    if not na(lastPrice)
        if close > lastPrice
            upTicks += 1
            upVol   += dVol
        else if close < lastPrice
            downTicks += 1
            downVol   += dVol
    lastPrice := close
    lastVol   := volume

    // Cumulative probability (the combined tick+volume metric)
    float pUp = f_probUp()
    float pDn = 100.0 - pUp

    // The two quantum timelines: which branch currently carries more weight?
    bool upLeading   = pUp > (50.0 + neutralZone)
    bool downLeading = pUp < (50.0 - neutralZone)
    string leadTxt   = upLeading ? "UP" : downLeading ? "DOWN" : "NEUTRAL"

    // Compact text gauge (10 blocks) visualizing the balance of the two timelines
    int filled = int(math.round(pUp / 10.0))
    string gauge = ""
    for i = 0 to 9
        gauge := gauge + (i < filled ? "?" : "?")

    // Label color based on dominant side
    color lblCol = upLeading ? color.new(color.green, 60) : downLeading ? color.new(color.red, 60) : color.new(color.blue, 70)

    string txt = "Up Ticks: "   + str.tostring(upTicks)   + " | Vol: " + str.tostring(upVol,   format.volume) +
             "\nDown Ticks: "   + str.tostring(downTicks) + " | Vol: " + str.tostring(downVol, format.volume) +
             "\nUP Prob: "      + str.tostring(pUp, "#.#") + "%   DOWN Prob: " + str.tostring(pDn, "#.#") + "%" +
             "\n[" + gauge + "]" +
             "\nTimeline: "     + leadTxt

    label.set_xy(live_label, bar_index, high + ta.tr * 0.5)
    label.set_text(live_label, txt)
    label.set_color(live_label, lblCol)

    // SUPERPOSITION: both timelines coexist while the bar is still forming.
    // The real candle is blended by the current relative weight of each side.
    quantumColor := showSuperposition ? color.new(color.from_gradient(pUp, 0.0, 100.0, downColorInput, upColorInput), 55) : na

    // Commit data when the bar closes
    if barstate.isconfirmed
        // Store this bar's data into history
        array.push(hUpTicks, float(upTicks))
        array.push(hDnTicks, float(downTicks))
        array.push(hUpVol,   upVol)
        array.push(hDnVol,   downVol)

        // Saved-label color follows the probability (same logic as the live label)
        color savedCol = upLeading ? color.new(color.green, 40) : downLeading ? color.new(color.red, 40) : color.new(color.gray, 20)

        label saved_lbl = label.new(bar_index, high, txt, color=savedCol, textcolor=color.white, style=label.style_label_down)
        array.push(history_labels, saved_lbl)

        // COLLAPSE (observation event): the winning timeline becomes the bar's REAL color
        quantumColor := upLeading ? color.new(upColorInput, 15) : downLeading ? color.new(downColorInput, 15) : color.new(neutralColorInput, 15)

        // Memory management
        if array.size(history_labels) > 490
            label.delete(array.shift(history_labels))
        if array.size(hUpTicks) > 490
            array.shift(hUpTicks)
            array.shift(hDnTicks)
            array.shift(hUpVol)
            array.shift(hDnVol)

// Paint the REAL candle: translucent blend while forming, solid color once collapsed.
// Historical bars keep native coloring (no live tick data exists for them).
barcolor(quantumColor, title="Quantum Timeline Color")

// Mark the exact moment of collapse (bar close) with a small diamond
plotshape(showCollapseMarker and barstate.isrealtime and barstate.isconfirmed, title="Quantum Collapse Marker", location=location.abovebar, style=shape.diamond, size=size.tiny, color=quantumColor)
