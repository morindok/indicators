length = input.int(14, title="Wave Length", minval=1)

// ›—„Ê·: WMA(2*WMA(n/2) - WMA(n), sqrt(n))
wma_half = ta.wma(close, math.floor(length / 2))
wma_full = ta.wma(close, length)
wma_diff = 2 * wma_half - wma_full
smooth_wave = ta.wma(wma_diff, math.floor(math.sqrt(length)))

// —”„ „ÊÃ
plot(smooth_wave, color=color.new(color.fuchsia, 0), linewidth=3, title="Zero-Lag Wave")
