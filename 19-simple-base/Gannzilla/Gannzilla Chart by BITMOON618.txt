f_anchor() =>
    float anch = na
    if i_anchorMode == "œ” Ì"
        anch := i_manualAnchor
    else if i_anchorMode == "¬Œ—Ì‰ High"
        anch := array.size(ph_price) > 0 ? array.get(ph_price, 0) : na
    else if i_anchorMode == "¬Œ—Ì‰ Low"
        anch := array.size(pl_price) > 0 ? array.get(pl_price, 0) : na
    else  // ¬Œ—Ì‰ ÅÌÊÊ 
        float lastH = array.size(ph_price) > 0 ? array.get(ph_price, 0) : na
        float lastL = array.size(pl_price) > 0 ? array.get(pl_price, 0) : na
        int   barH  = array.size(ph_bar)   > 0 ? array.get(ph_bar, 0)   : 0
        int   barL  = array.size(pl_bar)   > 0 ? array.get(pl_bar, 0)   : 0
        if na(lastH) and na(lastL)
            anch := na
        else if na(lastH)
            anch := lastL
        else if na(lastL)
            anch := lastH
        else
            anch := barH >= barL ? lastH : lastL
    anch

// Â— 0.25 = 45 œ—ÃÂ  |  Â— 0.5 = 90∞  |  Â— 1.0 = 180∞  |  Â— 2.0 = 360∞
// Resistance: (?P + n◊0.25)≤
// Support:    (?P - n◊0.25)≤
// ?????????????????????????????????????????????
f_sq9(price, steps) =>
    sqrtP = math.sqrt(price)
    math.pow(sqrtP + steps * 0.25, 2)

f_sq9dn(price, steps) =>
    sqrtP = math.sqrt(price)
    val   = sqrtP - steps * 0.25
    val > 0 ? math.pow(val, 2) : na
