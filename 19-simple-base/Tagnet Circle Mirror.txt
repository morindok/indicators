// ---  «»⁄ —”„ œ«Ì—ÂùÂ« »« ÷—Ì» „ﬁÌ«” „”«Õ  ---
draw_tangent_circles(t_start, p_start, t_end, p_end, col1, col2, m1, m2) =>
    float dt = float(t_end - t_start)
    float dp = p_end - p_start

    float ux = dt == 0 ? 1 : dt / math.abs(dt)
    float uy = dp == 0 ? 1 : dp / math.abs(dp)
    float nx = -uy
    float ny = ux

    float r_t = math.abs(dt) / 4.0
    float r_p = math.abs(dp) / 4.0

    float c1t = float(t_start) + dt * 0.25
    float c1p = p_start + dp * 0.25
    float c2t = float(t_start) + dt * 0.75
    float c2p = p_start + dp * 0.75

    int N = 120
    pts1 = array.new<chart.point>()
    pts2 = array.new<chart.point>()

    for i = 0 to N - 1
        float angle = 2.0 * math.pi * i / N
        float ca = math.cos(angle)
        float sa = math.sin(angle)

        array.push(pts1, chart.point.from_time(int(c1t + (r_t * m1) * (ca * ux + sa * nx)), c1p + (r_p * m1) * (ca * uy + sa * ny)))
        array.push(pts2, chart.point.from_time(int(c2t + (r_t * m2) * (ca * ux + sa * nx)), c2p + (r_p * m2) * (ca * uy + sa * ny)))

    polyline.new(pts1, closed=true, xloc=xloc.bar_time, line_color=col1, line_width=2)
    polyline.new(pts2, closed=true, xloc=xloc.bar_time, line_color=col2, line_width=2)

if barstate.islastconfirmedhistory
    // „Õ«›Ÿ  œ— »—«»— „Ã„Ê⁄ ’›—
    float total_body = body_t1 + body_t2
    total_body := total_body == 0 ? 1 : total_body

    // „Õ«”»Â ÷—«Ì» („”«Õ  „ ‰«”» »« »«œÌ ò‰œ· Â„«‰ ‰ﬁÿÂ)
    float m1 = math.sqrt(2 * (body_t1 / total_body))
    float m2 = math.sqrt(2 * (body_t2 / total_body))

    // —‰êùÂ« »Â ¬»Ì Ê ﬁ—„“ »—ê—œ«‰œÂ ‘œ (Ì« Â— —‰ê œ·ŒÊ«Â)
    color col1 = color.blue

    //  ﬁ«—‰ ‰”»  »Â p2

    draw_tangent_circles(t1, p1_mirrored_p2, t2, p2, color.new(col1, 50), color.new(col2, 50), m1, m2)
