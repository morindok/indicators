//@version=5
indicator("Time Machine{Morindok}", overlay=true, max_lines_count = 500)

// 1. œ—Ì«›  “„«‰ œÊ ‰ﬁÿÂ Ê ”ÿÕ ﬁÌ„ Ì «‰ Œ«»Ì
t1 = input.time(0, "Point 1", confirm=true)
t2 = input.time(0, "Point 2", confirm=true)
selected_level = input.price(0.0, "Selected Level", confirm=true)

//  ‰ŸÌ„«  „ÊÌ‰ê
ma_mult = input.float(1.0, "MA Length Multiplier", minval=0.1, step=0.1, group="MA Settings")
use_manual_len = input.bool(false, "Use Manual MA Length", group="MA Settings")
manual_len = input.int(50, "Manual MA Length", minval=1, group="MA Settings")
ma_ext_mult = input.float(1.618, "MA Extension Rate", step=0.1, group="MA Settings")

t_min = math.min(t1, t2)
t_max = math.max(t1, t2)
in_range = time >= t_min and time <= t_max

// 2. –ŒÌ—Â ‘„«—Â ò‰œ·ùÂ«
var int b1 = na
var int b2 = na

if time == t1
    b1 := bar_index
if time == t2
    b2 := bar_index

// 3. ¬—«ÌÂùÂ« »—«Ì –ŒÌ—Â  «—ÌŒçÂ ò«„·
var a_close = array.new_float()
var a_time  = array.new_int()
var a_high  = array.new_float()
var a_low   = array.new_float()

array.push(a_close, close)
array.push(a_time, time)
array.push(a_high, high)
array.push(a_low, low)

// ‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹ —”„ ”ÿÕ ﬁÌ„ Ì «‰ Œ«»Ì ‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹
var line mid_level = na

if in_range
    if na(mid_level)
        mid_level := line.new(t_min, selected_level, t_max, selected_level, xloc=xloc.bar_time, extend=extend.both, color=color.yellow, width=2, style=line.style_solid)
    else
        line.set_y1(mid_level, selected_level)
        line.set_y2(mid_level, selected_level)

// ‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹ —”„ ò‰œ·ùÂ«Ì „ ﬁ«—‰ »« plotcandle ‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹‹
float o_sym = not na(selected_level) ? 2.0 * selected_level - open : na
float c_sym = not na(selected_level) ? 2.0 * selected_level - close : na
float h_sym = not na(selected_level) ? 2.0 * selected_level - low : na
float l_sym = not na(selected_level) ? 2.0 * selected_level - high : na

color sym_col = c_sym >= o_sym ? color.rgb(38, 166, 154) : color.rgb(239, 83, 80)

plotcandle(o_sym, h_sym, l_sym, c_sym, title="Symmetric Candles", color=sym_col, wickcolor=sym_col, bordercolor=sym_col, display = display.none)

// 4. „Õ«”»«  œ— ¬Œ—Ì‰ ò‰œ·
if barstate.islast and not na(b1) and not na(b2)
    start_b = math.min(b1, b2)
    end_b = math.max(b1, b2)
    L = math.max(1, end_b - start_b)
    
    //  ⁄ÌÌ‰ ÿÊ· œÊ—Â „ÊÌ‰ê (œ” Ì Ì« ÅÊÌ«)
    int ma_length = use_manual_len ? manual_len : math.max(1, math.round(L * ma_mult)) 
    
    int size = array.size(a_close)
    
    a_cum = array.new_float(size, 0.0)
    float current_sum = 0.0
    for i = 0 to size - 1
        current_sum += array.get(a_close, i)
        array.set(a_cum, i, current_sum)
        
    var pts_ma = array.new<chart.point>()
    var pts_ma_full = array.new<chart.point>()
    array.clear(pts_ma)
    array.clear(pts_ma_full)
    
    int ext_bars = math.round(L * ma_ext_mult) 
    int calc_start = math.max(ma_length, start_b - ext_bars + L)
    
    for i = calc_start to end_b
        float sum_L = array.get(a_cum, i) - array.get(a_cum, i - ma_length)
        float ma = sum_L / ma_length
        int draw_time = array.get(a_time, i - L)
        
        chart.point pt = chart.point.from_time(draw_time, ma)
        array.push(pts_ma, pt)
        array.push(pts_ma_full, pt)
        
        if array.size(pts_ma) >= 9000
            polyline.new(pts_ma, line_color=color.blue, line_width=2, xloc=xloc.bar_time)
            array.clear(pts_ma)
            array.push(pts_ma, pt)
            
    if array.size(pts_ma) > 1
        polyline.new(pts_ma, line_color=color.blue, line_width=2, xloc=xloc.bar_time)

    time_start = array.get(a_time, start_b)
    time_end   = array.get(a_time, end_b)
    time_center = math.round((time_start + time_end) / 2)
    
    if array.size(pts_ma_full) > 0
        var pts_mirror = array.new<chart.point>()
        array.clear(pts_mirror)
        
        for j = 0 to array.size(pts_ma_full) - 1
            chart.point orig = array.get(pts_ma_full, j)
            int t_mirror = math.round(2 * time_center - orig.time)
            float y_mirror = orig.price
            array.push(pts_mirror, chart.point.from_time(t_mirror, y_mirror))
            
        var pts_mirror_chunk = array.new<chart.point>()
        array.clear(pts_mirror_chunk)
        for j = 0 to array.size(pts_mirror) - 1
            array.push(pts_mirror_chunk, array.get(pts_mirror, j))
            if array.size(pts_mirror_chunk) >= 9000
                polyline.new(pts_mirror_chunk, line_color=color.fuchsia, line_width=2, xloc=xloc.bar_time)
                array.clear(pts_mirror_chunk)
                array.push(pts_mirror_chunk, array.get(pts_mirror, j))
        if array.size(pts_mirror_chunk) > 1
            polyline.new(pts_mirror_chunk, line_color=color.fuchsia, line_width=2, xloc=xloc.bar_time)

    float midline_price = selected_level
    
    if array.size(pts_ma_full) > 0
        var pts_mirror_price = array.new<chart.point>()
        array.clear(pts_mirror_price)
        
        for j = 0 to array.size(pts_ma_full) - 1
            chart.point orig = array.get(pts_ma_full, j)
            float y_mirror_price = 2.0 * midline_price - orig.price
            array.push(pts_mirror_price, chart.point.from_time(orig.time, y_mirror_price))
            
        var pts_mirror_price_chunk = array.new<chart.point>()
        array.clear(pts_mirror_price_chunk)
        for j = 0 to array.size(pts_mirror_price) - 1
            array.push(pts_mirror_price_chunk, array.get(pts_mirror_price, j))
            if array.size(pts_mirror_price_chunk) >= 9000
                polyline.new(pts_mirror_price_chunk, line_color=color.white, line_width=2, xloc=xloc.bar_time)
                array.clear(pts_mirror_price_chunk)
                array.push(pts_mirror_price_chunk, array.get(pts_mirror_price, j))
        if array.size(pts_mirror_price_chunk) > 1
            polyline.new(pts_mirror_price_chunk, line_color=color.white, line_width=2, xloc=xloc.bar_time)

    if array.size(pts_ma_full) > 0
        var pts_mirror_full = array.new<chart.point>()
        array.clear(pts_mirror_full)
        
        for j = 0 to array.size(pts_ma_full) - 1
            chart.point orig = array.get(pts_ma_full, j)
            int t_mirror_full = math.round(2 * time_center - orig.time)
            float y_mirror_full = 2.0 * midline_price - orig.price
            array.push(pts_mirror_full, chart.point.from_time(t_mirror_full, y_mirror_full))
            
        var pts_mirror_full_chunk = array.new<chart.point>()
        array.clear(pts_mirror_full_chunk)
        for j = 0 to array.size(pts_mirror_full) - 1
            array.push(pts_mirror_full_chunk, array.get(pts_mirror_full, j))
            if array.size(pts_mirror_full_chunk) >= 9000
                polyline.new(pts_mirror_full_chunk, line_color=color.purple, line_width=2, xloc=xloc.bar_time)
                array.clear(pts_mirror_full_chunk)
                array.push(pts_mirror_full_chunk, array.get(pts_mirror_full, j))
        if array.size(pts_mirror_full_chunk) > 1
            polyline.new(pts_mirror_full_chunk, line_color=color.purple, line_width=2, xloc=xloc.bar_time)

    // ”ÿÊÕ ›Ì»Ê «’·Ì
    var line price_1618       = na
    var line price_minus0618  = na
    var line price_0786       = na 
    var line price_0618       = na
    var line price_0382       = na
    var line price_0214       = na 
    
    // ”ÿÊÕ ›Ì»Ê „ ﬁ«— 
    var line sym_price_1618       = na
    var line sym_price_minus0618  = na
    var line sym_price_0786       = na 
    var line sym_price_0618       = na
    var line sym_price_0382       = na
    var line sym_price_0214       = na 

    var line time_minus0618   = na
    var line time_minus1618   = na
    var line time_1618        = na
    
    float highest = array.get(a_high, start_b)
    float lowest  = array.get(a_low, start_b)
    for i = start_b + 1 to end_b
        highest := math.max(highest, array.get(a_high, i))
        lowest  := math.min(lowest, array.get(a_low, i))
    float range_p = highest - lowest
    
    // 1.618
    float lvl_1618 = lowest + 1.618 * range_p
    float sym_lvl_1618 = 2.0 * selected_level - lvl_1618
    if na(price_1618)
        price_1618 := line.new(x1=time_start, y1=lvl_1618, x2=time_end, y2=lvl_1618, xloc=xloc.bar_time, extend=extend.both, color=color.green, width=1, style=line.style_dashed)
        sym_price_1618 := line.new(x1=time_start, y1=sym_lvl_1618, x2=time_end, y2=sym_lvl_1618, xloc=xloc.bar_time, extend=extend.both, color=color.green, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_1618, time_start, lvl_1618), line.set_xy2(price_1618, time_end, lvl_1618)
        line.set_xy1(sym_price_1618, time_start, sym_lvl_1618), line.set_xy2(sym_price_1618, time_end, sym_lvl_1618)
        
    // 0.786
    float lvl_0786 = lowest + 0.786 * range_p
    float sym_lvl_0786 = 2.0 * selected_level - lvl_0786
    if na(price_0786)
        price_0786 := line.new(x1=time_start, y1=lvl_0786, x2=time_end, y2=lvl_0786, xloc=xloc.bar_time, extend=extend.both, color=color.teal, width=1, style=line.style_dashed)
        sym_price_0786 := line.new(x1=time_start, y1=sym_lvl_0786, x2=time_end, y2=sym_lvl_0786, xloc=xloc.bar_time, extend=extend.both, color=color.teal, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_0786, time_start, lvl_0786), line.set_xy2(price_0786, time_end, lvl_0786)
        line.set_xy1(sym_price_0786, time_start, sym_lvl_0786), line.set_xy2(sym_price_0786, time_end, sym_lvl_0786)
    
    // 0.618
    float lvl_0618 = lowest + 0.618 * range_p
    float sym_lvl_0618 = 2.0 * selected_level - lvl_0618
    if na(price_0618)
        price_0618 := line.new(x1=time_start, y1=lvl_0618, x2=time_end, y2=lvl_0618, xloc=xloc.bar_time, extend=extend.both, color=color.blue, width=1, style=line.style_dashed)
        sym_price_0618 := line.new(x1=time_start, y1=sym_lvl_0618, x2=time_end, y2=sym_lvl_0618, xloc=xloc.bar_time, extend=extend.both, color=color.blue, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_0618, time_start, lvl_0618), line.set_xy2(price_0618, time_end, lvl_0618)
        line.set_xy1(sym_price_0618, time_start, sym_lvl_0618), line.set_xy2(sym_price_0618, time_end, sym_lvl_0618)
    
    // 0.382
    float lvl_0382 = lowest + 0.382 * range_p
    float sym_lvl_0382 = 2.0 * selected_level - lvl_0382
    if na(price_0382)
        price_0382 := line.new(x1=time_start, y1=lvl_0382, x2=time_end, y2=lvl_0382, xloc=xloc.bar_time, extend=extend.both, color=color.orange, width=1, style=line.style_dashed)
        sym_price_0382 := line.new(x1=time_start, y1=sym_lvl_0382, x2=time_end, y2=sym_lvl_0382, xloc=xloc.bar_time, extend=extend.both, color=color.orange, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_0382, time_start, lvl_0382), line.set_xy2(price_0382, time_end, lvl_0382)
        line.set_xy1(sym_price_0382, time_start, sym_lvl_0382), line.set_xy2(sym_price_0382, time_end, sym_lvl_0382)
        
    // 0.214
    float lvl_0214 = lowest + 0.214 * range_p
    float sym_lvl_0214 = 2.0 * selected_level - lvl_0214
    if na(price_0214)
        price_0214 := line.new(x1=time_start, y1=lvl_0214, x2=time_end, y2=lvl_0214, xloc=xloc.bar_time, extend=extend.both, color=color.maroon, width=1, style=line.style_dashed)
        sym_price_0214 := line.new(x1=time_start, y1=sym_lvl_0214, x2=time_end, y2=sym_lvl_0214, xloc=xloc.bar_time, extend=extend.both, color=color.maroon, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_0214, time_start, lvl_0214), line.set_xy2(price_0214, time_end, lvl_0214)
        line.set_xy1(sym_price_0214, time_start, sym_lvl_0214), line.set_xy2(sym_price_0214, time_end, sym_lvl_0214)
    
    // -0.618
    float lvl_minus0618 = lowest - 0.618 * range_p
    float sym_lvl_minus0618 = 2.0 * selected_level - lvl_minus0618
    if na(price_minus0618)
        price_minus0618 := line.new(x1=time_start, y1=lvl_minus0618, x2=time_end, y2=lvl_minus0618, xloc=xloc.bar_time, extend=extend.both, color=color.red, width=1, style=line.style_dashed)
        sym_price_minus0618 := line.new(x1=time_start, y1=sym_lvl_minus0618, x2=time_end, y2=sym_lvl_minus0618, xloc=xloc.bar_time, extend=extend.both, color=color.red, width=1, style=line.style_dotted)
    else
        line.set_xy1(price_minus0618, time_start, lvl_minus0618), line.set_xy2(price_minus0618, time_end, lvl_minus0618)
        line.set_xy1(sym_price_minus0618, time_start, sym_lvl_minus0618), line.set_xy2(sym_price_minus0618, time_end, sym_lvl_minus0618)
    
    int idx_minus0618 = start_b - math.round(0.618 * L)
    if idx_minus0618 >= 0
        t_minus0618 = array.get(a_time, idx_minus0618)
        if na(time_minus0618)
            time_minus0618 := line.new(x1=t_minus0618, y1=lowest, x2=t_minus0618, y2=highest, xloc=xloc.bar_time, extend=extend.both, color=color.purple, width=1, style=line.style_dashed)
        else
            line.set_xy1(time_minus0618, t_minus0618, lowest)
            line.set_xy2(time_minus0618, t_minus0618, highest)
            
    int idx_minus1618 = start_b - math.round(1.618 * L)
    if idx_minus1618 >= 0
        t_minus1618 = array.get(a_time, idx_minus1618)
        if na(time_minus1618)
            time_minus1618 := line.new(x1=t_minus1618, y1=lowest, x2=t_minus1618, y2=highest, xloc=xloc.bar_time, extend=extend.both, color=color.purple, width=1, style=line.style_dashed)
        else
            line.set_xy1(time_minus1618, t_minus1618, lowest)
            line.set_xy2(time_minus1618, t_minus1618, highest)

    int idx_1618 = start_b + math.round(1.618 * L)
    if idx_1618 < array.size(a_time)
        t_1618 = array.get(a_time, idx_1618)
        if na(time_1618)
            time_1618 := line.new(x1=t_1618, y1=lowest, x2=t_1618, y2=highest, xloc=xloc.bar_time, extend=extend.both, color=color.purple, width=1, style=line.style_dashed)
        else
            line.set_xy1(time_1618, t_1618, lowest)
            line.set_xy2(time_1618, t_1618, highest)
