continueToIntersect = input.bool(title="«œ«„Â «”ÅÌ—«·ùÂ«  « “„«‰ »—ŒÊ—œ »Â ÌòœÌê—", defval=false)
useSpiralEnds   = input.bool(title=" ﬁ«—‰ œÊ„ ‰”»  »Â «‰ Â«Ì «”ÅÌ—«·ùÂ« (»Ã«Ì ”ÿÊÕ)", defval=false)
mirrorEndpoints = input.bool(title=" ﬁ«—‰ „” ﬁÌ„ Ê ”«œÂ ‰”»  »Â «‰ Â«Ì «”ÅÌ—«·ùÂ«", defval=false)

// --- „ €Ì—Â«Ì –ŒÌ—Âù”«“Ì Ê „Õ«”»« Ì ---
var int totalBarCount = 0
var int greenBarCount = 0 
var int redBarCount   = 0 

var float totalHeight = 0.0
var float greenHeight = 0.0
var float redHeight   = 0.0

var float max_p = na
var float min_p = na

var int   end_t = na
var float end_p = na
var int   start_t = na
var float start_p = na

var all_t = array.new_int()
var all_c = array.new_float()
array.push(all_t, time)
array.push(all_c, close)

time_start = math.min(t1, t2)
time_end   = math.max(t1, t2)
in_range   = time >= time_start and time <= time_end

plot(upperLevel, color=color.new(color.blue, 50), title="„ÕÊ— »«·«", linewidth=2)
plot(lowerLevel, color=color.new(color.red, 50), title="„ÕÊ— Å«ÌÌ‰", linewidth=2)

centerLevel = (upperLevel + lowerLevel) / 2.0
plot(centerLevel, color=color.new(color.gray, 50), title="„—ò“ œÊ ”ÿÕ", style=plot.style_cross)

if in_range
    float c_height = high - low
    totalBarCount += 1
    totalHeight   += c_height
    
    if close > open
        greenBarCount += 1
        greenHeight   += c_height
    else if close < open
        redBarCount   += 1
        redHeight     += c_height
    
    max_p := na(max_p) ? high : math.max(max_p, high)
    min_p := na(min_p) ? low : math.min(min_p, low)
    
    end_t := time
    end_p := hl2
    
    if na(start_t)
        start_t := time
        start_p := hl2

//  Ê«»⁄ ò„òÌ  ﬁ«ÿ⁄
ccw(x1, y1, x2, y2, x3, y3) =>
    (y3 - y1) * (x2 - x1) > (y2 - y1) * (x3 - x1)

intersect(x1, y1, x2, y2, x3, y3, x4, y4) =>
    ccw(x1, y1, x3, y3, x4, y4) != ccw(x2, y2, x3, y3, x4, y4) and ccw(x1, y1, x2, y2, x3, y3) != ccw(x1, y1, x2, y2, x4, y4)

if barstate.islast
    float dt = math.abs(t2 - t1)
    float dp = math.abs(max_p - min_p)
   
    if dt == 0
        dt := timeframe.in_seconds("") * 1000.0
    if dp == 0
        dp := 1.0

    float totalDistanceUp   = 0.0
    float totalDistanceDown = 0.0
    
    int size = array.size(all_t)
    int start_idx = math.max(0, size - 9000)
    
    for i = start_idx to size - 1
        int t_i = array.get(all_t, i)
        float c_i = array.get(all_c, i)
        
        if t_i >= time_start and t_i <= time_end
            float m_center = 2 * centerLevel - c_i
            float m_up_base   = 2 * upperLevel - m_center
            float m_down_base = 2 * lowerLevel - m_center
            
            totalDistanceUp   += math.abs(m_up_base - c_i)
            totalDistanceDown += math.abs(m_down_base - c_i)

    float divUp   = 1.0
    float divDown = 1.0
    
    if divisorMode == " ⁄œ«œ: ò· ò‰œ·ùÂ«"
        divUp   := totalBarCount
        divDown := totalBarCount
    else if divisorMode == " ⁄œ«œ: ò‰œ·ùÂ«Ì ”»“"
        divUp   := greenBarCount
        divDown := greenBarCount
    else if divisorMode == " ⁄œ«œ: ò‰œ·ùÂ«Ì ﬁ—„“"
        divUp   := redBarCount
        divDown := redBarCount
    else if divisorMode == " ⁄œ«œ:  ›òÌòù‘œÂ (»«·«:”»“° Å«ÌÌ‰:ﬁ—„“)"
        divUp   := greenBarCount
        divDown := redBarCount
    else if divisorMode == "«— ›«⁄: ò· ò‰œ·ùÂ«"
        divUp   := totalHeight
        divDown := totalHeight
    else if divisorMode == "«— ›«⁄: ò‰œ·ùÂ«Ì ”»“"
        divUp   := greenHeight
        divDown := greenHeight
    else if divisorMode == "«— ›«⁄: ò‰œ·ùÂ«Ì ﬁ—„“"
        divUp   := redHeight
        divDown := redHeight
    else if divisorMode == "«— ›«⁄:  ›òÌòù‘œÂ (»«·«:”»“° Å«ÌÌ‰:ﬁ—„“)"
        divUp   := greenHeight
        divDown := redHeight

    float baseLenUp   = divUp > 0 ? totalDistanceUp / divUp : totalDistanceUp
    float baseLenDown = divDown > 0 ? totalDistanceDown / divDown : totalDistanceDown
    
    float targetLenUp   = baseLenUp * multUp
    float targetLenDown = baseLenDown * multDown

    // --- —”„ «”ÅÌ—«·ùÂ« ---
    float b_coeff = math.log(1.6180339887) / (math.pi / 2)
    float a = 0.02
    int dir = reverseDir ? -1 : 1
    
    float final_p_up   = end_p
    float final_p_down = start_p

    pts_spiral_up = array.new<chart.point>()
    pts_spiral_down = array.new<chart.point>()
    
    // ¬—«ÌÂùÂ«Ì ò„òÌ »—«Ì –ŒÌ—Â „Œ ’«  ‰—„«·ù‘œÂ (ÃÂ  »——”Ì  ﬁ«ÿ⁄ œﬁÌﬁ)
    hist_up_t = array.new_float()
    hist_up_p = array.new_float()
    hist_down_t = array.new_float()
    hist_down_p = array.new_float()
    
    if not na(end_t) and not na(start_t)
        array.push(pts_spiral_up, chart.point.from_time(end_t, end_p))
        array.push(pts_spiral_down, chart.point.from_time(start_t, start_p))
        
        array.push(hist_up_t, (end_t - time_start) / dt)
        array.push(hist_up_p, (end_p - min_p) / dp)
        array.push(hist_down_t, (start_t - time_start) / dt)
        array.push(hist_down_p, (start_p - min_p) / dp)
        
        float acc_len_up = 0.0
        float acc_len_down = 0.0
        
        float prev_t_up = end_t
        float prev_p_up = end_p
        float prev_t_down = start_t
        float prev_p_down = start_p
        
        bool up_done = targetLenUp <= 0
        bool down_done = targetLenDown <= 0
        bool is_intersected = false
       
        for i = 1 to 4000
            float theta = i * 0.1
            float r = a * math.exp(b_coeff * theta)
            
            // „Õ«”»« 
            int curr_t_up = math.round(end_t + r * dt * math.cos(theta))
            float curr_p_up = end_p + (dir * r * dp * math.sin(theta))
            float seg_up = math.sqrt(math.pow(((curr_t_up - prev_t_up) / dt) * dp, 2) + math.pow(curr_p_up - prev_p_up, 2))
            
            int curr_t_down = math.round(start_t + r * dt * math.cos(theta))
            float curr_p_down = start_p - (dir * r * dp * math.sin(theta))
            float seg_down = math.sqrt(math.pow(((curr_t_down - prev_t_down) / dt) * dp, 2) + math.pow(curr_p_down - prev_p_down, 2))
            
            float n_curr_t_up = (curr_t_up - time_start) / dt
            float n_curr_p_up = (curr_p_up - min_p) / dp
            float n_prev_t_up = (prev_t_up - time_start) / dt
            float n_prev_p_up = (prev_p_up - min_p) / dp
            
            float n_curr_t_down = (curr_t_down - time_start) / dt
            float n_curr_p_down = (curr_p_down - min_p) / dp
            float n_prev_t_down = (prev_t_down - time_start) / dt
            float n_prev_p_down = (prev_p_down - min_p) / dp
            
            if continueToIntersect
                int len_down = array.size(hist_down_t)
                if len_down > 1
                    for j = 1 to len_down - 1
                        if intersect(n_prev_t_up, n_prev_p_up, n_curr_t_up, n_curr_p_up, array.get(hist_down_t, j-1), array.get(hist_down_p, j-1), array.get(hist_down_t, j), array.get(hist_down_p, j))
                            is_intersected := true
                            break
                            
                int len_up = array.size(hist_up_t)
                if not is_intersected and len_up > 1
                    for j = 1 to len_up - 1
                        if intersect(n_prev_t_down, n_prev_p_down, n_curr_t_down, n_curr_p_down, array.get(hist_up_t, j-1), array.get(hist_up_p, j-1), array.get(hist_up_t, j), array.get(hist_up_p, j))
                            is_intersected := true
                            break
            else
                if acc_len_up > targetLenUp
                    up_done := true
                if acc_len_down > targetLenDown
                    down_done := true
            
            if (continueToIntersect and is_intersected) or (not continueToIntersect and up_done and down_done)
                break
                
            if not up_done or continueToIntersect
                array.push(pts_spiral_up, chart.point.from_time(curr_t_up, curr_p_up))
                array.push(hist_up_t, n_curr_t_up)
                array.push(hist_up_p, n_curr_p_up)
                acc_len_up += seg_up
                final_p_up := curr_p_up
                prev_t_up  := curr_t_up
                prev_p_up  := curr_p_up
                
            if not down_done or continueToIntersect
                array.push(pts_spiral_down, chart.point.from_time(curr_t_down, curr_p_down))
                array.push(hist_down_t, n_curr_t_down)
                array.push(hist_down_p, n_curr_p_down)
                acc_len_down += seg_down
                final_p_down := curr_p_down
                prev_t_down  := curr_t_down
                prev_p_down  := curr_p_down

        if array.size(pts_spiral_up) > 1
            polyline.new(pts_spiral_up, xloc = xloc.bar_time, line_color = color.fuchsia, line_width = 2)
        if array.size(pts_spiral_down) > 1
            polyline.new(pts_spiral_down, xloc = xloc.bar_time, line_color = color.yellow, line_width = 2)

    // --- —”„  ﬁ«—‰ùÂ«Ì œÊê«‰Â ---
    pts_mirrored_up   = array.new<chart.point>()
    pts_mirrored_down = array.new<chart.point>()
    
    float refUp   = useSpiralEnds ? final_p_up   : upperLevel
    float refDown = useSpiralEnds ? final_p_down : lowerLevel
    
    for i = start_idx to size - 1
        int t_i = array.get(all_t, i)
        float c_i = array.get(all_c, i)
        
        float m_center = 2 * centerLevel - c_i
        float m_up   = 2 * refUp - m_center
        float m_down = 2 * refDown - m_center
        
        array.push(pts_mirrored_up, chart.point.from_time(t_i, m_up))
        array.push(pts_mirrored_down, chart.point.from_time(t_i, m_down))
            
    polyline.new(pts_mirrored_up, xloc=xloc.bar_time, line_color=color.green, line_width=2)
    polyline.new(pts_mirrored_down, xloc=xloc.bar_time, line_color=color.orange, line_width=2)

    // ---  ﬁ«—‰ ”«œÂ Ê „” ﬁÌ„ ---
    if mirrorEndpoints
        int chart_start_t = array.get(all_t, 0)
        line.new(x1=chart_start_t, y1=final_p_up, x2=time, y2=final_p_up, xloc=xloc.bar_time, color=color.new(color.fuchsia, 50), style=line.style_dashed, width=2)
        line.new(x1=chart_start_t, y1=final_p_down, x2=time, y2=final_p_down, xloc=xloc.bar_time, color=color.new(color.yellow, 50), style=line.style_dashed, width=2)
        
        pts_mirrored_end_up   = array.new<chart.point>()
        pts_mirrored_end_down = array.new<chart.point>()
        
        for i = start_idx to size - 1
            int t_i = array.get(all_t, i)
            float c_i = array.get(all_c, i)
            
            float m_end_up   = 2 * final_p_up - c_i
            float m_end_down = 2 * final_p_down - c_i
            
            array.push(pts_mirrored_end_up, chart.point.from_time(t_i, m_end_up))
            array.push(pts_mirrored_end_down, chart.point.from_time(t_i, m_end_down))
            
        polyline.new(pts_mirrored_end_up, xloc=xloc.bar_time, line_color=color.new(color.fuchsia, 20), line_width=1)
        polyline.new(pts_mirrored_end_down, xloc=xloc.bar_time, line_color=color.new(color.yellow, 20), line_width=1)

        polyline.new(pts_mirrored_end_down, xloc=xloc.bar_time, line_color=color.new(color.yellow, 20), line_width=1)


    // --- »——”Ì ”Ìê‰«·ùÂ«Ì Œ—Ìœ Ê ›—Ê‘ ---
    bool  signalFound = false
    int   signalTime  = na
    float signalPrice = na
    bool  isBuySignal = false

    int spiralUpSize   = array.size(pts_spiral_up)
    int spiralDownSize = array.size(pts_spiral_down)
    int mirrorSize     = array.size(pts_mirrored_up)
    
    // »——”Ì »—ŒÊ—œ «”ÅÌ—«· »‰›‘ (›—Ê‘: «“ »«·« »Â Å«ÌÌ‰ Œÿ ”»“)
    if spiralUpSize > 1 and mirrorSize > 1
        for j = 1 to spiralUpSize - 1
            if signalFound
                break
                
            chart.point p_prev = array.get(pts_spiral_up, j - 1)
            chart.point p_curr = array.get(pts_spiral_up, j)

            int max_t = math.max(p_prev.time, p_curr.time)
            
            for i = 1 to mirrorSize - 1
                int t_prev = array.get(all_t, i - 1 + start_idx)
                
                // »ÂÌ‰Âù”«“Ì: «ê— “„«‰ ò‰œ· «“ Õœ«òÀ— “„«‰ «”ÅÌ—«· ê–‘ ° «“ Õ·ﬁÂ Œ«—Ã ‘Ê (çÊ‰ “„«‰ùÂ« ’⁄ÊœÌ Â” ‰œ)
                if t_prev > max_t
                    break
                    
                int t_curr = array.get(all_t, i + start_idx)
                
                // »ÂÌ‰Âù”«“Ì: «ê— “„«‰ ò‰œ· Â‰Ê“ »Â Õœ«ﬁ· “„«‰ «”ÅÌ—«· ‰—”ÌœÂ° »—Ê »Â ò‰œ· »⁄œÌ
                if t_curr < min_t
                    continue
                    
                float m_up_prev = array.get(pts_mirrored_up, i - 1).price
                float m_up_curr = array.get(pts_mirrored_up, i).price
                
                if (p_prev.time <= t_curr and p_curr.time >= t_prev) or (p_curr.time <= t_curr and p_prev.time >= t_prev)
                    if p_prev.price > m_up_prev and p_curr.price <= m_up_curr
                        signalFound := true
                        isBuySignal := false
                        signalTime  := t_curr
                        signalPrice := p_curr.price
                        break

    if spiralDownSize > 1 and mirrorSize > 1 and not signalFound
        for j = 1 to spiralDownSize - 1
            if signalFound
                break
                
            chart.point p_prev = array.get(pts_spiral_down, j - 1)
            chart.point p_curr = array.get(pts_spiral_down, j)
            
            int min_t = math.min(p_prev.time, p_curr.time)
            int max_t = math.max(p_prev.time, p_curr.time)
            
            for i = 1 to mirrorSize - 1
                int t_prev = array.get(all_t, i - 1 + start_idx)
                
                if t_prev > max_t
                    break
                    
                int t_curr = array.get(all_t, i + start_idx)
                
                if t_curr < min_t
                    continue

                float m_down_curr = array.get(pts_mirrored_down, i).price
                
                if (p_prev.time <= t_curr and p_curr.time >= t_prev) or (p_curr.time <= t_curr and p_prev.time >= t_prev)
                    if p_prev.price < m_down_prev and p_curr.price >= m_down_curr
                        signalFound := true
                        isBuySignal := true
                        signalTime  := t_curr
                        signalPrice := p_curr.price
                        break
                            
    if signalFound
        if isBuySignal
            label.new(x=signalTime, y=signalPrice, text="BUY", xloc=xloc.bar_time, color=color.green, textcolor=color.white, style=label.style_label_up, size=size.normal)
        else
            label.new(x=signalTime, y=signalPrice, text="SELL", xloc=xloc.bar_time, color=color.red, textcolor=color.white, style=label.style_label_down, size=size.normal)
