//@version=5
indicator("Two Tangent Circles", overlay=true, max_polylines_count=2)

t1 = input.time(0, title="Point 1", confirm=true, inline="p1")
p1 = input.price(0, title="", confirm=true, inline="p1")
t2 = input.time(0, title="Point 2", confirm=true, inline="p2")
p2 = input.price(0, title="", confirm=true, inline="p2")

if barstate.islastconfirmedhistory
    float dt = float(t2 - t1)
    float dp = p2 - p1

    float ux = dt / math.abs(dt)
    float uy = dp / math.abs(dp)
    float nx = -uy
    float ny = ux

    float r_t = math.abs(dt) / 4.0
    float r_p = math.abs(dp) / 4.0

    float c1t = float(t1) + dt * 0.25
    float c1p = p1 + dp * 0.25
    float c2t = float(t1) + dt * 0.75
    float c2p = p1 + dp * 0.75

    int N = 120
    pts1 = array.new<chart.point>()
    pts2 = array.new<chart.point>()

    for i = 0 to N - 1
        float angle = 2.0 * math.pi * i / N
        float ca = math.cos(angle)
        float sa = math.sin(angle)

        array.push(pts1, chart.point.from_time(int(c1t + r_t * (ca * ux + sa * nx)), c1p + r_p * (ca * uy + sa * ny)))
        array.push(pts2, chart.point.from_time(int(c2t + r_t * (ca * ux + sa * nx)), c2p + r_p * (ca * uy + sa * ny)))

    polyline.new(pts1, closed=true, xloc=xloc.bar_time, line_color=color.blue, line_width=2)
    polyline.new(pts2, closed=true, xloc=xloc.bar_time, line_color=color.red, line_width=2)
