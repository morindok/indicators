//@version=5
indicator("Live Tick Counter{Morindok} by @BITMOON618", overlay=true, max_labels_count=500)

// --- ÊäÙíãÇÊ æÑæÏí ---
int lookback = input.int(50, "ÊÚÏÇÏ ˜äÏáåÇí ĞÔÊå ÈÑÇí ãÍÇÓÈå ÇÍÊãÇá", minval=10, maxval=490, tooltip="ÊÚÏÇÏ ˜äÏáåÇí ÈÓÊå ÔÏå ÇÎíÑ ˜å ÈÑÇí ãÍÇÓÈå ÇÍÊãÇá ÏÑ äÙÑ ÑİÊå ãíÔæäÏ.")

// --- ãÊÛíÑåÇí ÔãÇÑäÏå Êí˜ ---
varip float lastPrice = na
varip int   upTicks   = 0
varip int   downTicks = 0

// --- ÂÑÇíååÇ ÈÑÇí ĞÎíÑå ÏíÊÇí ĞÔÊå ÌåÊ ãÍÇÓÈå ÇÍÊãÇá ---
var array<int> hist_up = array.new<int>()
var array<int> hist_down = array.new<int>()
var array<bool> hist_green = array.new<bool>()

// --- ãÊÛíÑåÇí ĞÎíÑå ÇÍÊãÇá (ÈÑÇí ÌáæíÑí ÇÒ ãÍÇÓÈå ãÌÏÏ ÏÑ åÑ Êí˜) ---
var float prob_up_ticks = 50.0
var float prob_down_ticks = 50.0
var float prob_price_up = 50.0
var float prob_price_down = 50.0
var int valid_candles = 0

// --- áíÈáåÇ ---
var label[] history_labels = array.new_label()
var label live_label = label.new(na, na, "", color=color.new(color.blue, 70), textcolor=color.white, style=label.style_label_down)
var label prob_label = label.new(na, na, "", color=color.new(color.purple, 20), textcolor=color.white, style=label.style_label_down)

// --- ÑíÓÊ ˜ÑÏä ÔãÇÑäÏååÇ ÏÑ Çæáíä Êí˜ åÑ ˜äÏá ÌÏíÏ ---
if barstate.isnew
    upTicks   := 0
    downTicks := 0
    lastPrice := open

// --- İŞØ ÏÑ áÇíæ ãÇÑ˜Ê ãÍÇÓÈå ÔæÏ ---
if barstate.isrealtime
    
    // ÔãÇÑÔ Êí˜åÇ
    if not na(lastPrice)
        if close > lastPrice
            upTicks += 1
        else if close < lastPrice
            downTicks += 1
    
    lastPrice := close
    
    // ÂÏíÊ áíÈá ãÊÍÑ˜ Ñæí ˜äÏá ÌÇÑí (Êí˜åÇí áÍÙåÇí)
    label.set_xy(live_label, bar_index, high + ta.tr * 0.5)
    label.set_text(live_label, "Up: " + str.tostring(upTicks) + "\nDown: " + str.tostring(downTicks))

    // ÂÏíÊ áíÈá ÇÍÊãÇá (İŞØ äãÇíÔ ÏíÊÇí ãÍÇÓÈå ÔÏå ÏÑ ÈÓÊå ÔÏä ˜äÏá ŞÈáí)
    label.set_xy(prob_label, bar_index, high + ta.tr * 1.5)
    string prob_text = "ÇÍÊãÇá ÊÌãÚí (ÈÑ ÇÓÇÓ " + str.tostring(valid_candles) + " ˜äÏá ÇÎíÑ):\n" +
         "?? ÕÚæÏ (İÔÇÑ Êí˜): " + str.tostring(prob_up_ticks, "#.##") + "%\n" +
         "?? ÑíÒÔ (İÔÇÑ Êí˜): " + str.tostring(prob_down_ticks, "#.##") + "%\n" +
         "???????????????\n" +
         "?? ÇÍÊãÇá ÕÚæÏ (ŞíãÊ): " + str.tostring(prob_price_up, "#.##") + "%\n" +
         "?? ÇÍÊãÇá ÑíÒÔ (ŞíãÊ): " + str.tostring(prob_price_down, "#.##") + "%"
    label.set_text(prob_label, prob_text)

    // ?? ËÈÊ ŞØÚí æ ÏÇÆãí ÏíÊÇí ˜äÏá ÏÑ áÍÙå ÈÓÊå ÔÏä (ÂÎÑíä Êí˜) ??
    if barstate.isconfirmed
        // 1. ËÈÊ áíÈá ÙÇåÑí
        saved_lbl = label.new(bar_index, high, "Up: " + str.tostring(upTicks) + "\nDown: " + str.tostring(downTicks), 
                              color=color.new(color.gray, 20), textcolor=color.white, style=label.style_label_down)
        array.push(history_labels, saved_lbl)
        
        // ãÏíÑíÊ ÍÇİÙå áíÈáåÇ
        if array.size(history_labels) > 490
            label.delete(array.shift(history_labels))

        // 2. ËÈÊ ÏíÊÇ ÏÑ ÂÑÇíååÇ ÈÑÇí ãÍÇÓÈå ÇÍÊãÇá
        array.push(hist_up, upTicks)
        array.push(hist_down, downTicks)
        array.push(hist_green, close > open)
        
        // ãÏíÑíÊ ÍÇİÙå ÂÑÇíååÇí ÇÍÊãÇá (ÈÑ ÇÓÇÓ Lookback)
        if array.size(hist_up) > lookback
            array.shift(hist_up)
            array.shift(hist_down)
            array.shift(hist_green)
            
        // 3. ãÍÇÓÈå ãÌÏÏ ÇÍÊãÇáÇÊ (İŞØ í˜ÈÇÑ ÏÑ ÈÓÊå ÔÏä ˜äÏá ÇÌÑÇ ãíÔæÏ)
        int arr_size = array.size(hist_up)
        valid_candles := arr_size
        
        if arr_size > 0
            // ãÍÇÓÈå ÇÍÊãÇá ÈÑ ÇÓÇÓ ÎÇáÕ Êí˜åÇ
            int sum_up = array.sum(hist_up)
            int sum_down = array.sum(hist_down)
            int total_ticks = sum_up + sum_down
            
            prob_up_ticks := total_ticks > 0 ? (sum_up / total_ticks) * 100 : 50.0
            prob_down_ticks := total_ticks > 0 ? (sum_down / total_ticks) * 100 : 50.0
            
            // ãÍÇÓÈå ÇÍÊãÇá ÈÑ ÇÓÇÓ äÊíÌå ˜äÏáåÇ (æÒäÏåí ÔÏå)
            int green_weight = 0
            int red_weight = 0
            for i = 0 to arr_size - 1
                int candle_ticks = array.get(hist_up, i) + array.get(hist_down, i)
                if array.get(hist_green, i)
                    green_weight += candle_ticks
                else
                    red_weight += candle_ticks
                    
            int total_weight = green_weight + red_weight
            prob_price_up := total_weight > 0 ? (green_weight / total_weight) * 100 : 50.0
            prob_price_down := total_weight > 0 ? (red_weight / total_weight) * 100 : 50.0
