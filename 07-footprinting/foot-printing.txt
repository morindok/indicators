//@version=5
indicator("Pro Footprint & Imbalance", overlay=false)

// ---  ‰ŸÌ„«  Ê „ €Ì—Â«Ì «Ê·ÌÂ ---
var int BINS = 10 // «›“«Ì‘ ”ÿÊÕ ﬁÌ„ Ì »Â 10 »—«Ì œﬁ  »Ì‘ —
var table fpTable = table.new(position.middle_center, 5, BINS + 2, bgcolor=color.new(color.black, 85), border_width=2, border_color=color.gray)
var table analysisTable = table.new(position.bottom_right, 2, 4, bgcolor=color.new(color.black, 90), border_width=1, border_color=color.white)

// ¬—«ÌÂ »—«Ì ‰êÂœ«—Ì »«ò”ùÂ«Ì —”„ ‘œÂ —ÊÌ ç«—  «’·Ì
var box[] imbBoxes = array.new_box()

if barstate.islast or barstate.isrealtime
    // Å«ò ò—œ‰ »«ò”ùÂ«Ì ﬁœÌ„Ì «Ì„»«·«‰”
    if array.size(imbBoxes) > 0
        for i = 0 to array.size(imbBoxes) - 1
            box.delete(array.get(imbBoxes, i))
        array.clear(imbBoxes)
        
    float sum_delta_5bars = 0.0
    float total_vol_5bars = 0.0
    int bullish_bars = 0
    int bearish_bars = 0

    // --- „Õ«”»Â Ê —”„ ÃœÊ· ›Ê ùÅ—Ì‰  ---
    for i = 0 to 4
        col = 4 - i
        h = high[i], l = low[i], c = close[i], o = open[i], v = volume[i]
        
        total_vol_5bars += v
        if c > o
            bullish_bars += 1
        else if c < o
            bearish_bars += 1

        range_len = h - l
        step = range_len == 0 ? 0.0001 : range_len / BINS
        float bar_delta = 0.0
        
        // Âœ— ” Ê‰ùÂ«
        header_text = i == 0 ? "LIVE" : "T-" + str.tostring(i)
        table.cell(fpTable, col, 0, text=header_text, text_color=color.yellow, text_size=size.normal, bgcolor=color.new(color.blue, 70))
        
        for j = 0 to BINS - 1
            level_low = l + (j * step)
            level_high = level_low + step
            
            is_in_body = (level_high >= math.min(o, c)) and (level_low <= math.max(o, c))
            level_vol = (v / BINS) * (is_in_body ? 1.5 : 0.8)
            
            buy_pct = (c > o) ? 0.6 : 0.4
            if level_high <= c and c > o
                buy_pct := 0.75 
            if level_low >= c and c < o
                buy_pct := 0.25 
                
            buy_vol = level_vol * buy_pct
            sell_vol = level_vol * (1 - buy_pct)
            bar_delta += (buy_vol - sell_vol)
            
            cell_text = str.tostring(math.round(sell_vol)) + " x " + str.tostring(math.round(buy_vol))
            bg_col = buy_vol > sell_vol ? color.new(color.green, 70) : color.new(color.red, 70)
            
            //  ‘ŒÌ’ Imbalance Ê —”„ »«ò” —ÊÌ ç«—  «’·Ì (›ﬁÿ »—«Ì ò‰œ·ùÂ«Ì «ŒÌ—)
            if i <= 1 // —”„ «Ì„»«·«‰” »—«Ì ò‰œ· ·«ÌÊ Ê ò‰œ· ﬁ»·Ì
                if buy_vol > sell_vol * 2.5 // Imbalance Œ—Ìœ
                    b = box.new(left=bar_index[i], top=level_high, right=bar_index[i]+1, bottom=level_low, border_color=color.green, bgcolor=color.new(color.green, 85),force_overlay = true)
                    array.push(imbBoxes, b)
                else if sell_vol > buy_vol * 2.5 // Imbalance ›—Ê‘
                    b = box.new(left=bar_index[i], top=level_high, right=bar_index[i]+1, bottom=level_low, border_color=color.red, bgcolor=color.new(color.red, 85),force_overlay = true)
                    array.push(imbBoxes, b)

            row_idx = BINS - j
            table.cell(fpTable, col, row_idx, text=cell_text, text_color=color.white, bgcolor=bg_col, text_size=size.small)
            
        sum_delta_5bars += bar_delta
        delta_color = bar_delta > 0 ? color.lime : color.red
        table.cell(fpTable, col, BINS + 1, text="?: " + str.tostring(math.round(bar_delta)), text_color=delta_color, text_size=size.normal)

    // --- ÃœÊ· ¬‰«·Ì“ Õ—›Âù«Ì ---
    table.cell(analysisTable, 0, 0, text="Pro Analysis", text_color=color.yellow, text_size=size.normal, bgcolor=color.new(color.gray, 70))
    table.cell(analysisTable, 1, 0, text="Value", text_color=color.yellow, text_size=size.normal, bgcolor=color.new(color.gray, 70))
    
    table.cell(analysisTable, 0, 1, text="Net Delta (5 Bars):", text_color=color.white)
    table.cell(analysisTable, 1, 1, text=str.tostring(math.round(sum_delta_5bars)), text_color=(sum_delta_5bars > 0 ? color.lime : color.red))
    
    table.cell(analysisTable, 0, 2, text="Bar Control:", text_color=color.white)
    control_text = bullish_bars > bearish_bars ? "Buyers" : (bearish_bars > bullish_bars ? "Sellers" : "Neutral")
    table.cell(analysisTable, 1, 2, text=control_text, text_color=(bullish_bars > bearish_bars ? color.lime : color.red))

    table.cell(analysisTable, 0, 3, text="Trend Strength:", text_color=color.white)
    strength = math.abs(sum_delta_5bars) > (total_vol_5bars * 0.2) ? "Strong" : "Weak/Ranging"
    table.cell(analysisTable, 1, 3, text=strength, text_color=color.orange)
