//@version=5
indicator("Gann Fractal Predictor", overlay=true, max_lines_count=500)

// Fractal settings
n = input.int(title="Fractal Periods", defval=2, minval=2)
fractalsToAnalyze = input.int(10, "Number of Fractals to Analyze", minval=2, maxval=50)

// Gann settings
levelMultiplier = input.float(1.0, "Level Multiplier", minval=0.1, maxval=10.0, step=0.1)
gannSquareMultiplier = input.float(0.01, "Gann Square Multiplier", minval=0.001, maxval=1.0, step=0.001)
projectionLimit = input.float(5.0, "Projection Limit", minval=1.0, maxval=20.0, step=0.5)

// Start time input
startTime = input.time(timestamp("2024-01-01 00:00"), "Analysis Start Time", confirm=true)

// UpFractal detection
bool upflagDownFrontier = true
bool upflagUpFrontier0 = true
bool upflagUpFrontier1 = true
bool upflagUpFrontier2 = true
bool upflagUpFrontier3 = true
bool upflagUpFrontier4 = true

for i = 1 to n
    upflagDownFrontier := upflagDownFrontier and (high[n-i] < high[n])
    upflagUpFrontier0 := upflagUpFrontier0 and (high[n+i] < high[n])
    upflagUpFrontier1 := upflagUpFrontier1 and (high[n+1] <= high[n] and high[n+i + 1] < high[n])
    upflagUpFrontier2 := upflagUpFrontier2 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+i + 2] < high[n])
    upflagUpFrontier3 := upflagUpFrontier3 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+i + 3] < high[n])
    upflagUpFrontier4 := upflagUpFrontier4 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+4] <= high[n] and high[n+i + 4] < high[n])

flagUpFrontier = upflagUpFrontier0 or upflagUpFrontier1 or upflagUpFrontier2 or upflagUpFrontier3 or upflagUpFrontier4
upFractal = (upflagDownFrontier and flagUpFrontier)

// DownFractal detection
bool downflagDownFrontier = true
bool downflagUpFrontier0 = true
bool downflagUpFrontier1 = true
bool downflagUpFrontier2 = true
bool downflagUpFrontier3 = true
bool downflagUpFrontier4 = true

for i = 1 to n
    downflagDownFrontier := downflagDownFrontier and (low[n-i] > low[n])
    downflagUpFrontier0 := downflagUpFrontier0 and (low[n+i] > low[n])
    downflagUpFrontier1 := downflagUpFrontier1 and (low[n+1] >= low[n] and low[n+i + 1] > low[n])
    downflagUpFrontier2 := downflagUpFrontier2 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+i + 2] > low[n])
    downflagUpFrontier3 := downflagUpFrontier3 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+i + 3] > low[n])
    downflagUpFrontier4 := downflagUpFrontier4 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+4] >= low[n] and low[n+i + 4] > low[n])

flagDownFrontier = downflagUpFrontier0 or downflagUpFrontier1 or downflagUpFrontier2 or downflagUpFrontier3 or downflagUpFrontier4
downFractal = (downflagDownFrontier and flagDownFrontier)

// Store fractals BEFORE startTime
var float[] fractalPrices = array.new_float(0)
var int[] fractalTimes = array.new_int(0)

// Collect fractals BEFORE startTime
if time[n] < startTime
    if upFractal
        array.push(fractalPrices, high[n])
        array.push(fractalTimes, time[n])
    
    if downFractal
        array.push(fractalPrices, low[n])
        array.push(fractalTimes, time[n])

// Gann functions (identical to original)
gannSquare(price, timeRange) => price * math.sqrt(timeRange / 360) * gannSquareMultiplier
gannAngle(p1, p2, t1, t2) => (p2 - p1) / math.max(math.abs(t2 - t1), 1)

var line[] gannLines = array.new_line(0)
var float nextSupport = na
var float nextResistance = na

// Analysis at last bar (identical logic to original)
if barstate.islast
    int totalFractals = array.size(fractalPrices)
    
    if totalFractals >= fractalsToAnalyze
        // Clear old lines
        if array.size(gannLines) > 0
            for i = 0 to array.size(gannLines) - 1
                line.delete(array.get(gannLines, i))
            array.clear(gannLines)
        
        int startIndex = totalFractals - fractalsToAnalyze
        float totalAngle = 0.0
        int validAngles = 0
        float priceRange = 0.0
        
        // Draw lines between fractals (identical to original pivot logic)
        for i = startIndex to totalFractals - 2
            p1 = array.get(fractalPrices, i)
            p2 = array.get(fractalPrices, i + 1)
            t1 = array.get(fractalTimes, i)
            t2 = array.get(fractalTimes, i + 1)
            
            angle = gannAngle(p1, p2, t1, t2)
            totalAngle := totalAngle + angle
            validAngles := validAngles + 1
            priceRange := priceRange + math.abs(p2 - p1)
            
            x1 = t1
            x2 = t2
            y1 = p1
            y2 = p2
            
            gannLine = line.new(x1, y1, x2, y2, color=color.new(color.blue, 30), width=1, xloc=xloc.bar_time)
            array.push(gannLines, gannLine)
            
            timeDiff = math.abs(t2 - t1)
            gannSq = gannSquare(p1, timeDiff)
            midPrice = (p1 + p2) / 2
            
            for mult = 1 to 3
                levelUp = midPrice + (gannSq * mult)
                levelDown = midPrice - (gannSq * mult)
                sqLine1 = line.new(x1, levelUp, x2, levelUp, color=color.new(color.orange, 70), width=1, style=line.style_dotted, xloc=xloc.bar_time)
                sqLine2 = line.new(x1, levelDown, x2, levelDown, color=color.new(color.orange, 70), width=1, style=line.style_dotted, xloc=xloc.bar_time)
                array.push(gannLines, sqLine1)
                array.push(gannLines, sqLine2)
        
        if validAngles > 0
            avgAngle = totalAngle / validAngles
            avgPriceMove = priceRange / validAngles
            lastPrice = array.get(fractalPrices, totalFractals - 1)
            lastTime = array.get(fractalTimes, totalFractals - 1)
            currentTime = time
            timeDiff = (currentTime - lastTime) / (1000 * 60 * 60 * 24)
            
            projectionFactor = math.min(timeDiff / 10.0, projectionLimit)
            
            nextResistance := lastPrice + (avgPriceMove * 0.382 * projectionFactor * levelMultiplier)
            nextSupport := lastPrice - (avgPriceMove * 0.382 * projectionFactor * levelMultiplier)
            
            fibLevels = array.from(0.236, 0.382, 0.5, 0.618, 0.786, 1.0, 1.272, 1.618)
            
            for i = 0 to array.size(fibLevels) - 1
                fibMult = array.get(fibLevels, i)
                resLevel = lastPrice + (avgPriceMove * fibMult * projectionFactor * 0.5 * levelMultiplier)
                supLevel = lastPrice - (avgPriceMove * fibMult * projectionFactor * 0.5 * levelMultiplier)
                resLine = line.new(lastTime, resLevel, currentTime, resLevel, color=color.new(color.red, 60), width=1, extend=extend.right, style=line.style_dashed, xloc=xloc.bar_time)
                supLine = line.new(lastTime, supLevel, currentTime, supLevel, color=color.new(color.green, 60), width=1, extend=extend.right, style=line.style_dashed, xloc=xloc.bar_time)
                array.push(gannLines, resLine)
                array.push(gannLines, supLine)
            
            mainResLine = line.new(lastTime, nextResistance, currentTime, nextResistance, color=color.red, width=3, extend=extend.right, xloc=xloc.bar_time)
            mainSupLine = line.new(lastTime, nextSupport, currentTime, nextSupport, color=color.green, width=3, extend=extend.right, xloc=xloc.bar_time)
            array.push(gannLines, mainResLine)
            array.push(gannLines, mainSupLine)

plotshape(downFractal and time[n] < startTime, style=shape.triangledown, location=location.belowbar, offset=-n, color=#F44336, size=size.small)
plotshape(upFractal and time[n] < startTime, style=shape.triangleup, location=location.abovebar, offset=-n, color=#009688, size=size.small)

plot(nextResistance, "Next Resistance", color=color.red, linewidth=2, style=plot.style_circles)
plot(nextSupport, "Next Support", color=color.green, linewidth=2, style=plot.style_circles)
