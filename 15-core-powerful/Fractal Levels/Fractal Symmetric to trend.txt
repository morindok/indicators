f_check_and_label(int t, float p, array<int> times, array<float> highs, array<float> lows, string styleType) =>
    int left = 0
    int right = array.size(times) - 1
    int ans = -1
    
    // Ã” ÃÊÌ ò‰œ·Ì òÂ “„«‰ ¬‰ œﬁÌﬁ« —ÊÌ ‰ﬁÿÂ Ì« »·«›«’·Â »⁄œ «“ ¬‰ »«‘œ
    while left <= right
        int mid = int((left + right) / 2)
        int mid_time = array.get(times, mid)
        if mid_time == t
            ans := mid
            break
        else if mid_time < t
            left := mid + 1
        else
            ans := mid // –ŒÌ—Â ò‰œ·Ì òÂ “„«‰‘ Ã·Ê — «“ ‰ﬁÿÂ Âœ› «” 
            right := mid - 1
            
    if ans >= 0
        // »——”Ì »—ŒÊ—œ ﬁÌ„  »Â ”ﬁ› Ì« ò› ò‰œ· ÅÌœ« ‘œÂ
        float bHigh = array.get(highs, ans)
        float bLow = array.get(lows, ans)
        if p <= bHigh and p >= bLow
            string lblStyle = styleType == "down" ? label.style_label_down : label.style_label_up
            lbl = label.new(array.get(times, ans), p, "Reversal\nSignal", xloc=xloc.bar_time, style=lblStyle, color=color.red, textcolor=color.white, size=size.small)
            array.push(drawnLabels, lbl)
