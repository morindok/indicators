bool upflagUpFrontier3  = true
bool upflagUpFrontier4  = true
for i = 1 to n
    upflagDownFrontier := upflagDownFrontier and (high[n-i] < high[n])
    upflagUpFrontier0  := upflagUpFrontier0  and (high[n+i] < high[n])

    upflagUpFrontier4  := upflagUpFrontier4  and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+4] <= high[n] and high[n+i+4] < high[n])

bool downflagUpFrontier0b  = true

bool downflagUpFrontier2b  = true

    downflagUpFrontier1b  := downflagUpFrontier1b  and (mLow[n+1] >= mLow[n] and mLow[n+i+1] > mLow[n])
    downflagUpFrontier2b  := downflagUpFrontier2b  and (mLow[n+1] >= mLow[n] and mLow[n+2] >= mLow[n] and mLow[n+i+2] > mLow[n])

linePrice(srcT, dstT, srcPrice, dstPrice, kT) =>
    dstT == srcT ? srcPrice : srcPrice + float(kT - srcT) / float(dstT - srcT) * (dstPrice - srcPrice)

    upCnt1 = 0
    dnCnt1 = 0
    upBx1  = 0
    dnBx1  = 0

    for d = 1 to mirrorBar1
        srcIdx = mirrorBar1 - d
        dstIdx = mirrorBar1 + d
        if dstIdx >= total1
            break
        srcT = array.get(timeArr1, srcIdx)
        dstT = array.get(timeArr1, dstIdx)

        if array.get(upArr1, srcIdx)
            // crossSystem: «“ ﬁÌ„  „ ﬁ«—‰ (”Ì” „?) ‘—Ê⁄ ? »Â ﬁÌ„  «’·Ì (”Ì” „?) Œ „

                        hit := k
                        break
                if hit != -1
                    symIdx = 2 * mirrorBar1 - srcIdx
                    symT   = symIdx >= 0 and symIdx < total1 ? array.get(timeArr1, symIdx) : array.get(timeArr1, srcIdx)
                    box.new(left=symT, top=array.get(highArr1, hit), right=last_bar_time,
