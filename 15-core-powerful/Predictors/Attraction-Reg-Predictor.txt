//????????????????????????????????????
// Inputs
//????????????????????????????????????
length = input.int(
     100,
     "Regression Length",
     minval = 10,
     maxval = 2000)

source = input.source(
     hlc3,
     "Regression Source")

useLog = input.bool(
     true,
     "Logarithmic Symmetry",
     tooltip = "»—«Ì »«“«—Â«Ì œ«—«Ì Õ—ò  œ—’œÌ »“—ê „‰«”»ù — «” .")

channelMultiplier = input.float(
     2.0,
     "Channel Multiplier",
     minval = 0.1,
     step = 0.1)

showMirror = input.bool(
     true,
     "Show Mirrored Candles")

showChannel = input.bool(
     true,
     "Show Regression Channel")

showEventLines = input.bool(
     true,
     "Show Pivot Symmetry Lines")

showConnector = input.bool(
     true,
     "Connect Two Event Lines")

     color.new(color.aqua, 20),
     "Mirror Bull Color")

     color.new(color.fuchsia, 20),
     "Mirror Bear Color")

     color.orange,
     "Optimal Line Color")

     color.blue,
     "Original Price Event Line")

mirrorLineColor = input.color(
     color.fuchsia,
     "Mirrored Price Event Line")

     color.white,
     "Connector Color")

attractionColor = input.color(
     color.lime,
     "Prediction Line Color")

//????????????????????????????????????
// Regression
//????????????????????????????????????
safeSource = math.max(source, syminfo.mintick)

regressionSource = useLog
     ? math.log(safeSource)
     : safeSource

fitNowRaw = ta.linreg(
     regressionSource,
     length,
     0)

basis = useLog
     ? math.exp(fitNowRaw)
     : fitNowRaw

enoughData = not na(basis)

//????????????????????????????????????
// Mathematical Helpers
//????????????????????????????????????
mirrorAroundBasis(float price) =>
    safePrice = math.max(price, syminfo.mintick)
    reflected = useLog
         ? (basis * basis) / safePrice
         : (2.0 * basis) - safePrice
    math.max(reflected, syminfo.mintick)

midpointPrice(float price1, float price2) =>
    safePrice1 = math.max(price1, syminfo.mintick)
    safePrice2 = math.max(price2, syminfo.mintick)
    useLog
         ? math.sqrt(safePrice1 * safePrice2)
         : (safePrice1 + safePrice2) / 2.0

reflectForward(float startPrice, float midPrice) =>
    safeStart = math.max(startPrice, syminfo.mintick)
    safeMid = math.max(midPrice, syminfo.mintick)
    reflected = useLog 
         ? (safeMid * safeMid) / safeStart 
         : (2.0 * safeMid) - safeStart
    math.max(reflected, syminfo.mintick)

//????????????????????????????????????
// Mirrored OHLC
//????????????????????????????????????
mirrorOpen = enoughData ? mirrorAroundBasis(open) : na
mirrorHigh = enoughData ? mirrorAroundBasis(low) : na
mirrorLow = enoughData ? mirrorAroundBasis(high) : na
mirrorClose = enoughData ? mirrorAroundBasis(close) : na

mirrorBull = mirrorClose >= mirrorOpen
mirrorCandleColor = mirrorBull ? bullColor : bearColor

plotcandle(
     showMirror ? mirrorOpen : na,
     showMirror ? mirrorHigh : na,
     showMirror ? mirrorLow : na,
     showMirror ? mirrorClose : na,
     title = "Mirrored OHLC",
     color = mirrorCandleColor,
     wickcolor = mirrorCandleColor,
     bordercolor = mirrorCandleColor)

//????????????????????????????????????

//????????????????????????????????????

     basis,
     title = "Optimal Regression Line",
     color = basisColor,
     linewidth = 2)

//????????????????????????????????????
// Regression channel
//????????????????????????????????????
residual = useLog
     ? math.log(safeSource / basis)
     : safeSource - basis

deviation = ta.stdev(residual, length)

upperChannel = useLog
     ? basis * math.exp(channelMultiplier * deviation)
     : basis + channelMultiplier * deviation

lowerChannel = useLog
     ? basis * math.exp(-channelMultiplier * deviation)
     : basis - channelMultiplier * deviation

upperPlot = plot(showChannel ? upperChannel : na, title = "Upper Band", color = color.new(basisColor, 45), linewidth = 1)
lowerPlot = plot(showChannel ? lowerChannel : na, title = "Lower Band", color = color.new(basisColor, 45), linewidth = 1)
fill(upperPlot, lowerPlot, title = "Regression Channel", color = showChannel ? color.new(basisColor, 92) : na)

//????????????????????????????????????

        midX = reEntryTime

        originalEndPrice := reflectForward(startY, midY)
        
        if showEventLines
            originalEventLine := line.new(
                 x1 = startX,
                 y1 = startY,

                 y2 = originalEndPrice,

                 width = 2)

        originalEventReady := true
        originalState := 0

// »——”Ì «ò” —Ì„ Å«ÌÌ‰ »«‰œ
else if enoughData and originalState == -1
    // «» œ« ÅÌÊÊ  (Å«ÌÌ‰ù —Ì‰ ‰ﬁÿÂ) ¬ÅœÌ  „Ìù‘Êœ
    if low < originalPivotPrice
        originalPivotPrice := low
        originalPivotTime := time
        originalBasisAtPivot := basis

    if high >= lowerChannel 
        reEntryTime = time
        reEntryPrice = lowerChannel
        
        startX = originalPivotTime
        startY = originalBasisAtPivot
        midX = reEntryTime
        midY = reEntryPrice
        
        originalEndTime := midX + (midX - startX)
        originalEndPrice := reflectForward(startY, midY)
