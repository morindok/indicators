//@version=5
indicator("Spirals by Candle Heights - Price Mirror (End Pivot)", overlay = true)

t1 = input.time(0, "Start Time", confirm = true)
t2 = input.time(0, "End Time", confirm = true)
opt_center = input.bool(false, "Center Both Spirals")
opt_swap = input.bool(false, "Swap Positions")
opt_continue = input.bool(false, "Continue Spirals")
opt_total_sum = input.bool(false, "Use Total Sum for Length")

var float green_sum = 0.0
var float red_sum = 0.0
var float max_p = na
var float min_p = na

var time_arr = array.new_int()
var price_arr = array.new_float()

// ¬—«ÌÂùÂ«Ì –ŒÌ—Â ò· œÌ «Ì ç«—  »—«Ì —”„ ê–‘ Â
var all_t = array.new_int()
var all_c = array.new_float()
var all_h = array.new_float()
var all_l = array.new_float()

array.push(all_t, time)
array.push(all_c, close)
array.push(all_h, high)
array.push(all_l, low)

in_range = time >= math.min(t1, t2) and time <= math.max(t1, t2)

if in_range
    float h = high - low
    if close > open
        green_sum += h
    if close < open
        red_sum += h
       
    max_p := na(max_p) ? high : math.max(max_p, high)
    min_p := na(min_p) ? low : math.min(min_p, low)
   
    array.push(time_arr, time)
    array.push(price_arr, (high + low) / 2.0)

var float p_end_r = na
var float p_end_g = na
var float p_end_r_sym = na
var float p_end_g_sym = na

if barstate.islast and array.size(time_arr) > 0
    int size = array.size(time_arr)
   
    float center_start = array.get(price_arr, 0)
    float center_end = array.get(price_arr, size - 1)
   
    int center_idx = math.floor(size / 2)
    int t_center = array.get(time_arr, center_idx)
    float p_center = array.get(price_arr, center_idx)

    float dt = math.abs(t2 - t1)
    float dp = math.abs(max_p - min_p)
   
    if dt == 0
        dt := timeframe.in_seconds("") * 1000.0
    if dp == 0
        dp := 1.0

    float b_coeff = math.log(1.618) / (math.pi / 2)
    float a = 0.02
   
    int t_base_r = math.min(t1, t2)
    float p_base_r = center_start
    int t_base_g = math.max(t1, t2)
    float p_base_g = center_end
   
    if opt_swap
        t_base_r := math.max(t1, t2)
        p_base_r := center_end
        t_base_g := math.min(t1, t2)
        p_base_g := center_start
       
    if opt_center
        t_base_r := t_center
        p_base_r := p_center
        t_base_g := t_center
        p_base_g := p_center

    float total_sum = red_sum + green_sum
    float target_r = opt_total_sum ? total_sum : red_sum
    float target_g = opt_total_sum ? total_sum : green_sum

    //  —”Ì„ «”ÅÌ—«· ﬁ—„“ Ê «”ÅÌ—«· „ ﬁ«—‰ ¬‰
    if target_r > 0 or opt_continue
        pts_red = array.new<chart.point>()
        pts_red_sym = array.new<chart.point>()
        array.push(pts_red, chart.point.from_time(t_base_r, p_base_r))
        array.push(pts_red_sym, chart.point.from_time(t_base_r, p_base_r))
        float acc_len_r = 0.0
        float prev_t = t_base_r
        float prev_p = p_base_r
        p_end_r := p_base_r 
        p_end_r_sym := p_base_r 
       
        for i = 1 to 2000
            float theta = i * 0.1
            float r = a * math.exp(b_coeff * theta)
            int curr_t = math.round(t_base_r + r * dt * math.cos(theta))
            float curr_p = p_base_r + r * dp * math.sin(theta)
            float curr_p_sym = p_base_r - r * dp * math.sin(theta) // ‰ﬁÿÂ „ ﬁ«—‰
            float seg = math.sqrt(math.pow(((curr_t - prev_t) / dt) * dp, 2) + math.pow(curr_p - prev_p, 2))
           
            if opt_continue or acc_len_r <= target_r
                array.push(pts_red, chart.point.from_time(curr_t, curr_p))
                array.push(pts_red_sym, chart.point.from_time(curr_t, curr_p_sym))
                acc_len_r += seg
                p_end_r := curr_p
                p_end_r_sym := curr_p_sym
            else
                break
            prev_t := curr_t
            prev_p := curr_p
        polyline.new(pts_red, xloc = xloc.bar_time, line_color = color.red)
        polyline.new(pts_red_sym, xloc = xloc.bar_time, line_color = color.new(color.red, 40)) // «”ÅÌ—«· „ ﬁ«—‰

    //  —”Ì„ «”ÅÌ—«· ”»“ Ê «”ÅÌ—«· „ ﬁ«—‰ ¬‰
    if target_g > 0 or opt_continue
        pts_green = array.new<chart.point>()
        pts_green_sym = array.new<chart.point>()
        array.push(pts_green, chart.point.from_time(t_base_g, p_base_g))
        array.push(pts_green_sym, chart.point.from_time(t_base_g, p_base_g))
        float acc_len_g = 0.0
        float prev_t = t_base_g
        float prev_p = p_base_g
        p_end_g := p_base_g
        p_end_g_sym := p_base_g
       
        for i = 1 to 2000
            float theta = i * 0.1
            float r = a * math.exp(b_coeff * theta)
            int curr_t = math.round(t_base_g + r * dt * math.cos(-theta))
            float curr_p = p_base_g + r * dp * math.sin(-theta)
            float curr_p_sym = p_base_g - r * dp * math.sin(-theta) // ‰ﬁÿÂ „ ﬁ«—‰
            float seg = math.sqrt(math.pow(((curr_t - prev_t) / dt) * dp, 2) + math.pow(curr_p - prev_p, 2))
           
            if opt_continue or acc_len_g <= target_g
                array.push(pts_green, chart.point.from_time(curr_t, curr_p))
                array.push(pts_green_sym, chart.point.from_time(curr_t, curr_p_sym))
                acc_len_g += seg
                p_end_g := curr_p
                p_end_g_sym := curr_p_sym
            else
                break
            prev_t := curr_t
            prev_p := curr_p
        polyline.new(pts_green, xloc = xloc.bar_time, line_color = color.green)
        polyline.new(pts_green_sym, xloc = xloc.bar_time, line_color = color.new(color.green, 40)) // «”ÅÌ—«· „ ﬁ«—‰

    //  —”Ì„ „ ﬁ«—‰ ﬁÌ„ ùÂ« »—«Ì ò· ç«— 
    int total_bars = array.size(all_t)
    int start_idx = math.max(0, total_bars - 9999)

    // ﬁÌ„ ùÂ«Ì „ ﬁ«—‰ »—«Ì «”ÅÌ—«· ﬁ—„“ «’·Ì
    if not na(p_end_r)
        pts_cr = array.new<chart.point>()
        pts_hr = array.new<chart.point>()
        pts_lr = array.new<chart.point>()
        for i = start_idx to total_bars - 1
            int t_i = array.get(all_t, i)
            array.push(pts_cr, chart.point.from_time(t_i, 2 * p_end_r - array.get(all_c, i)))
            array.push(pts_hr, chart.point.from_time(t_i, 2 * p_end_r - array.get(all_l, i))) // low ﬁ—Ì‰Â „Ìù‘Êœ Ã«Ì high
            array.push(pts_lr, chart.point.from_time(t_i, 2 * p_end_r - array.get(all_h, i))) // high ﬁ—Ì‰Â „Ìù‘Êœ Ã«Ì low
        polyline.new(pts_cr, xloc = xloc.bar_time, line_color = color.new(color.red, 30))
        polyline.new(pts_hr, xloc = xloc.bar_time, line_color = color.new(color.red, 60))
        polyline.new(pts_lr, xloc = xloc.bar_time, line_color = color.new(color.red, 60))

    // ﬁÌ„ ùÂ«Ì „ ﬁ«—‰ »—«Ì «”ÅÌ—«· ﬁ—„“ ﬁ—Ì‰Â
    if not na(p_end_r_sym)
        pts_cr_sym = array.new<chart.point>()
        pts_hr_sym = array.new<chart.point>()
        pts_lr_sym = array.new<chart.point>()
        for i = start_idx to total_bars - 1
            int t_i = array.get(all_t, i)
            array.push(pts_cr_sym, chart.point.from_time(t_i, 2 * p_end_r_sym - array.get(all_c, i)))
            array.push(pts_hr_sym, chart.point.from_time(t_i, 2 * p_end_r_sym - array.get(all_l, i)))
            array.push(pts_lr_sym, chart.point.from_time(t_i, 2 * p_end_r_sym - array.get(all_h, i)))
        polyline.new(pts_cr_sym, xloc = xloc.bar_time, line_color = color.new(color.maroon, 40))
        polyline.new(pts_hr_sym, xloc = xloc.bar_time, line_color = color.new(color.maroon, 70))
        polyline.new(pts_lr_sym, xloc = xloc.bar_time, line_color = color.new(color.maroon, 70))

    // ﬁÌ„ ùÂ«Ì „ ﬁ«—‰ »—«Ì «”ÅÌ—«· ”»“ «’·Ì
    if not na(p_end_g)
        pts_cg = array.new<chart.point>()
        pts_hg = array.new<chart.point>()
        pts_lg = array.new<chart.point>()
        for i = start_idx to total_bars - 1
            int t_i = array.get(all_t, i)
            array.push(pts_cg, chart.point.from_time(t_i, 2 * p_end_g - array.get(all_c, i)))
            array.push(pts_hg, chart.point.from_time(t_i, 2 * p_end_g - array.get(all_l, i)))
            array.push(pts_lg, chart.point.from_time(t_i, 2 * p_end_g - array.get(all_h, i)))
        polyline.new(pts_cg, xloc = xloc.bar_time, line_color = color.new(color.green, 30))
        polyline.new(pts_hg, xloc = xloc.bar_time, line_color = color.new(color.green, 60))
        polyline.new(pts_lg, xloc = xloc.bar_time, line_color = color.new(color.green, 60))

    // ﬁÌ„ ùÂ«Ì „ ﬁ«—‰ »—«Ì «”ÅÌ—«· ”»“ ﬁ—Ì‰Â
    if not na(p_end_g_sym)
        pts_cg_sym = array.new<chart.point>()
        pts_hg_sym = array.new<chart.point>()
        pts_lg_sym = array.new<chart.point>()
        for i = start_idx to total_bars - 1
            int t_i = array.get(all_t, i)
            array.push(pts_cg_sym, chart.point.from_time(t_i, 2 * p_end_g_sym - array.get(all_c, i)))
            array.push(pts_hg_sym, chart.point.from_time(t_i, 2 * p_end_g_sym - array.get(all_l, i)))
            array.push(pts_lg_sym, chart.point.from_time(t_i, 2 * p_end_g_sym - array.get(all_h, i)))
        polyline.new(pts_cg_sym, xloc = xloc.bar_time, line_color = color.new(color.teal, 40))
        polyline.new(pts_hg_sym, xloc = xloc.bar_time, line_color = color.new(color.teal, 70))
        polyline.new(pts_lg_sym, xloc = xloc.bar_time, line_color = color.new(color.teal, 70))
