rot_x(cx, cy, rx, ry, a, theta) =>
    raw_px = int(math.round(cx + rx * math.cos(a + theta)))
    math.min(bar_index + 499, math.max(bar_index - 4999, math.max(0, raw_px)))

rot_y(cx, cy, rx, ry, a, theta) =>
    cy + ry * math.sin(a + theta)

drawPoly(cx, cy, rx, ry, n, offset, theta, col) =>
    pts = array.new<chart.point>()
    for i = 0 to n
        a = 2 * math.pi * i / n + offset
        px = rot_x(cx, cy, rx, ry, a, theta)
        py = rot_y(cx, cy, rx, ry, a, theta)
        pts.push(chart.point.from_index(px, py))
    polyline.new(pts, closed=true, line_color=col, line_width=2)

drawStar(cx, cy, rx, ry, theta, col) =>
    pts = array.new<chart.point>()
    for i = 0 to 9
        r = i % 2 == 0 ? 1.0 : 0.45
        a = 2 * math.pi * i / 10 - math.pi / 2
        px = rot_x(cx, cy, rx * r, ry * r, a, theta)
        py = rot_y(cx, cy, rx * r, ry * r, a, theta)
        pts.push(chart.point.from_index(px, py))
    polyline.new(pts, closed=true, line_color=col, line_width=2)

get_exact_bounds(cx, cy, rx, ry, theta, type, find_min) =>
    int n = 60
    float offset = 0.0
    bool isStar = false
    if type == "Rectangle"
        n := 4
        offset := math.pi / 4
    else if type == "Triangle"
        n := 3
        offset := -math.pi / 2
    else if type == "Diamond"
        n := 4
        offset := -math.pi / 2
    else if type == "Pentagon"
        n := 5
        offset := -math.pi / 2
    else if type == "Hexagon"
        n := 6
        offset := 0.0
    else if type == "Octagon"
        n := 8
        offset := -math.pi / 8
    else if type == "Star"
        isStar := true
    int best_x = find_min ? 9999999 : -9999999
    float best_y = na
    if isStar
        for i = 0 to 9
            r = i % 2 == 0 ? 1.0 : 0.45
            a = 2 * math.pi * i / 10 - math.pi / 2
            px = rot_x(cx, cy, rx * r, ry * r, a, theta)
            py = rot_y(cx, cy, rx * r, ry * r, a, theta)
            if (find_min and px < best_x) or (not find_min and px > best_x)
                best_x := px
                best_y := py
    else
        for i = 0 to n
            a = 2 * math.pi * i / n + offset
            px = rot_x(cx, cy, rx, ry, a, theta)
            py = rot_y(cx, cy, rx, ry, a, theta)
            if (find_min and px < best_x) or (not find_min and px > best_x)
                best_x := px
                best_y := py
    [best_x, best_y]

point_in_shape(cx, cy, rx, ry, theta, type, px, py) =>
    dx = float(px - cx)
    dy = py - cy
    cos_t = math.cos(-theta)
    sin_t = math.sin(-theta)
    lx = dx * cos_t - dy * sin_t
    ly = dx * sin_t + dy * cos_t
    nx = rx > 0 ? math.abs(lx) / rx : 1.0
    ny = ry > 0 ? math.abs(ly) / ry : 1.0
    result = false
    if type == "Ellipse"
        result := (nx * nx + ny * ny) <= 1.0
    else if type == "Diamond"
        result := (nx + ny) <= 1.0
    else if type == "Rectangle"
        result := nx <= 1.0 and ny <= 1.0
    else
        result := (nx * nx + ny * ny) <= 1.0
    result

get_shape_dy(dx, rx, ry, type) =>
    float dy = ry
    nx = rx > 0 ? math.abs(dx) / rx : 1.0
    if type == "Ellipse" or type == "Pentagon" or type == "Hexagon" or type == "Octagon" or type == "Star"
        dy := ry * math.sqrt(math.max(0.0, 1.0 - nx * nx))
    else if type == "Diamond" or type == "Triangle"
        dy := ry * math.max(0.0, 1.0 - nx)
    else if type == "Rectangle"
        dy := ry
    dy

    int exact_exit_idx = na
    int max_search = math.round(r_x2)

    int last_inside_idx = na
    for i = max_search to 0
        int current_idx = int(e_bar) + i
        int bars_back = bar_index - current_idx
        if bars_back >= 0 and bars_back <= 4999
            float c_val = close[bars_back]
            bool is_close_inside = point_in_shape(e_bar, e_p, r_x2, r_y2, theta, shape_type, current_idx, c_val)
            if is_close_inside and na(last_inside_idx)

    bars_back_2 = bar_index - valid_exit_idx
    if bars_back_2 >= 0 and bars_back_2 <= 4999
        c_val2 = close[bars_back_2]

    if not na(exact_enter_idx) and not na(valid_exit_idx) and not na(white_y1) and not na(white_y2)
        line.new(exact_enter_idx, white_y1, valid_exit_idx, white_y2, color=color.white, width=2, extend=extend.right)

            float c_val = close[bars_back]
            float dyn_lv = s_level + (active_slope * (current_idx - piv_bar))
            float sym_price = 2 * dyn_lv - c_val
            bool sym_inside_shape1 = point_in_shape(s_bar, s_p, r_x1, r_y1, theta, shape_type, current_idx, sym_price)
            if sym_inside_shape1 and na(rev_enter_idx)
                rev_enter_idx := current_idx

    int rev_last_inside_idx = na
    for i = max_search to 0
        int current_idx = int(e_bar) + i
        int bars_back = bar_index - current_idx
        if bars_back >= 0 and bars_back <= 4999
            float c_val = close[bars_back]
            float dyn_lv = s_level + (active_slope * (current_idx - piv_bar))
            float sym_price = 2 * dyn_lv - c_val
            bool sym_inside_shape2 = point_in_shape(e_bar, e_p, r_x2, r_y2, theta, shape_type, current_idx, sym_price)
            if sym_inside_shape2 and na(rev_last_inside_idx)
                rev_last_inside_idx := current_idx

    if not na(rev_last_inside_idx)

    if na(rev_exit_idx)
        rev_exit_idx := math.round(e_bar + r_x2)

    float orange_y1 = na
    float orange_y2 = na

    if not na(rev_enter_idx)
        int bb1 = bar_index - rev_enter_idx
        if bb1 >= 0 and bb1 <= 4999
            orange_y1 := close[bb1]

    int bb2 = bar_index - valid_rev_exit

        line.new(rev_enter_idx, orange_y1, valid_rev_exit, orange_y2, color=color.orange, width=2, extend=extend.right)

if barstate.islast
    active_slope_final = use_slope ? (auto_slope ? final_auto_slope : manual_slope) : 0.0
    shape_angle = (use_slope and rotate_shapes) ? math.atan(active_slope_final) : 0.0

    sym_pts = array.new<chart.point>()
    loop_limit = math.min(bar_index, 4999)
    for i = loop_limit to 0
        idx = bar_index - i
        past_close = close[i]
        dyn_level = sym_level + (active_slope_final * (idx - pivot_bar))
        sym_c = 2 * dyn_level - past_close
        sym_pts.push(chart.point.from_index(idx, sym_c))
    polyline.new(sym_pts, closed=false, line_color=color.new(color.purple, 0), line_width=2)

    if not na(start_bar) and not na(end_bar)
        bars_to_start = bar_index - start_bar
        bars_to_end = bar_index - end_bar
        loop_start_box = math.min(bars_to_start, 4999)
        loop_end_box = math.max(bars_to_end, 0)

        final_acc_total = 0.0
        final_max_dist = 0.0

        if loop_start_box >= loop_end_box
            for i = loop_start_box to loop_end_box
                idx = bar_index - i
                past_close = close[i]
                dyn_level = sym_level + (active_slope_final * (idx - pivot_bar))
                sym_c = 2 * dyn_level - past_close
                dist = math.abs(past_close - sym_c)
                final_acc_total += dist
                final_max_dist := math.max(final_max_dist, dist)

        rx1 = math.min(math.round(final_acc_total / (2 * math.pi) * width_factor_1), 450)
        ry1 = final_max_dist / 2 * height_factor_1
        rx2 = math.min(math.round(final_acc_total / (2 * math.pi) * width_factor_2), 450)
        ry2 = final_max_dist / 2 * height_factor_2

        draw_elements(start_bar, start_price, end_bar, end_price, rx1, ry1, rx2, ry2, shape_angle, active_slope_final, pivot_bar, sym_level)

        if mirror_shapes
            sym_start_price = 2 * (sym_level + active_slope_final * (start_bar - pivot_bar)) - start_price
            sym_end_price = 2 * (sym_level + active_slope_final * (end_bar - pivot_bar)) - end_price
            draw_elements(start_bar, sym_start_price, end_bar, sym_end_price, rx1, ry1, rx2, ry2, -shape_angle, active_slope_final, pivot_bar, sym_level)
