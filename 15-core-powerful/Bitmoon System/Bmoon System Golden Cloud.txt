//@version=6
indicator("Bmoon System Golden Cloud{Morindok}", overlay=true, max_bars_back=4000)

// ==========================================
// 1.  ‰ŸÌ„«  “„«‰ Ê „Õ«”»Â »«“Â «‰ Œ«»Ì (Selected Interval)
// ==========================================
selectedTime1 = input.time(0, title="P1 (Upper) Reference Time", confirm=true)
selectedTime2 = input.time(0, title="P2 (Lower) Reference Time", confirm=true)

var int refBarIndex1 = na
var int refBarIndex2 = na

if time <= selectedTime1
    refBarIndex1 := bar_index
if time <= selectedTime2
    refBarIndex2 := bar_index

int interval_length = 50 // „ﬁœ«— ÅÌ‘ù›—÷
if not na(refBarIndex1) and not na(refBarIndex2) and refBarIndex1 != refBarIndex2
    interval_length := math.max(1, math.abs(refBarIndex2 - refBarIndex1))

// ==========================================
// 2.  ‰ŸÌ„«  Â” Â Dynamic Spine 
// ==========================================
grp_dyn = "Dynamic Spine Core"
step_threshold_in = input.float(1.0, "Step Threshold (%)", group=grp_dyn) / 100.0
length_multiplier = input.float(2.5, "Length Multiplier", group=grp_dyn)
initial_length = input.int(5, "Initial Length", minval=1, group=grp_dyn)
srcCenter = input.source(close, "Base Source", group=grp_dyn)

calculate_dynamic_line(float source) =>
    var float last_price = source
    var int current_length = initial_length
    var int actual_segment_length = initial_length 
    var bool is_upward = true
    var int step_count = 0
    var float line_level = source
    var int step_start_bar = bar_index
    
    float price_change = last_price != 0 ? (source - last_price) / last_price : 0.0
    
    if math.abs(price_change) > step_threshold_in
        bool new_direction = price_change > 0
        if new_direction != is_upward
            actual_segment_length := math.max(1, bar_index - step_start_bar) 
            step_count := 1
            is_upward := new_direction
            line_level := source
            step_start_bar := bar_index
            current_length := initial_length
        else
            step_count += 1
            int bars_in_step = bar_index - step_start_bar
            current_length := int(math.round(initial_length * math.pow(length_multiplier, step_count - 1)))
            if bars_in_step >= current_length
                actual_segment_length := math.max(1, bar_index - step_start_bar) 
                line_level := source
                step_start_bar := bar_index
        last_price := source
        
    int elapsed_in_step = bar_index - step_start_bar
    [line_level, actual_segment_length, elapsed_in_step]

[dynamic_midline, dyn_period, elapsed_bars] = calculate_dynamic_line(srcCenter)

// ==========================================
// 3. „Õ«”»Â ÿÊ· œ«Ì‰«„Ìò »— «”«” ›«’·Â ›—«ò «· Œ«„
// ==========================================
grp_cycle = "Cycle Period Mode"
// ê“Ì‰Â Selected Interval «÷«›Â ‘œ
cycleMode = input.string("Selected Interval", "Cycle Mode", options=["Selected Interval", "Step Dynamic", "Fractal Distance", "Manual"], group=grp_cycle)
manualCycle = input.int(50, "Manual Cycle Period", minval=1, group=grp_cycle)
n_frac = input.int(100, title="Fractal Lookback (n)", minval=2, group=grp_cycle)

bool dyn_upDnFront = true, bool dyn_upUp0 = true, bool dyn_upUp1 = true, bool dyn_upUp2 = true, bool dyn_upUp3 = true, bool dyn_upUp4 = true
bool dyn_dnDnFront = true, bool dyn_dnUp0 = true, bool dyn_dnUp1 = true, bool dyn_dnUp2 = true, bool dyn_dnUp3 = true, bool dyn_dnUp4 = true

for i = 1 to n_frac
    dyn_upDnFront := dyn_upDnFront and (high[n_frac-i] < high[n_frac])
    dyn_upUp0 := dyn_upUp0 and (high[n_frac+i] < high[n_frac])
    dyn_upUp1 := dyn_upUp1 and (high[n_frac+1] <= high[n_frac] and high[n_frac+i+1] < high[n_frac])
    dyn_upUp2 := dyn_upUp2 and (high[n_frac+1] <= high[n_frac] and high[n_frac+2] <= high[n_frac] and high[n_frac+i+2] < high[n_frac])
    dyn_upUp3 := dyn_upUp3 and (high[n_frac+1] <= high[n_frac] and high[n_frac+2] <= high[n_frac] and high[n_frac+3] <= high[n_frac] and high[n_frac+i+3] < high[n_frac])
    dyn_upUp4 := dyn_upUp4 and (high[n_frac+1] <= high[n_frac] and high[n_frac+2] <= high[n_frac] and high[n_frac+3] <= high[n_frac] and high[n_frac+4] <= high[n_frac] and high[n_frac+i+4] < high[n_frac])
    
    dyn_dnDnFront := dyn_dnDnFront and (low[n_frac-i] > low[n_frac])
    dyn_dnUp0 := dyn_dnUp0 and (low[n_frac+i] > low[n_frac])
    dyn_dnUp1 := dyn_dnUp1 and (low[n_frac+1] >= low[n_frac] and low[n_frac+i+1] > low[n_frac])
    dyn_dnUp2 := dyn_dnUp2 and (low[n_frac+1] >= low[n_frac] and low[n_frac+2] >= low[n_frac] and low[n_frac+i+2] > low[n_frac])
    dyn_dnUp3 := dyn_dnUp3 and (low[n_frac+1] >= low[n_frac] and low[n_frac+2] >= low[n_frac] and low[n_frac+3] >= low[n_frac] and low[n_frac+i+3] > low[n_frac])
    dyn_dnUp4 := dyn_dnUp4 and (low[n_frac+1] >= low[n_frac] and low[n_frac+2] >= low[n_frac] and low[n_frac+3] >= low[n_frac] and low[n_frac+4] >= low[n_frac] and low[n_frac+i+4] > low[n_frac])

bool rawUpFractal = (dyn_upDnFront and (dyn_upUp0 or dyn_upUp1 or dyn_upUp2 or dyn_upUp3 or dyn_upUp4))
bool rawDownFractal = (dyn_dnDnFront and (dyn_dnUp0 or dyn_dnUp1 or dyn_dnUp2 or dyn_dnUp3 or dyn_dnUp4))

var int lastUpBar = na
var int lastDownBar = na
if rawUpFractal
    lastUpBar := bar_index - n_frac
if rawDownFractal
    lastDownBar := bar_index - n_frac

int fractal_dist_length = 14 
if not na(lastUpBar) and not na(lastDownBar)
    fractal_dist_length := math.max(1, math.abs(lastUpBar - lastDownBar))

// ==========================================
// 4. „Õ«”»«  Spiral Spine 
// ==========================================
grp_spiral = "Spiral Spine Settings"
radius_scale = input.float(0.15, "Radius Scale", step=0.05, group=grp_spiral)

// «‰ Œ«» ÿÊ· œÊ—Â »— «”«” Õ«·  «‰ Œ«»ù‘œÂ (Å‘ Ì»«‰Ì «“ Õ«·  ÃœÌœ)
int period_len = cycleMode == "Selected Interval" ? interval_length : (cycleMode == "Fractal Distance" ? fractal_dist_length : (cycleMode == "Step Dynamic" ? math.max(1, dyn_period) : manualCycle))
int current_step = bar_index % period_len 

float raw_vol = ta.cum(high - low) - ta.valuewhen(current_step == 0, ta.cum(high - low), 0)
float smooth_vol = ta.wma(raw_vol, 10)

float progress_pi = (float(current_step) / period_len) * math.pi
float wave = math.sin(progress_pi)

float rad_1 = math.toradians(137.5)
float rad_2 = math.toradians(222.5)

float upper_spine = dynamic_midline + (smooth_vol * radius_scale * wave * math.sin(rad_1))
float lower_spine = dynamic_midline + (smooth_vol * radius_scale * wave * math.sin(rad_2))

plot(dynamic_midline, "Dynamic Midline", color=color.new(color.gray, 50), linewidth=2, style=plot.style_stepline)
p_upper = plot(upper_spine, "Upper Spine Band", color=color.new(color.red, 60))
p_lower = plot(lower_spine, "Lower Spine Band", color=color.new(color.blue, 60))
fill(p_upper, p_lower, color=color.new(color.purple, 90), title="Spine Spiral Area")

// ==========================================
// 5. „Ê Ê—  ﬁ«—‰ ›—«ò «· (Dual Mirror)
// ==========================================
n = n_frac

baseMaLen = input.int(5, title="Base Price MA Length")
pOpen  = ta.sma(open, baseMaLen)
pHigh  = ta.sma(high, baseMaLen)
pLow   = ta.sma(low, baseMaLen)
pClose = ta.sma(close, baseMaLen)

centralMirrorLevel = dynamic_midline
mirrorLevel1       = upper_spine
mirrorLevel2       = lower_spine

cOpen  = float(2.0 * centralMirrorLevel - pOpen)
cHigh  = float(2.0 * centralMirrorLevel - pLow)
cLow   = float(2.0 * centralMirrorLevel - pHigh)
cClose = float(2.0 * centralMirrorLevel - pClose)

mHigh1 = float(2.0 * mirrorLevel1 - cLow)
mLow1  = float(2.0 * mirrorLevel1 - cHigh)
mHigh2 = float(2.0 * mirrorLevel2 - cLow)
mLow2  = float(2.0 * mirrorLevel2 - cHigh)

// ==========================================
// 6.  ‘ŒÌ’ ›—«ò «·ùÂ« —ÊÌ œ«œÂùÂ«Ì «”„Ê  ‘œÂ
// ==========================================
bool u_dn = true, bool u_u0 = true, bool u_u1 = true, bool u_u2 = true, bool u_u3 = true, bool u_u4 = true
bool d_dn = true, bool d_u0 = true, bool d_u1 = true, bool d_u2 = true, bool d_u3 = true, bool d_u4 = true

for i = 1 to n
    u_dn := u_dn and (pHigh[n-i] < pHigh[n])
    u_u0 := u_u0 and (pHigh[n+i] < pHigh[n])
    u_u1 := u_u1 and (pHigh[n+1] <= pHigh[n] and pHigh[n+i+1] < pHigh[n])
    u_u2 := u_u2 and (pHigh[n+1] <= pHigh[n] and pHigh[n+2] <= pHigh[n] and pHigh[n+i+2] < pHigh[n])
    u_u3 := u_u3 and (pHigh[n+1] <= pHigh[n] and pHigh[n+2] <= pHigh[n] and pHigh[n+3] <= pHigh[n] and pHigh[n+i+3] < pHigh[n])
    u_u4 := u_u4 and (pHigh[n+1] <= pHigh[n] and pHigh[n+2] <= pHigh[n] and pHigh[n+3] <= pHigh[n] and pHigh[n+4] <= pHigh[n] and pHigh[n+i+4] < pHigh[n])
    
    d_dn := d_dn and (pLow[n-i] > pLow[n])
    d_u0 := d_u0 and (pLow[n+i] > pLow[n])
    d_u1 := d_u1 and (pLow[n+1] >= pLow[n] and pLow[n+i+1] > pLow[n])
    d_u2 := d_u2 and (pLow[n+1] >= pLow[n] and pLow[n+2] >= pLow[n] and pLow[n+i+2] > pLow[n])
    d_u3 := d_u3 and (pLow[n+1] >= pLow[n] and pLow[n+2] >= pLow[n] and pLow[n+3] >= pLow[n] and pLow[n+i+3] > pLow[n])
    d_u4 := d_u4 and (pLow[n+1] >= pLow[n] and pLow[n+2] >= pLow[n] and pLow[n+3] >= pLow[n] and pLow[n+4] >= pLow[n] and pLow[n+i+4] > pLow[n])

flagUp = u_u0 or u_u1 or u_u2 or u_u3 or u_u4
flagDn = d_u0 or d_u1 or d_u2 or d_u3 or d_u4

upFractal1 = u_dn and flagUp and time[n] < selectedTime1
downFractal1 = d_dn and flagDn and time[n] < selectedTime1
upFractal2 = u_dn and flagUp and time[n] < selectedTime2
downFractal2 = d_dn and flagDn and time[n] < selectedTime2

var upFracIdx1 = array.new_int(0)
var dnFracIdx1 = array.new_int(0)
if upFractal1
    array.push(upFracIdx1, int(bar_index - n))
if downFractal1
    array.push(dnFracIdx1, int(bar_index - n))

var upFracIdx2 = array.new_int(0)
var dnFracIdx2 = array.new_int(0)
if upFractal2
    array.push(upFracIdx2, int(bar_index - n))
if downFractal2
    array.push(dnFracIdx2, int(bar_index - n))

bool p1Up = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(upFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p1Dn = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(dnFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p2Up = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(upFracIdx2, 2 * refBarIndex2 - bar_index) >= 0
bool p2Dn = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(dnFracIdx2, 2 * refBarIndex2 - bar_index) >= 0

// ==========================================
// 7. „Ê Ê—  ﬁ«ÿ⁄ Ê  Ê·Ìœ ŒÿÊÿ «” « Ìò
// ==========================================
var float[] activeWhiteLines = array.new_float()

var int yPrevX = na, var float yPrevY = na
var int yCurrX = na, var float yCurrY = na
var int gPrevX = na, var float gPrevY = na
var int gCurrX = na, var float gCurrY = na
bool triggerIntersectionCheck = false

if p1Up
    yPrevX := yCurrX, yPrevY := yCurrY, yCurrX := bar_index, yCurrY := mHigh1
    triggerIntersectionCheck := true
if p2Up
    yPrevX := yCurrX, yPrevY := yCurrY, yCurrX := bar_index, yCurrY := mHigh2
    triggerIntersectionCheck := true

if p1Dn
    gPrevX := gCurrX, gPrevY := gCurrY, gCurrX := bar_index, gCurrY := mLow1
    triggerIntersectionCheck := true
if p2Dn
    gPrevX := gCurrX, gPrevY := gCurrY, gCurrX := bar_index, gCurrY := mLow2
    triggerIntersectionCheck := true

if triggerIntersectionCheck and not na(yPrevX) and not na(yCurrX) and not na(gPrevX) and not na(gCurrX)
    if (yCurrX != yPrevX) and (gCurrX != gPrevX)
        float m1 = (yCurrY - yPrevY) / (yCurrX - yPrevX)
        float m2 = (gCurrY - gPrevY) / (gCurrX - gPrevX)
        if m1 != m2
            float ix = (m1 * yPrevX - m2 * gPrevX + gPrevY - yPrevY) / (m1 - m2)
            if ix >= math.max(yPrevX, gPrevX) and ix <= math.min(yCurrX, gCurrX)
                float iy = m1 * (ix - yPrevX) + yPrevY
                array.push(activeWhiteLines, iy) 

// ==========================================
// 8. ”Ìê‰«·ùœÂÌ
// ==========================================
bool buySignal = false
bool sellSignal = false

if array.size(activeWhiteLines) > 0
    int idx = array.size(activeWhiteLines) - 1
    while idx >= 0
        float lineLevel = array.get(activeWhiteLines, idx)
        if high >= lineLevel and low <= lineLevel
            if close[1] < lineLevel or (close[1] == lineLevel and open < lineLevel)
                sellSignal := true
            else if close[1] > lineLevel or (close[1] == lineLevel and open > lineLevel)
                buySignal := true
            array.remove(activeWhiteLines, idx)
        idx -= 1

plotshape(buySignal, title="Buy Signal", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.normal)
plotshape(sellSignal, title="Sell Signal", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.normal)
