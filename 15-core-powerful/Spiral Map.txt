// --- Custom atan2 Function ---
custom_atan2(y, x) =>
    float angle = 0.0
    if x > 0
        angle := math.atan(y / x)
    else if x < 0 and y >= 0
        angle := math.atan(y / x) + math.pi
    else if x < 0 and y < 0
        angle := math.atan(y / x) - math.pi
    else if x == 0 and y > 0
        angle := math.pi / 2
    else if x == 0 and y < 0
        angle := -math.pi / 2
    angle

// ---  «»⁄ «” Œ—«Ã œﬁÌﬁ “«ÊÌÂ  ﬁ«ÿ⁄ œÊ «”ÅÌ—«· ---
findIntersection(float[] x1Arr, float[] y1Arr, float[] x2Arr, float[] y2Arr, int cx, float cy) =>
    float intersectAngle = na
    bool found = false
    int s1 = array.size(x1Arr)
    int s2 = array.size(x2Arr)
    if s1 > 10 and s2 > 10
        for i = 5 to s1 - 2
            if found
                break
            float xA = array.get(x1Arr, i)
            float yA = array.get(y1Arr, i)
            float xB = array.get(x1Arr, i+1)
            float yB = array.get(y1Arr, i+1)
            for j = 5 to s2 - 2
                float xC = array.get(x2Arr, j)
                float yC = array.get(y2Arr, j)
                float xD = array.get(x2Arr, j+1)
                float yD = array.get(y2Arr, j+1)
                
                float den = (xA - xB) * (yC - yD) - (yA - yB) * (xC - xD)
                if den != 0
                    float t = ((xA - xC) * (yC - yD) - (yA - yC) * (xC - xD)) / den
                    float u = -((xA - xB) * (yA - yC) - (yA - yB) * (xA - xC)) / den
                    if t >= 0 and t <= 1 and u >= 0 and u <= 1
                        float px = xA + t * (xB - xA)
                        float py = yA + t * (yB - yA)
                        
                        float angRad = custom_atan2(py - cy, px - cx)
                        float angDeg = angRad * 180.0 / math.pi
                        if angDeg < 0
                            angDeg += 360.0
                        intersectAngle := angDeg
                        found := true
                        break
    intersectAngle

// ---  «»⁄ —”„ «”ÅÌ—«· »ÂÌ‰Âù‘œÂ »« Å«—«„ — ÃÂ  X ---
drawSmoothMappedSpiral(float targetRadius, float targetArcLen, int cx, float cy, bool isCw, color c, polyline oldPoly, float targetDeg, int[] idxArr, float[] priceArr, float[] cumDArr, bool findIntersects, color lblColor, string lblPrefix, float dirX) =>
    polyline newPoly = na
    float labelX = na
    float labelY = na
    int chartBarIdx = na
    float chartCandlePrice = na
    float targetRad = targetDeg * math.pi / 180.0
    float thetaEnd = 0.0
    float[] xCoords = array.new_float()
    float[] yCoords = array.new_float()
    
    if targetRadius > 0 and targetArcLen > 0 and array.size(idxArr) > 0
        float targetRatio = targetArcLen / targetRadius
        float baseArcLen = 0.0
        float lastPx = 0.0
        float lastPy = 0.0
        float finalBaseRadius = 1.0
        float stepCalc = 0.01
        
        for i = 1 to 10000
            float theta = i * stepCalc
            float r = spType == "Archimedean" ? theta : (spType == "Logarithmic" ? math.pow(1.1, theta) : math.pow(1.05, theta))
            float px = r * math.cos(theta)
            float py = r * math.sin(theta)
            if i > 1
                baseArcLen := baseArcLen + math.sqrt(math.pow(px - lastPx, 2) + math.pow(py - lastPy, 2))
            float baseRadius = math.sqrt(px*px + py*py)
            thetaEnd := theta
            finalBaseRadius := baseRadius
            if baseRadius > 0 and (baseArcLen / baseRadius) >= targetRatio
                break
            lastPx := px
            lastPy := py

        float visualScale = 80.0 / finalBaseRadius
        
        var pts = array.new<chart.point>()
        array.clear(pts)
        int drawSteps = 250
        float drawStep = thetaEnd / drawSteps
        
        for i = 0 to drawSteps
            float t = i * drawStep
            float r = spType == "Archimedean" ? t : (spType == "Logarithmic" ? math.pow(1.1, t) : math.pow(1.05, t))
            float aDir = isCw ? -t : t
            
            float rawX = (r * math.cos(aDir)) * visualScale * scaleX * dirX
            float rawY = (r * math.sin(aDir)) * visualScale * scaleY
            
            int pX_coord = cx + int(math.round(rawX))
            float pY_coord = cy + rawY
            
            array.push(xCoords, float(pX_coord))
            array.push(yCoords, pY_coord)
            
            if pX_coord > bar_index + 450
                pX_coord := bar_index + 450
                
            array.push(pts, chart.point.from_index(pX_coord, pY_coord))
            
        if not na(oldPoly)
            polyline.delete(oldPoly)
        newPoly := polyline.new(pts, curved=true, line_color=c, line_width=2)
        
        float closestThetaDiff = 999999.0
        for i = 0 to array.size(idxArr) - 1
            float targetD = array.get(cumDArr, i)
            float mappedTheta = (targetD / targetArcLen) * thetaEnd
            
            if math.abs(mappedTheta - targetRad) < closestThetaDiff
                closestThetaDiff := math.abs(mappedTheta - targetRad)
                float rMap = spType == "Archimedean" ? mappedTheta : (spType == "Logarithmic" ? math.pow(1.1, mappedTheta) : math.pow(1.05, mappedTheta))
                float aMapDir = isCw ? -mappedTheta : mappedTheta
                
                float rLblX = (rMap * math.cos(aMapDir)) * visualScale * scaleX * dirX
                float rLblY = (rMap * math.sin(aMapDir)) * visualScale * scaleY
                
                int finalLblX = cx + int(math.round(rLblX))
                if finalLblX > bar_index + 450
                    finalLblX := bar_index + 450
                    
                labelX := finalLblX
                labelY := cy + rLblY
                chartBarIdx := array.get(idxArr, i)
                chartCandlePrice := array.get(priceArr, i)

        if findIntersects and showIntersections
            float baseRad = targetRad % (2 * math.pi)
            int k = 0
            while true
                float tk = baseRad + k * 2 * math.pi
                if tk > thetaEnd
                    break
                
                float expectedD = tk * (targetArcLen / thetaEnd)
                float minDist = 999999.0
                int bestI = -1
                for i = 0 to array.size(cumDArr) - 1
                    float d = array.get(cumDArr, i)
                    if math.abs(d - expectedD) < minDist
                        minDist := math.abs(d - expectedD)
                        bestI := i
                
                if bestI != -1
                    int intersectBar = array.get(idxArr, bestI)
                    float intersectPrice = array.get(priceArr, bestI)
                    label lbl = label.new(intersectBar, intersectPrice, text=lblPrefix + " Ray Int (" + str.tostring(k) + ")\n" + str.tostring(intersectPrice, "#.##"), color=color.new(lblColor, 50), textcolor=color.white, style=label.style_label_left, force_overlay=true)
                    array.push(intersectLabels, lbl)
                k := k + 1

    [newPoly, labelX, labelY, chartBarIdx, chartCandlePrice, thetaEnd, xCoords, yCoords]
