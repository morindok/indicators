get_shape_y_offsets(ry, theta, type) =>
    float max_y = -999999.0
    float min_y = 999999.0
    int n = 60
    float offset = 0.0
    if type == "Rectangle"
        n := 4
        offset := math.pi/4
    else if type == "Diamond"
        n := 4
        offset := -math.pi/2
    else if type == "Triangle"
        n := 3
        offset := -math.pi/2
    else if type == "Pentagon"
        n := 5
        offset := -math.pi/2
    else if type == "Hexagon"
        n := 6
        offset := 0.0
    else if type == "Octagon"
        n := 8
        offset := -math.pi/8

    if type == "Star"
        for i = 0 to 9
            r = i % 2 == 0 ? 1.0 : 0.45
            a = 2 * math.pi * i / 10 - math.pi / 2
            py = (ry * r) * math.sin(a + theta)
            max_y := math.max(max_y, py)
            min_y := math.min(min_y, py)
    else
        for i = 0 to n
            a = 2 * math.pi * i / n + offset
            py = ry * math.sin(a + theta)
            max_y := math.max(max_y, py)
            min_y := math.min(min_y, py)
    [max_y, min_y]

check_touch(c_p, sym_c_p, c_b, sym_c_b, ry_c, rx_c, theta, w1, h1, w2, h2) =>
    float dy = c_p - sym_c_p
    float dx = float(c_b - sym_c_b)
    float r_norm = rx_c > 0 ? ry_c / rx_c : ry_c
    float dy_norm = dy
    float dx_as_price = r_norm > 0 ? dx * r_norm : 0.0
    float dist = math.sqrt(dy_norm * dy_norm + dx_as_price * dx_as_price)
    float combined_r = ry_c * (h1 + h2) / 2 * 1.0
    bool touching = dist <= combined_r
    float exact = touching ? dist / math.max(combined_r, 0.000001) * ry_c * 2 : ry_c * 2
    [touching, exact]

rot_x(cx, cy, rx, ry, a, theta) =>
    raw_px = int(math.round(cx + rx * math.cos(a + theta)))
    math.min(bar_index + 499, math.max(bar_index - 4999, math.max(0, raw_px)))

rot_y(cx, cy, rx, ry, a, theta) =>
    cy + ry * math.sin(a + theta)

drawPoly(cx, cy, rx, ry, n, offset, theta, col) =>
    pts = array.new<chart.point>()
    for i = 0 to n
        a = 2 * math.pi * i / n + offset
        px = rot_x(cx, cy, rx, ry, a, theta)
        py = rot_y(cx, cy, rx, ry, a, theta)
        pts.push(chart.point.from_index(px, py))
    polyline.new(pts, closed=true, line_color=col, line_width=2)

drawStar(cx, cy, rx, ry, theta, col) =>
    pts = array.new<chart.point>()
    for i = 0 to 9
        r = i % 2 == 0 ? 1.0 : 0.45
        a = 2 * math.pi * i / 10 - math.pi / 2
        px = rot_x(cx, cy, rx * r, ry * r, a, theta)
        py = rot_y(cx, cy, rx * r, ry * r, a, theta)
        pts.push(chart.point.from_index(px, py))
    polyline.new(pts, closed=true, line_color=col, line_width=2)

drawSingleShape(cx, cy, rx, ry, theta, col) =>
    if shape_type == "Rectangle"
        drawPoly(float(cx), cy, float(rx), ry, 4, math.pi/4, theta, col)
    else if shape_type == "Triangle"
        drawPoly(float(cx), cy, float(rx), ry, 3, -math.pi/2, theta, col)
    else if shape_type == "Diamond"
        drawPoly(float(cx), cy, float(rx), ry, 4, -math.pi/2, theta, col)
    else if shape_type == "Pentagon"
        drawPoly(float(cx), cy, float(rx), ry, 5, -math.pi/2, theta, col)
    else if shape_type == "Hexagon"
        drawPoly(float(cx), cy, float(rx), ry, 6, 0.0, theta, col)
    else if shape_type == "Octagon"
        drawPoly(float(cx), cy, float(rx), ry, 8, -math.pi/8, theta, col)
    else if shape_type == "Star"
        drawStar(float(cx), cy, float(rx), ry, theta, col)
    else
        drawPoly(float(cx), cy, float(rx), ry, 60, 0.0, theta, col)
