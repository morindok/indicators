//@version=6
indicator("Bitmoon System{Morindok}", overlay=true)
n = input.int(100, title="Periods", minval=2)

// ??? Bitmoon Inputs ???
centralMirrorLevel = input.price(0, title="Central Mirror Level", confirm=true)
mirrorLevel1 = input.price(0, title="P1 Mirror Level", confirm=true)
selectedTime1 = input.time(0, title="P1 Reference Time", confirm=true)
mirrorLevel2 = input.price(0, title="P2 Mirror Level", confirm=true)
selectedTime2 = input.time(0, title="P2 Reference Time", confirm=true)

// ??? Dynamic Level Bounds ???
sc_p1 = input.price(0, "Sym Level 1", confirm=true)
sc_p2 = input.price(0, "Sym Level 2", confirm=true)
dynHi_mode = input.bool(true, "Dyn-Hi Dynamic", group="Dynamic Levels")
dynLo_mode = input.bool(true, "Dyn-Lo Dynamic", group="Dynamic Levels")

sc_pHi = math.max(sc_p1, sc_p2)
sc_pLo = math.min(sc_p1, sc_p2)

// ??? Bitmoon Core ???
cHigh = float(2.0 * centralMirrorLevel - low)
cLow  = float(2.0 * centralMirrorLevel - high)

bool u_dn = true, bool u_u0 = true, bool u_u1 = true, bool u_u2 = true, bool u_u3 = true, bool u_u4 = true
bool d_dn = true, bool d_u0 = true, bool d_u1 = true, bool d_u2 = true, bool d_u3 = true, bool d_u4 = true

for i = 1 to n
    u_dn := u_dn and (high[n-i] < high[n])
    u_u0 := u_u0 and (high[n+i] < high[n])
    u_u1 := u_u1 and (high[n+1] <= high[n] and high[n+i+1] < high[n])
    u_u2 := u_u2 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+i+2] < high[n])
    u_u3 := u_u3 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+i+3] < high[n])
    u_u4 := u_u4 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+3] <= high[n] and high[n+4] <= high[n] and high[n+i+4] < high[n])
    d_dn := d_dn and (low[n-i] > low[n])
    d_u0 := d_u0 and (low[n+i] > low[n])
    d_u1 := d_u1 and (low[n+1] >= low[n] and low[n+i+1] > low[n])
    d_u2 := d_u2 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+i+2] > low[n])
    d_u3 := d_u3 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+i+3] > low[n])
    d_u4 := d_u4 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+4] >= low[n] and low[n+i+4] > low[n])

flagUp = u_u0 or u_u1 or u_u2 or u_u3 or u_u4
flagDn = d_u0 or d_u1 or d_u2 or d_u3 or d_u4

upFractal1   = u_dn and flagUp and time[n] < selectedTime1
downFractal1 = d_dn and flagDn and time[n] < selectedTime1
upFractal2   = u_dn and flagUp and time[n] < selectedTime2
downFractal2 = d_dn and flagDn and time[n] < selectedTime2

var int refBarIndex1 = na
var int refBarIndex2 = na
if time <= selectedTime1
    refBarIndex1 := bar_index
if time <= selectedTime2
    refBarIndex2 := bar_index

var upFracIdx1 = array.new_int(0)
var dnFracIdx1 = array.new_int(0)
if upFractal1
    array.push(upFracIdx1, int(bar_index - n))
if downFractal1
    array.push(dnFracIdx1, int(bar_index - n))

var upFracIdx2 = array.new_int(0)
var dnFracIdx2 = array.new_int(0)
if upFractal2
    array.push(upFracIdx2, int(bar_index - n))
if downFractal2
    array.push(dnFracIdx2, int(bar_index - n))

bool p1Up = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(upFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p1Dn = not na(refBarIndex1) and bar_index > refBarIndex1 and array.indexof(dnFracIdx1, 2 * refBarIndex1 - bar_index) >= 0
bool p2Up = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(upFracIdx2, 2 * refBarIndex2 - bar_index) >= 0
bool p2Dn = not na(refBarIndex2) and bar_index > refBarIndex2 and array.indexof(dnFracIdx2, 2 * refBarIndex2 - bar_index) >= 0

// ??? Dynamic Levels ???
var float dynHi = sc_pHi
var float dynLo = sc_pLo

var int   hiState     = 0
var float hiFlatLevel = na
var int   loState     = 0
var float loFlatLevel = na

var float[] hLines   = array.new_float()
var bool[]  hLineHit = array.new_bool()
var line[]  hLineObj = array.new_line()

addHLine(lvl, col) =>
    if math.abs(dynHi - lvl) > syminfo.mintick and math.abs(dynLo - lvl) > syminfo.mintick
        ln = line.new(bar_index, lvl, bar_index + 1, lvl, extend=extend.right, color=col, width=1, style=line.style_solid)
        array.push(hLines,   lvl)
        array.push(hLineHit, false)
        array.push(hLineObj, ln)

purgeInvalidLines() =>
    if array.size(hLines) > 0
        i = array.size(hLines) - 1
        while i >= 0
            float lvl = array.get(hLines, i)
            if math.abs(dynHi - lvl) <= syminfo.mintick or math.abs(dynLo - lvl) <= syminfo.mintick
                line.delete(array.get(hLineObj, i))
                array.remove(hLines,   i)
                array.remove(hLineHit, i)
                array.remove(hLineObj, i)
            i -= 1

bool anySignal = p1Up or p1Dn or p2Up or p2Dn

if anySignal
    if dynHi_mode
        float newHi = high >= dynHi ? high : low
        if newHi > dynHi
            if hiState == 0 and not na(hiFlatLevel)
                addHLine(hiFlatLevel, color.aqua)
            hiState     := 1
            hiFlatLevel := na
        else if newHi < dynHi
            if hiState == 1
                hiState     := 0
                hiFlatLevel := dynHi
            else if hiState == -1
                if not na(hiFlatLevel)
                    addHLine(hiFlatLevel, color.aqua)
                hiState     := 0
                hiFlatLevel := dynHi
            else if hiState == 0
                hiState := -1
        else
            if hiState == 1 or hiState == -1
                hiState     := 0
                hiFlatLevel := dynHi
        dynHi := newHi
    else
        dynHi := sc_pHi

    if dynLo_mode
        float newLo = low <= dynLo ? low : high
        if newLo < dynLo
            if loState == 0 and not na(loFlatLevel)
                addHLine(loFlatLevel, color.yellow)
            loState     := -1
            loFlatLevel := na
        else if newLo > dynLo
            if loState == -1
                loState     := 0
                loFlatLevel := dynLo
            else if loState == 1
                if not na(loFlatLevel)
                    addHLine(loFlatLevel, color.yellow)
                loState     := 0
                loFlatLevel := dynLo
            else if loState == 0
                loState := 1
        else
            if loState == 1 or loState == -1
                loState     := 0
                loFlatLevel := dynLo
        dynLo := newLo
    else
        dynLo := sc_pLo

    purgeInvalidLines()

// ??? ÓíäÇáÏåí ???
bool buySignal  = false
bool sellSignal = false

if array.size(hLines) > 0
    for i = 0 to array.size(hLines) - 1
        if not array.get(hLineHit, i)
            float lvl = array.get(hLines, i)
            if high >= lvl and low <= lvl
                array.set(hLineHit, i, true)
                if close[1] > lvl
                    buySignal := true
                else if close[1] < lvl
                    sellSignal := true

plotshape(buySignal,  title="Buy",  location=location.belowbar, style=shape.triangleup,   color=color.green, size=size.small)
plotshape(sellSignal, title="Sell", location=location.abovebar, style=shape.triangledown, color=color.red,   size=size.small)

if buySignal
    alert("ÓíäÇá ÎÑíÏ: ÞíãÊ ÇÒ ÈÇáÇ ÈÇ ÎØ ÇÝÞí ÈÑÎæÑÏ ˜ÑÏ", alert.freq_once_per_bar)
if sellSignal
    alert("ÓíäÇá ÝÑæÔ: ÞíãÊ ÇÒ Çííä ÈÇ ÎØ ÇÝÞí ÈÑÎæÑÏ ˜ÑÏ", alert.freq_once_per_bar)
