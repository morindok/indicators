//@version=5
indicator("Symmetric Candles + Both Boxes (Fixed) {Morindok}", overlay=true, max_lines_count=500, max_boxes_count=50, max_labels_count=500)

t1 = input.time(0, "ÒãÇä ÔÑæÚ ÈÇÒå", confirm=true)
t2 = input.time(0, "ÒãÇä ÇíÇä ÈÇÒå", confirm=true)

p1 = input.price(0, "ÓØÍ Çæá (ÈÇáÇ/Çííä)", confirm=true)
p2 = input.price(0, "ÓØÍ Ïæã (ÈÇáÇ/Çííä)", confirm=true)

adj_opp     = input.bool(true,  "ÈÑ ÇÓÇÓ ÝÇÕáå ãÏá ? ÈÇ ãÏá ? ãÊÖÇÏ")
adj_between = input.bool(false, "ÝÇÕáå 2 ãÏá äÓÈÊ Èå åã (ãÓÊÞá - ˜á ÇÑÊ)")

reflect(src, pivot) => 2 * pivot - src

high_lvl = math.max(p1, p2)
low_lvl  = math.min(p1, p2)

in_range    = time >= t1 and time <= t2
ended_range = in_range[1] and not in_range

var int[]   a_idx = array.new_int()
var float[] a_o   = array.new_float()
var float[] a_h   = array.new_float()
var float[] a_l   = array.new_float()
var float[] a_c   = array.new_float()

var float lowest_in_range  = na
var float highest_in_range = na
var float highest_orig     = na
var float lowest_orig      = na
var float mid_lvl_indep    = na

if in_range
    array.push(a_idx, bar_index)
    array.push(a_o, open)
    array.push(a_h, high)
    array.push(a_l, low)
    array.push(a_c, close)

    mid_initial  = (high_lvl + low_lvl) / 2.0
    m1_H_init    = reflect(low,  mid_initial)
    m1_L_init    = reflect(high, mid_initial)
    m1_dL_init   = reflect(m1_H_init, low_lvl)
    m1_uH_init   = reflect(m1_L_init, high_lvl)

    lowest_in_range  := na(lowest_in_range)  ? m1_dL_init : math.min(lowest_in_range,  m1_dL_init)
    highest_in_range := na(highest_in_range) ? m1_uH_init : math.max(highest_in_range, m1_uH_init)
    highest_orig     := na(highest_orig)     ? high       : math.max(highest_orig, high)
    lowest_orig      := na(lowest_orig)      ? low        : math.min(lowest_orig,  low)
    mid_lvl_indep    := (highest_orig + lowest_orig) / 2.0

calc_m2(m1_val, m2_raw, m2_opp_raw) =>
    adj_opp ? m1_val + (m1_val - m2_opp_raw) : m2_raw

mid_lvl = mid_lvl_indep

m1_O = reflect(open,  mid_lvl)
m1_H = reflect(low,   mid_lvl)
m1_L = reflect(high,  mid_lvl)
m1_C = reflect(close, mid_lvl)

m1_uO = reflect(m1_O, high_lvl)
m1_uH = reflect(m1_L, high_lvl)
m1_uL = reflect(m1_H, high_lvl)
m1_uC = reflect(m1_C, high_lvl)

m1_dO = reflect(m1_O, low_lvl)
m1_dH = reflect(m1_L, low_lvl)
m1_dL = reflect(m1_H, low_lvl)
m1_dC = reflect(m1_C, low_lvl)

// plotcandle åÇ ÍÐÝ ÔÏä (ÝÞØ ÓíäÇá äãÇíÔ ÏÇÏå ãíÔå)

var line  signal_line      = na
var bool  signal_triggered = false

if ended_range
    signal_triggered := false

    final_mid_up = lowest_in_range
    final_mid_dn = highest_in_range

    float top_m1_top = 0.0
    float top_m1_bot = 0.0
    float top_m2_top = 0.0
    float top_m2_bot = 0.0
    float bot_m1_top = 0.0
    float bot_m1_bot = 0.0
    float bot_m2_top = 0.0
    float bot_m2_bot = 0.0

    size_arr = array.size(a_idx)

    if size_arr > 0
        for i = 0 to size_arr - 1
            idx = array.get(a_idx, i)
            o_v = array.get(a_o, i)
            h_v = array.get(a_h, i)
            l_v = array.get(a_l, i)
            c_v = array.get(a_c, i)

            m1_O_up    = reflect(o_v, final_mid_up)
            m1_H_up    = reflect(l_v, final_mid_up)
            m1_L_up    = reflect(h_v, final_mid_up)
            m1_C_up    = reflect(c_v, final_mid_up)
            m1_uO_main = reflect(m1_O_up, high_lvl)
            m1_uH_main = reflect(m1_L_up, high_lvl)
            m1_uL_main = reflect(m1_H_up, high_lvl)
            m1_uC_main = reflect(m1_C_up, high_lvl)

            m2_uO_tmp_main = reflect(o_v, high_lvl)
            m2_uH_tmp_main = reflect(l_v, high_lvl)
            m2_uL_tmp_main = reflect(h_v, high_lvl)
            m2_uC_tmp_main = reflect(c_v, high_lvl)
            m2_dO_tmp_main = reflect(o_v, low_lvl)
            m2_dH_tmp_main = reflect(l_v, low_lvl)
            m2_dL_tmp_main = reflect(h_v, low_lvl)
            m2_dC_tmp_main = reflect(c_v, low_lvl)

            m2_uO_raw_up = reflect(m2_uO_tmp_main, final_mid_up)
            m2_uH_raw_up = reflect(m2_uL_tmp_main, final_mid_up)
            m2_uL_raw_up = reflect(m2_uH_tmp_main, final_mid_up)
            m2_uC_raw_up = reflect(m2_uC_tmp_main, final_mid_up)
            m2_dO_raw_up = reflect(m2_dO_tmp_main, final_mid_up)
            m2_dH_raw_up = reflect(m2_dL_tmp_main, final_mid_up)
            m2_dL_raw_up = reflect(m2_dH_tmp_main, final_mid_up)
            m2_dC_raw_up = reflect(m2_dC_tmp_main, final_mid_up)

            m2_uO_main = calc_m2(m1_uO_main, m2_uO_raw_up, m2_dO_raw_up)
            m2_uH_main = calc_m2(m1_uH_main, m2_uH_raw_up, m2_dH_raw_up)
            m2_uL_main = calc_m2(m1_uL_main, m2_uL_raw_up, m2_dL_raw_up)
            m2_uC_main = calc_m2(m1_uC_main, m2_uC_raw_up, m2_dC_raw_up)

            // draw_candle åÇí up ÍÐÝ ÔÏä

            top_m1_top := i == 0 ? m1_uH_main : math.max(top_m1_top, m1_uH_main)
            top_m1_bot := i == 0 ? m1_uL_main : math.min(top_m1_bot, m1_uL_main)
            top_m2_top := i == 0 ? m2_uH_main : math.max(top_m2_top, m2_uH_main)
            top_m2_bot := i == 0 ? m2_uL_main : math.min(top_m2_bot, m2_uL_main)

            m1_O_dn    = reflect(o_v, final_mid_dn)
            m1_H_dn    = reflect(l_v, final_mid_dn)
            m1_L_dn    = reflect(h_v, final_mid_dn)
            m1_C_dn    = reflect(c_v, final_mid_dn)
            m1_dO_main = reflect(m1_O_dn, low_lvl)
            m1_dH_main = reflect(m1_L_dn, low_lvl)
            m1_dL_main = reflect(m1_H_dn, low_lvl)
            m1_dC_main = reflect(m1_C_dn, low_lvl)

            m2_dO_raw_dn = reflect(m2_dO_tmp_main, final_mid_dn)
            m2_dH_raw_dn = reflect(m2_dL_tmp_main, final_mid_dn)
            m2_dL_raw_dn = reflect(m2_dH_tmp_main, final_mid_dn)
            m2_dC_raw_dn = reflect(m2_dC_tmp_main, final_mid_dn)
            m2_uO_raw_dn = reflect(m2_uO_tmp_main, final_mid_dn)
            m2_uH_raw_dn = reflect(m2_uL_tmp_main, final_mid_dn)
            m2_uL_raw_dn = reflect(m2_uH_tmp_main, final_mid_dn)
            m2_uC_raw_dn = reflect(m2_uC_tmp_main, final_mid_dn)

            m2_dO_main = calc_m2(m1_dO_main, m2_dO_raw_dn, m2_uO_raw_dn)
            m2_dH_main = calc_m2(m1_dH_main, m2_dH_raw_dn, m2_uH_raw_dn)
            m2_dL_main = calc_m2(m1_dL_main, m2_dL_raw_dn, m2_uL_raw_dn)
            m2_dC_main = calc_m2(m1_dC_main, m2_dC_raw_dn, m2_uC_raw_dn)

            // draw_candle åÇí dn ÍÐÝ ÔÏä

            bot_m1_top := i == 0 ? m1_dH_main : math.max(bot_m1_top, m1_dH_main)
            bot_m1_bot := i == 0 ? m1_dL_main : math.min(bot_m1_bot, m1_dL_main)
            bot_m2_top := i == 0 ? m2_dH_main : math.max(bot_m2_top, m2_dH_main)
            bot_m2_bot := i == 0 ? m2_dL_main : math.min(bot_m2_bot, m2_dL_main)

        is_top_m1_highest = top_m1_top > top_m2_top
        final_top_box_top = is_top_m1_highest ? top_m1_top : top_m2_top
        final_top_box_bot = is_top_m1_highest ? top_m1_bot : top_m2_bot

        is_bot_m1_lowest  = bot_m1_bot < bot_m2_bot
        final_bot_box_top = is_bot_m1_lowest ? bot_m1_top : bot_m2_top
        final_bot_box_bot = is_bot_m1_lowest ? bot_m1_bot : bot_m2_bot

        // ÈÇ˜ÓåÇ æ ÎØæØ ÓÈÒ/ÞÑãÒ ÍÐÝ ÔÏä

        dy_green = final_top_box_bot - final_top_box_top
        dy_red   = final_bot_box_top - final_bot_box_bot
        diff     = dy_green - dy_red
        
        dx_ratio = diff != 0.0 ? (final_bot_box_bot - final_top_box_top) / diff : 0.0
        
        x_int_float = float(t1) + dx_ratio * float(t2 - t1)
        y_int       = final_top_box_top + dx_ratio * dy_green

        xi = int(math.round(x_int_float))
        // signal_line äÇãÑÆí æáí ÈÑÇí ãÍÇÓÈå ÓíäÇá äå ÏÇÔÊå ãíÔå
        signal_line := line.new(x1=xi, y1=y_int, x2=xi + 1, y2=y_int, xloc=xloc.bar_time, color=color.new(color.yellow, 100), width=1, style=line.style_dashed, extend=extend.right)

if not signal_triggered and not na(signal_line) and time > t2
    line_y = line.get_y1(signal_line)

    if m1_uL <= line_y and m1_uC > line_y
        label.new(x=bar_index, y=m1_uL, text="Reversal", color=color.red,   textcolor=color.white, style=label.style_label_up,   size=size.small)
        signal_triggered := true

    else if m1_dH >= line_y and m1_dC < line_y
        label.new(x=bar_index, y=m1_dH, text="Reversal",  color=color.green, textcolor=color.white, style=label.style_label_down, size=size.small)
        signal_triggered := true
