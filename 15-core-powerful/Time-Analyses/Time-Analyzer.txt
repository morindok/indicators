// „ €Ì—Â«

var int bars_in_range = 0

// ???????????????????????????????????????????????????????????????
// ‘„«—‘ ò‰œ·ùÂ«Ì »«“Â Å«ÌÂ
// ???????????????????????????????????????????????????????????????
in_range = time >= x1 and time <= x2

    if na(range_start_bar)
        range_start_bar := bar_index

// ???????????????????????????????????????????????????????????????
// —”„ Œÿ ⁄„ÊœÌ »« »—ç”»
// ???????????????????????????????????????????????????????????????
draw_time_line(int target_bar, color line_color, string label_text, string tooltip_text) =>
    if bar_index == target_bar and range_completed
        // —”„ Œÿ ⁄„ÊœÌ
        line.new(bar_index, low, bar_index, high, extend=extend.both, color=line_color, style=line.style_dashed, width=line_width)
        
        // —”„ »—ç”»
        if show_labels
            label_size = switch label_size_input
                "tiny" => size.tiny
                "small" => size.small
                "normal" => size.normal
                "large" => size.large
            
            label.new(bar_index, high, label_text, color=color.new(line_color, 80), textcolor=line_color, style=label.style_label_down, size=label_size,tooltip=tooltip_text)

// ???????????????????????????????????????????????????????????????

    if fib_618

        draw_time_line(target, color.new(color.orange, 0), "0.618 ?", "ç—ŒÂ ÿ·«ÌÌ - «Õ „«·  €ÌÌ— »«·«")
    
    if fib_786

    if fib_1000

    if fib_1618
        target = range_end_bar + math.round(base_length * 1.618)

    if mult_2x
