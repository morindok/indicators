//@version=5
indicator("Simulated Price Repaint (Polyline)", overlay=true, max_bars_back=5000)

lookbackPeriod   = input.int(20, "Lookback Period", minval=1)
attractionStr    = input.float(0.2, "Attraction Strength", step=0.1)
volatilityFactor = input.float(0.5, "Volatility Factor", step=0.1)

var float[] a_close = array.new_float()
var float[] a_high  = array.new_float()
var float[] a_low   = array.new_float()

// äíÇÒí Èå ÐÎíÑå ÒãÇä äíÓÊ¡ ÇÒ bar_index ÇÓÊÝÇÏå ãí˜äíã
max_points = 1000

// ãÏíÑíÊ ÂÑÇíååÇ
if barstate.isnew
    array.unshift(a_close, close)
    array.unshift(a_high, high)
    array.unshift(a_low, low)
else
    if array.size(a_close) > 0
        array.set(a_close, 0, close)
        array.set(a_high, 0, high)
        array.set(a_low, 0, low)

if array.size(a_close) > max_points
    array.pop(a_close)
    array.pop(a_high)
    array.pop(a_low)

var polyline p_sim = na

if barstate.islast
    int size = array.size(a_close)
    
    if size > 1
        float[] sim_price = array.new_float(size, na)
        
        for i = size - 1 to 0
            float weightedSum = 0.0
            float totalWeight = 0.0
            int limit = math.min(lookbackPeriod, size - 1 - i)
            
            for j = 0 to math.max(0, limit)
                if i + j < size
                    float weight = lookbackPeriod - j
                    weightedSum += array.get(a_close, i + j) * weight
                    totalWeight += weight
            
            float basePrice = totalWeight > 0 ? (weightedSum / totalWeight) : array.get(a_close, i)
            
            float maxVol = 0.0
            for j = 0 to math.max(0, limit)
                if i + j < size
                    float vol = array.get(a_high, i + j) - array.get(a_low, i + j)
                    if vol > maxVol
                        maxVol := vol
            
            float volatility = maxVol * 0.1
            float noise = math.random(-1.0, 1.0)
            
            if i == size - 1
                array.set(sim_price, i, array.get(a_close, i))
            else
                float prev_sim = array.get(sim_price, i + 1)
                float attraction = (basePrice - prev_sim) * attractionStr
                float current_sim = prev_sim + attraction + (volatility * noise * volatilityFactor)
                array.set(sim_price, i, current_sim)

        chart.point[] pts_sim = array.new<chart.point>()
        
        // ÇÓÊÝÇÏå ÇÒ bar_index ÈÑÇí ÑÓã ÏÞíÞ äÞÇØ
        for i = size - 1 to 0
            float val = array.get(sim_price, i)
            if not na(val)
                array.push(pts_sim, chart.point.from_index(bar_index - i, val))
                
        if not na(p_sim)
            polyline.delete(p_sim)
            
        if array.size(pts_sim) > 1
            p_sim := polyline.new(pts_sim, line_color=color.blue, line_width=2)
