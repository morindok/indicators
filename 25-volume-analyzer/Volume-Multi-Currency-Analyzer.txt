// ================== توابع ==================
f_ma(src, len, type) =>
    float out = na
    if type == "SMA"
        out := ta.sma(src, len)
    else if type == "EMA"
        out := ta.ema(src, len)
    else if type == "WMA"
        out := ta.wma(src, len)
    else if type == "VWMA"
        out := ta.vwma(src, len)
    else
        out := ta.rma(src, len)
    out

f_detrend(sym, len, type, norm) =>
    [c, m] = request.security(sym, timeframe.period, [close, f_ma(close, len, type)])
    diff = c - m
    result = norm == "درصد" ? diff / m * 100 : diff / ta.stdev(diff, len)
    result

f_lastPivotVal(len) =>
    var float pv = na
    ph = ta.pivothigh(high, len, len)
    pl = ta.pivotlow(low, len, len)
    if not na(ph)
        pv := ph
    if not na(pl)
        pv := pl
    pv

f_barsSincePivot(len) =>
    var int bsp = 0
    ph = ta.pivothigh(high, len, len)
    pl = ta.pivotlow(low, len, len)
    if not na(ph) or not na(pl)
        bsp := 0
    else
        bsp := bsp + 1
    bsp

f_flowMetrics(sym, tf, maLen, sLen, pLen, chLen) =>
    [c, v, m, atrVal, linNow, linPrev, pv, bsp] = request.security(sym, tf, [close, volume, f_ma(close, maLen, maType), ta.atr(14), ta.linreg(close, sLen, 0), ta.linreg(close, sLen, 1), f_lastPivotVal(pLen), f_barsSincePivot(pLen)])
    priceChangePct = (c - c[chLen]) / c[chLen] * 100
    volChangePct = (v - v[chLen]) / v[chLen] * 100
    dollarVol = c * v
    dollarVolChangePct = (dollarVol - dollarVol[chLen]) / dollarVol[chLen] * 100
    slope = linNow - linPrev
    angleDeg = math.atan(slope / atrVal) * 180 / math.pi
    ratioToMA = (c - m) / m * 100
    pivotHeightPct = na(pv) ? 0.0 : (c - pv) / pv * 100
    [priceChangePct, volChangePct, dollarVolChangePct, angleDeg, ratioToMA, pivotHeightPct, bsp, dollarVol]

f_buySellMetrics(sym, tf, len) =>
    [c, h, l, v] = request.security(sym, tf, [close, high, low, volume])
    rangeVal = h - l
    clv = rangeVal == 0 ? 0.0 : ((c - l) - (h - c)) / rangeVal
    mfv = clv * v
    cmf = ta.sma(mfv, len) / ta.sma(v, len)
    upVol = c > c[1] ? v : c < c[1] ? 0.0 : v * 0.5
    downVol = c < c[1] ? v : c > c[1] ? 0.0 : v * 0.5
    upVolAvg = ta.sma(upVol, len)
    downVolAvg = ta.sma(downVol, len)
    upDownRatio = (upVolAvg - downVolAvg) / (upVolAvg + downVolAvg)
    obvDelta = c > c[1] ? v : c < c[1] ? -v : 0.0
    obvLine = ta.cum(obvDelta)
    obvSlope = ta.linreg(obvLine, len, 0) - ta.linreg(obvLine, len, 1)
    obvAtr = ta.atr(len)
    obvSlopeNorm = obvAtr == 0 ? 0.0 : obvSlope / obvAtr
    [cmf, upDownRatio, clv, obvSlopeNorm]

f_validationMetrics(sym, tf, rLen, aLen, chLen) =>
    [c, rsiVal, atrNow] = request.security(sym, tf, [close, ta.rsi(close, rLen), ta.atr(aLen)])
    atrChangePct = (atrNow - atrNow[chLen]) / atrNow[chLen] * 100
    priceDir = c - c[chLen]
    [rsiVal, atrChangePct, priceDir]

f_cleanSym(s) =>
    out = s
    out := str.replace(out, "BINANCE:", "")
    out := str.replace(out, "BYBIT:", "")
    out := str.replace(out, "OKX:", "")
    out := str.replace(out, "BITGET:", "")
    out := str.replace(out, "USDT", "")
    out

f_zsafe(x, arr) =>
    m = array.avg(arr)
    sd = array.stdev(arr)
    result = sd == 0 ? 0.0 : (x - m) / sd
    result

f_voteTrend(ang, rat) =>
    v1 = ang > 0 ? 1 : ang < 0 ? -1 : 0
    v2 = rat > 0 ? 1 : rat < 0 ? -1 : 0
    total = v1 + v2
    result = total > 0 ? 1 : total < 0 ? -1 : 0
    result

f_voteMomentum(rsiVal, upTh, dnTh) =>
    result = rsiVal > upTh ? 1 : rsiVal < dnTh ? -1 : 0
    result

f_voteVolatility(atrChgPct, priceDir) =>
    result = atrChgPct > 0 and priceDir > 0 ? 1 : atrChgPct > 0 and priceDir < 0 ? -1 : 0
    result

f_voteFlow(bsRaw) =>
    result = bsRaw > 0 ? 1 : bsRaw < 0 ? -1 : 0
    result

f_voteRotation(scoreVal) =>
    result = scoreVal > 0 ? 1 : scoreVal < 0 ? -1 : 0
    result

// توابع انتقال‌یافته به فضای سراسری (Global Scope) برای رفع خطای سینتکس
f_voteText(v) =>
    v > 0 ? "▲" : v < 0 ? "▼" : "–"

f_signalLabel(buyC, sellC) =>
    agree = math.max(buyC, sellC)
    string sig = "خنثی"
    if agree >= strongAgreeCount and buyC > sellC
        sig := "خرید قطعی"
    else if agree >= strongAgreeCount and sellC > buyC
        sig := "فروش قطعی"
    else if agree >= probableAgreeCount and buyC > sellC
        sig := "خرید محتمل"
    else if agree >= probableAgreeCount and sellC > buyC
        sig := "فروش محتمل"
    sig

f_signalColor(buyC, sellC) =>
    agree = math.max(buyC, sellC)
    color sigColor = color.new(color.gray, 70)
    if agree >= strongAgreeCount and buyC > sellC
        sigColor := color.new(color.green, 60)
    else if agree >= strongAgreeCount and sellC > buyC
        sigColor := color.new(color.red, 60)
    else if agree >= probableAgreeCount and buyC > sellC
        sigColor := color.new(color.green, 82)
    else if agree >= probableAgreeCount and sellC > buyC
        sigColor := color.new(color.red, 82)
    sigColor
