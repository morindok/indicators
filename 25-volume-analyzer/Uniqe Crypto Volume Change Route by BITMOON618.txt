// =================== توابع محاسباتی ===================
f_flow(sym) =>
    [o, h, l, c0, c1, c2, v, vAvg] = request.security(sym, timeframe.period, [open, high, low, close, close[1], close[2], volume, ta.sma(volume, volLen)], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off)
    
    // محاسبه حجم پولی تقریبی به تتر (Volume * Close Price)
    usdt_vol = v * c0 
    
    pct = c1 != 0 ? (c0 - c1) / c1 * 100 : 0.0
    volRatio = vAvg > 0 ? v / vAvg : 1.0
    momentum = c1 != 0 ? (c1 - c2) / c2 * 100 : 0.0
    
    range_hl = h - l
    buy_vol = range_hl != 0 ? v * ((c0 - l) / range_hl) : (c0 >= o ? v : 0)
    sell_vol = range_hl != 0 ? v * ((h - c0) / range_hl) : (c0 < o ? v : 0)
    
    buyer_power = sell_vol > 0 ? (buy_vol / sell_vol) : (buy_vol > 0 ? 5.0 : 1.0)
    buyer_power := math.round(buyer_power * 10) / 10
    
    rawScore = (pct * 0.7 + momentum * 0.3) * volRatio * (buyer_power > 0 ? math.sqrt(buyer_power) : 1.0)
    
    [pct, volRatio, rawScore, buyer_power, usdt_vol]
