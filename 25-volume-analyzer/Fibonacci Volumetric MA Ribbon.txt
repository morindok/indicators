//@version=5
indicator("Fibonacci Volumetric MA Ribbon", overlay=true)

// ==========================================
// 1. تعریف دوره‌های فیبوناچی برای مووینگ‌ها
// ==========================================
len1 = 8
len2 = 13
len3 = 21
len4 = 34
len5 = 55
len6 = 89
len7 = 144

// محاسبه مووینگ اوریج‌های نمایی (EMA)
ma1 = ta.ema(close, len1)
ma2 = ta.ema(close, len2)
ma3 = ta.ema(close, len3)
ma4 = ta.ema(close, len4)
ma5 = ta.ema(close, len5)
ma6 = ta.ema(close, len6)
ma7 = ta.ema(close, len7)

// رسم مووینگ‌ها روی چارت با طیف رنگی
plot(ma1, color=#00FF00, title="EMA 8", linewidth=2)
plot(ma2, color=#32CD32, title="EMA 13")
plot(ma3, color=#ADFF2F, title="EMA 21")
plot(ma4, color=#FFD700, title="EMA 34")
plot(ma5, color=#FFA500, title="EMA 55")
plot(ma6, color=#FF4500, title="EMA 89")
plot(ma7, color=#FF0000, title="EMA 144", linewidth=2)

// ==========================================
// 2. محاسبه فضای خالی (گپ) ترکیبی با حجم
// ==========================================
// برای جلوگیری از اعداد نجومی، حجم را نرمالایز می‌کنیم (نسبت حجم فعلی به میانگین 50 دوره‌ای)
avg_vol = ta.sma(volume, 50)
norm_vol = volume / nz(avg_vol, 1) 

// محاسبه گپ حجمی بین هر دو مووینگ (Volume-Weighted Gap)
gap1_2 = (ma1 - ma2) * norm_vol
gap2_3 = (ma2 - ma3) * norm_vol
gap3_4 = (ma3 - ma4) * norm_vol
gap4_5 = (ma4 - ma5) * norm_vol
gap5_6 = (ma5 - ma6) * norm_vol
gap6_7 = (ma6 - ma7) * norm_vol

// محاسبه مجموع فضای خالی کل شبکه مووینگ‌ها
total_volumetric_gap = gap1_2 + gap2_3 + gap3_4 + gap4_5 + gap5_6 + gap6_7

// میانگین متحرکِ گپ کل برای تشخیص قدرت لحظه‌ای
signal_gap = ta.sma(total_volumetric_gap, 20)

// ==========================================
// 3. تشخیص روند و ساختار
// ==========================================
// بررسی هم‌راستایی ساختاری مووینگ‌ها
aligned_up = ma1 > ma2 and ma2 > ma3 and ma3 > ma4 and ma4 > ma5
aligned_dn = ma1 < ma2 and ma2 < ma3 and ma3 < ma4 and ma4 < ma5

var string trend_status = "رنج / بدون روند"
var color trend_color = color.gray

if aligned_up and total_volumetric_gap > 0
    // اگر هم‌راستا صعودی باشند و فضای خالی با حجم در حال گسترش باشد
    if total_volumetric_gap > signal_gap
        trend_status := "صعودی قوی (حجم بالا)"
        trend_color := color.new(color.green, 0)
    else
        trend_status := "صعودی ضعیف (کاهش حجم/گپ)"
        trend_color := color.new(color.lime, 30)

else if aligned_dn and total_volumetric_gap < 0
    // اگر هم‌راستا نزولی باشند
    if total_volumetric_gap < signal_gap
        trend_status := "نزولی قوی (حجم بالا)"
        trend_color := color.new(color.red, 0)
    else
        trend_status := "نزولی ضعیف (کاهش حجم/گپ)"
        trend_color := color.new(color.maroon, 30)
else
    trend_status := "رنج / درگیری مووینگ‌ها"
    trend_color := color.new(color.gray, 0)

// ==========================================
// 4. رسم جدول اطلاعات
// ==========================================
var table info_table = table.new(position.top_right, 2, 5, border_width=2, border_color=color.black)

if barstate.islast
    // هدر جدول
    table.cell(info_table, 0, 0, "شاخص بررسی", bgcolor=color.new(color.blue, 70), text_color=color.white, text_size=size.normal)
    table.cell(info_table, 1, 0, "وضعیت لحظه‌ای", bgcolor=color.new(color.blue, 70), text_color=color.white, text_size=size.normal)

    // ردیف روند کلی
    table.cell(info_table, 0, 1, "روند کلی سیستم", bgcolor=color.new(color.gray, 80), text_color=color.white)
    table.cell(info_table, 1, 1, trend_status, bgcolor=trend_color, text_color=color.white, text_size=size.large)

    // ردیف وضعیت مووینگ‌های کوتاه مدت (8 تا 34)
    short_term_gap = gap1_2 + gap2_3 + gap3_4
    short_color = short_term_gap > 0 ? color.green : color.red
    table.cell(info_table, 0, 2, "فشار حجمی کوتاه‌مدت", bgcolor=color.new(color.gray, 80), text_color=color.white)
    table.cell(info_table, 1, 2, short_term_gap > 0 ? "خریداران مسلط" : "فروشندگان مسلط", bgcolor=short_color, text_color=color.white)

    // ردیف وضعیت مووینگ‌های میان/بلند مدت (34 تا 144)
    long_term_gap = gap4_5 + gap5_6 + gap6_7
    long_color = long_term_gap > 0 ? color.green : color.red
    table.cell(info_table, 0, 3, "فشار حجمی بلندمدت", bgcolor=color.new(color.gray, 80), text_color=color.white)
    table.cell(info_table, 1, 3, long_term_gap > 0 ? "سرمایه ورودی" : "سرمایه خروجی", bgcolor=long_color, text_color=color.white)

    // امتیاز کل حجمی (نماینده عددی قدرت روند)
    table.cell(info_table, 0, 4, "امتیاز خالص گپ حجمی", bgcolor=color.new(color.gray, 80), text_color=color.white)
    table.cell(info_table, 1, 4, str.tostring(math.round(total_volumetric_gap, 2)), bgcolor=color.new(color.black, 20), text_color=total_volumetric_gap > 0 ? color.green : color.red)
