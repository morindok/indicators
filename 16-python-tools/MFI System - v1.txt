cached_data = {}
last_update = None


def fetch_crypto_data():
    """œ—Ì«›  œ«œÂùÂ«Ì 30 «—“ ò—ÌÅ Ê"""
    global cached_data, last_update
    data_dict = {}

    for symbol in CRYPTO_SYMBOLS:
        try:
            ticker = yf.Ticker(symbol)
            # œ—Ì«›  œ«œÂùÂ«Ì 5 œﬁÌﬁÂù«Ì »—«Ì 2 —Ê“ ê–‘ Â
            df = ticker.history(period='2d', interval='5m')
            if not df.empty:
                data_dict[symbol] = df
        except Exception as e:
            print(f"Error fetching {symbol}: {e}")
            continue

    cached_data = data_dict
    last_update = datetime.now()
    return data_dict


def calculate_money_flow(df):
    """„Õ«”»Â Ã—Ì«‰ ÅÊ·"""
    if df.empty:
        return 50  # „ﬁœ«— ÅÌ‘ù›—÷ œ— ’Ê—  Œ«·Ì »Êœ‰ œÌ «

    typical_price = (df['High'] + df['Low'] + df['Close']) / 3
    raw_money_flow = typical_price * df['Volume']

    positive_flow = raw_money_flow[typical_price > typical_price.shift(1)].sum()
    negative_flow = raw_money_flow[typical_price < typical_price.shift(1)].sum()

    if negative_flow == 0:
        return 100

    money_flow_ratio = positive_flow / negative_flow
    mfi = 100 - (100 / (1 + money_flow_ratio))

    return mfi


def calculate_volume_profile(df, bins=20):
    """„Õ«”»Â Å—Ê›«Ì· ÕÃ„"""
    if df.empty:
        return pd.DataFrame()

    price_range = np.linspace(df['Low'].min(), df['High'].max(), bins)
    volume_profile = []

    for i in range(len(price_range) - 1):
        low = price_range[i]
        high = price_range[i + 1]
        mid = (low + high) / 2

        mask = (df['Low'] <= high) & (df['High'] >= low)
        vol = df.loc[mask, 'Volume'].sum()

        volume_profile.append({
            'Price': mid,
            'Volume': vol
        })

    return pd.DataFrame(volume_profile)
