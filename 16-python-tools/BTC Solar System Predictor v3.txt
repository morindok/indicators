def extract_cosmic_forces(prices):
    x = np.arange(len(prices))
    poly_coeffs = np.polyfit(x, prices, deg=2)
    black_hole_gravity = np.polyval(poly_coeffs, x)
    detrended = prices - black_hole_gravity

    N = len(detrended)
    yf_fft = fft(detrended)
    xf_fft = fftfreq(N, 1)

    pos_idx = np.where(xf_fft > 0)[0]
    xf_pos = xf_fft[pos_idx]
    yf_pos = np.abs(yf_fft[pos_idx])
    phases = np.angle(yf_fft[pos_idx])

    top_indices = np.argsort(yf_pos)[-PLANET_COUNT:]

    planets = []
    max_price = np.max(prices)
    for i, idx in enumerate(top_indices):
        inclination = (i + 1) * (np.pi / (PLANET_COUNT + 1)) - (np.pi / 2)
        planets.append({
            'mass': yf_pos[idx] / (N / 2),
            'frequency': xf_pos[idx],
            'phase': phases[idx],
            'radius': max_price * (0.3 + (i + 1) * 0.15),
            'inclination': inclination
        })

    return poly_coeffs, planets


def calc_trajectory(poly_coeffs, planets, start_t, steps):
    t_array = np.arange(start_t, start_t + steps)
    trend = np.polyval(poly_coeffs, t_array)

    harmonics = np.zeros(steps)
    for planet in planets:
        harmonics += planet['mass'] * np.cos(2 * np.pi * planet['frequency'] * t_array + planet['phase'])

    return t_array, trend + harmonics


def calculate_statistics(actual, fitted):
    # „Õ«”»«  ò·«”Ìò Œÿ«
    mae = np.mean(np.abs(actual - fitted))
    rmse = np.sqrt(np.mean((actual - fitted) ** 2))

    ss_res = np.sum((actual - fitted) ** 2)
    ss_tot = np.sum((actual - np.mean(actual)) ** 2)
    r_squared = 1 - (ss_res / ss_tot) if ss_tot != 0 else 0
    correlation = np.corrcoef(actual, fitted)[0, 1]

    # „Õ«”»Â Win Rate (œﬁ  ÃÂ ùêÌ—Ì ”Ìê‰«·)
    actual_diff = np.diff(actual)
    fitted_diff = np.diff(fitted)

    actual_dir = np.sign(actual_diff)
    fitted_dir = np.sign(fitted_diff)

    # ›ﬁÿ “„«‰ùÂ«ÌÌ òÂ ﬁÌ„  Ê«ﬁ⁄«  €ÌÌ— ò—œÂ —« »—«Ì »—œ Õ”«» „Ìùò‰Ì„
    valid_moves = actual_dir != 0
    wins = np.sum((actual_dir[valid_moves] == fitted_dir[valid_moves]))
    total_valid = np.sum(valid_moves)

    win_rate = (wins / total_valid) * 100 if total_valid > 0 else 0

    return mae, rmse, r_squared, correlation, win_rate
