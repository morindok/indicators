def generate_large_spiral_coords(size, reverse_dir=False):
    coords = {}
    r, c = 0, 0
    coords[0] = (c, r)

    if reverse_dir:
        dr, dc = [0, 1, 0, -1], [-1, 0, 1, 0]
    else:
        dr, dc = [0, 1, 0, -1], [1, 0, -1, 0]

    dir_idx, step_length, current_step, direction_changes = 0, 1, 0, 0

    for i in range(1, size * size):
        r, c = r + dr[dir_idx], c + dc[dir_idx]
        coords[i] = (c, r)
        current_step += 1
        if current_step == step_length:
            current_step, dir_idx = 0, (dir_idx + 1) % 4
            direction_changes += 1
            if direction_changes % 2 == 0: step_length += 1
    return coords


def get_spiral_window(target_size, offset_x, offset_y, reverse_dir):
    large_size = target_size + 40
    if large_size % 2 == 0: large_size += 1

    matrix = [[-1] * large_size for _ in range(large_size)]
    r, c = large_size // 2, large_size // 2
    matrix[r][c] = 0

    if reverse_dir:
        dr, dc = [0, -1, 0, 1], [-1, 0, 1, 0]
    else:
        dr, dc = [0, -1, 0, 1], [1, 0, -1, 0]

    dir_idx, step_length, current_step, direction_changes = 0, 1, 0, 0
    for i in range(1, large_size * large_size):
        r, c = r + dr[dir_idx], c + dc[dir_idx]
        if 0 <= r < large_size and 0 <= c < large_size: matrix[r][c] = i
        current_step += 1
        if current_step == step_length:
            current_step, dir_idx = 0, (dir_idx + 1) % 4
            direction_changes += 1
            if direction_changes % 2 == 0: step_length += 1

    center = large_size // 2
    r_start = center - target_size // 2 + offset_y
    c_start = center - target_size // 2 + offset_x

    window = []
    for row in range(r_start, r_start + target_size):
        row_vals = []
        for col in range(c_start, c_start + target_size):
            row_vals.append(matrix[row][col])
        window.append(row_vals)
    return window

@app.callback(
    Output('offsets-store', 'data'),
    [Input('graph-left', 'relayoutData'),
     Input('graph-right', 'relayoutData')],
    [State('offsets-store', 'data')]
)
def update_offsets_from_drag(relayout_left, relayout_right, current_offsets):
    ctx = dash.callback_context
    if not ctx.triggered:
        return current_offsets

    trigger_id = ctx.triggered[0]['prop_id'].split('.')[0]
    relayout_data = relayout_left if trigger_id == 'graph-left' else relayout_right

    if relayout_data and 'shapes[0].x0' in relayout_data:
        x0 = relayout_data['shapes[0].x0']
        x1 = relayout_data['shapes[0].x1']
        y0 = relayout_data['shapes[0].y0']
        y1 = relayout_data['shapes[0].y1']

        center_x = round((x0 + x1) / 2)
        center_y = round((y0 + y1) / 2)

        if trigger_id == 'graph-left':
            current_offsets['asc_x'] = center_x
            current_offsets['asc_y'] = center_y
        else:
            current_offsets['desc_x'] = center_x
            current_offsets['desc_y'] = center_y

    return current_offsets


@app.callback(
    [Output('gann-table-container', 'children'),
     Output('graph-left', 'figure'),
     Output('graph-right', 'figure')],
    [Input('asc-center', 'value'), Input('asc-step', 'value'), Input('asc-factor', 'value'),
     Input('desc-center', 'value'), Input('desc-step', 'value'), Input('desc-factor', 'value'),
     Input('grid-size', 'value'),
     Input('offsets-store', 'data')]
)
def update_all(asc_center, asc_step, asc_factor, desc_center, desc_step, desc_factor, size, offsets):
    if None in [asc_center, asc_step, asc_factor, desc_center, desc_step, desc_factor, size]:
        return html.Div(), go.Figure(), go.Figure()

    if size % 2 == 0: size += 1

    asc_ox, asc_oy = offsets['asc_x'], offsets['asc_y']
    desc_ox, desc_oy = offsets['desc_x'], offsets['desc_y']

    large_size = size + 10

    def create_spiral_fig(reverse_dir, current_ox, current_oy, color):
        coords = generate_large_spiral_coords(large_size, reverse_dir)
        x_vals = [coords[i][0] for i in range(large_size * large_size)]
        y_vals = [coords[i][1] for i in range(large_size * large_size)]

        fig = go.Figure()
        fig.add_trace(go.Scatter(x=x_vals, y=y_vals, mode='lines+markers', marker=dict(size=4, color=color),
                                 line=dict(width=1, color=color)))

        half = size / 2
        fig.add_shape(
            type="rect",
            x0=current_ox - half, y0=current_oy - half,
            x1=current_ox + half, y1=current_oy + half,
            line=dict(color="blue", width=3, dash="dash"),
            fillcolor="rgba(0,0,255,0.1)",
            editable=True
        )

        fig.update_layout(
            xaxis=dict(showgrid=False, zeroline=True, scaleanchor="y", scaleratio=1),
            yaxis=dict(showgrid=False, zeroline=True),
            margin=dict(l=10, r=10, t=10, b=10),
            dragmode="pan"
        )
        return fig

    fig_left = create_spiral_fig(False, asc_ox, asc_oy, '#38a169')
    fig_right = create_spiral_fig(True, desc_ox, desc_oy, '#e53e3e')

    indices_left = get_spiral_window(size, asc_ox, -asc_oy, reverse_dir=False)
    indices_right = get_spiral_window(size, desc_ox, -desc_oy, reverse_dir=True)

    vals_left = [[asc_center + (indices_left[r][c] * asc_step * asc_factor) for c in range(size)] for r in range(size)]
    vals_right = [[desc_center - (indices_right[r][c] * desc_step * desc_factor) for c in range(size)] for r in
                  range(size)]

    # ÅÌœ« ò—œ‰ ‰ﬁÿÂ „„«” (Global Minimum Difference)
    min_global_diff = float('inf')
    tangent_cells = set()

    for r in range(size):
        for c in range(size):
            diff = abs(vals_left[r][c] - vals_right[r][c])
            if diff < min_global_diff:
                min_global_diff = diff
                tangent_cells = {(r, c)}
            elif diff == min_global_diff:
                tangent_cells.add((r, c))

    # —‰œ— ò—œ‰ ÃœÊ·
    rows = []
    for r in range(size):
        cells = []
        for c in range(size):
            val = vals_left[r][c]
            is_tangent = (r, c) in tangent_cells

            # «” «Ì· ”·Ê·ùÂ«Ì „„«” ‘œÂ (‰ﬁ«ÿ »«“ê‘ Ì ê‰)
            bg_color = '#ef4444' if is_tangent else '#ffffff'  # ﬁ—„“ »—«Ì ‰ﬁÿÂ »«“ê‘ Ì
            text_color = '#ffffff' if is_tangent else '#000000'
            font_weight = 'bold' if is_tangent else 'normal'

            cells.append(html.Td(f"{val:g}",
                                 style={'border': '1px solid #e2e8f0', 'padding': '10px', 'textAlign': 'center',
                                        'backgroundColor': bg_color, 'color': text_color, 'fontWeight': font_weight}))

        # Ãœ«ò‰‰œÂ Ê”ÿ
        cells.append(html.Td("", style={'width': '10px', 'backgroundColor': '#cbd5e0'}))

        for c in range(size):
            val = vals_right[r][c]
            is_tangent = (r, c) in tangent_cells

            bg_color = '#ef4444' if is_tangent else '#ffffff'
            text_color = '#ffffff' if is_tangent else '#000000'
            font_weight = 'bold' if is_tangent else 'normal'

            cells.append(html.Td(f"{val:g}",
                                 style={'border': '1px solid #e2e8f0', 'padding': '10px', 'textAlign': 'center',
                                        'backgroundColor': bg_color, 'color': text_color, 'fontWeight': font_weight}))

        rows.append(html.Tr(cells))

    table = html.Table(rows, style={'borderCollapse': 'collapse', 'backgroundColor': 'white'})

    return table, fig_left, fig_right
