def generate_spiral_indices(size, reverse_dir=False):
    matrix = [[-1] * size for _ in range(size)]
    r, c = size // 2, size // 2
    matrix[r][c] = 0

    if reverse_dir:
        dr, dc = [0, -1, 0, 1], [-1, 0, 1, 0]
    else:
        dr, dc = [0, -1, 0, 1], [1, 0, -1, 0]

    dir_idx, step_length, current_step, direction_changes = 0, 1, 0, 0

    for i in range(1, size * size):
        r, c = r + dr[dir_idx], c + dc[dir_idx]
        if 0 <= r < size and 0 <= c < size: matrix[r][c] = i
        current_step += 1
        if current_step == step_length:
            current_step, dir_idx = 0, (dir_idx + 1) % 4
            direction_changes += 1
            if direction_changes % 2 == 0: step_length += 1
    return matrix

def update_table(asc_center, asc_step, desc_center, desc_step, size):
    if None in [asc_center, asc_step, desc_center, desc_step, size]:
        return html.Div("áØÝÇð ÊãÇãí ãÞÇÏíÑ ÑÇ æÇÑÏ ˜äíÏ.", style={'color': 'gray'})

    if size % 2 == 0: size += 1

    indices_left = generate_spiral_indices(size, reverse_dir=False)
    indices_right = generate_spiral_indices(size, reverse_dir=True)

    vals_left = [[asc_center + (indices_left[r][c] * asc_step) for c in range(size)] for r in range(size)]
    vals_right = [[desc_center - (indices_right[r][c] * desc_step) for c in range(size)] for r in range(size)]

    closest_row_for_col = []
    for c in range(size):
        min_diff = float('inf')
        best_r = -1
        for r in range(size):
            diff = abs(vals_left[r][c] - vals_right[r][c])
            if diff < min_diff:
                min_diff = diff
                best_r = r
        closest_row_for_col.append(best_r)

    rows = []
    for r in range(size):
        cells = []

        # ÇÓíÑÇá 
        for c in range(size):
            val = vals_left[r][c]
            is_center = (indices_left[r][c] == 0)
            is_closest = (r == closest_row_for_col[c])

            if is_closest:
                bg_color = '#fef08a'
                border_color = '#ca8a04'
            else:
                bg_color = '#e6fffa' if is_center else '#ffffff'
                border_color = '#38a169' if is_center else '#e2e8f0'

            cell_content = html.Div(f"{val:g}",
                                    style={'color': '#2f855a' if not is_closest else '#854d0e', 'fontWeight': 'bold',
                                           'fontSize': '14px'})
            cells.append(html.Td(cell_content,
                                 style={'border': f'2px solid {border_color}', 'padding': '10px', 'textAlign': 'center',
                                        'backgroundColor': bg_color, 'minWidth': '60px', 'borderRadius': '4px'}))

        # ããÇÓ
        cells.append(html.Td("", style={'width': '10px', 'backgroundColor': '#cbd5e0', 'borderRadius': '2px'}))

        # ÇÓíÑÇá ÑÇÓÊ
        for c in range(size):
            val = vals_right[r][c]
            is_center = (indices_right[r][c] == 0)
            is_closest = (r == closest_row_for_col[c])

            if is_closest:
                bg_color = '#fef08a'
                border_color = '#ca8a04'
            else:
                bg_color = '#fff5f5' if is_center else '#ffffff'
                border_color = '#e53e3e' if is_center else '#e2e8f0'

            cell_content = html.Div(f"{val:g}",
                                    style={'color': '#c53030' if not is_closest else '#854d0e', 'fontWeight': 'bold',
                                           'fontSize': '14px'})
            cells.append(html.Td(cell_content,
                                 style={'border': f'2px solid {border_color}', 'padding': '10px', 'textAlign': 'center',
                                        'backgroundColor': bg_color, 'minWidth': '60px', 'borderRadius': '4px'}))

        rows.append(html.Tr(cells))

    # --- ÈÎÔ ãíÇäíä ÓÊæäåÇ ---
    avg_cells = []
    left_col_avgs = []
    right_col_avgs = []

    # ãíÇäíä ÓÊæäåÇí ÌÏæá 
    for c in range(size):
        col_vals = [vals_left[r][c] for r in range(size)]
        first_digits = [str(v).lstrip('-')[0] for v in col_vals]
        most_common_digit = Counter(first_digits).most_common(1)[0][0]
        filtered_vals = [v for v, fd in zip(col_vals, first_digits) if fd == most_common_digit]

        col_avg = sum(filtered_vals) / len(filtered_vals) if filtered_vals else 0
        left_col_avgs.append(col_avg)

    avg_cells.append(html.Td("", style={'width': '10px'}))

    # ãíÇäíä ÓÊæäåÇí ÌÏæá ÑÇÓÊ

        col_vals = [vals_right[r][c] for r in range(size)]
        first_digits = [str(v).lstrip('-')[0] for v in col_vals]
        most_common_digit = Counter(first_digits).most_common(1)[0][0]
        filtered_vals = [v for v, fd in zip(col_vals, first_digits) if fd == most_common_digit]
