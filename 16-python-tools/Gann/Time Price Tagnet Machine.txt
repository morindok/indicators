    children=[
        html.H1("„«‘Ì‰ “„«‰ ê«‰:  ﬁ«ÿ⁄ „ÿ·ﬁ ﬁÌ„  Ê “„«‰"),

        html.Div([
            html.Label("ﬁÌ„  Å«ÌÂ »Ì ùòÊÌ‰ (Base Price): "),

@app.callback(
    [Output('time-display', 'children'),
     Output('resistance-display', 'children'),
     Output('support-display', 'children')],
    [Input('clock-interval', 'n_intervals'),
     Input('base-price-input', 'value')]
)
def update_absolute_points(n, base_price):
    if not base_price or base_price <= 0:
        return "Invalid Price", "-", "-"

    # œ—Ì«›  “„«‰ œﬁÌﬁ ›⁄·Ì
    now = datetime.datetime.now()

    # „Õ«”»Â “„«‰ »Â ⁄‰Ê«‰ Ìò ò”— ÅÌÊ” Â «“ —Ê“ (œﬁ   « „Ì·ÌùÀ«‰ÌÂ »—«Ì Õ—ò  ‰—„)
    midnight = now.replace(hour=0, minute=0, second=0, microsecond=0)
    seconds_passed = (now - midnight).total_seconds()

    # 86400 À«‰ÌÂ œ— —Ê“ œ«—Ì„. ‰ê«‘  »Â ÷—Ì» ê«‰ (Õœ«òÀ— 2 »—«Ì 360 œ—ÃÂ)
    # F_t = (seconds / 86400) * 2
    f_t = (seconds_passed / 86400.0) * 2.0

    # „Õ«”»Â —Ì«÷Ì ê«‰
    sqrt_p = math.sqrt(base_price)

    resistance = (sqrt_p + f_t) ** 2
    support = (sqrt_p - f_t) ** 2

    time_str = f"“„«‰ ›⁄·Ì: {now.strftime('%H:%M:%S')} | ÷—Ì» “„«‰Ì ê«‰: {f_t:.4f}"
    res_str = f"„ﬁ«Ê„  „ÿ·ﬁ: {resistance:.4f} $"
    sup_str = f"Õ„«Ì  „ÿ·ﬁ: {support:.4f} $"

    return time_str, res_str, sup_str
