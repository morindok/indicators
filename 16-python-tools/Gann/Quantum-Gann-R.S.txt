def calculate_quantum_orbitals(peak, trough):
    """„Õ«”»Â „œ«—Â« »« œﬁ  90 œ—ÃÂ (Cardinal Cross) »œÊ‰ ‰ÊÌ“ ŒÿÌ"""
    targets = []

    sqrt_peak = math.sqrt(peak)
    sqrt_trough = math.sqrt(trough)

    # 1. Fibonacci Macro Fractals
    diff = peak - trough
    fib_levels = {
        0.5: "Fibo 0.5", 0.618: "Fibo 0.618", 0.786: "Fibo 0.786",
        1.272: "Fibo 1.272", 1.618: "Fibo 1.618", 2.0: "Fibo 2.0",
        2.618: "Fibo 2.618", 4.236: "Fibo 4.236"
    }
    macro_fibs = []
    for fib, name in fib_levels.items():
        macro_fibs.append({"price": peak - (diff * fib), "name": f"Macro {name} (Down)"})
        macro_fibs.append({"price": trough + (diff * fib), "name": f"Macro {name} (Up)"})

    # 2. SQ9 Quantum Orbits (90∞ Increments -> Factor 0.5)
    # factor 0.5 = 90 deg | 1.0 = 180 deg | 2.0 = 360 deg

    for step in range(1, 25):  # „Õ«”»Â  « 6 ç—Œ‘ ò«„· (24 ê«„ 90 œ—ÃÂ «Ì)
        factor = step * 0.5
        degrees = int(factor * 180)

        # === TARGETS DOWN (From Peak) ===
        price_dn = (sqrt_peak - factor) ** 2
        if price_dn > 0:
            resonance, fib_name = check_resonance(price_dn, macro_fibs)
            orbit_type, stars = classify_orbit(degrees)

            targets.append({
                "Target Price": price_dn,
                "Degrees": f"-{degrees}∞",
                "Geometry": orbit_type,
                "Resonance": fib_name if resonance else "-",
                "Power Level": stars
            })

        # === TARGETS UP (From Trough) ===
        price_up = (sqrt_trough + factor) ** 2
        resonance_up, fib_name_up = check_resonance(price_up, macro_fibs)
        orbit_type_up, stars_up = classify_orbit(degrees)

        targets.append({
            "Target Price": price_up,
            "Degrees": f"+{degrees}∞",
            "Geometry": orbit_type_up,
            "Resonance": fib_name_up if resonance_up else "-",
            "Power Level": stars_up
        })

    return targets


def classify_orbit(degrees):
    """ÿ»ﬁÂù»‰œÌ «Â„Ì  “Ê«Ì« œ— „—»⁄ 9 ê«‰"""
    if degrees == 1440:
        return "Singularity Orbit", "?? MACRO DROP (1440∞)"
    elif degrees % 360 == 0:
        return "Full Cycle Orbit", "???"
    elif degrees % 180 == 0:
        return "Half Cycle Pivot", "??"
    elif degrees % 90 == 0:
        return "Cardinal Cross (90∞)", "? (Intraday/Swing)"
    return "Minor", "-"


def check_resonance(sq9_price, macro_fibs, tol_pct=0.002):
    for fib in macro_fibs:
        if abs(sq9_price - fib["price"]) / fib["price"] <= tol_pct:
            return True, f"?? Res: {fib['name']}"
    return False, None
