class LiveDataEngine:
    def __init__(self, maxlen=500):
        self.lock = threading.Lock()
        self.price = None
        self.prev_price = None
        self.volume_buy = 0.0
        self.volume_sell = 0.0
        self.price_hist = deque(maxlen=maxlen)
        self.time_hist = deque(maxlen=maxlen)
        self.vol_hist = deque(maxlen=maxlen)
        self.delta_hist = deque(maxlen=60)
        self.kline_open = None
        self.kline_high = None
        self.kline_low = None
        self.kline_close = None
        self.kline_vol = 0.0
        self.kline_trades = 0
        self.connected = False
        self.error_msg = ""
        self.tick_count = 0
        self.source = "œ— Õ«· Ã” ÃÊ..."
        self._stop = False
        self._last_kline_ts = time.time()

    def _prefill_history(self):
        """œ—Ì«›   «—ÌŒçÂ ê–‘ Â «“ CoinGecko »—«Ì Ã·ÊêÌ—Ì «“ Œÿ ’«› ‘œ‰ ‰„Êœ«— œ— ·ÕŸÂ «Ã—«"""
        try:
            url = "https://api.coingecko.com/api/v3/coins/bitcoin/market_chart?vs_currency=usd&days=1"
            res = requests.get(url, timeout=10)

            if res.status_code == 200:
                data = res.json()
                prices = data.get("prices", [])
                volumes = data.get("total_volumes", [])

                prices = prices[-120:]
                volumes = volumes[-120:]

                with self.lock:
                    for i in range(len(prices)):
                        ts = prices[i][0] / 1000.0
                        c = float(prices[i][1])
                        v = float(volumes[i][1]) if i < len(volumes) else 0.0

                        self.time_hist.append(ts)
                        self.price_hist.append(c)
                        self.vol_hist.append(v)

                        is_green = True
                        if i > 0:
                            is_green = c >= float(prices[i - 1][1])

                        self.delta_hist.append(v * (0.001 if is_green else -0.001))

                    if prices:
                        self.price = float(prices[-1][1])
        except Exception as e:
            print(f"CoinGecko Prefill Error: {e}")
            pass

    def start(self):
        self._stop = False
        t = threading.Thread(target=self._run_loop, daemon=True)
        t.start()

    def stop(self):
        self._stop = True

    def _run_loop(self):
        self._prefill_history()
        session = requests.Session()
        use_binance = True

        while not self._stop:
            try:
                now = time.time()
                p = None

                if use_binance:
                    res = session.get("https://api.binance.com/api/v3/ticker/24hr?symbol=BTCUSDT", timeout=3)
                    if res.status_code == 200:
                        data = res.json()
                        p = float(data["lastPrice"])
                        self.source = "Binance"
                    else:
                        use_binance = False

                if not use_binance:
                    res = session.get("https://api.coingecko.com/api/v3/simple/price?ids=bitcoin&vs_currencies=usd",
                                      timeout=5)
                    if res.status_code == 200:
                        data = res.json()
                        p = float(data["bitcoin"]["usd"])
                        self.source = "CoinGecko"
                    else:
                        raise Exception(f"HTTP {res.status_code}")

                if p is not None:
                    with self.lock:
                        self.connected = True
                        self.error_msg = ""

                        if self.price != p:
                            self.prev_price = self.price or p
                            self.price = p
                            self.tick_count += 1

                            self.price_hist.append(p)
                            self.time_hist.append(now)

                            diff = p - self.prev_price
                            pseudo_vol = abs(diff) * 1.5
                            self.vol_hist.append(pseudo_vol)

                            if diff < 0:
                                self.volume_sell += pseudo_vol
                            else:
                                self.volume_buy += pseudo_vol

                            self.delta_hist.append(self.volume_buy - self.volume_sell)
                            self.volume_buy = 0.0
                            self.volume_sell = 0.0

                            if now - self._last_kline_ts >= 60 or self.kline_open is None:
                                self.kline_open = p
                                self.kline_high = p
                                self.kline_low = p
                                self.kline_close = p
                                self.kline_vol = 0
                                self.kline_trades = 0
                                self._last_kline_ts = now
                            else:
                                self.kline_high = max(self.kline_high, p)
                                self.kline_low = min(self.kline_low, p)
                                self.kline_close = p
                                self.kline_vol += pseudo_vol
                                self.kline_trades += 1

            except Exception as e:
                with self.lock:
                    self.connected = False
                    self.error_msg = str(e)[:30]
                    use_binance = False

            time.sleep(2.5 if use_binance else 10)

def gann_confluence(price, base, step, n_rings=6, tol=0.9):
    score = 0
    hits = []
    if base <= 0 or step <= 0: return 0, []
    for ring in range(1, n_rings + 1):
        rw = 6 / (ring + 1)
        for ang, info in GANN_ANGLES.items():
            gp = base + (ang / 360) * step * ring
            dp = abs(gp - price) / price * 100
            if dp <= tol:
                prox = 1 - dp / tol
                s = info["weight"] * rw * prox
                score += s
                hits.append({"angle": ang, "ring": ring, "label": info["label"],
                             "gann_p": gp, "diff_pct": dp, "score": s, "type": info["type"]})
    return score, hits


def fibo_confluence(price, sw_lo, sw_hi, tol=0.9):
    score = 0
    hits = []
    if sw_lo <= 0 or sw_hi <= sw_lo: return 0, []
    rng = sw_hi - sw_lo
    for fib in FIBO_LEVELS:
        ret_p = sw_hi - rng * fib
        dp = abs(ret_p - price) / price * 100
        if dp <= tol:
            w = FIBO_WEIGHTS.get(fib, 1)
            prox = 1 - dp / tol
            score += w * prox
            hits.append({"level": fib, "price": ret_p, "diff": dp})
    return score, hits


def signal_label(total):
    if total >= 15: return "STRONG REVERSAL", RED_ZONE
    if total >= 8:  return "LIKELY ZONE", ORANGE
    if total >= 4:  return "WATCH", BLUE_ACC
    return "", TEXT_DIM


def scan_zones(base, step, sw_lo, sw_hi, p_min, p_max):
    zones = []
    sz = max(50, int((p_max - p_min) / 600))
    for price in range(int(p_min), int(p_max), sz):
        gs, gh = gann_confluence(price, base, step)
        fs, fh = fibo_confluence(price, sw_lo, sw_hi)
        total = gs + fs * 1.5
        if total >= 3:
            zones.append({"price": price, "gann": gs, "fibo": fs,
                          "total": total, "gann_hits": gh, "fib_hits": fh})
    zones.sort(key=lambda x: x["price"])
    merged = []
    i = 0
    while i < len(zones):
        cluster = [zones[i]]
        j = i + 1
        while j < len(zones) and zones[j]["price"] - zones[j - 1]["price"] < 600:
            cluster.append(zones[j])
            j += 1
        best = max(cluster, key=lambda x: x["total"])
        merged.append(best)
        i = j
    return sorted(merged, key=lambda x: -x["total"])


def price_to_angle(price, base, step):
    if base <= 0 or step <= 0: return 0
    return ((price - base) / step * 360) % 360


def momentum_signal(prices, window=20):
    if len(prices) < window: return 0, "??"
    arr = list(prices)[-window:]
    mom = (arr[-1] - arr[0]) / arr[0] * 100
    accel = arr[-1] - 2 * arr[-window // 2] + arr[0]
    return mom, accel


def nearest_gann(price, base, step, n_rings=6):
    best = None
    best_d = 99999
    if base <= 0 or step <= 0: return None, None, None
    for ring in range(1, n_rings + 1):
        for ang, info in GANN_ANGLES.items():
            gp = base + (ang / 360) * step * ring
            d = abs(gp - price)
            if d < best_d:
                best_d = d
                best = gp
                best_info = {"angle": ang, "ring": ring, "type": info["type"], "label": info["label"]}
    return best, best_d, best_info


def compute_delta_score(delta_hist):
    if not delta_hist: return 0, ""
    dl = list(delta_hist)[-10:]
    cumulative = sum(dl)
    if cumulative > 0.5:
        return cumulative, "BUY PRESSURE"
    elif cumulative < -0.5:
        return cumulative, "SELL PRESSURE"
    return cumulative, "NEUTRAL"
