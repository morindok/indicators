def gann_confluence(price, base, step, n_rings=6, tol=0.9):
    score = 0; hits = []
    if base <= 0 or step <= 0: return 0, []
    for ring in range(1, n_rings+1):
        rw = 6 / (ring + 1)
        for ang, info in GANN_ANGLES.items():
            gp = base + (ang / 360) * step * ring
            dp = abs(gp - price) / price * 100
            if dp <= tol:
                prox = 1 - dp / tol
                s = info["weight"] * rw * prox
                score += s
                hits.append({"angle": ang, "ring": ring, "label": info["label"],
                              "gann_p": gp, "diff_pct": dp, "score": s,
                              "type": info["type"]})
    return score, hits

def fibo_confluence(price, swing_lo, swing_hi, tol=0.9):
    score = 0; hits = []
    if swing_lo <= 0 or swing_hi <= swing_lo: return 0, []
    rng = swing_hi - swing_lo
    for fib in FIBO_LEVELS:
        ret_p = swing_hi - rng * fib
        dp = abs(ret_p - price) / price * 100
        if dp <= tol:
            w = FIBO_WEIGHTS.get(fib, 1)
            prox = 1 - dp / tol
            score += w * prox
            hits.append({"level": fib, "price": ret_p, "diff": dp})
    return score, hits

def signal_label(total):
    if total >= 15: return ("STRONG REVERSAL", RED_ZONE)
    if total >= 8:  return ("LIKELY ZONE",     "#ff9900")
    if total >= 4:  return ("WATCH",           BLUE_ACC)
    return ("",     TEXT_DIM)

def scan_reversal_zones(base, step, swing_lo, swing_hi, price_min, price_max):
    zones = []
    step_sz = max(100, int((price_max - price_min) / 400))
    for price in range(int(price_min), int(price_max), step_sz):
        gs, gh = gann_confluence(price, base, step)
        fs, fh = fibo_confluence(price, swing_lo, swing_hi)
        total = gs + fs * 1.5
        if total >= 3:
            zones.append({"price": price, "gann": gs, "fibo": fs,
                           "total": total, "gann_hits": gh, "fib_hits": fh})
    # «œ€«„ clusterùÂ«Ì ‰“œÌò
    zones.sort(key=lambda x: x["price"])
    merged = []
    i = 0
    while i < len(zones):
        cluster = [zones[i]]
        j = i + 1
        while j < len(zones) and zones[j]["price"] - zones[j-1]["price"] < 800:
            cluster.append(zones[j]); j += 1
        best = max(cluster, key=lambda x: x["total"])
        merged.append(best); i = j
    return sorted(merged, key=lambda x: -x["total"])

def price_to_angle(price, base, step):
    if base <= 0 or step <= 0: return 0
    return ((price - base) / step * 360) % 360

def get_wheel_prices(base, step, rotation, n_rings=3):
    results = []
    for ring in range(1, n_rings+1):
        for ang, info in GANN_ANGLES.items():
            eff = (ang - rotation) % 360
            price = base + (eff / 360) * step * ring
            results.append({"ring": ring, "angle": ang, "price": price,
                             "label": info["label"], "type": info["type"],
                             "desc": info["desc"]})
    return results

def build_wheel(base, step, rotation, current=None, swing_lo=None, swing_hi=None):
    fig = go.Figure()
    R_INNER = 0.15
    R_RINGS = [0.35, 0.58, 0.80]
    R_LABEL = 0.93
    theta = np.linspace(0, 2*np.pi, 300)

    for r in R_RINGS:
        fig.add_trace(go.Scatter(x=r*np.cos(theta), y=r*np.sin(theta),
            mode="lines", line=dict(color=RING_COL, width=1.5),
            hoverinfo="skip", showlegend=False))

    fig.add_trace(go.Scatter(x=R_INNER*np.cos(theta), y=R_INNER*np.sin(theta),
        mode="lines", line=dict(color=GOLD_DIM, width=2),
        fill="toself", fillcolor="rgba(201,168,76,0.08)",
        hoverinfo="skip", showlegend=False))

    for ang, info in GANN_ANGLES.items():
        rad = math.radians(90 - (ang + rotation))
        c, s = math.cos(rad), math.sin(rad)
        col = ANGLE_COLORS[info["type"]]
        w   = 2 if info["type"]=="major" else (1.5 if info["type"]=="mid" else 0.7)
        dk  = "dot" if info["type"]=="minor" else "solid"
        fig.add_trace(go.Scatter(x=[R_INNER*c, R_RINGS[-1]*c],
            y=[R_INNER*s, R_RINGS[-1]*s], mode="lines",
            line=dict(color=col, width=w, dash=dk),
            hoverinfo="skip", showlegend=False))
        fig.add_trace(go.Scatter(x=[R_LABEL*c], y=[R_LABEL*s], mode="text",
            text=[f"{ang}∞"],
            textfont=dict(color=col, size=8 if info["type"]=="minor" else 10,
                          family="monospace"),
            hoverinfo="skip", showlegend=False))

    for item in get_wheel_prices(base, step, rotation):
        r   = R_RINGS[item["ring"]-1]
        rad = math.radians(90 - (item["angle"] + rotation))
        c, s = math.cos(rad), math.sin(rad)
        col = ANGLE_COLORS[item["type"]]
        sz  = 10 if item["type"]=="major" else (8 if item["type"]=="mid" else 6)
        # »——”Ì confluence
        gs, _ = gann_confluence(item["price"], base, step)
        fs, _ = fibo_confluence(item["price"], swing_lo or 0, swing_hi or 0)
        total = gs + fs * 1.5
        marker_col = RED_ZONE if total >= 15 else ("#ff9900" if total >= 8 else col)
        fig.add_trace(go.Scatter(x=[r*c], y=[r*s],
            mode="markers+text",
            marker=dict(color=marker_col, size=sz*1.4, symbol="circle",
                        line=dict(color=DARK_BG, width=1)),
            text=[f"${item['price']:,.0f}"],
            textposition="middle center",
            textfont=dict(color=DARK_BG, size=6, family="monospace"),
            hovertemplate=(f"<b>{item['label']}</b><br>"
                           f"Õ·ﬁÂ {item['ring']}<br>"
                           f"ﬁÌ„ : ${item['price']:,.0f}<br>"
                           f"Confluence: {total:.1f}<br>"
                           f"{item['desc']}<extra></extra>"),
            showlegend=False))

    if current and base > 0:
        curr_ang = price_to_angle(current, base, step)
        rad = math.radians(90 - (curr_ang + rotation))
        c, s = math.cos(rad), math.sin(rad)
        for r in R_RINGS:
            fig.add_trace(go.Scatter(x=[r*c], y=[r*s], mode="markers",
                marker=dict(color=GREEN_ZONE, size=11, symbol="diamond",

            mode="lines", line=dict(color=GREEN_ZONE, width=2.5),

        text=[f"{rotation:.0f}∞"],

    return fig

    prices = [z["price"] for z in zones]
    totals = [z["total"] for z in zones]
    gann_s = [z["gann"] for z in zones]
    fibo_s = [z["fibo"] for z in zones]
    labels = []
    for z in zones:
        sig, _ = signal_label(z["total"])
        gh = sorted(z["gann_hits"], key=lambda x: -x["score"])[:2]

        labels.append(f"${z['price']:,}<br>{sig}<br>{gstr} {fstr}<br>«„ Ì«“: {z['total']:.1f}")

    fig = go.Figure()

    # ‰Ê«— confluence
    fig.add_trace(go.Bar(x=prices, y=totals, name="«„ Ì«“ ò·",
