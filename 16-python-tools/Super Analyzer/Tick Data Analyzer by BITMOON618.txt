# ==================== Bybit API با Retry و Cache ====================

class BybitAPIClient:
    """کلاینت حرفه‌ای بایبیت با retry و cache"""
    
    def __init__(self):
        self.session = requests.Session()
        self.last_request_time = 0
        self.min_request_interval = 1.5  # حداقل فاصله بین درخواست‌ها (ثانیه)
        self.cache = {}
        self.cache_expiry = 30  # انقضای کش (ثانیه)
        
        # تنظیم retry strategy
        retry_strategy = Retry(
            total=3,
            backoff_factor=1,
            status_forcelist=[429, 500, 502, 503, 504],
            allowed_methods=["GET"]
        )
        
        adapter = HTTPAdapter(max_retries=retry_strategy)
        self.session.mount("https://", adapter)
        self.session.mount("http://", adapter)
        
        # Headers حرفه‌ای
        self.headers = {
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
            'Accept': 'application/json, text/plain, */*',
            'Accept-Language': 'en-US,en;q=0.9,fa;q=0.8',
            'Accept-Encoding': 'gzip, deflate, br',
            'Connection': 'keep-alive',
            'Sec-Fetch-Dest': 'empty',
            'Sec-Fetch-Mode': 'cors',
            'Sec-Fetch-Site': 'cross-site',
        }
    
    def _rate_limit(self):
        """اعمال rate limiting"""
        current_time = time.time()
        time_since_last = current_time - self.last_request_time
        
        if time_since_last < self.min_request_interval:
            sleep_time = self.min_request_interval - time_since_last
            time.sleep(sleep_time)
        
        self.last_request_time = time.time()
    
    def _get_cache_key(self, symbol, interval, limit):
        """تولید کلید کش"""
        return f"{symbol}_{interval}_{limit}"
    
    def _is_cache_valid(self, key):
        """بررسی اعتبار کش"""
        if key not in self.cache:
            return False
        
        timestamp, _ = self.cache[key]
        return (time.time() - timestamp) < self.cache_expiry
    
    def get_klines(self, symbol="BTCUSDT", interval="5", limit=200):
        """دریافت کندل‌ها با retry و cache"""
        cache_key = self._get_cache_key(symbol, interval, limit)
        
        # بررسی کش
        if self._is_cache_valid(cache_key):
            print(f"✅ استفاده از کش برای {symbol}")
            _, cached_data = self.cache[cache_key]
            return cached_data
        
        url = "https://api.bybit.com/v5/market/kline"
        params = {
            "category": "linear",
            "symbol": symbol,
            "interval": interval,
            "limit": limit
        }
        
        max_retries = 3
        for attempt in range(max_retries):
            try:
                # اعمال rate limiting
                self._rate_limit()
                
                print(f" درخواست به API (تلاش {attempt + 1}/{max_retries})...")
                
                response = self.session.get(
                    url, 
                    params=params, 
                    headers=self.headers, 
                    timeout=15
                )
                
                if response.status_code == 403:
                    print(f"❌ خطای 403 - IP ممکنه بلاک شده باشه")
                    print("💡 پیشنهاد: از VPN یا Proxy استفاده کنید")
                    time.sleep(5)
                    continue
                
                if response.status_code == 429:
                    print(f"⚠️ Rate limit - صبر کنید...")
                    retry_after = int(response.headers.get('Retry-After', 5))
                    time.sleep(retry_after)
                    continue
                
                if response.status_code != 200:
                    print(f" HTTP Error: {response.status_code}")
                    print(f"Response: {response.text[:200]}")
                    time.sleep(2)
                    continue
                
                try:
                    data = response.json()
                except json.JSONDecodeError as e:
                    print(f"❌ JSON Decode Error: {e}")
                    return pd.DataFrame()
                
                if not isinstance(data, dict):
                    print(f" Unexpected response format")
                    return pd.DataFrame()
                
                if data.get('retCode') != 0:
                    print(f"❌ API Error: {data.get('retMsg', 'Unknown')}")
                    return pd.DataFrame()
                
                if 'result' not in data or 'list' not in data['result']:
                    print(f"❌ Missing result/list in response")
                    return pd.DataFrame()
                
                klines = data['result']['list']
                
                if not klines:
                    print("⚠️ No klines data received")
                    return pd.DataFrame()
                
                df = pd.DataFrame(klines, columns=[
                    'startTime', 'open', 'high', 'low', 'close', 'volume', 'turnover'
                ])
                df['startTime'] = pd.to_datetime(df['startTime'].astype(int), unit='ms')
                for col in ['open', 'high', 'low', 'close', 'volume', 'turnover']:
                    df[col] = pd.to_numeric(df[col], errors='coerce')
                df = df.sort_values('startTime').reset_index(drop=True)
                
                # ذخیره در کش
                self.cache[cache_key] = (time.time(), df)
                
                print(f"✅ دریافت موفق: {len(df)} کندل")
                return df
                
            except requests.exceptions.RequestException as e:
                print(f"❌ Request Error (attempt {attempt + 1}): {e}")
                if attempt < max_retries - 1:
                    wait_time = (attempt + 1) * 3
                    print(f" صبر به مدت {wait_time} ثانیه...")
                    time.sleep(wait_time)
                else:
                    print("❌ تمام تلاش‌ها ناموفق بود")
                    return pd.DataFrame()
            
            except Exception as e:
                print(f"❌ Unexpected Error: {e}")
                import traceback
                traceback.print_exc()
                return pd.DataFrame()
        
        return pd.DataFrame()

# ==================== Tick Analysis Engine ====================

class TickAnalyzer:
    
    def enhanced_analysis(self, klines_df):
        if klines_df is None or klines_df.empty:
            return pd.DataFrame()
        
        results = []
        
        for idx, candle in klines_df.iterrows():
            start = candle['startTime']
            
            estimated_ticks = max(20, int(candle['volume'] / 0.01))
            estimated_ticks = min(estimated_ticks, 500)
            
            np.random.seed(int(candle['startTime'].timestamp()) % 10000)
            
            body = abs(candle['close'] - candle['open'])
            upper_shadow = candle['high'] - max(candle['open'], candle['close'])
            lower_shadow = min(candle['open'], candle['close']) - candle['low']
            total_range = candle['high'] - candle['low']
            
            if total_range == 0 or pd.isna(total_range):
                total_range = 0.01
            
            if pd.isna(body) or pd.isna(upper_shadow) or pd.isna(lower_shadow):
                base_bull_ratio = 0.5
            elif candle['close'] >= candle['open']:
                base_bull_ratio = 0.55 + (body / total_range) * 0.2
                if lower_shadow > body:
                    base_bull_ratio += 0.1
            else:
                base_bull_ratio = 0.45 - (body / total_range) * 0.2
                if upper_shadow > body:
                    base_bull_ratio -= 0.1
            
            base_bull_ratio = np.clip(base_bull_ratio, 0.1, 0.9)
            
            pos_ticks = int(estimated_ticks * base_bull_ratio)
            neg_ticks = estimated_ticks - pos_ticks
            
            total_vol = candle['volume'] if not pd.isna(candle['volume']) else 0
            pos_volume = total_vol * base_bull_ratio * np.random.uniform(0.8, 1.2)
            neg_volume = total_vol - pos_volume
            
            noise = np.random.normal(0, 0.05)
            pos_ticks = max(0, int(pos_ticks * (1 + noise)))
            neg_ticks = max(0, estimated_ticks - pos_ticks)
            
            tick_ratio = pos_ticks / (pos_ticks + neg_ticks) if (pos_ticks + neg_ticks) > 0 else 0.5
            volume_ratio = pos_volume / (pos_volume + neg_volume) if (pos_volume + neg_volume) > 0 else 0.5
            
            results.append({
                'time': candle['startTime'],
                'open': candle['open'],
                'high': candle['high'],
                'low': candle['low'],
                'close': candle['close'],
                'volume': candle['volume'],
                'positive_ticks': pos_ticks,
                'negative_ticks': neg_ticks,
                'total_ticks': pos_ticks + neg_ticks,
                'positive_volume': round(pos_volume, 4),
                'negative_volume': round(neg_volume, 4),
                'tick_ratio': round(tick_ratio, 4),
                'volume_ratio': round(volume_ratio, 4),
                'net_ticks': pos_ticks - neg_ticks,
                'net_volume': round(pos_volume - neg_volume, 4),
                'candle_direction': 1 if candle['close'] >= candle['open'] else -1
            })
        
        return pd.DataFrame(results)
    
    def generate_signals(self, analysis_df):
        if analysis_df is None or analysis_df.empty:
            return pd.DataFrame()
        
        df = analysis_df.copy()
        
        df['price_sma'] = df['close'].rolling(5).mean()
        df['volume_pressure'] = df['net_volume'] / df['volume'].replace(0, 1)
        df['tick_momentum'] = df['net_ticks'].rolling(3).sum()
        
        delta = df['net_ticks'].diff()
        gain = delta.where(delta > 0, 0).rolling(14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
        rs = gain / loss.replace(0, 1)
        df['tick_rsi'] = 100 - (100 / (1 + rs))
        
        df['signal_score'] = 0
        
        df.loc[df['tick_ratio'] > 0.60, 'signal_score'] += 2
        df.loc[df['tick_ratio'] < 0.40, 'signal_score'] -= 2
        df.loc[df['volume_pressure'] > 0.3, 'signal_score'] += 2
        df.loc[df['volume_pressure'] < -0.3, 'signal_score'] -= 2
        df.loc[df['tick_momentum'] > 10, 'signal_score'] += 1
        df.loc[df['tick_momentum'] < -10, 'signal_score'] -= 1
        df.loc[df['tick_rsi'] < 30, 'signal_score'] += 2
        df.loc[df['tick_rsi'] > 70, 'signal_score'] -= 2
        
        price_up = df['close'] > df['price_sma']
        tick_pressure_down = df['volume_pressure'] < 0
        price_down = df['close'] < df['price_sma']
        tick_pressure_up = df['volume_pressure'] > 0
        
        df.loc[price_down & tick_pressure_up, 'signal_score'] += 3
        df.loc[price_up & tick_pressure_down, 'signal_score'] -= 3
        
        df['signal'] = 'HOLD'
        
        # 🔥 خرید قوی: نسبت تیک مثبت >= 70%
        strong_buy_mask = (df['signal_score'] >= 4) & (df['tick_ratio'] >= 0.70)
        df.loc[strong_buy_mask, 'signal'] = 'STRONG_BUY'
        
        # خرید معمولی
        buy_mask = (df['signal_score'] >= 2) & (df['signal_score'] < 4) & (df['signal'] == 'HOLD')
        df.loc[buy_mask, 'signal'] = 'BUY'
        
        # 🔥 فروش قوی: نسبت تیک منفی >= 70% (یعنی tick_ratio <= 30%)
        strong_sell_mask = (df['signal_score'] <= -4) & (df['tick_ratio'] <= 0.30)
        df.loc[strong_sell_mask, 'signal'] = 'STRONG_SELL'
        
        # فروش معمولی
        sell_mask = (df['signal_score'] <= -2) & (df['signal_score'] > -4) & (df['signal'] == 'HOLD')
        df.loc[sell_mask, 'signal'] = 'SELL'
        
        return df

            )
        
        signals_section = html.Div([
            html.Div([html.H3("⚡ سیگنال‌های ویژه (آخرین سیگنال‌های خرید و فروش)")],

                html.Table([

                    ])),
                    html.Tbody(signal_table_rows)
                ], className="signals-table"),

        ], style={'marginBottom': '25px'})
    else:

    status = f"✅ {datetime.now().strftime('%H:%M:%S')} | {symbol} | {len(analysis_df)} کندل"
