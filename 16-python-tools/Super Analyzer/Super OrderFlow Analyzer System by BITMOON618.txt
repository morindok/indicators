def hex_to_rgba(hex_color, alpha=0.1):
    hex_color = hex_color.lstrip('#')
    if len(hex_color) == 6:
        r = int(hex_color[0:2], 16)
        g = int(hex_color[2:4], 16)
        b = int(hex_color[4:6], 16)
        return f"rgba({r}, {g}, {b}, {alpha})"
    return hex_color


class NobitexOrderbookAnalyzer:
    def __init__(self, symbol="BTCIRT", max_history=200):
        self.symbol = symbol
        self.base_url = "https://apiv2.nobitex.ir/v3/orderbook"
        self.max_history = max_history

        self.usd_rate = self.fetch_usd_rate()
        if not self.usd_rate:
            self.usd_rate = DEFAULT_USD_TO_IRR

        self.unit_multiplier = 1
        self.last_usd_rate_update = 0

        # قیمت مرجع جهانی BTC/USD (بایننس با fallback به CoinGecko) برای مقایسه با قیمت مؤثر نوبیتکس
        self.reference_price = None
        self.reference_source = None
        self.last_reference_update = 0
        self.reference_history = deque(maxlen=max_history)
        self.premium_history = deque(maxlen=max_history)

        # آفست کالیبراسیون: اختلاف (قیمت مرجع - قیمت خام نوبیتکس) که به همه قیمت‌های
        # مطلق تبدیل‌شده از نوبیتکس اضافه می‌شود تا با بایننس/CoinGecko هم‌راستا شوند.
        # فقط هنگام دریافت قیمت مرجع تازه بازتنظیم می‌شود (نه هر چرخه)، تا نوسانات
        # واقعی اردربوک نوبیتکس بین دو بروزرسانی از بین نرود.
        self.price_calibration_offset = 0.0
        self._recalibrate_offset = True

        self.price_history = deque(maxlen=max_history)
        self.spread_history = deque(maxlen=max_history)
        self.imbalance_history = deque(maxlen=max_history)
        self.bid_depth_history = deque(maxlen=max_history)
        self.ask_depth_history = deque(maxlen=max_history)
        self.mid_price_history = deque(maxlen=max_history)
        self.volume_history = deque(maxlen=max_history)
        self.buy_pressure_history = deque(maxlen=max_history)
        self.sell_pressure_history = deque(maxlen=max_history)
        self.delta_history = deque(maxlen=max_history)
        self.vwap_history = deque(maxlen=max_history)
        self.timestamps = deque(maxlen=max_history)

        self.current_stats = None
        self.current_orderbook = None
        self.current_signals = []
        self.footprint_data = None
        self.orderflow_data = None

        self.footprint_history = deque(maxlen=max_history)
        self.orderflow_history = deque(maxlen=max_history)

        self.prev_bids = {}
        self.prev_asks = {}
        self.prev_time = 0

    def fetch_usd_rate(self):
        try:
            url = "https://apiv2.nobitex.ir/market/stats?srcCurrency=usdt&dstCurrency=irt"
            response = requests.get(url, timeout=5)
            if response.status_code == 200:
                data = response.json()
                if data.get("status") == "ok" and "usdt-irt" in data.get("stats", {}):
                    return float(data["stats"]["usdt-irt"]["latest"])
        except Exception:
            pass
        return None

    def update_usd_rate_if_needed(self):
        current_time = time.time()
        if current_time - self.last_usd_rate_update > 60:
            new_rate = self.fetch_usd_rate()
            if new_rate and new_rate > 0:
                self.usd_rate = new_rate
                self.last_usd_rate_update = current_time

    def fetch_reference_price(self):
        """
        قیمت مرجع جهانی BTC/USD را می‌گیرد تا با قیمت مؤثر نوبیتکس مقایسه/کالیبره شود.

        توجه: این متد و `update_reference_price_if_needed` دیگر به‌صورت خودکار در حلقهٔ اصلی
        اپ فراخوانی نمی‌شوند، چون در محیط فعلی دسترسی شبکه به api.binance.com و
        api.coingecko.com وجود ندارد و صبر کردن برای timeout هر چرخه را کند می‌کرد.
        به‌جای آن، قیمت مرجع از طریق ورودی دستی «قیمت لحظه‌ای بایننس» در بالای صفحه
        تنظیم می‌شود (کالبک `update_manual_reference_price`). این دو متد فقط برای زمانی
        نگه داشته شده‌اند که دسترسی شبکه در دسترس باشد (مثلاً با VPN).
        """
        try:
            response = requests.get(
                "https://api.binance.com/api/v3/ticker/price",
                params={"symbol": "BTCUSDT"}, timeout=5
            )
            if response.status_code == 200:
                price = float(response.json().get("price"))
                if price > 0:
                    return price, "Binance"
        except Exception:
            pass

        try:
            response = requests.get(
                "https://api.coingecko.com/api/v3/simple/price",
                params={"ids": "bitcoin", "vs_currencies": "usd"}, timeout=5
            )
            if response.status_code == 200:
                price = response.json().get("bitcoin", {}).get("usd")
                if price:
                    return float(price), "CoinGecko"
        except Exception:
            pass

        return None, None

    def update_reference_price_if_needed(self):
        current_time = time.time()
        if current_time - self.last_reference_update > 5:
            price, source = self.fetch_reference_price()
            if price and price > 0:
                self.reference_price = price
                self.reference_source = source
                self.last_reference_update = current_time
                self._recalibrate_offset = True  # آفست را در چرخه محاسباتی بعدی بازتنظیم کن

    def _raw_to_usd(self, price_amount):
        """تبدیل خام IRR/IRT به USD، بدون اعمال آفست کالیبراسیون (برای مقادیر دلتا مثل اسپرد)."""
        effective_divisor = self.usd_rate * self.unit_multiplier
        if effective_divisor <= 0:
            return 0
        return price_amount / effective_divisor

    def to_usd(self, price_amount):
        """تبدیل قیمت مطلق IRR/IRT به USD + اعمال آفست کالیبراسیون نسبت به بایننس/CoinGecko."""
        return self._raw_to_usd(price_amount) + self.price_calibration_offset

    def fetch_orderbook(self):
        url = f"{self.base_url}/{self.symbol}"
        try:
            response = requests.get(url, timeout=10)
            if response.status_code == 200:
                return response.json()
            return None
        except Exception:
            return None

    def parse_orderbook(self, data):
        if not data or data.get("status") != "ok":
            return None

        bids = [[float(price), float(volume)] for price, volume in data.get("bids", [])]
        asks = [[float(price), float(volume)] for price, volume in data.get("asks", [])]

        bids.sort(key=lambda x: x[0], reverse=True)
        asks.sort(key=lambda x: x[0])

        return {
            "bids": bids,
            "asks": asks,
            "last_trade_price": float(data.get("lastTradePrice", 0)),
            "last_update": data.get("lastUpdate", 0)
        }

    def calculate_statistics(self, orderbook):
        bids = orderbook["bids"]
        asks = orderbook["asks"]

        if not bids or not asks:
            return None

        best_bid_irr = bids[0][0]
        best_ask_irr = asks[0][0]
        spread_irr = best_ask_irr - best_bid_irr
        mid_price_irr = (best_bid_irr + best_ask_irr) / 2

        # کالیبراسیون با قیمت مرجع جهانی: فقط وقتی یک قیمت مرجع تازه رسیده (هر ~۵ ثانیه)
        # آفست بازتنظیم می‌شود؛ در فاصله بین دو بروزرسانی، آفست ثابت می‌ماند تا نوسانات
        # واقعی اردربوک نوبیتکس (که پایه سیگنال‌های این اپ است) دست‌نخورده باقی بمانند.
        raw_mid_price = self._raw_to_usd(mid_price_irr)
        if self._recalibrate_offset and self.reference_price and self.reference_price > 0 and raw_mid_price > 0:
            self.price_calibration_offset = self.reference_price - raw_mid_price
            self._recalibrate_offset = False

        best_bid = self.to_usd(best_bid_irr)
        best_ask = self.to_usd(best_ask_irr)
        spread = self._raw_to_usd(spread_irr)  # اسپرد یک اختلاف قیمتی است؛ آفست کالیبراسیون نباید رویش اعمال شود
        mid_price = self.to_usd(mid_price_irr)
        last_trade = self.to_usd(orderbook["last_trade_price"])
        spread_pct = (spread / mid_price) * 100 if mid_price else 0

        total_bid_volume = sum(b[1] for b in bids)
        total_ask_volume = sum(a[1] for a in asks)
        total_volume = total_bid_volume + total_ask_volume

        if total_volume > 0:
            imbalance = (total_bid_volume - total_ask_volume) / total_volume
        else:
            imbalance = 0

        vwap_bid = self._weighted_avg_price(bids)
        vwap_ask = self._weighted_avg_price(asks)
        vwap_bid_usd = self.to_usd(vwap_bid)
        vwap_ask_usd = self.to_usd(vwap_ask)

        bid_depth_1pct = self._calculate_depth(bids, mid_price_irr, 0.01)
        bid_depth_5pct = self._calculate_depth(bids, mid_price_irr, 0.05)
        ask_depth_1pct = self._calculate_depth(asks, mid_price_irr, 0.01, is_ask=True)
        ask_depth_5pct = self._calculate_depth(asks, mid_price_irr, 0.05, is_ask=True)

        avg_bid_size = statistics.mean([b[1] for b in bids]) if bids else 0
        avg_ask_size = statistics.mean([a[1] for a in asks]) if asks else 0

        large_bids = [b for b in bids if b[1] > avg_bid_size * 2]
        large_asks = [a for a in asks if a[1] > avg_ask_size * 2]

        top5_bid_vol = sum(b[1] for b in bids[:5])
        top5_ask_vol = sum(a[1] for a in asks[:5])
        buy_pressure = top5_bid_vol / (top5_bid_vol + top5_ask_vol) if (top5_bid_vol + top5_ask_vol) > 0 else 0.5
        sell_pressure = 1 - buy_pressure

        delta = self._calculate_delta(bids, asks)

        return {
            "timestamp": time.time(),
            "best_bid": best_bid,
            "best_ask": best_ask,
            "spread": spread,
            "spread_pct": spread_pct,
            "mid_price": mid_price,
            "mid_price_raw": raw_mid_price,
            "calibration_offset": self.price_calibration_offset,
            "mid_price_irr": mid_price_irr,
            "last_trade_price": last_trade,
            "total_bid_volume": total_bid_volume,
            "total_ask_volume": total_ask_volume,
            "total_volume": total_volume,
            "imbalance": imbalance,
            "bid_depth_1pct": bid_depth_1pct,
            "bid_depth_5pct": bid_depth_5pct,
            "ask_depth_1pct": ask_depth_1pct,
            "ask_depth_5pct": ask_depth_5pct,
            "vwap_bid": vwap_bid_usd,
            "vwap_ask": vwap_ask_usd,
            "bid_levels": len(bids),
            "ask_levels": len(asks),
            "avg_bid_size": avg_bid_size,
            "avg_ask_size": avg_ask_size,
            "large_bids_count": len(large_bids),
            "large_asks_count": len(large_asks),
            "large_bids_volume": sum(b[1] for b in large_bids),
            "large_asks_volume": sum(a[1] for a in large_asks),
            "buy_pressure": buy_pressure,
            "sell_pressure": sell_pressure,
            "delta": delta,
            "bids": bids,
            "asks": asks,
            "best_bid_irr": best_bid_irr,
            "best_ask_irr": best_ask_irr,
        }

    def _calculate_depth(self, orders, mid_price, pct_range, is_ask=False):
        if is_ask:
            limit = mid_price * (1 + pct_range)
            return sum(o[1] for o in orders if o[0] <= limit)
        else:
            limit = mid_price * (1 - pct_range)
            return sum(o[1] for o in orders if o[0] >= limit)

    def _weighted_avg_price(self, orders):
        if not orders:
            return 0
        total_vol = sum(o[1] for o in orders)
        if total_vol == 0:
            return 0
        return sum(o[0] * o[1] for o in orders) / total_vol

    def _calculate_delta(self, bids, asks):
        """
        برآورد خالص تغییر «فشار» اردربوک بین دو اسنپ‌شات (Book-Pressure Delta).
        توجه علمی مهم: این معیار Trade Delta واقعی (خرید/فروش تیک‌به‌تیک) نیست،
        چون نوبیتکس داده معاملات با برچسب Aggressor Side نمی‌دهد؛ این فقط یک
        تقریب از جهت نقدینگی اردربوک است.

        نسخه قبلی فقط افزایش حجم را می‌شمرد و کاهش/حذف سطوح (لغو سفارش یا پر شدن آن)
        را نادیده می‌گرفت که تصویر را ناقص می‌کرد. این نسخه خالص تغییرات
        (هم افزایش، هم کاهش) را روی اجتماع سطوح فعلی و قبلی محاسبه می‌کند.
        """
        current_bids = {b[0]: b[1] for b in bids[:20]}
        current_asks = {a[0]: a[1] for a in asks[:20]}

        bid_net = 0.0
        for price in set(current_bids) | set(self.prev_bids):
            bid_net += current_bids.get(price, 0.0) - self.prev_bids.get(price, 0.0)

        ask_net = 0.0
        for price in set(current_asks) | set(self.prev_asks):
            ask_net += current_asks.get(price, 0.0) - self.prev_asks.get(price, 0.0)

        self.prev_bids = current_bids
        self.prev_asks = current_asks

        return bid_net - ask_net

    def calculate_footprint(self, stats):
        bids = stats["bids"]
        asks = stats["asks"]

        size_ranges = [
            (0, 0.001, "Micro"),
            (0.001, 0.01, "Small"),
            (0.01, 0.1, "Medium"),
            (0.1, 1.0, "Large"),
            (1.0, float('inf'), "Whale")
        ]

        footprint = {
            "bid_distribution": defaultdict(float),
            "ask_distribution": defaultdict(float),
            "bid_count": defaultdict(int),
            "ask_count": defaultdict(int),
            "cluster_analysis": [],
            "absorption_zones": [],
            "iceberg_hints": [],
        }

        for min_s, max_s, label in size_ranges:
            bid_vol = sum(b[1] for b in bids if min_s <= b[1] < max_s)
            ask_vol = sum(a[1] for a in asks if min_s <= a[1] < max_s)
            bid_cnt = sum(1 for b in bids if min_s <= b[1] < max_s)
            ask_cnt = sum(1 for a in asks if min_s <= a[1] < max_s)

            footprint["bid_distribution"][label] = bid_vol
            footprint["ask_distribution"][label] = ask_vol
            footprint["bid_count"][label] = bid_cnt
            footprint["ask_count"][label] = ask_cnt

        for i in range(len(bids) - 2):
            cluster_vol = sum(b[1] for b in bids[i:i + 3])
            cluster_price_range = bids[i][0] - bids[i + 2][0]
            if cluster_vol > stats["total_bid_volume"] * 0.15 and cluster_price_range < stats["best_bid_irr"] * 0.002:
                footprint["cluster_analysis"].append({
                    "side": "bid",
                    "price": self.to_usd(bids[i][0]),
                    "volume": cluster_vol,
                    "levels": 3,
                    "strength": "Strong" if cluster_vol > stats["total_bid_volume"] * 0.25 else "Medium"
                })

        for i in range(len(asks) - 2):
            cluster_vol = sum(a[1] for a in asks[i:i + 3])
            cluster_price_range = asks[i + 2][0] - asks[i][0]
            if cluster_vol > stats["total_ask_volume"] * 0.15 and cluster_price_range < stats["best_ask_irr"] * 0.002:
                footprint["cluster_analysis"].append({
                    "side": "ask",
                    "price": self.to_usd(asks[i][0]),
                    "volume": cluster_vol,
                    "levels": 3,
                    "strength": "Strong" if cluster_vol > stats["total_ask_volume"] * 0.25 else "Medium"
                })

        bid_sizes = [b[1] for b in bids[:10]]
        ask_sizes = [a[1] for a in asks[:10]]

        for size, count in self._find_repeated_sizes(bid_sizes).items():
            if count >= 3 and size > stats["avg_bid_size"]:
                footprint["iceberg_hints"].append({
                    "side": "bid",
                    "size": size,
                    "count": count,
                    "confidence": "High" if count >= 5 else "Medium"
                })

        for size, count in self._find_repeated_sizes(ask_sizes).items():
            if count >= 3 and size > stats["avg_ask_size"]:
                footprint["iceberg_hints"].append({
                    "side": "ask",
                    "size": size,
                    "count": count,
                    "confidence": "High" if count >= 5 else "Medium"
                })

        if len(self.footprint_history) > 0:
            prev = self.footprint_history[-1]
            if abs(stats["mid_price"] - self.mid_price_history[-1]) / stats["mid_price"] < 0.001:
                if stats["total_volume"] > statistics.mean(self.volume_history) * 1.5:
                    footprint["absorption_zones"].append({
                        "price": stats["mid_price"],
                        "volume": stats["total_volume"],
                        "type": "Bid Absorption" if stats["imbalance"] > 0 else "Ask Absorption"
                    })

        return footprint

    def _find_repeated_sizes(self, sizes, tolerance=0.0001):
        groups = defaultdict(int)
        for s in sizes:
            key = round(s / tolerance) * tolerance
            groups[key] += 1
        return {k: v for k, v in groups.items() if v >= 2}

    def calculate_orderflow(self, stats):
        orderflow = {
            "flow_direction": "Neutral",
            "flow_strength": 0,
            "aggressive_buyers": 0,
            "aggressive_sellers": 0,
            "passive_buyers": 0,
            "passive_sellers": 0,
            "market_depth_ratio": 0,
            "liquidity_score": 0,
            "resistance_levels": [],
            "support_levels": [],
            "momentum": 0,
        }

        if stats["imbalance"] > 0.4:
            orderflow["flow_direction"] = "Strong Buy"
            orderflow["flow_strength"] = stats["imbalance"]
        elif stats["imbalance"] > 0.2:
            orderflow["flow_direction"] = "Buy"
            orderflow["flow_strength"] = stats["imbalance"]
        elif stats["imbalance"] < -0.4:
            orderflow["flow_direction"] = "Strong Sell"
            orderflow["flow_strength"] = abs(stats["imbalance"])
        elif stats["imbalance"] < -0.2:
            orderflow["flow_direction"] = "Sell"
            orderflow["flow_strength"] = abs(stats["imbalance"])

        bid_aggression = sum(b[1] for b in stats["bids"][:3]) / stats["total_bid_volume"] if stats[
                                                                                                 "total_bid_volume"] > 0 else 0
        ask_aggression = sum(a[1] for a in stats["asks"][:3]) / stats["total_ask_volume"] if stats[
                                                                                                 "total_ask_volume"] > 0 else 0

        orderflow["aggressive_buyers"] = bid_aggression
        orderflow["aggressive_sellers"] = ask_aggression
        orderflow["passive_buyers"] = 1 - bid_aggression
        orderflow["passive_sellers"] = 1 - ask_aggression

        if stats["ask_depth_1pct"] > 0:
            orderflow["market_depth_ratio"] = stats["bid_depth_1pct"] / stats["ask_depth_1pct"]

        spread_score = max(0, 1 - stats["spread_pct"] / 0.5)
        volume_score = min(1, stats["total_volume"] / 1.0)
        depth_score = min(1, (stats["bid_depth_1pct"] + stats["ask_depth_1pct"]) / 0.5)
        orderflow["liquidity_score"] = (spread_score * 0.4 + volume_score * 0.3 + depth_score * 0.3) * 100

        bids = stats["bids"]
        asks = stats["asks"]

        for i in range(min(10, len(bids))):
            if bids[i][1] > stats["avg_bid_size"] * 3:
                orderflow["support_levels"].append({
                    "price": self.to_usd(bids[i][0]),
                    "volume": bids[i][1],
                    "strength": "Strong" if bids[i][1] > stats["avg_bid_size"] * 5 else "Medium"
                })

        for i in range(min(10, len(asks))):
            if asks[i][1] > stats["avg_ask_size"] * 3:
                orderflow["resistance_levels"].append({
                    "price": self.to_usd(asks[i][0]),
                    "volume": asks[i][1],
                    "strength": "Strong" if asks[i][1] > stats["avg_ask_size"] * 5 else "Medium"
                })

        if len(self.mid_price_history) >= 3:
            changes = [self.mid_price_history[i] - self.mid_price_history[i - 1]
                       for i in range(1, len(self.mid_price_history))]
            if len(changes) >= 2:
                orderflow["momentum"] = changes[-1] - changes[-2]

        return orderflow

    def update_history(self, stats):
        self.price_history.append(stats["mid_price"])
        self.spread_history.append(stats["spread_pct"])
        self.imbalance_history.append(stats["imbalance"])
        self.bid_depth_history.append(stats["bid_depth_1pct"])
        self.ask_depth_history.append(stats["ask_depth_1pct"])
        self.mid_price_history.append(stats["mid_price"])
        self.volume_history.append(stats["total_volume"])
        self.buy_pressure_history.append(stats["buy_pressure"])
        self.sell_pressure_history.append(stats["sell_pressure"])
        self.delta_history.append(stats["delta"])
        self.vwap_history.append((stats["vwap_bid"] + stats["vwap_ask"]) / 2)
        self.timestamps.append(time.strftime('%H:%M:%S'))

        if self.reference_price and self.reference_price > 0:
            premium_pct = (stats["mid_price"] - self.reference_price) / self.reference_price * 100
            self.reference_history.append(self.reference_price)
            self.premium_history.append(premium_pct)

    def calculate_trends(self):
        trends = {}
        if len(self.mid_price_history) >= 2:
            trends["price_change"] = self.mid_price_history[-1] - self.mid_price_history[-2]
            trends["price_change_pct"] = (trends["price_change"] / self.mid_price_history[-2]) * 100 if \
            self.mid_price_history[-2] > 0 else 0

        if len(self.spread_history) >= 5:
            trends["avg_spread"] = statistics.mean(self.spread_history)
            trends["spread_volatility"] = statistics.stdev(self.spread_history) if len(self.spread_history) > 1 else 0

        if len(self.imbalance_history) >= 5:
            trends["avg_imbalance"] = statistics.mean(self.imbalance_history)
            trends["imbalance_trend"] = "Bullish" if trends["avg_imbalance"] > 0.1 else "Bearish" if trends[
                                                                                                         "avg_imbalance"] < -0.1 else "Neutral"

        if len(self.volume_history) >= 5:
            trends["avg_volume"] = statistics.mean(self.volume_history)
            trends["volume_trend"] = "Increasing" if self.volume_history[-1] > trends["avg_volume"] else "Decreasing"

        if len(self.delta_history) >= 5:
            trends["avg_delta"] = statistics.mean(self.delta_history)
            trends["delta_trend"] = "Accumulation" if trends["avg_delta"] > 0 else "Distribution"

        return trends

    def generate_signals(self, stats, trends, footprint, orderflow):
        signals = []

        # بررسی معقول بودن قیمت محاسبه‌شده - محافظ در برابر انتخاب اشتباه واحد (تومان/ریال)
        if stats["mid_price"] > 0 and (stats["mid_price"] < 1000 or stats["mid_price"] > 500000):
            signals.append(("danger",
                             "🚫 قیمت محاسبه‌شده (${:,.0f}) خارج از بازه منطقی BTC است - واحد قیمت (تومان/ریال) را از بالای صفحه بررسی کنید".format(
                                 stats["mid_price"]), 0))

        if self.reference_price and self.reference_price > 0 and stats.get("calibration_offset") is not None:
            offset_pct = (stats["calibration_offset"] / self.reference_price) * 100
            if abs(offset_pct) > 5:
                signals.append(("warning",
                                 "💱 آفست کالیبراسیون نسبت به {} غیرعادی است ({:+.2f}%, ${:+,.2f}) - نرخ دلار/واحد قیمت را بررسی کنید".format(
                                     self.reference_source or "قیمت مرجع", offset_pct, stats["calibration_offset"]), 1))

        if stats["spread_pct"] > 0.5:
            signals.append(("warning", "⚠️ اسپرد بالا ({:.4f}%) - نقدشوندگی پایین".format(stats["spread_pct"]), 3))
        elif stats["spread_pct"] < 0.1:
            signals.append(("success", "✅ اسپرد کم ({:.4f}%) - نقدشوندگی بالا".format(stats["spread_pct"]), 1))

        if stats["imbalance"] > 0.5:
            signals.append(("success", "🟢 فشار خرید بسیار قوی (Imbalance: {:.2f})".format(stats["imbalance"]), 1))
        elif stats["imbalance"] > 0.3:
            signals.append(("success", "🟢 فشار خرید قوی (Imbalance: {:.2f})".format(stats["imbalance"]), 2))
        elif stats["imbalance"] < -0.5:
            signals.append(("danger", "🔴 فشار فروش بسیار قوی (Imbalance: {:.2f})".format(stats["imbalance"]), 1))
        elif stats["imbalance"] < -0.3:
            signals.append(("danger", "🔴 فشار فروش قوی (Imbalance: {:.2f})".format(stats["imbalance"]), 2))

        if "avg_delta" in trends:
            if stats["delta"] > trends["avg_delta"] * 2 and stats["delta"] > 0:
                signals.append(("success", "📈 Delta مثبت قوی - جریان خرید در حال افزایش", 2))
            elif stats["delta"] < trends["avg_delta"] * 2 and stats["delta"] < 0:
                signals.append(("danger", "📉 Delta منفی قوی - جریان فروش در حال افزایش", 2))

        if footprint["iceberg_hints"]:
            for hint in footprint["iceberg_hints"]:
                if hint["confidence"] == "High":
                    side_emoji = "🟢" if hint["side"] == "bid" else "🔴"
                    signals.append(("info", "{} Iceberg Order {}: {:.6f} BTC x {}".format(
                        side_emoji, hint["side"].upper(), hint["size"], hint["count"]), 2))

        if footprint["absorption_zones"]:
            for zone in footprint["absorption_zones"]:
                signals.append(("warning", "⚡ Zone Absorption در ${:,.2f} - حجم: {:.6f} BTC".format(
                    zone["price"], zone["volume"]), 2))

        if orderflow["flow_direction"] in ["Strong Buy", "Strong Sell"]:
            signals.append(("info", "🌊 Order Flow: {} (Strength: {:.2f})".format(
                orderflow["flow_direction"], orderflow["flow_strength"]), 1))

        if orderflow["liquidity_score"] < 30:
            signals.append(
                ("danger", "💧 نقدشوندگی بسیار پایین (Score: {:.1f}/100)".format(orderflow["liquidity_score"]), 2))
        elif orderflow["liquidity_score"] > 70:
            signals.append(("success", "💧 نقدشوندگی عالی (Score: {:.1f}/100)".format(orderflow["liquidity_score"]), 3))

        if stats["last_trade_price"] > 0:
            deviation = (stats["mid_price"] - stats["last_trade_price"]) / stats["last_trade_price"] * 100
            if deviation > 0.5:

            elif deviation < -0.5:

        if stats["mid_price"] < stats["vwap_bid"]:
            signals.append(("success", "🟢 قیمت زیر VWAP بید - فرصت خرید بالقوه", 3))
        elif stats["mid_price"] > stats["vwap_ask"]:
            signals.append(("danger", "💎 قیمت بالای VWAP اسک - فشار فروش بالقوه", 3))

        signals.sort(key=lambda x: x[2])
        return [(s[0], s[1]) for s in signals]

    def calculate_comprehensive_trend(self):
        result = {"score": 0.0, "direction": "خنثی", "color_key": "text_muted",
                  "factors": [], "confidence": "کم"}

        if len(self.mid_price_history) < 5 or not self.current_stats:
            return result

        prices = list(self.mid_price_history)
        n = len(prices)
        xs = list(range(n))
        x_mean = statistics.mean(xs)
        y_mean = statistics.mean(prices)
        denom = sum((x - x_mean) ** 2 for x in xs)
        slope = sum((xs[i] - x_mean) * (prices[i] - y_mean) for i in range(n)) / denom if denom > 0 else 0
        slope_pct = (slope / y_mean) * 100 if y_mean else 0

        score = 0.0
        factors = []

        price_score = max(-25, min(25, slope_pct * 800))
        score += price_score
        factors.append(("شیب قیمت میانی", f"{slope_pct:+.5f}%/نمونه", price_score))

        avg_imbalance = statistics.mean(list(self.imbalance_history)[-15:]) if self.imbalance_history else 0
        imbalance_score = max(-20, min(20, avg_imbalance * 40))
        score += imbalance_score
        factors.append(("نامتعادلی بید/اسک", f"{avg_imbalance:+.3f}", imbalance_score))

        recent_delta = list(self.delta_history)[-15:] if self.delta_history else []
        avg_delta = statistics.mean(recent_delta) if recent_delta else 0
        delta_ref = statistics.mean([abs(d) for d in self.delta_history]) if self.delta_history else 0
        delta_score = max(-20, min(20, (avg_delta / delta_ref) * 20)) if delta_ref > 0 else 0
        score += delta_score
        factors.append(("Delta حجم", f"{avg_delta:+.6f} BTC", delta_score))

        avg_buy_p = statistics.mean(list(self.buy_pressure_history)[-15:]) if self.buy_pressure_history else 0.5
        pressure_score = max(-15, min(15, (avg_buy_p - 0.5) * 60))
        score += pressure_score
        factors.append(("فشار خرید/فروش", f"{avg_buy_p * 100:.1f}% خرید", pressure_score))

        stats = self.current_stats
        vwap_score = 0
        if stats["mid_price"] < stats["vwap_bid"]:
            vwap_score = 10
        elif stats["mid_price"] > stats["vwap_ask"]:
            vwap_score = -10
        score += vwap_score
        vwap_label = "زیر VWAP بید" if vwap_score > 0 else ("بالای VWAP اسک" if vwap_score < 0 else "داخل بازه VWAP")
        factors.append(("موقعیت نسبت به VWAP", vwap_label, vwap_score))

        volume_score = 0
        if len(self.volume_history) >= 10:
            recent_vol = statistics.mean(list(self.volume_history)[-5:])
            older_vol = statistics.mean(list(self.volume_history)[-10:-5])
            if older_vol > 0:
                vol_change_pct = (recent_vol - older_vol) / older_vol * 100
                volume_score = max(-10, min(10, vol_change_pct / 5))
        score += volume_score
        vol_label = "افزایشی" if volume_score > 0 else ("کاهشی" if volume_score < 0 else "ثابت")
        factors.append(("روند حجم معاملات", vol_label, volume_score))

        score = max(-100, min(100, score))

        if score >= 40:
            direction, color_key = "📈 صعودی قوی", "bid"
        elif score >= 15:
            direction, color_key = "📈 صعودی", "success"
        elif score <= -40:
            direction, color_key = "📉 نزولی قوی", "ask"
        elif score <= -15:
            direction, color_key = "📉 نزولی", "danger"
        else:
            direction, color_key = "➖ خنثی / بدون روند مشخص", "text_muted"

        confidence = "بالا" if n >= 30 else ("متوسط" if n >= 15 else "کم")

        result.update({"score": score, "direction": direction, "color_key": color_key,
                       "factors": factors, "confidence": confidence})
        return result

    def update(self):
        data = self.fetch_orderbook()
        if not data:
            return False

        orderbook = self.parse_orderbook(data)
        if not orderbook:
            return False

        stats = self.calculate_statistics(orderbook)
        if not stats:
            return False

        self.current_stats = stats
        self.current_orderbook = orderbook

        self.footprint_data = self.calculate_footprint(stats)
        self.orderflow_data = self.calculate_orderflow(stats)

        self.footprint_history.append(self.footprint_data)
        self.orderflow_history.append(self.orderflow_data)

        self.update_history(stats)
        trends = self.calculate_trends()
        self.current_signals = self.generate_signals(stats, trends, self.footprint_data, self.orderflow_data)

        return True


class FootprintAnalyzer:
    def __init__(self, max_history=50):
        self.max_history = max_history
        self.footprint_history = deque(maxlen=max_history)
        self.timestamps = deque(maxlen=max_history)

    def calculate_aggressive_volume(self, bids, asks, levels=5):
        aggressive_buyers = sum(b[1] for b in bids[:levels]) if bids else 0
        aggressive_sellers = sum(a[1] for a in asks[:levels]) if asks else 0
        return aggressive_buyers, aggressive_sellers

    def update(self, orderbook_data, mid_price_usd):
        if not orderbook_data:
            return None

        bids = orderbook_data.get("bids", [])
        asks = orderbook_data.get("asks", [])

        agg_buyers, agg_sellers = self.calculate_aggressive_volume(bids, asks, levels=5)
        total_volume = agg_buyers + agg_sellers

        imbalance = (agg_buyers - agg_sellers) / total_volume if total_volume > 0 else 0

        if agg_buyers > agg_sellers * 1.2:
            candle_type = "bullish"
        elif agg_sellers > agg_buyers * 1.2:
            candle_type = "bearish"
        else:
            candle_type = "neutral"

        is_hvn = False
        if len(self.footprint_history) >= 5:
            avg_vol = statistics.mean([fp["total_volume"] for fp in self.footprint_history])
            is_hvn = total_volume > avg_vol * 0.8

        footprint = {
            "aggressive_buyers": agg_buyers,
            "aggressive_sellers": agg_sellers,
            "total_volume": total_volume,
            "imbalance": imbalance,
            "candle_type": candle_type,
            "is_hvn": is_hvn,
            "price": mid_price_usd,
            "timestamp": time.strftime('%H:%M:%S')
        }

        self.footprint_history.append(footprint)
        self.timestamps.append(footprint["timestamp"])

        return footprint


class OrderbookFootprintBuilder:
    def __init__(self, max_candles=18, tick_pct=0.0006, depth_levels=25):
        self.max_candles = max_candles
        self.tick_pct = tick_pct
        self.depth_levels = depth_levels
        self.candles = deque(maxlen=max_candles)

    def add_snapshot(self, stats, usd_rate, price_offset=0.0):
        if not stats:
            return None

        bids = stats["bids"][:self.depth_levels]
        asks = stats["asks"][:self.depth_levels]
        mid_usd = stats["mid_price"]
        if mid_usd <= 0:
            return None

        tick_size = max(mid_usd * self.tick_pct, 0.01)

        levels = defaultdict(lambda: [0.0, 0.0])

        for price_irr, vol in bids:
            price_usd = price_irr / usd_rate + price_offset
            lvl = round(math.floor(price_usd / tick_size) * tick_size, 2)
            levels[lvl][0] += vol

        for price_irr, vol in asks:
            price_usd = price_irr / usd_rate + price_offset
            lvl = round(math.floor(price_usd / tick_size) * tick_size, 2)
            levels[lvl][1] += vol

        if not levels:
            return None

        price_levels = list(levels.keys())
        high = max(price_levels) + tick_size / 2
        low = min(price_levels) - tick_size / 2
        open_price = self.candles[-1]["close"] if self.candles else mid_usd
        close_price = mid_usd

        totals = {lvl: v[0] + v[1] for lvl, v in levels.items()}
        hvn_level = max(totals, key=totals.get)
        avg_total = statistics.mean(totals.values()) if totals else 0
        is_hvn_candle = avg_total > 0 and totals[hvn_level] > avg_total * 1.8

        buy_total = sum(v[0] for v in levels.values())
        sell_total = sum(v[1] for v in levels.values())

        candle = {
            "time": time.strftime('%H:%M:%S'),
            "open": open_price,
            "high": high,
            "low": low,
            "close": close_price,
            "levels": {lvl: (v[0], v[1]) for lvl, v in levels.items()},
            "tick_size": tick_size,
            "hvn_level": hvn_level,
            "is_hvn_candle": is_hvn_candle,
            "buy_total": buy_total,
            "sell_total": sell_total,
            "delta": buy_total - sell_total,
        }
        self.candles.append(candle)
        return candle

class VolumeProfileAnalyzer:
    def __init__(self, num_levels=40):
        self.num_levels = num_levels
        self.profile = {}
        self.poc = 0.0
        self.va_high = 0.0
        self.va_low = 0.0
        self.hvn_levels = []
        self.lvn_levels = []

    def calculate(self, prices, volumes):
        if not prices or not volumes or len(prices) < 10:
            return None

        min_p = min(prices)
        max_p = max(prices)
        if min_p == max_p:
            return None

        level_size = (max_p - min_p) / self.num_levels
        if level_size == 0:
            level_size = 0.01

        self.profile = {}
        for p, v in zip(prices, volumes):
            level = round(((p - min_p) // level_size) * level_size + min_p, 2)
            self.profile[level] = self.profile.get(level, 0.0) + v

        if not self.profile:
            return None

        # محاسبه POC (Point of Control)
        self.poc = max(self.profile, key=self.profile.get)

        # محاسبه Value Area (70%)
        sorted_levels = sorted(self.profile.keys(), key=lambda x: abs(x - self.poc))
        total_vol = sum(self.profile.values())
        target_vol = total_vol * 0.70
        cum_vol = 0
        va_levels = []
        for lvl in sorted_levels:
            cum_vol += self.profile[lvl]
            va_levels.append(lvl)
            if cum_vol >= target_vol:
                break
        self.va_high = max(va_levels) if va_levels else self.poc
        self.va_low = min(va_levels) if va_levels else self.poc

        # شناسایی HVN و LVN بر اساس انحراف معیار
        vols = list(self.profile.values())
        mean_vol = np.mean(vols)
        std_vol = np.std(vols)

        self.hvn_levels = []
        self.lvn_levels = []
        for lvl, vol in self.profile.items():
            if vol > mean_vol + 0.5 * std_vol:
                self.hvn_levels.append(
                    {'price': lvl, 'volume': vol, 'strength': 'Strong' if vol > mean_vol + 1.5 * std_vol else 'Medium'})
            elif vol < mean_vol - 0.5 * std_vol:
                self.lvn_levels.append(
                    {'price': lvl, 'volume': vol, 'strength': 'Strong' if vol < mean_vol - 1.5 * std_vol else 'Medium'})

        self.hvn_levels.sort(key=lambda x: x['volume'], reverse=True)
        self.lvn_levels.sort(key=lambda x: x['volume'])

        return self.profile

# کلاس جدید: سیستم آماری پیشرفته + برایند نهایی تمام تب‌ها
# ==========================================
class StatisticalSynthesizer:
    """
    به‌جای وزن‌دهی دلبخواهی (که در نسخه‌های قبلی روی امتیازها اعمال می‌شد)،
    این کلاس از دو ابزار آماری استاندارد استفاده می‌کند:

    1) Z-Score خودکالیبره: هر شاخص (Imbalance، Delta، فشار خرید، اسپرد) نسبت به
       میانگین و انحراف‌معیار خودِ همان نشست معاملاتی نرمال‌سازی می‌شود، نه
       بازه‌های ثابت و دلخواه.

    2) ضریب همبستگی پیرسون + آزمون معناداری با تبدیل فیشر (Fisher z-transform):
       رابطه بین Imbalance/Delta در لحظه t و تغییر قیمت در لحظه t+1 (lag-1)
       سنجیده می‌شود تا معلوم شود آیا این شاخص‌ها واقعاً «پیش‌بینی‌کننده»
       بوده‌اند یا صرفاً هم‌بسته لحظه‌ای. اگر p-value < 0.05 باشد رابطه از نظر
       آماری معنادار تلقی می‌شود (بدون نیاز به scipy).

    خروجی نهایی یک «رأی‌گیری» بین سیگنال‌های مستقل چهار تب است، نه جمع ساده
    امتیازها؛ سطح اطمینان هم بر اساس میزان توافق رأی‌ها و هم معناداری آماری
    همبستگی‌های بالا تعیین می‌شود.
    """

    def __init__(self):
        self.history_composite = deque(maxlen=200)

    @staticmethod
    def zscore(current, series):
        series = list(series)
        if len(series) < 5:
            return 0.0
        mu = statistics.mean(series)
        sigma = statistics.pstdev(series) if len(series) > 1 else 0.0
        if sigma == 0:
            return 0.0
        return (current - mu) / sigma

    @staticmethod
    def _norm_cdf(x):
        return 0.5 * (1 + math.erf(x / math.sqrt(2)))

    @classmethod
    def pearson_with_pvalue(cls, x, y):
        """همبستگی پیرسون + p-value دوطرفه با روش تبدیل فیشر (برای n >= 10 قابل‌اعتماد)."""
        n = min(len(x), len(y))
        if n < 10:
            return 0.0, 1.0, n
        x = list(x)[-n:]
        y = list(y)[-n:]
        try:
            r = statistics.correlation(x, y)  # Python 3.10+
        except Exception:
            mx, my = statistics.mean(x), statistics.mean(y)
            sx, sy = statistics.pstdev(x), statistics.pstdev(y)
            if sx == 0 or sy == 0:
                return 0.0, 1.0, n
            cov = sum((xi - mx) * (yi - my) for xi, yi in zip(x, y)) / n
            r = cov / (sx * sy)

        r = max(-0.999999, min(0.999999, r))
        z = 0.5 * math.log((1 + r) / (1 - r))
        se = 1 / math.sqrt(n - 3) if n > 3 else 1.0
        z_stat = z / se if se > 0 else 0.0
        p_value = 2 * (1 - cls._norm_cdf(abs(z_stat)))
        return r, max(0.0, min(1.0, p_value)), n

    def evaluate_predictive_power(self, imbalance_hist, delta_hist, price_hist):
        """آیا Imbalance/Delta واقعاً پیش‌بینی‌کننده تغییر قیمت لحظه بعد بوده‌اند؟"""
        results = {}
        prices = list(price_hist)
        if len(prices) < 12:
            return results

        price_changes = [prices[i + 1] - prices[i] for i in range(len(prices) - 1)]

        imb = list(imbalance_hist)[:-1][-len(price_changes):]
        if len(imb) == len(price_changes) and len(imb) >= 10:
            r, p, n = self.pearson_with_pvalue(imb, price_changes)
            results['imbalance_vs_price'] = {'r': r, 'p_value': p, 'n': n,
                                              'significant': p < 0.05, 'x': imb, 'y': price_changes}

        delta = list(delta_hist)[:-1][-len(price_changes):]
        if len(delta) == len(price_changes) and len(delta) >= 10:
            r, p, n = self.pearson_with_pvalue(delta, price_changes)
            results['delta_vs_price'] = {'r': r, 'p_value': p, 'n': n,
                                          'significant': p < 0.05, 'x': delta, 'y': price_changes}
        return results

    def collect_votes(self, trend, orderflow, footprint, vp_state, current_price):
        """رأی‌گیری مستقل هر تب: جهت (-1/0/+1)، قدرت (0..1) و دلیل."""
        votes = []

        t_dir = 1 if trend['score'] > 15 else (-1 if trend['score'] < -15 else 0)
        votes.append({'source': 'روند جامع (Overview)', 'direction': t_dir,
                      'strength': min(1.0, abs(trend['score']) / 100), 'detail': trend['direction']})

        if orderflow:
            of_dir = 1 if 'Buy' in orderflow.get('flow_direction', '') else (
                -1 if 'Sell' in orderflow.get('flow_direction', '') else 0)
            votes.append({'source': 'Order Flow', 'direction': of_dir,
                          'strength': min(1.0, orderflow.get('flow_strength', 0)),
                          'detail': orderflow.get('flow_direction', 'Neutral')})

        if footprint:
            bid_ice = sum(1 for h in footprint.get('iceberg_hints', []) if h['side'] == 'bid')
            ask_ice = sum(1 for h in footprint.get('iceberg_hints', []) if h['side'] == 'ask')
            fp_dir = 1 if bid_ice > ask_ice else (-1 if ask_ice > bid_ice else 0)
            votes.append({'source': 'Footprint (Iceberg)', 'direction': fp_dir,
                          'strength': min(1.0, (bid_ice + ask_ice) / 5),
                          'detail': f"{bid_ice} Bid-Iceberg / {ask_ice} Ask-Iceberg"})

        if vp_state and vp_state.get('poc'):
            poc_dev = ((current_price - vp_state['poc']) / vp_state['poc'] * 100) if vp_state['poc'] else 0
            vp_dir = -1 if poc_dev > 1 else (1 if poc_dev < -1 else 0)
            votes.append({'source': 'پروفایل حجم (POC)', 'direction': vp_dir,
                          'strength': min(1.0, abs(poc_dev) / 5),
                          'detail': f"انحراف از POC: {poc_dev:+.2f}%"})

        return votes

    def composite_verdict(self, votes, predictive):
        if not votes:
            return {'score': 0.0, 'label': 'داده ناکافی', 'agreement': 0.0, 'confidence': 'کم', 'n_votes': 0}

        weighted_sum = sum(v['direction'] * v['strength'] for v in votes)
        max_possible = sum(v['strength'] for v in votes) or 1
        score = (weighted_sum / max_possible) * 100

        directions = [v['direction'] for v in votes if v['direction'] != 0]
        agree_ratio = 0.0
        if directions:
            pos = sum(1 for d in directions if d > 0)
            neg = sum(1 for d in directions if d < 0)
            agree_ratio = max(pos, neg) / len(directions)

        sig_count = sum(1 for v in predictive.values() if v.get('significant'))
        if agree_ratio >= 0.75 and sig_count >= 1:
            confidence = 'بالا'
        elif agree_ratio >= 0.6:
            confidence = 'متوسط'
        else:
            confidence = 'کم'

        if score >= 30:
            label = '📈 برایند صعودی'
        elif score <= -30:
            label = '📉 برایند نزولی'
        else:
            label = '➖ برایند خنثی / سیگنال‌های متناقض'

        self.history_composite.append(score)

        return {'score': score, 'label': label, 'agreement': agree_ratio * 100,
                'confidence': confidence, 'n_votes': len(votes)}

@app.callback(
    Output("manual-reference-status", "children"),
    [Input("manual-reference-price-input", "value")]
)
def update_manual_reference_price(value):
    """
    چون در این محیط دسترسی شبکه به بایننس/CoinGecko وجود ندارد، قیمت مرجع جهانی به‌صورت
    دستی از کاربر گرفته می‌شود (مثلاً از سایت بایننس یا هر منبع دیگری که در دسترس است).
    با هر تغییر این عدد، آفست کالیبراسیون در چرخه محاسباتی بعدی بازتنظیم می‌شود و از
    همان‌جا در سراسر اپ (Overview، Footprint، Order Flow، LVN/HVN) اعمال می‌گردد.
    """
    if value is None or value <= 0:
        analyzer.reference_price = None
        analyzer.reference_source = None
        return "قیمت لحظه‌ای بایننس را وارد کنید تا قیمت نوبیتکس با آن هم‌راستا شود."

    analyzer.reference_price = float(value)
    analyzer.reference_source = "دستی (Manual)"
    analyzer._recalibrate_offset = True
    return f"✅ ثبت شد: ${float(value):,.2f} — آفست در چرخه محاسباتی بعدی اعمال می‌شود."

def update_lvn_tab(n):
    if len(analyzer.mid_price_history) < 20:
        return [dash.no_update] * 10

    prices = list(analyzer.mid_price_history)[-100:]
    volumes = list(analyzer.volume_history)[-100:]

    profile = vp_analyzer.calculate(prices, volumes)
    if not profile:
        return [dash.no_update] * 10

    current_price = analyzer.current_stats['mid_price'] if analyzer.current_stats else (prices[-1] if prices else 0)
    if not current_price:
        return [dash.no_update] * 10

    # 1. نمودار پروفایل حجم
    fig_profile = make_subplots(rows=1, cols=2, shared_yaxes=True, horizontal_spacing=0.02,
                                column_widths=[0.35, 0.65],
                                subplot_titles=('📊 پروفایل حجم', '📈 روند قیمت با HVN/LVN'))

    if vp_analyzer.profile:
        vol_levels = sorted(vp_analyzer.profile.keys())
        vol_values = [vp_analyzer.profile[l] for l in vol_levels]

        bar_colors = []
        for level in vol_levels:
            is_hvn = any(abs(h['price'] - level) < 0.001 for h in vp_analyzer.hvn_levels)
            is_lvn = any(abs(lv['price'] - level) < 0.001 for lv in vp_analyzer.lvn_levels)
            if is_hvn:
                bar_colors.append(COLORS['hvn'])
            elif is_lvn:
                bar_colors.append(COLORS['warning'])
            else:
                bar_colors.append(COLORS['accent'])

        fig_profile.add_trace(
            go.Bar(y=vol_levels, x=vol_values, orientation='h', marker_color=bar_colors, name='Volume',
                   showlegend=False), row=1, col=1)

        dist = ((lvn['price'] - current_price) / current_price * 100) if current_price else 0

             html.Th("Distance", style={'color': COLORS['text_muted'], 'padding': '10px', 'textAlign': 'left'})])),
        html.Tbody(lvn_rows if lvn_rows else [html.Tr([html.Td("LVN یافت نشد", colSpan=4,

        cum_vol = np.cumsum(volumes)

@app.callback(
    [Output("syn-verdict-card", "children"),
     Output("syn-agreement-card", "children"),
     Output("syn-significance-card", "children"),
     Output("syn-votes-table", "children"),
     Output("syn-zscore-chart", "figure"),
     Output("syn-composite-history-chart", "figure"),
     Output("syn-imbalance-scatter", "figure"),
     Output("syn-delta-scatter", "figure")],
    [Input("interval-component", "n_intervals")]
)
def update_synthesis_tab(n):
    stats = analyzer.current_stats
    if not stats or len(analyzer.mid_price_history) < 12:
        empty_fig = go.Figure()
        empty_fig.update_layout(
            plot_bgcolor=COLORS['card'], paper_bgcolor=COLORS['card'], font_color=COLORS['text_muted'],
            annotations=[dict(text="در انتظار داده کافی برای تحلیل آماری...", xref="paper", yref="paper",
                              x=0.5, y=0.5, showarrow=False)])
        waiting = html.Div("در انتظار داده کافی...", style={'color': COLORS['text_muted']})
        return waiting, waiting, waiting, waiting, empty_fig, empty_fig, empty_fig, empty_fig

    trend = analyzer.calculate_comprehensive_trend()
    orderflow = analyzer.orderflow_data
    footprint = analyzer.footprint_data
    vp_state = {'poc': vp_analyzer.poc}

    votes = synth_analyzer.collect_votes(trend, orderflow, footprint, vp_state, stats['mid_price'])
    predictive = synth_analyzer.evaluate_predictive_power(
        analyzer.imbalance_history, analyzer.delta_history, analyzer.mid_price_history)
    verdict = synth_analyzer.composite_verdict(votes, predictive)

    verdict_color = COLORS['bid'] if verdict['score'] > 0 else COLORS['ask'] if verdict['score'] < 0 else COLORS['text_muted']
    verdict_card = html.Div([
        html.Div(verdict['label'], style={'color': verdict_color, 'fontWeight': 'bold', 'fontSize': '26px'}),
        html.Div(f"امتیاز ترکیبی: {verdict['score']:+.1f} از 100 | بر اساس {verdict['n_votes']} سیگنال مستقل",
                 style={'color': COLORS['text_muted'], 'fontSize': '12px', 'marginTop': '6px'}),
        html.Div(f"سطح اطمینان: {verdict['confidence']}",
                 style={'color': COLORS['warning'], 'fontSize': '13px', 'marginTop': '6px', 'fontWeight': 'bold'}),
    ])

    agreement_card = html.Div([
        html.Div(f"{verdict['agreement']:.0f}%",
                 style={'color': COLORS['accent'], 'fontWeight': 'bold', 'fontSize': '28px'}),
        html.Div("از سیگنال‌های جهت‌دار هم‌راستا هستند", style={'color': COLORS['text_muted'], 'fontSize': '12px'}),
    ])

    sig_lines = []
    for key, res in predictive.items():
        label = "Imbalance ↔ ΔPrice" if key == 'imbalance_vs_price' else "Delta ↔ ΔPrice"
        sig_color = COLORS['success'] if res['significant'] else COLORS['text_muted']
        sig_lines.append(html.Div([
            html.Span(f"{label}: ", style={'color': COLORS['text_muted'], 'fontSize': '12px'}),
            html.Span(f"r={res['r']:+.3f}, p={res['p_value']:.3f} (n={res['n']}) ",
                      style={'color': sig_color, 'fontSize': '12px', 'fontWeight': 'bold'}),
            html.Span("✓ معنادار" if res['significant'] else "✗ غیرمعنادار",
                      style={'color': sig_color, 'fontSize': '11px'}),
        ], style={'marginBottom': '6px'}))
    if not sig_lines:
        sig_lines = [html.Div("داده کافی برای آزمون همبستگی موجود نیست",
                              style={'color': COLORS['text_muted'], 'fontSize': '12px'})]
    significance_card = html.Div(sig_lines)

    votes_rows = []
    for v in votes:
        dir_label = "🟢 صعودی" if v['direction'] > 0 else ("🔴 نزولی" if v['direction'] < 0 else "⚪ خنثی")
        dir_color = COLORS['bid'] if v['direction'] > 0 else (COLORS['ask'] if v['direction'] < 0 else COLORS['text_muted'])
        votes_rows.append(html.Tr([
            html.Td(v['source'], style={'padding': '8px'}),
            html.Td(dir_label, style={'padding': '8px', 'color': dir_color, 'fontWeight': 'bold'}),
            html.Td(f"{v['strength']:.2f}", style={'padding': '8px'}),
            html.Td(v['detail'], style={'padding': '8px', 'color': COLORS['text_muted'], 'fontSize': '12px'}),
        ], style={'borderBottom': f'1px solid {COLORS["card_border"]}'}))

    votes_table = html.Table([
        html.Thead(html.Tr([
            html.Th("منبع سیگنال (تب)", style={'color': COLORS['text_muted'], 'padding': '10px', 'textAlign': 'left'}),
            html.Th("جهت", style={'color': COLORS['text_muted'], 'padding': '10px', 'textAlign': 'left'}),
            html.Th("قدرت (0-1)", style={'color': COLORS['text_muted'], 'padding': '10px', 'textAlign': 'left'}),
            html.Th("جزئیات", style={'color': COLORS['text_muted'], 'padding': '10px', 'textAlign': 'left'}),
        ])),
        html.Tbody(votes_rows if votes_rows else [html.Tr([html.Td("داده‌ای موجود نیست", colSpan=4,
                                                                   style={'textAlign': 'center', 'padding': '20px',
                                                                          'color': COLORS['text_muted']})])])
    ], style={'width': '100%', 'color': COLORS['text'], 'fontSize': '13px'})

    z_imbalance = synth_analyzer.zscore(stats['imbalance'], analyzer.imbalance_history)
    z_delta = synth_analyzer.zscore(stats['delta'], analyzer.delta_history)
    z_buy_pressure = synth_analyzer.zscore(stats['buy_pressure'], analyzer.buy_pressure_history)
    z_spread = synth_analyzer.zscore(stats['spread_pct'], analyzer.spread_history)

    z_labels = ['Imbalance', 'Delta', 'Buy Pressure', 'Spread %']
    z_values = [z_imbalance, z_delta, z_buy_pressure, z_spread]
    z_colors = [COLORS['bid'] if z > 0 else COLORS['ask'] for z in z_values]

    fig_z = go.Figure()
    fig_z.add_trace(go.Bar(x=z_labels, y=z_values, marker_color=z_colors,
                           text=[f"{z:+.2f}σ" for z in z_values], textposition='outside'))
    fig_z.add_hline(y=2, line_dash="dot", line_color=COLORS['warning'], opacity=0.5)
    fig_z.add_hline(y=-2, line_dash="dot", line_color=COLORS['warning'], opacity=0.5)
    fig_z.update_layout(plot_bgcolor=COLORS['card'], paper_bgcolor=COLORS['card'], font_color=COLORS['text'],
                        title="انحراف از میانگین نشست (به واحد انحراف معیار σ)",
                        yaxis_title="Z-Score", margin=dict(l=40, r=20, t=50, b=40), showlegend=False)

    fig_composite_hist = go.Figure()
    hist_scores = list(synth_analyzer.history_composite)
    if hist_scores:
        colors_hist = [COLORS['bid'] if s > 0 else COLORS['ask'] for s in hist_scores]
        fig_composite_hist.add_trace(go.Scatter(y=hist_scores, mode='lines+markers',
                                                 line=dict(color=COLORS['pink'], width=2),
                                                 marker=dict(size=5, color=colors_hist)))
    fig_composite_hist.add_hline(y=0, line_dash="dash", line_color="#64748b")
    fig_composite_hist.update_layout(plot_bgcolor=COLORS['card'], paper_bgcolor=COLORS['card'],
                                     font_color=COLORS['text'], title="روند امتیاز برایند ترکیبی",
                                     yaxis_title="Score (-100..100)", margin=dict(l=40, r=20, t=50, b=40))

    def make_scatter(res, title, color):
        fig = go.Figure()
        if res:
            x = res['x']
            y = res['y']
            fig.add_trace(go.Scatter(x=x, y=y, mode='markers', marker=dict(color=color, size=7, opacity=0.6)))
            if len(x) >= 2 and statistics.pstdev(x) > 0:
                slope = res['r'] * (statistics.pstdev(y) / statistics.pstdev(x))
                intercept = statistics.mean(y) - slope * statistics.mean(x)
                x_line = [min(x), max(x)]
                y_line = [slope * xi + intercept for xi in x_line]
                fig.add_trace(go.Scatter(x=x_line, y=y_line, mode='lines',
                                         line=dict(color=COLORS['warning'], width=2, dash='dash'), name='خط رگرسیون'))
            sig_txt = "✓ معنادار" if res['significant'] else "✗ غیرمعنادار (احتیاط لازم است)"
            fig.update_layout(
                title=f"{title}<br><sub>r={res['r']:+.3f} | p={res['p_value']:.3f} | n={res['n']} | {sig_txt}</sub>")
        else:
            fig.update_layout(title=title,
                              annotations=[dict(text="داده کافی موجود نیست", xref="paper", yref="paper",
                                                x=0.5, y=0.5, showarrow=False, font=dict(color=COLORS['text_muted']))])
        fig.update_layout(plot_bgcolor=COLORS['card'], paper_bgcolor=COLORS['card'], font_color=COLORS['text'],
                          showlegend=False, margin=dict(l=40, r=20, t=70, b=40))
        return fig

    fig_imb_scatter = make_scatter(predictive.get('imbalance_vs_price'),
                                    "Imbalance(t) در برابر ΔPrice(t+1)", COLORS['accent'])
    fig_delta_scatter = make_scatter(predictive.get('delta_vs_price'),
                                      "Delta(t) در برابر ΔPrice(t+1)", COLORS['purple'])

    return (verdict_card, agreement_card, significance_card, votes_table,
            fig_z, fig_composite_hist, fig_imb_scatter, fig_delta_scatter)
