def fetch_btc_data():
    """œ—Ì«›  ﬁÌ„  »Ì ùòÊÌ‰ »«  «Ì„ù›—Ì„ ? œﬁÌﬁÂ"""
    try:
        btc = yf.download("BTC-USD", period="5d", interval="5m")
        if btc.empty:
            raise ValueError("œ«œÂù«Ì œ—Ì«›  ‰‘œ")
        btc = btc.dropna(subset=['Close'])
        btc.index = btc.index.tz_localize(None)
        return btc
    except Exception as e:
        print(f"?? Œÿ« œ— œ—Ì«›  œ«œÂ: {e}")
        # œ«œÂùÂ«Ì ‘»ÌÂù”«“Ì ‘œÂ »Â ⁄‰Ê«‰ fallback
        dates = pd.date_range(end=datetime.now(), periods=500, freq='5min')
        np.random.seed(42)
        prices = 67000 + np.cumsum(np.random.randn(500) * 150)
        return pd.DataFrame({
            'Open': prices, 'High': prices + np.random.rand(500)*200,
            'Low': prices - np.random.rand(500)*200,
            'Close': prices, 'Volume': np.random.randint(100, 5000, 500)
        }, index=dates)


def calculate_stats(df):
    """„Õ«”»Â ¬„«—Â«Ì ò·ÌœÌ"""
    close = df['Close']
    current = close.iloc[-1]
    prev = close.iloc[-2] if len(close) > 1 else current
    change = current - prev
    change_pct = (change / prev) * 100 if prev != 0 else 0

    return {
        "current_price": current,
        "change": change,
        "change_pct": change_pct,
        "high_5d": close.max(),
        "low_5d": close.min(),
        "volatility": close.pct_change().std() * 100,
        "avg_price": close.mean(),
        "volume_avg": df['Volume'].mean() if 'Volume' in df.columns else 0,
        "rsi_14": calculate_rsi(close, 14),
        "ma_20": close.rolling(20).mean().iloc[-1] if len(close) >= 20 else current,
        "ma_50": close.rolling(50).mean().iloc[-1] if len(close) >= 50 else current,
        "bollinger_upper": close.rolling(20).mean().iloc[-1] + 2 * close.rolling(20).std().iloc[-1] if len(close) >= 20 else current * 1.05,
        "bollinger_lower": close.rolling(20).mean().iloc[-1] - 2 * close.rolling(20).std().iloc[-1] if len(close) >= 20 else current * 0.95,
        "total_points": len(close),
    }


def calculate_rsi(series, period=14):
    """„Õ«”»Â RSI"""
    delta = series.diff()
    gain = delta.where(delta > 0, 0).rolling(period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(period).mean()
    rs = gain / loss
    return 100 - (100 / (1 + rs)).iloc[-1] if not rs.empty else 50

def build_3d_scene(df, frame=0):

    min_p, max_p = close.min(), close.max()
    normalized = (close - min_p) / (max_p - min_p) if max_p != min_p else np.ones_like(close)

    current_price = close[frame % len(close)]
    price_factor = normalized[frame % len(close)]

    fig = go.Figure()

    # ? ”Ì«Âç«·Â / »Ì ùòÊÌ‰ „—ò“Ì
    btc_size = 15 + (current_price / max_p) * 35

        x=[0], y=[0], z=[0],
        mode='markers',
        marker=dict(
            size=btc_size,
            color=[GOLD],
            colorscale=[[0, '#FF8C00'], [0.5, '#FFD700'], [1, '#FFF8DC']],
            opacity=0.95,
            line=dict(color='#FFD700', width=2),
            symbol='diamond',
        ),
        text=[f'? BTC<br>${current_price:,.0f}'],
        hoverinfo='text',
        name='? »Ì ùòÊÌ‰',
    ))

    #  Â«·Â ‰Ê—«‰Ì »Ì ùòÊÌ‰
    theta_halo = np.linspace(0, 2*np.pi, 60)
    halo_r = 0.8
    fig.add_trace(go.Scatter3d(
        x=halo_r * np.cos(theta_halo),
        y=halo_r * np.sin(theta_halo),
        z=np.zeros_like(theta_halo),
        mode='lines',
        line=dict(color='#FFD700', width=3, dash='dot'),
        opacity=0.6,

        showlegend=False,

    # ?? ”Ì«—«

            hovertext=f'{planet["name"]} ({planet["name_en"]})<br>›«’·Â: {r:.2f} AU',

def build_price_chart(df):
    """‰„Êœ«— ﬁÌ„  »Ì ùòÊÌ‰"""
    close = df['Close']
    ma20 = close.rolling(20).mean()
    ma50 = close.rolling(50).mean()

    fig = go.Figure()

    fig.add_trace(go.Scatter(
        x=df.index, y=close,
        mode='lines',
        line=dict(color=GOLD, width=2),
        name='ﬁÌ„ ',
        fill='tozeroy',
        fillcolor='rgba(255,215,0,0.05)',
    ))

    if len(close) >= 20:
        fig.add_trace(go.Scatter(
            x=df.index, y=ma20,
            mode='lines',
            line=dict(color=ACCENT, width=1.5, dash='dash'),
            name='MA20',
        ))

    if len(close) >= 50:
        fig.add_trace(go.Scatter(
            x=df.index, y=ma50,
            mode='lines',
            line=dict(color='#ff6b6b', width=1.5, dash='dash'),
            name='MA50',
        ))

    fig.update_layout(
        paper_bgcolor='rgba(0,0,0,0)',
        plot_bgcolor='rgba(0,0,0,0)',
        font=dict(color=TEXT_COLOR, size=11),
        margin=dict(l=50, r=20, t=30, b=30),
        height=220,
        xaxis=dict(
            showgrid=True, gridcolor='rgba(255,255,255,0.05)',
            zeroline=False, tickfont=dict(size=9),
        ),
        yaxis=dict(
            showgrid=True, gridcolor='rgba(255,255,255,0.05)',
            zeroline=False, tickformat='$,.0f',
            tickfont=dict(size=9),
        ),
        legend=dict(
            orientation='h', yanchor='bottom', y=1.02,
            xanchor='right', x=1, font=dict(size=9),
        ),
        showlegend=True,
    )

    return fig

def build_volume_chart(df):
    if 'Volume' not in df.columns:
        return go.Figure()

    colors = ['#00d4ff' if df['Close'].iloc[i] >= df['Open'].iloc[i]

def build_rsi_chart(df):
    close = df['Close']
    delta = close.diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    rs = gain / loss
    rsi = 100 - (100 / (1 + rs))

    fig = go.Figure()
    fig.add_trace(go.Scatter(
        x=df.index, y=rsi,
        mode='lines',
        line=dict(color='#a855f7', width=1.5),
        name='RSI(14)',
    ))

    # ŒÿÊÿ „—Ã⁄
    fig.add_hline(y=70, line_dash="dash", line_color="#ff4444", opacity=0.5,
                  annotation_text="«‘»«⁄ Œ—Ìœ")
    fig.add_hline(y=30, line_dash="dash", line_color="#00ff88", opacity=0.5,
                  annotation_text="«‘»«⁄ ›—Ê‘")
    fig.add_hrect(y0=30, y1=70, fillcolor="rgba(168,85,247,0.05)", line_width=0)

    fig.update_layout(
        paper_bgcolor='rgba(0,0,0,0)',
        plot_bgcolor='rgba(0,0,0,0)',
        font=dict(color=TEXT_COLOR, size=10),
        margin=dict(l=50, r=20, t=10, b=30),
        height=130,
        xaxis=dict(showgrid=False, zeroline=False, tickfont=dict(size=8)),
        yaxis=dict(showgrid=False, zeroline=False, range=[0, 100],
                   tickfont=dict(size=8)),
        showlegend=False,
    )
    return fig

@app.callback(
    [Output('scene-3d', 'figure'),
     Output('store-frame', 'data'),
     Output('status-text', 'children'),
     Output('interval-3d', 'disabled')],
    [Input('interval-3d', 'n_intervals'),
     Input('btn-toggle', 'n_clicks')],
    [State('store-frame', 'data'),
     State('store-running', 'data'),
     State('store-data', 'data')],
)
def animate_scene(n_intervals, btn_clicks, current_frame, is_running, data_json):
    """«‰Ì„Ì‘‰ ’Õ‰Â ”Âù»⁄œÌ"""
    if data_json is None:
        df = fetch_btc_data()
    else:
        df = pd.read_json(data_json)
        df.index = pd.to_datetime(df.index)

    running = is_running
    if btn_clicks and btn_clicks % 2 == 1:
        running = not is_running

    if not running:
        fig = build_3d_scene(df, current_frame)
        return fig, current_frame, "?? „ Êﬁ›", True

    frame = (current_frame + 1) % max(len(df), 1)
    fig = build_3d_scene(df, frame)

    price = df['Close'].iloc[frame % len(df)]
    status = f"?? ›—Ì„ {frame} | ${price:,.0f} | {datetime.now().strftime('%H:%M:%S')}"

    return fig, frame, status, False
