# ?? TimesFM Model Loader (Singleton)

_model = None
_model_loaded = False
_model_error = None

def get_timesfm_model():
    global _model, _model_loaded, _model_error
    if _model_loaded:
        return _model, _model_error
    try:
        import timesfm
        from timesfm.timesfm_2p5.timesfm_2p5_torch import TimesFM_2p5_200M_torch
        print("? Loading TimesFM 2.5 from HuggingFace...")
        m = TimesFM_2p5_200M_torch.from_pretrained("google/timesfm-2.5-200m-pytorch")
        m.compile(timesfm.ForecastConfig(
            max_context=512,
            max_horizon=48,
            normalize_inputs=True,
            use_continuous_quantile_head=True,
            force_flip_invariance=True,
            infer_is_positive=True,
            fix_quantile_crossing=True,
        ))
        _model = m
        _model_error = None
        print("? TimesFM 2.5 loaded successfully!")
    except Exception as e:
        _model = None
        _model_error = str(e)
        print(f"?? TimesFM load failed: {e}")
    _model_loaded = True
    return _model, _model_error

# ?? Gann Support/Resistance Overlay
# ==========================================
def gann_sr_from_close(close_forecast, atr):
    """
     »œÌ· ÅÌ‘ù»Ì‰Ì close »Â support/resistance »« ›—„Ê· Gann Square of 9
    resistance = (sqrt(close) + 1)^2
    support    = (sqrt(close) - 1)^2
    »« ATR scaling »—«Ì ”ÿÊÕ Ê«ﬁ⁄ù»Ì‰«‰Â
    """
    sq = np.sqrt(np.maximum(close_forecast, 1e-6))
    atr_scale = atr / close_forecast * 0.5
    delta = np.maximum(sq * atr_scale, 0.5)
    resistance = (sq + delta) ** 2
    support = (sq - delta) ** 2
    return resistance, support


def fractal_band(high_fc, low_fc, quantile_low, quantile_high):
    """
    »«‰œÂ«Ì ›—«ò «·Ì «“ quantileùÂ«Ì TimesFM
    quantile_low  = »«‰œ ???  (Õ„«Ì  «Õ „«·«)
    quantile_high = »«‰œ ???  („ﬁ«Ê„  «Õ „«·«)
    """
    phi = 1.6180339887
    R = high_fc - low_fc
    band_up = high_fc + R * (phi - 1) * (quantile_high - 0.5)
    band_dn = low_fc - R * (phi - 1) * (0.5 - quantile_low)
    return band_up, band_dn

# ?? Fallback Predictor (»œÊ‰ TimesFM)
# ==========================================
def statistical_forecast(close_series, n_steps=24):
    """
    ›«·ù»ò ÂÊ‘„‰œ:  —òÌ» EMA + Linear Trend + Volatility cone
    »Â — «“ —Ê‘ ﬁ»·Ì (flat momentum)
    """
    arr = np.array(close_series)
    n = len(arr)
    if n < 10:
        return np.full(n_steps, arr[-1]), np.full(n_steps, arr[-1] * 0.99), np.full(n_steps, arr[-1] * 1.01)

    # Linear trend (OLS —ÊÌ ?? ‰ﬁÿÂ ¬Œ—)
    window = min(48, n)
    x = np.arange(window)
    y = arr[-window:]
    slope, intercept = np.polyfit(x, y, 1)

    # EMA ¬Œ—
    alpha = 2 / (14 + 1)
    ema = arr[-1]
    for v in arr[-14:]:
        ema = alpha * v + (1 - alpha) * ema

    # Volatility
    returns = np.diff(np.log(np.maximum(arr[-window:], 1e-6)))
    vol = np.std(returns) * arr[-1]

    forecasts = []
    lower = []
    upper = []
    for i in range(1, n_steps + 1):
        trend_val = slope * (window + i) + intercept
        ema_val = ema
        blended = 0.4 * trend_val + 0.6 * ema_val
        conf = vol * np.sqrt(i) * 1.645  # ??? CI
        forecasts.append(blended)
        lower.append(blended - conf)
        upper.append(blended + conf)

    return np.array(forecasts), np.array(lower), np.array(upper)

# ?? Core Forecasting Engine
# ==========================================
def run_forecast(df, start_idx, horizon=24):
    model, model_err = get_timesfm_model()
    use_timesfm = (model is not None)

    context_window = min(512, start_idx)
    context_df = df.iloc[start_idx - context_window: start_idx]

    close_series = context_df['close'].values.astype(float)
    high_series = context_df['high'].values.astype(float)
    low_series = context_df['low'].values.astype(float)

    # ATR (14 ò‰œ·)
    atr_window = min(14, len(context_df))
    atr = np.mean(high_series[-atr_window:] - low_series[-atr_window:])
    last_close = close_series[-1]

    # ==========================================
    # „—Õ·Â ?: ÅÌ‘ù»Ì‰Ì close »« TimesFM Ì« Fallback
    # ==========================================
    if use_timesfm:
        try:
            # TimesFM —ÊÌ close° high° low »Â ’Ê—  Ãœ«ê«‰Â
            inputs = [close_series, high_series, low_series]
            point_fc, quantile_fc = model.forecast(horizon=horizon, inputs=inputs)

            close_fc = point_fc[0]   # ÅÌ‘ù»Ì‰Ì close
            high_fc  = point_fc[1]   # ÅÌ‘ù»Ì‰Ì high
            low_fc   = point_fc[2]   # ÅÌ‘ù»Ì‰Ì low

            # quantileùÂ«: [mean, q10, q20, ..., q90] ? ?? „ﬁœ«—
            q10_close = quantile_fc[0, :, 1]   # ??? quantile »—«Ì close
            q90_close = quantile_fc[0, :, 9]   # ??? quantile

            use_timesfm_ok = True
            method = "TimesFM 2.5 ?"
        except Exception as e:
            use_timesfm_ok = False
            method = f"Fallback (TimesFM error: {str(e)[:40]})"
            close_fc, q10_close, q90_close = statistical_forecast(close_series, horizon)
            high_fc = close_fc * (1 + atr / last_close * 0.5)
            low_fc = close_fc * (1 - atr / last_close * 0.5)
    else:
        use_timesfm_ok = False
        method = f"Statistical Fallback (TimesFM unavailable)"
        close_fc, q10_close, q90_close = statistical_forecast(close_series, horizon)
        high_fc = close_fc * (1 + atr / last_close * 0.5)
        low_fc = close_fc * (1 - atr / last_close * 0.5)

    # ==========================================
    # „—Õ·Â ?: Gann S/R Overlay
    # ==========================================
    gann_res, gann_sup = gann_sr_from_close(close_fc, atr)

    # ==========================================
    # „—Õ·Â ?: Fractal Quantile Bands
    # ==========================================
    if use_timesfm and use_timesfm_ok:
        fractal_up, fractal_dn = fractal_band(high_fc, low_fc, q10_close, q90_close)
    else:
        fractal_up = q90_close
        fractal_dn = q10_close

    # ==========================================
    # „—Õ·Â ?: ”«Œ ‰ ò‰œ·ùÂ«Ì ¬Ì‰œÂ
    # ==========================================
    if start_idx < len(df):
        last_dt = df['date'].iloc[start_idx - 1]
    else:
        last_dt = df['date'].iloc[-1]

    #  ‘ŒÌ’ timeframe (›«’·Â »Ì‰ ò‰œ·ùÂ«)
    if len(df) > 1:
        tf_delta = df['date'].iloc[-1] - df['date'].iloc[-2]
    else:
        tf_delta = pd.Timedelta(hours=1)

    future_times = [last_dt + tf_delta * (i + 1) for i in range(horizon)]

    # œ«œÂùÂ«Ì Ê«ﬁ⁄Ì »—«Ì „ﬁ«Ì”Â («êÂ „ÊÃÊœ »Êœ)
    actual_data = []
    for i in range(horizon):
        real_idx = start_idx + i
        if real_idx < len(df):
            actual_data.append({
                'high': df['high'].iloc[real_idx],
                'low': df['low'].iloc[real_idx],
                'close': df['close'].iloc[real_idx],
            })
        else:
            actual_data.append(None)

    results = []
    candles = []

    for i in range(horizon):
        t = future_times[i]
        c_fc = float(close_fc[i])
        h_fc = float(high_fc[i])
        l_fc = float(low_fc[i])
        g_r = float(gann_res[i])
        g_s = float(gann_sup[i])
        f_u = float(fractal_up[i])
        f_d = float(fractal_dn[i])

        # open ò‰œ· = close ﬁ»·Ì
        if i == 0:
            o_fc = last_close
        else:
            o_fc = float(close_fc[i - 1])

        # «ÿ„Ì‰«‰ «“ „‰ÿﬁ OHLC
        h_fc = max(h_fc, c_fc, o_fc)
        l_fc = min(l_fc, c_fc, o_fc)

        candles.append({
            'date': t,
            'open': round(o_fc, 2),
            'high': round(h_fc, 2),
            'low': round(l_fc, 2),
            'close': round(c_fc, 2),
        })

        act = actual_data[i]
        act_h = round(act['high'], 2) if act else '-'
        act_l = round(act['low'], 2) if act else '-'
        act_c = round(act['close'], 2) if act else '-'

        # Œÿ« «êÂ œ«œÂ Ê«ﬁ⁄Ì œ«—Ì„
        if act:
            err_pct = abs(c_fc - act['close']) / act['close'] * 100
            err_str = f"{err_pct:.2f}%"
        else:
            err_str = '-'

        results.append({
            '“„«‰': t.strftime('%Y-%m-%d %H:%M'),
            'ÅÌ‘ù»Ì‰Ì Close': round(c_fc, 2),
            'Gann „ﬁ«Ê„ ': round(g_r, 2),
            'Gann Õ„«Ì ': round(g_s, 2),
            '»«‰œ ›—«ò «· »«·«': round(f_u, 2),
            '»«‰œ ›—«ò «· Å«ÌÌ‰': round(f_d, 2),
            'High Ê«ﬁ⁄Ì': act_h,
            'Low Ê«ﬁ⁄Ì': act_l,
            'Œÿ« %': err_str,
        })

    return {
        'candles': candles,
        'results': results,
        'gann_res': [float(x) for x in gann_res],
        'gann_sup': [float(x) for x in gann_sup],
        'fractal_up': [float(x) for x in fractal_up],
        'fractal_dn': [float(x) for x in fractal_dn],
        'times': [t.strftime('%Y-%m-%d %H:%M') for t in future_times],
        'method': method,
    }
