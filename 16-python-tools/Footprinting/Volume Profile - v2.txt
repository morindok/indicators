import yfinance as yf
import pandas as pd
import numpy as np
import dash
from dash import dcc, html, Input, Output
import plotly.graph_objs as go
from plotly.subplots import make_subplots

app = dash.Dash(__name__, suppress_callback_exceptions=True)


# Fetch Bitcoin 5-minute data
def fetch_btc_data():
    try:
        btc = yf.Ticker("BTC-USD")
        data = btc.history(period="2d", interval="5m")
        data.reset_index(inplace=True)
        data.columns = [col.lower() if col != 'Datetime' else 'datetime' for col in data.columns]
        return data
    except Exception as e:
        print(f"Data fetch error: {e}")
        return pd.DataFrame()


# Advanced Order Flow Analysis
class OrderFlowAnalyzer:
    def __init__(self, df):
        self.df = df.copy()
        self.calculate_all_indicators()

    def calculate_all_indicators(self):
        # 1. Support/Resistance
        self.df['support'] = self.df['low'].rolling(15, min_periods=1).min()
        self.df['resistance'] = self.df['high'].rolling(15, min_periods=1).max()

        # 2. Volume Profile
        self.df['volume_profile'] = self.df['volume'].rolling(20, min_periods=1).sum()

        # 3. Key Levels Detection
        self.df['high_20'] = self.df['high'].rolling(20, min_periods=1).max()
        self.df['low_20'] = self.df['low'].rolling(20, min_periods=1).min()
        self.df['vwap'] = self.calculate_vwap()

        # 4. Absorption Detection
        self.df['absorption_score'] = self.calculate_absorption()

        # 5. Gamma Exposure Zones
        self.df['gamma_zone'] = self.detect_gamma_zones()

        # 6. Buy Zone Detection
        self.df['buy_zone'] = self.detect_buy_zones()

        # 7. Entry Signals
        self.df['entry_signal'] = self.generate_entry_signals()

    def calculate_vwap(self):
        typical_price = (self.df['high'] + self.df['low'] + self.df['close']) / 3
        cumulative_tp = (typical_price * self.df['volume']).cumsum()
        cumulative_vol = self.df['volume'].cumsum().replace(0, np.nan)
        return cumulative_tp / cumulative_vol

    def calculate_absorption(self):
        volume_ma = self.df['volume'].rolling(10, min_periods=1).mean()
        price_range = self.df['high'] - self.df['low']
        range_ma = price_range.rolling(10, min_periods=1).mean().replace(0, np.nan)
        absorption = (self.df['volume'] > 2 * volume_ma) & (price_range < 0.5 * range_ma)
        return absorption.astype(int) * 100

    def detect_gamma_zones(self):
        zones = pd.Series(0, index=self.df.index)
        volume_nodes = self.df['volume'] > self.df['volume'].quantile(0.8)
        consolidation = (self.df['high'] - self.df['low']) < (
                    self.df['high'].rolling(10, min_periods=1).mean() - self.df['low'].rolling(10,
                                                                                               min_periods=1).mean()) * 0.3
        zones[volume_nodes & consolidation] = 1
        return zones

    def detect_buy_zones(self):
        zones = pd.Series(False, index=self.df.index)
        strong_absorption = self.df['absorption_score'] > 50
        near_support = (self.df['low'] - self.df['support']) < (self.df['high'] - self.df['low'] + 1e-9) * 0.2
        bullish = self.df['close'] > self.df['open']
        zones[strong_absorption & near_support & bullish] = True
        return zones

    def generate_entry_signals(self):
        signals = pd.Series(0, index=self.df.index)
        entry_condition = (
                (self.df['buy_zone'] == True) &
                (self.df['absorption_score'] > 30) &
                (self.df['close'] > self.df['vwap']) &
                (self.df['volume'] > self.df['volume'].rolling(10, min_periods=1).mean() * 1.5)
        )
        signals[entry_condition] = 1
        return signals


# Create comprehensive trading chart
def create_full_chart(df):
    if df.empty:
        return go.Figure()

    analyzer = OrderFlowAnalyzer(df)
    df = analyzer.df

    fig = make_subplots(
        rows=4, cols=2,
        specs=[
            [{"rowspan": 2, "colspan": 1}, {"type": "bar"}],
            [None, {"type": "bar"}],
            [{"colspan": 2}, None],
            [{"colspan": 2}, None],
        ],
        subplot_titles=(
            'Price Action with Buy Zones', 'Volume Profile',
            'Order Flow DOM', 'Footprint Chart',
            'VWAP & Key Levels', 'Absorption & Entry Signals'
        ),
        vertical_spacing=0.08,
        horizontal_spacing=0.05,
        row_heights=[0.35, 0.35, 0.15, 0.15]
    )

    # 1. Main Price Chart
    fig.add_trace(
        go.Candlestick(
            x=df['datetime'], open=df['open'], high=df['high'], low=df['low'], close=df['close'],
            increasing_line_color='rgb(0, 255, 0)', decreasing_line_color='rgb(128, 0, 128)',
            increasing_fillcolor='rgb(0, 255, 0)', decreasing_fillcolor='rgb(128, 0, 128)',
            name='BTC/USD'
        ), row=1, col=1
    )

    # 2. Buy Zones
    buy_zones = df[df['buy_zone'] == True]
    if not buy_zones.empty:
        for idx, row in buy_zones.iterrows():
            fig.add_shape(
                type="rect", x0=row['datetime'] - pd.Timedelta(minutes=5), x1=row['datetime'] + pd.Timedelta(minutes=5),
                y0=row['low'] * 0.998, y1=row['low'] * 1.005,
                fillcolor="rgba(0, 255, 0, 0.2)", line=dict(color="green", width=2), row=1, col=1
            )

    # 3. VWAP, Support, Resistance
    fig.add_trace(go.Scatter(x=df['datetime'], y=df['vwap'], mode='lines', name='VWAP',
                             line=dict(color='yellow', width=2, dash='dash')), row=1, col=1)
    fig.add_trace(go.Scatter(x=df['datetime'], y=df['support'], mode='lines', name='Support',
                             line=dict(color='cyan', width=1.5, dash='dot')), row=1, col=1)
    fig.add_trace(go.Scatter(x=df['datetime'], y=df['resistance'], mode='lines', name='Resistance',
                             line=dict(color='red', width=1.5, dash='dot')), row=1, col=1)

    # 4. Entry & Absorption Markers
    entry_signals = df[df['entry_signal'] == 1]
    if not entry_signals.empty:
        fig.add_trace(
            go.Scatter(x=entry_signals['datetime'], y=entry_signals['low'] * 0.995, mode='markers', name='Entry',
                       marker=dict(symbol='circle', size=12, color='rgb(128, 0, 128)',
                                   line=dict(color='white', width=2))), row=1, col=1)

    absorption = df[df['absorption_score'] > 30]
    if not absorption.empty:
        fig.add_trace(
            go.Scatter(x=absorption['datetime'], y=absorption['high'] * 1.002, mode='markers', name='Absorption',
                       marker=dict(symbol='circle', size=8, color='rgb(0, 255, 0)')), row=1, col=1)

    # 5. Volume Profile (FIXED)
    min_price = df['low'].min()
    max_price = df['high'].max()
    if min_price == max_price:
        max_price += 1  # Prevent linspace error if all prices are the same

    bins = np.linspace(min_price, max_price, 30)
    df['price_bin'] = pd.cut(df['close'], bins, include_lowest=True)
    volume_profile = df.groupby('price_bin', observed=True)['volume'].sum().dropna()

    if not volume_profile.empty:
        # Calculate midpoints safely for all pandas versions
        midpoints = [interval.mid for interval in volume_profile.index]
        fig.add_trace(
            go.Bar(y=midpoints, x=volume_profile.values, orientation='h', marker_color='rgba(255, 165, 0, 0.6)',
                   showlegend=False), row=1, col=2)

    # 6. DOM
    last_price = df['close'].iloc[-1]
    bid_levels = np.linspace(last_price * 0.998, last_price, 10)
    ask_levels = np.linspace(last_price, last_price * 1.002, 10)
    recent_vol = df['volume'].tail(10).values
    bid_sizes = np.random.randint(recent_vol.min(), recent_vol.max() + 1, 10) if len(
        recent_vol) > 0 else np.random.randint(10, 100, 10)
    ask_sizes = np.random.randint(recent_vol.min(), recent_vol.max() + 1, 10) if len(
        recent_vol) > 0 else np.random.randint(10, 100, 10)

    fig.add_trace(
        go.Bar(y=bid_levels, x=bid_sizes, orientation='h', marker_color='rgba(0, 255, 0, 0.7)', showlegend=False),
        row=2, col=2)
    fig.add_trace(
        go.Bar(y=ask_levels, x=ask_sizes, orientation='h', marker_color='rgba(255, 0, 0, 0.7)', showlegend=False),
        row=2, col=2)

    # 7. Footprint Chart (last 20 candles)
    recent_df = df.tail(20)
    fig.add_trace(
        go.Scatter(
            x=recent_df['datetime'], y=recent_df['close'], mode='markers+text',
            text=recent_df['volume'].astype(int).astype(str), textposition='middle right',
            marker=dict(size=15, color=['rgba(128, 0, 128, 0.8)' if c < o else 'rgba(0, 255, 0, 0.8)' for c, o in
                                        zip(recent_df['close'], recent_df['open'])], symbol='square'),
            name='Footprint', showlegend=False
        ), row=2, col=2
    )

    # 8. Absorption Score
    fig.add_trace(go.Scatter(x=df['datetime'], y=df['absorption_score'], mode='lines', name='Absorption Score',
                             line=dict(color='orange', width=2), fill='tozeroy', fillcolor='rgba(255, 165, 0, 0.2)'),
                  row=3, col=1)

    # 9. Entry Signals Chart
    fig.add_trace(go.Scatter(x=df['datetime'], y=df['entry_signal'] * 100, mode='markers', name='Signals',
                             marker=dict(symbol='triangle-up', size=15, color='rgb(0, 255, 0)')), row=4, col=1)

    fig.update_layout(
        title='Bitcoin Order Flow Trading System', template='plotly_dark', height=1200,
        paper_bgcolor='rgb(20, 20, 20)', plot_bgcolor='rgb(20, 20, 20)', font=dict(color='white'),
        legend=dict(orientation="h", yanchor="bottom", y=1.02, xanchor="right", x=1)
    )

    return fig


app.layout = html.Div([
    html.H1('Bitcoin Order Flow Trading System', style={'textAlign': 'center', 'color': 'white'}),
    html.Div(id='market-info', style={'textAlign': 'center', 'color': 'yellow', 'marginBottom': '20px'}),
    dcc.Graph(id='trading-chart', style={'height': '800px'}),
    dcc.Interval(id='interval-component', interval=60 * 1000, n_intervals=0)
], style={'backgroundColor': 'rgb(20, 20, 20)', 'padding': '20px'})


@app.callback(
    [Output('trading-chart', 'figure'), Output('market-info', 'children')],
    Input('interval-component', 'n_intervals')
)
def update_dashboard(n):
    df = fetch_btc_data()
    if df.empty:
        return go.Figure(), "Failed to fetch data. Check internet connection."

    fig = create_full_chart(df)
    last_close = df['close'].iloc[-1]
    return fig, f"Current BTC Price: ${last_close:,.2f} | 5-Minute Timeframe"


if __name__ == '__main__':
    app.run(debug=True, port=8050)
