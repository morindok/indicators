def fetch_ohlcv():
    """œ—Ì«›  œ«œÂùÌ ﬁÌ„  Ê ÕÃ„ »Ì ùòÊÌ‰ «“ Ì«ÂÊ ›«Ì‰‰”."""
    df = yf.download(
        tickers=TICKER,
        period=PERIOD,
        interval=INTERVAL,
        progress=False,
        auto_adjust=True,
    )

    if df is None or df.empty:
        raise ValueError("œ«œÂù«Ì «“ Yahoo Finance œ—Ì«›  ‰‘œ")

    # yfinance ê«ÂÌ ” Ê‰ùÂ«Ì ç‰œ”ÿÕÌ »—„Ìùê—œ«‰œ
    if isinstance(df.columns, pd.MultiIndex):
        df.columns = df.columns.get_level_values(0)

    df = df.reset_index()
    time_col = "Datetime" if "Datetime" in df.columns else "Date"

    out = pd.DataFrame({
        "time": pd.to_datetime(df[time_col]),
        "price": df["Close"].astype(float),
        "volume": df["Volume"].astype(float),
    }).dropna()

    out = out.sort_values("time").reset_index(drop=True)

    if out.empty:
        raise ValueError("Å” «“ Å«òù”«“Ì° œ«œÂùÌ „⁄ »—Ì »«ﬁÌ ‰„«‰œ")

    return out

def classify_rolling_volume_price_trend(df, window=TREND_WINDOW):
    """
    »—«Ì Â— ‰ﬁÿÂ° »« «” ›«œÂ «“ ¬Œ—Ì‰ `window` ‰ﬁÿÂ° —«»ÿÂùÌ ÕÃ„ Ê ﬁÌ„  —«
    „Ìù”‰Ãœ (‰—„«·ù”«“Ì z-score° ‘Ì» —ê—”ÌÊ‰ ŒÿÌ Ê ÷—Ì» Â„»” êÌ) Ê »—ç”»
    bull / bear / neutral  Ê·Ìœ „Ìùò‰œ.
    """
    n = len(df)
    labels = ["neutral"] * n
    corr_values = [0.0] * n

    volumes = df["volume"].values
    prices = df["price"].values

    for i in range(n):
        start = max(0, i - window + 1)
        x = volumes[start:i + 1]
        y = prices[start:i + 1]

        if len(x) < 4:
            continue

        x_std = x.std()
        y_std = y.std()

        if x_std == 0 or y_std == 0:
            continue

        x_norm = (x - x.mean()) / x_std
        y_norm = (y - y.mean()) / y_std

        slope = np.polyfit(x_norm, y_norm, 1)[0]
        corr = float(np.corrcoef(x_norm, y_norm)[0, 1])
        price_change_pct = ((y[-1] - y[0]) / y[0]) * 100 if y[0] else 0.0

        corr_values[i] = corr

        if slope > 0.2 and corr > 0.2 and price_change_pct > 0:
            labels[i] = "bull"
        elif slope < -0.2 and corr < -0.2 and price_change_pct < 0:
            labels[i] = "bear"
        else:
            labels[i] = "neutral"

    return labels, corr_values

def build_statistical_analysis(df):
    prices = df["price"].values
    volumes = df["volume"].values

    last_price = prices[-1]
    first_price = prices[0]
    max_price = prices.max()
    min_price = prices.min()
    avg_price = prices.mean()
    avg_volume = volumes.mean()
    price_std = prices.std()
    volatility_pct = (price_std / avg_price) * 100 if avg_price else 0.0
    price_change_pct = ((last_price - first_price) / first_price) * 100 if first_price else 0.0

    recent_prices = prices[-5:]
    short_momentum = (
        ((recent_prices[-1] - recent_prices[0]) / recent_prices[0]) * 100
        if len(recent_prices) >= 2 and recent_prices[0] else 0.0
    )

    bullish_volume = 0.0
    bearish_volume = 0.0
    for i in range(1, len(df)):
        if prices[i] >= prices[i - 1]:
            bullish_volume += volumes[i]
        else:
            bearish_volume += volumes[i]

    total_dir_volume = bullish_volume + bearish_volume
    volume_bias = ((bullish_volume - bearish_volume) / total_dir_volume) * 100 if total_dir_volume else 0.0

    if volumes.std() != 0 and prices.std() != 0:
        price_volume_corr = float(np.corrcoef(volumes, prices)[0, 1])
    else:
        price_volume_corr = 0.0

    range_pct = ((max_price - min_price) / avg_price) * 100 if avg_price else 0.0

    recommendation = "‰êÂœ«—Ì / ’»—"
    signal_class = "wait"
    summary = "»«“«— œ— Ê÷⁄Ì  Œ‰ÀÌ ﬁ—«— œ«—œ Ê »Â — «”   «  «ÌÌœ »Ì‘ — —Ê‰œ° »« «Õ Ì«ÿ ⁄„· ‘Êœ."

    if short_momentum > 0.35 and price_volume_corr > 0.2 and volume_bias > 8 and price_change_pct > 0:
        recommendation = "ÅÌ‘‰Â«œ ¬„«—Ì: Œ—Ìœ òÊ «Âù„œ "
        signal_class = "buy"
        summary = "„Ê„‰ Ê„ òÊ «Âù„œ  „À»  «” ° ÕÃ„ ’⁄ÊœÌ »— ÕÃ„ ‰“Ê·Ì €·»Â œ«—œ Ê Â„»” êÌ ﬁÌ„  Ê ÕÃ„ ‰‘«‰Â Õ„«Ì  «“ —Ê‰œ ’⁄ÊœÌ «” ."
    elif short_momentum < -0.35 and price_volume_corr < -0.1 and volume_bias < -8 and price_change_pct < 0:
        recommendation = "ÅÌ‘‰Â«œ ¬„«—Ì: ›—Ê‘ / ò«Â‘ —Ì”ò"
        signal_class = "sell"
        summary = "›‘«— ›—Ê‘ œ— œ«œÂùÂ«Ì «ŒÌ— »Ì‘ — «” ° ÕÃ„ ‰“Ê·Ì €«·» ‘œÂ Ê ”«Œ «— òÊ «Âù„œ  «“ ÷⁄› ﬁÌ„  Õ„«Ì  „Ìùò‰œ."
    elif volatility_pct > 1.8 and abs(short_momentum) < 0.25:
        recommendation = "ÅÌ‘‰Â«œ ¬„«—Ì: ’»— »—«Ì ‘ò” "
        signal_class = "wait"
        summary = "‰Ê”«‰ »«·«”  «„« ÃÂ  —Ê‰œ Ê«÷Õ ‰Ì” ∫ Ê—Êœ ⁄ÃÊ·«‰Â „Ìù Ê«‰œ —Ì”òùÅ–Ì— »«‘œ."

    stats = [
        f"»Ì‘ —Ì‰ ﬁÌ„ : {fmt_currency(max_price)}",
        f"ò„ —Ì‰ ﬁÌ„ : {fmt_currency(min_price)}",
        f"„Ì«‰êÌ‰ ﬁÌ„ : {fmt_currency(avg_price)}",
        f"«‰Õ—«› „⁄Ì«— ﬁÌ„ : {fmt_currency(price_std)}",
        f"œ«„‰Â ‰Ê”«‰ ‰”»Ì: {range_pct:.2f}%",
        f"„Ê„‰ Ê„ 5 ‰ﬁÿÂ ¬Œ—: {short_momentum:.2f}%",
        f"„Ì«‰êÌ‰ ÕÃ„: {fmt_number(avg_volume)}",
        f"ÕÃ„ ’⁄ÊœÌ / ‰“Ê·Ì: {fmt_number(bullish_volume)} / {fmt_number(bearish_volume)}",
        f"”ÊêÌ—Ì ÕÃ„: {volume_bias:.2f}%",
        f"Â„»” êÌ ﬁÌ„  Ê ÕÃ„: {price_volume_corr:.2f}",
    ]

    return {
        "stats": stats,
        "recommendation": recommendation,
        "signal_class": signal_class,
        "summary": summary,
        "metrics": {
            "last_price": last_price,
            "price_change_pct": price_change_pct,
            "avg_volume": avg_volume,
            "volatility_pct": volatility_pct,
        },
    }


def analyze_middle_view_trend(df, window=TREND_WINDOW):
    """„ ‰ Œ·«’ÂùÌ ‰Ê«— —Ê‰œ° »— «”«” ¬Œ—Ì‰ Å‰Ã—Â «“ ‰„«Ì ÕÃ„-ﬁÌ„ ."""
    if len(df) < 4:
        return "œ«œÂ ò«›Ì »—«Ì  Õ·Ì· ÊÃÊœ ‰œ«—œ.", COLOR_NEUTRAL

    labels, corrs = classify_rolling_volume_price_trend(df, window=window)
    last_label = labels[-1]
    last_corr = corrs[-1]

    recent = df.tail(min(window, len(df)))
    price_change_pct = (
        ((recent["price"].iloc[-1] - recent["price"].iloc[0]) / recent["price"].iloc[0]) * 100
        if recent["price"].iloc[0] else 0.0
    )

    word = {"bull": "’⁄ÊœÌ", "bear": "‰“Ê·Ì", "neutral": "Œ‰ÀÌ"}[last_label]
    text = f" Õ·Ì· ‰„«Ì ÕÃ„-ﬁÌ„ : {word} | Â„»” êÌ: {last_corr:.2f} |  €ÌÌ— ﬁÌ„ : {price_change_pct:.2f}%"
    return text, LABEL_COLOR[last_label]

def fmt_currency(value):
    return f"${value:,.2f}"


def fmt_number(value):
    return f"{value:,.0f}"


def get_camera(view_mode):
    if view_mode == "volume_price":
        return dict(eye=dict(x=0.05, y=2.8, z=0.35))
    if view_mode == "time_price_volume":
        return dict(eye=dict(x=2.6, y=0.8, z=1.7))
    return dict(eye=dict(x=1.6, y=1.6, z=1.2))


def get_volume_direction_colors(df):
    prices = df["price"].values
    colors = [COLOR_NEUTRAL]
    for i in range(1, len(prices)):
        colors.append(COLOR_BULL if prices[i] >= prices[i - 1] else COLOR_BEAR)
    return colors

def build_3d_figure(df, view_mode, source_name="Yahoo Finance"):
    title_map = {
        "default": "Bitcoin 3D View: Time - Price - Volume",
        "volume_price": "Bitcoin 3D View: Volume - Price",
        "time_price_volume": "Bitcoin 3D View: Time - Price - Volume Angle",
    }

    hover_text = [
        f"Time: {t.strftime('%Y-%m-%d %H:%M')}<br>Price: {fmt_currency(p)}<br>Volume: {fmt_number(v)}"
        for t, p, v in zip(df["time"], df["price"], df["volume"])
    ]

    fig = go.Figure(data=[go.Scatter3d(
        x=df["time"], y=df["price"], z=df["volume"],
        mode="lines+markers",
        marker=dict(size=5, color=df["price"], colorscale="Viridis", colorbar=dict(title="Price")),
        line=dict(color="cyan", width=4),
        name="BTC 3D",
        text=hover_text,
        hoverinfo="text",
    )])

    fig.update_layout(
        **DARK_LAYOUT,
        title=f"{title_map.get(view_mode, title_map['default'])} ({source_name})",
        scene=dict(
            xaxis=dict(title="Time", gridcolor="rgba(255,255,255,0.08)"),
            yaxis=dict(title="Price (USD)", gridcolor="rgba(255,255,255,0.08)"),
            zaxis=dict(title="Volume", gridcolor="rgba(255,255,255,0.08)"),
            camera=get_camera(view_mode),
            bgcolor="rgba(0,0,0,0)",
        ),
        margin=dict(l=0, r=0, b=0, t=50),
        height=620,
    )
    return fig


def build_colored_price_time_figure(df, labels, source_name="Yahoo Finance"):
    """
    ‰„Êœ«— ´ﬁÌ„ -“„«‰ª »« „”Ì— —‰êÌ: Â— »Œ‘ «“ Œÿ »— «”«” ‰ ÌÃÂùÌ  Õ·Ì·
    €·ÿ«‰ ‰„«Ì ÕÃ„-ﬁÌ„  (”»“/ﬁ—„“/“—œ) —‰ê „ÌùêÌ—œ.
    """
    fig = go.Figure()
    times = df["time"].tolist()
    prices = df["price"].tolist()

    shown_legend = set()
    for i in range(1, len(df)):
        label = labels[i]
        color = LABEL_COLOR[label]
        show_legend = label not in shown_legend
        shown_legend.add(label)

        fig.add_trace(go.Scatter(
            x=[times[i - 1], times[i]],
            y=[prices[i - 1], prices[i]],
            mode="lines",
            line=dict(color=color, width=3.5),
            legendgroup=label,
            name=LABEL_TEXT[label],
            showlegend=show_legend,
            hoverinfo="skip",
        ))

    hover_text = [
        f"Time: {t.strftime('%Y-%m-%d %H:%M')}<br>Price: {fmt_currency(p)}<br>Ê÷⁄Ì : {LABEL_TEXT[l]}"
        for t, p, l in zip(times, prices, labels)
    ]

    fig.add_trace(go.Scatter(
        x=times, y=prices,
        mode="markers",
        marker=dict(
            size=7,
            color=[LABEL_COLOR[l] for l in labels],
            line=dict(color="rgba(255,255,255,0.3)", width=1),
        ),
        name="‰ﬁ«ÿ ﬁÌ„ ",
        showlegend=False,
        text=hover_text,
        hoverinfo="text",
    ))

    fig.update_layout(
        **DARK_LAYOUT,
        title=f"2D View: Price vs Time - —‰êù¬„Ì“Ì »— «”«”  Õ·Ì· ÕÃ„-ﬁÌ„  ({source_name})",

def build_time_price_volume_2d_figure(df, source_name="Yahoo Finance"):
    volumes = df["volume"].values
    min_v, max_v = volumes.min(), volumes.max()
    colors = get_volume_direction_colors(df)

    if max_v - min_v == 0:
        marker_size = [12] * len(df)
    else:
        marker_size = [9 + 24 * ((v - min_v) / (max_v - min_v)) for v in volumes]

    prices = df["price"].values
    hover_text = []
    for i, (t, p, v) in enumerate(zip(df["time"], prices, volumes)):
        direction = "Œ‰ÀÌ" if i == 0 else ("ÕÃ„ ’⁄ÊœÌ" if prices[i] >= prices[i - 1] else "ÕÃ„ ‰“Ê·Ì")
        hover_text.append(f"Time: {t.strftime('%Y-%m-%d %H:%M')}<br>Price: {fmt_currency(p)}<br>Volume: {fmt_number(v)}<br>Status: {direction}")

    fig = go.Figure(data=[go.Scatter(
        x=df["time"], y=df["price"],
        mode="lines+markers",
        line=dict(color="#9ecbff", width=2.5),
        marker=dict(size=marker_size, color=colors, opacity=0.9, line=dict(color="rgba(255,255,255,0.25)", width=1)),
        name="Time / Price / Volume",
        text=hover_text,
        hoverinfo="text",
    )])

    fig.update_layout(
        **DARK_LAYOUT,
        title=f"2D View: Time - Price - Volume ({source_name})",
        xaxis=dict(title="Time", gridcolor="rgba(255,255,255,0.08)"),
        yaxis=dict(title="Price (USD)", gridcolor="rgba(255,255,255,0.08)"),
        height=380,
        margin=dict(l=40, r=20, b=40, t=50),
    )
    return fig
