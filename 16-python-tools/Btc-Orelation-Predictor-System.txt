class CelestialBody:
    def __init__(self, name, mass, orbital_radius, orbital_period, color, size, influence_factor=1.0):
        self.name = name
        self.mass = mass
        self.orbital_radius = orbital_radius
        self.orbital_period = orbital_period
        self.color = color
        self.size = size
        self.influence_factor = influence_factor

        self.angle = np.random.uniform(0, 2*np.pi)
        self.x = orbital_radius * np.cos(self.angle)
        self.y = orbital_radius * np.sin(self.angle)
        self.angular_velocity = 2 * np.pi / orbital_period

    def update_position(self, time_step):
        self.angle += self.angular_velocity * time_step
        self.x = self.orbital_radius * np.cos(self.angle)
        self.y = self.orbital_radius * np.sin(self.angle)
        return self.x, self.y

    def get_gravitational_force(self, other_x, other_y):
        dx = self.x - other_x
        dy = self.y - other_y
        distance = np.sqrt(dx**2 + dy**2)
        force = (self.mass * self.influence_factor) / (distance**2 + 0.1)
        return force, dx/distance, dy/distance


class BlackHole:
    def __init__(self, mass=5000.0, schwarzschild_radius=0.5):
        self.mass = mass
        self.rs = schwarzschild_radius
        self.x = 0.0
        self.y = 0.0

    def get_gravitational_force(self, x, y):
        r = np.sqrt(x**2 + y**2)
        if r < self.rs:
            force = self.mass / (self.rs**2) * (1 + (self.rs - r)*10)
        else:
            force = self.mass / (r**2)
        lensing_factor = 1 + (self.rs / (r + 0.1))**2
        return force * lensing_factor, -x/r, -y/r

    def get_time_dilation(self, x, y):
        """Returns SCALAR time dilation factor - NO INDEXING NEEDED"""
        r = np.sqrt(x**2 + y**2)
        return float(np.sqrt(1 - self.rs / (r + self.rs + 0.1)))


class EarthBitcoin:
    def __init__(self, initial_price, initial_velocity=0.0):
        self.price = initial_price
        self.velocity = initial_velocity
        self.acceleration = 0.0
        self.x = 1.0
        self.y = 0.0
        self.angle = 0.0
        self.mass = 1.0

        self.price_history = [initial_price]
        self.velocity_history = [initial_velocity]
        self.position_history = [(1.0, 0.0)]

    def update(self, forces, dt, time_dilation=1.0, actual_price=None):
        total_force = 0
        for force, dir_x, dir_y in forces:
            total_force += force

        # Normalize force to prevent explosion
        self.acceleration = np.tanh(total_force * 0.001)
        self.velocity = 0.9 * self.velocity + 0.1 * self.acceleration

        # Price change as percentage return
        price_return = self.velocity * 0.001 * time_dilation
        cosmic_harmonic = np.sin(self.angle * 13) * 0.0002

        new_price = self.price * (1 + price_return + cosmic_harmonic)

        # Blend with actual price if available (calibration)
        if actual_price is not None:
            confidence = np.exp(-abs(self.acceleration) * 10)
            self.price = confidence * new_price + (1 - confidence) * actual_price
        else:
            self.price = new_price

        self.price = max(self.price, 1.0)

        self.angle += 0.005 * dt * time_dilation
        self.x = np.cos(self.angle)
        self.y = np.sin(self.angle)

        self.price_history.append(self.price)
        self.velocity_history.append(self.velocity)
        self.position_history.append((self.x, self.y))

        return self.price

def predict_future(n_future=10, n_iterations=10):
    all_predictions = np.zeros((n_iterations, n_future))
    all_confidence = np.zeros((n_iterations, n_future))

    current_price = closes[-1]

    for iteration in range(n_iterations):
        future_prices = np.zeros(n_future)
        confidence = np.zeros(n_future)

        temp_price = current_price
        temp_velocity = earth.velocity
        temp_angle = earth.angle

        saved_angles = [p.angle for p in all_planets]

        for f in range(n_future):
            for planet in all_planets:
                planet.update_position(dt)

            forces = []
            bh_force, bh_dx, bh_dy = sagittarius_a.get_gravitational_force(
                np.cos(temp_angle), np.sin(temp_angle))
            forces.append((bh_force, bh_dx, bh_dy))

            for planet in all_planets:
                p_force, p_dx, p_dy = planet.get_gravitational_force(
                    np.cos(temp_angle), np.sin(temp_angle))
                forces.append((p_force, p_dx, p_dy))

            total_force = sum([f[0] for f in forces])

            harmonic_component = np.sin(temp_angle * 13) * 0.0002
            gravity_component = np.tanh(total_force * 0.001) * 0.001
            noise = np.random.normal(0, np.std(np.diff(closes[-100:])) / closes[-1] * 0.3)

            price_return = harmonic_component + gravity_component + noise
            temp_price *= (1 + price_return)
            temp_velocity = np.tanh(total_force * 0.001)

            # FIX: get_time_dilation returns SCALAR, no [0] indexing!
            td = sagittarius_a.get_time_dilation(np.cos(temp_angle), np.sin(temp_angle))
            temp_angle += 0.005 * dt * td

            alignment = np.mean([np.cos(p.angle) for p in all_planets])
            confidence[f] = abs(alignment)
            future_prices[f] = temp_price

        for i, planet in enumerate(all_planets):
            planet.angle = saved_angles[i]

        all_predictions[iteration] = future_prices
        all_confidence[iteration] = confidence

    mean_prediction = np.mean(all_predictions, axis=0)
    std_prediction = np.std(all_predictions, axis=0)
    ci = 1.96 * std_prediction

    return mean_prediction, ci, all_predictions, all_confidence

mean_pred, ci, all_preds, all_conf = predict_future(10, 10)
