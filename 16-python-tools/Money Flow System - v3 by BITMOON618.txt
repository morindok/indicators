class CryptoDataManager:
    """Manages real-time crypto data fetching and analysis"""

    def __init__(self):
        self.data_cache = {}
        self.volume_history = {}
        self.money_flow_history = {}
        self.last_update = None
        self.update_lock = threading.Lock()

    def fetch_data(self, ticker, period='1mo', interval='5m'):
        """Fetch data from Yahoo Finance"""
        try:
            df = yf.download(ticker, period=period, interval=interval, progress=False)
            if df.empty:
                return None
            df = df.reset_index()
            df.columns = ['Date', 'Open', 'High', 'Low', 'Close', 'Volume']
            df['Ticker'] = ticker.replace('-USD', '')
            return df
        except Exception as e:
            print(f"Error fetching {ticker}: {e}")
            return None

    def calculate_indicators(self, df):
        """Calculate technical indicators for money flow analysis"""
        if df is None or df.empty:
            return None

        df = df.copy()

        # Volume indicators
        df['Volume_MA_20'] = df['Volume'].rolling(window=20).mean()
        df['Volume_MA_50'] = df['Volume'].rolling(window=50).mean()
        df['Volume_Ratio'] = df['Volume'] / (df['Volume_MA_20'] + 1e-9)
        df['Volume_Surge'] = df['Volume_Ratio'] > 2.0

        # Price indicators
        df['Price_Change'] = df['Close'].pct_change() * 100
        df['Price_MA_20'] = df['Close'].rolling(window=20).mean()
        df['Price_MA_50'] = df['Close'].rolling(window=50).mean()

        # Money Flow Index (MFI)
        typical_price = (df['High'] + df['Low'] + df['Close']) / 3
        raw_money_flow = typical_price * df['Volume']

        money_flow_positive = np.where(typical_price > typical_price.shift(1),
                                       raw_money_flow, 0)
        money_flow_negative = np.where(typical_price < typical_price.shift(1),
                                       raw_money_flow, 0)

        period = 14
        positive_sum = pd.Series(money_flow_positive).rolling(window=period).sum()
        negative_sum = pd.Series(money_flow_negative).rolling(window=period).sum()

        money_ratio = positive_sum / (negative_sum + 1e-9)
        df['MFI'] = 100 - (100 / (1 + money_ratio))

        # On-Balance Volume (OBV)
        df['OBV'] = np.where(df['Close'] > df['Close'].shift(1), df['Volume'],
                             np.where(df['Close'] < df['Close'].shift(1), -df['Volume'], 0)).cumsum()

        # Accumulation/Distribution Line (ADL)
        money_flow_multiplier = ((df['Close'] - df['Low']) - (df['High'] - df['Close'])) / (
                    df['High'] - df['Low'] + 1e-9)
        money_flow_volume = money_flow_multiplier * df['Volume']
        df['ADL'] = money_flow_volume.cumsum()

        # VWAP
        df['VWAP'] = (typical_price * df['Volume']).cumsum() / (df['Volume'].cumsum() + 1e-9)

        # Bollinger Bands
        df['BB_Upper'] = df['Price_MA_20'] + (df['Close'].rolling(window=20).std() * 2)
        df['BB_Lower'] = df['Price_MA_20'] - (df['Close'].rolling(window=20).std() * 2)

        # RSI
        delta = df['Close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
        rs = gain / (loss + 1e-9)
        df['RSI'] = 100 - (100 / (1 + rs))

        # ATR (for volatility)
        prev_close = df['Close'].shift(1)
        tr = pd.concat([
            df['High'] - df['Low'],
            (df['High'] - prev_close).abs(),
            (df['Low'] - prev_close).abs()
        ], axis=1).max(axis=1)
        df['ATR'] = tr.rolling(window=14).mean()
        df['ATR_Pct'] = (df['ATR'] / (df['Close'] + 1e-9)) * 100

        # Money Flow Direction
        df['Money_Flow_Direction'] = np.where(df['Close'] > df['Open'], 'Inflow',
                                              np.where(df['Close'] < df['Open'], 'Outflow', 'Neutral'))

        # Buy/Sell Pressure
        df['Buy_Pressure'] = (df['Close'] - df['Low']) / (df['High'] - df['Low'] + 1e-9) * 100
        df['Sell_Pressure'] = (df['High'] - df['Close']) / (df['High'] - df['Low'] + 1e-9) * 100

        return df

    def find_fractal_pivots(self, df, left=2, right=2):
        """Williams-fractal style pivot high/low detection (left/right bars must be lower/higher)"""
        if df is None or len(df) < (left + right + 1):
            return [], []

        highs = df['High'].values
        lows = df['Low'].values
        dates = df['Date'].values

        pivot_highs = []
        pivot_lows = []

        for i in range(left, len(df) - right):
            window_high = highs[i - left:i + right + 1]
            window_low = lows[i - left:i + right + 1]
            if highs[i] == window_high.max():
                pivot_highs.append((dates[i], highs[i]))
            if lows[i] == window_low.min():
                pivot_lows.append((dates[i], lows[i]))

        return pivot_highs, pivot_lows

    def get_support_resistance(self, df, left=2, right=2):
        """Nearest fractal-based support (below price) and resistance (above price)"""
        if df is None or df.empty:
            return None, None

        current_price = df['Close'].iloc[-1]
        pivot_highs, pivot_lows = self.find_fractal_pivots(df, left, right)

        resistances_above = [p for _, p in pivot_highs if p > current_price]
        supports_below = [p for _, p in pivot_lows if p < current_price]

        resistance = min(resistances_above) if resistances_above else df['High'].max()
        support = max(supports_below) if supports_below else df['Low'].min()

        return support, resistance

    def compute_trend_score(self, df):
        """
        Composite trend score in range -100 (strong bearish) .. +100 (strong bullish).
        Weights: MA20/50 cross 25, VWAP 15, RSI 20, MFI 20, OBV slope 10, ADL slope 10
        """
        if df is None or len(df) < 20:
            return None

        latest = df.iloc[-1]
        score = 0.0
        breakdown = {}

        # MA20 vs MA50 (weight 25)
        if pd.notna(latest['Price_MA_20']) and pd.notna(latest['Price_MA_50']):
            if latest['Price_MA_20'] > latest['Price_MA_50']:
                score += 25
                breakdown['ma_trend'] = '’⁄ÊœÌ'
            else:
                score -= 25
                breakdown['ma_trend'] = '‰“Ê·Ì'
        else:
            breakdown['ma_trend'] = '‰«„‘Œ’'

        # Price vs VWAP (weight 15)
        if pd.notna(latest['VWAP']):
            if latest['Close'] > latest['VWAP']:
                score += 15
                breakdown['vwap_trend'] = '»«·«Ì VWAP'
            else:
                score -= 15
                breakdown['vwap_trend'] = 'Å«ÌÌ‰ VWAP'
        else:
            breakdown['vwap_trend'] = '‰«„‘Œ’'

        # RSI (weight 20)
        if pd.notna(latest['RSI']):
            rsi = latest['RSI']
            if rsi > 70:
                score -= 10
                breakdown['rsi_status'] = f'«‘»«⁄ Œ—Ìœ ({rsi:.1f})'
            elif rsi < 30:
                score += 10
                breakdown['rsi_status'] = f'«‘»«⁄ ›—Ê‘ ({rsi:.1f})'
            elif rsi > 50:
                score += 20
                breakdown['rsi_status'] = f'„À»  ({rsi:.1f})'
            else:
                score -= 20
                breakdown['rsi_status'] = f'„‰›Ì ({rsi:.1f})'
        else:
            breakdown['rsi_status'] = '‰«„‘Œ’'

        # MFI (weight 20)
        if pd.notna(latest['MFI']):
            mfi = latest['MFI']
            if mfi > 80:
                score -= 10
                breakdown['mfi_status'] = f'«‘»«⁄ Œ—Ìœ ({mfi:.1f})'
            elif mfi < 20:
                score += 10
                breakdown['mfi_status'] = f'«‘»«⁄ ›—Ê‘ ({mfi:.1f})'
            elif mfi > 50:
                score += 20
                breakdown['mfi_status'] = f'Ê—Êœ ÅÊ· ({mfi:.1f})'
            else:
                score -= 20
                breakdown['mfi_status'] = f'Œ—ÊÃ ÅÊ· ({mfi:.1f})'
        else:
            breakdown['mfi_status'] = '‰«„‘Œ’'

        # OBV slope (weight 10)
        lookback = min(20, len(df) - 1)
        if lookback > 0:
            obv_now = df['OBV'].iloc[-1]
            obv_prev = df['OBV'].iloc[-lookback]
            if obv_now > obv_prev:
                score += 10
                breakdown['obv_slope'] = '’⁄ÊœÌ'
            else:
                score -= 10
                breakdown['obv_slope'] = '‰“Ê·Ì'
        else:
            breakdown['obv_slope'] = '‰«„‘Œ’'

        # ADL slope (weight 10)
        if lookback > 0:
            adl_now = df['ADL'].iloc[-1]
            adl_prev = df['ADL'].iloc[-lookback]
            if adl_now > adl_prev:
                score += 10
                breakdown['adl_slope'] = '’⁄ÊœÌ'
            else:
                score -= 10
                breakdown['adl_slope'] = '‰“Ê·Ì'
        else:
            breakdown['adl_slope'] = '‰«„‘Œ’'

        score = max(-100, min(100, score))

        if score >= 60:
            label, color = '’⁄ÊœÌ ﬁÊÌ', COLORS['accent_green']
        elif score >= 20:
            label, color = '’⁄ÊœÌ', COLORS['accent_green']
        elif score > -20:
            label, color = 'Œ‰ÀÌ', COLORS['accent_orange']
        elif score > -60:
            label, color = '‰“Ê·Ì', COLORS['accent_red']
        else:
            label, color = '‰“Ê·Ì ﬁÊÌ', COLORS['accent_red']

        return {
            'score': score,
            'label': label,
            'color': color,
            'breakdown': breakdown
        }

    def compute_btc_correlation(self, ticker, df, period='1mo', interval='5m'):
        """Pearson correlation of returns vs BTC over the same window"""
        if ticker == 'BTC-USD' or df is None or df.empty:
            return None

        btc_df = self.fetch_data('BTC-USD', period=period, interval=interval)
        if btc_df is None or btc_df.empty:
            return None

        merged = pd.merge(
            df[['Date', 'Close']].rename(columns={'Close': 'Close_asset'}),
            btc_df[['Date', 'Close']].rename(columns={'Close': 'Close_btc'}),
            on='Date', how='inner'
        )
        if len(merged) < 10:
            return None

        ret_asset = merged['Close_asset'].pct_change().dropna()
        ret_btc = merged['Close_btc'].pct_change().dropna()
        n = min(len(ret_asset), len(ret_btc))
        if n < 10:
            return None

        corr = np.corrcoef(ret_asset.tail(n), ret_btc.tail(n))[0, 1]
        return corr

    def update_all_data(self):
        """Update data for all cryptocurrencies"""
        with self.update_lock:
            print(f"[{datetime.now()}] Updating crypto data...")
            for ticker in CRYPTO_TICKERS:
                df = self.fetch_data(ticker)
                if df is not None:
                    df = self.calculate_indicators(df)
                    self.data_cache[ticker] = df

                    # Store volume history for trend analysis
                    if ticker not in self.volume_history:
                        self.volume_history[ticker] = deque(maxlen=500)
                    self.volume_history[ticker].append({
                        'time': df['Date'].iloc[-1],
                        'volume': df['Volume'].iloc[-1],
                        'price': df['Close'].iloc[-1],
                        'change': df['Price_Change'].iloc[-1]
                    })

            self.last_update = datetime.now()
            print(f"[{datetime.now()}] Data update complete!")

    def get_latest_snapshot(self):
        """Get latest snapshot of all cryptocurrencies"""
        snapshot = []
        for ticker, df in self.data_cache.items():
            if df is not None and not df.empty:
                latest = df.iloc[-1]
                snapshot.append({
                    'ticker': ticker.replace('-USD', ''),
                    'name': CRYPTO_NAMES.get(ticker, ticker),
                    'price': latest['Close'],
                    'change_24h': latest['Price_Change'],
                    'volume': latest['Volume'],
                    'volume_ratio': latest['Volume_Ratio'],
                    'mfi': latest['MFI'],
                    'rsi': latest['RSI'],
                    'money_flow': latest['Money_Flow_Direction'],
                    'buy_pressure': latest['Buy_Pressure'],
                    'sell_pressure': latest['Sell_Pressure'],
                    'vwap': latest['VWAP'],
                    'bb_position': (latest['Close'] - latest['BB_Lower']) / (
                                latest['BB_Upper'] - latest['BB_Lower'] + 1e-9) * 100
                })
        return pd.DataFrame(snapshot)

    def get_volume_leaders(self, n=10):
        """Get top N cryptocurrencies by volume"""
        snapshot = self.get_latest_snapshot()
        if snapshot.empty:
            return snapshot
        return snapshot.nlargest(n, 'volume')

    def get_money_flow_leaders(self, n=10, direction='inflow'):
        """Get top N by money flow direction"""
        snapshot = self.get_latest_snapshot()
        if snapshot.empty:
            return snapshot
        if direction == 'inflow':
            return snapshot[snapshot['money_flow'] == 'Inflow'].nlargest(n, 'buy_pressure')
        else:
            return snapshot[snapshot['money_flow'] == 'Outflow'].nlargest(n, 'sell_pressure')

    def get_volume_surge_alerts(self, threshold=2.0):
        """Get cryptocurrencies with volume surge"""
        snapshot = self.get_latest_snapshot()
        if snapshot.empty:
            return snapshot
        return snapshot[snapshot['volume_ratio'] > threshold].sort_values('volume_ratio', ascending=False)

@app.callback(
    [Output('data-store', 'data'),
     Output('total-volume', 'children'),
     Output('volume-leader', 'children'),
     Output('money-flow-dominant', 'children'),
     Output('surge-alerts', 'children')],
    [Input('interval-component', 'n_intervals'),
     Input('refresh-btn', 'n_clicks')]
)
def update_data(n_intervals, n_clicks):
    """Update all data periodically"""
    ctx = callback_context

    data_manager.update_all_data()
    snapshot = data_manager.get_latest_snapshot()

    if snapshot.empty:
        return {}, "N/A", "N/A", "N/A", "0"

    total_vol = snapshot['volume'].sum()
    top_vol = snapshot.nlargest(1, 'volume').iloc[0]

    inflow_count = len(snapshot[snapshot['money_flow'] == 'Inflow'])
    outflow_count = len(snapshot[snapshot['money_flow'] == 'Outflow'])
    dominant = "Inflow ??" if inflow_count > outflow_count else "Outflow ??"

    surge_alerts = data_manager.get_volume_surge_alerts()
    surge_count = len(surge_alerts)

    store_data = snapshot.to_dict('records')

    return (
        store_data,
        f"${total_vol / 1e9:.2f}B",
        f"{top_vol['ticker']} (${top_vol['volume'] / 1e6:.1f}M)",
        dominant,
        str(surge_count)
    )

@app.callback(
    Output('surge-alerts-table', 'children'),
    [Input('data-store', 'data')]
)
def update_surge_alerts(data):
    if not data:
        return html.Div("No alerts", style={'color': COLORS['text_secondary']})

    df = pd.DataFrame(data)
    alerts = df[df['volume_ratio'] > 2.0].sort_values('volume_ratio', ascending=False)

    if alerts.empty:
        return html.Div("No volume surge alerts detected",
                        style={'color': COLORS['text_secondary'], 'textAlign': 'center', 'padding': '20px'})

    table_header = [
        html.Thead(html.Tr([
            html.Th("Crypto", style={'color': COLORS['accent_cyan']}),
            html.Th("Price", style={'color': COLORS['accent_cyan']}),
            html.Th("Change", style={'color': COLORS['accent_cyan']}),
            html.Th("Volume", style={'color': COLORS['accent_cyan']}),
            html.Th("Volume Ratio", style={'color': COLORS['accent_cyan']}),
            html.Th("MFI", style={'color': COLORS['accent_cyan']}),
            html.Th("Signal", style={'color': COLORS['accent_cyan']})
        ], style={'backgroundColor': COLORS['bg_card_light']}))
    ]

    rows = []
    for _, row in alerts.iterrows():
        change_color = COLORS['accent_green'] if row['change_24h'] > 0 else COLORS['accent_red']
        signal = "STRONG BUY" if row['volume_ratio'] > 5 and row['change_24h'] > 0 else \
            "BUY" if row['volume_ratio'] > 3 and row['change_24h'] > 0 else \
                "SELL" if row['change_24h'] < 0 else "WATCH"

        rows.append(html.Tr([
            html.Td(row['ticker'], style={'fontWeight': 'bold'}),
            html.Td(f"${row['price']:.4f}"),
            html.Td(f"{row['change_24h']:.2f}%", style={'color': change_color}),
            html.Td(f"${row['volume'] / 1e6:.1f}M"),
            html.Td(f"{row['volume_ratio']:.1f}x", style={'color': COLORS['accent_orange']}),
            html.Td(f"{row['mfi']:.1f}"),
            html.Td(signal, style={'fontWeight': 'bold'})
        ], style={'borderBottom': f'1px solid {COLORS["border"]}'}))

    table_body = [html.Tbody(rows)]
    return dbc.Table(table_header + table_body, bordered=False,
                     style={'color': COLORS['text_primary']}, className="table-dark")
