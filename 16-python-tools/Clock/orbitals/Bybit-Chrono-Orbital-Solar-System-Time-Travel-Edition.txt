REST_CANDIDATES = ["https://api.bybit.com", "https://api.bytick.com", "https://api.bybit.kz"]
SESSION = requests.Session()
SESSION.headers.update({
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
    "Accept": "application/json",
    "Referer": "https://www.bybit.com/",
})
_ACTIVE_REST_BASE = {"url": None}


def bybit_get(path, params, timeout=10):
    cands = ([_ACTIVE_REST_BASE["url"]] if _ACTIVE_REST_BASE["url"] else []) + \
            [b for b in REST_CANDIDATES if b != _ACTIVE_REST_BASE["url"]]
    for base in cands:
        try:
            r = SESSION.get(f"{base}{path}", params=params, timeout=timeout)
            if r.status_code in (403, 451):
                continue
            r.raise_for_status()
            d = r.json()
            if d.get("retCode") == 0:
                _ACTIVE_REST_BASE["url"] = base
                return d
        except Exception:
            continue
    return None


def get_klines(symbol, interval, category="linear", limit=300, end_time=None):
    """دریافت کندل‌ها با پشتیبانی از end_time برای سفر در زمان."""
    params = {
        "category": category,
        "symbol": symbol,
        "interval": interval,
        "limit": limit,
    }
    if end_time is not None:
        # end_time باید datetime با tzinfo UTC باشد
        if end_time.tzinfo is None:
            end_time = pytz.utc.localize(end_time)
        params["end"] = int(end_time.timestamp() * 1000)

    d = bybit_get("/v5/market/kline", params)
    if not d or "list" not in (d.get("result") or {}):
        return pd.DataFrame()
    lst = d["result"]["list"]
    if not lst:
        return pd.DataFrame()
    df = pd.DataFrame(lst, columns=["ts", "open", "high", "low", "close", "volume", "turnover"])
    df["ts"] = pd.to_datetime(df["ts"].astype(int), unit="ms", utc=True)
    for c in ["open", "high", "low", "close", "volume"]:
        df[c] = df[c].astype(float)
    return df.sort_values("ts").reset_index(drop=True)


def build_yearly_proxy(monthly_df):
    if monthly_df.empty:
        return None
    tail = monthly_df.tail(12)
    if tail.empty:
        return None
    return {
        "open": float(tail["open"].iloc[0]),
        "close": float(tail["close"].iloc[-1]),
        "high": float(tail["high"].max()),
        "low": float(tail["low"].min()),
        "volume": float(tail["volume"].sum()),
    }

PLANETS = OrderedDict([
    ("1",   {"label": "1m",  "planet": "Mercury ☿", "minutes": 1}),
    ("5",   {"label": "5m",  "planet": "Venus ♀",   "minutes": 5}),
    ("15",  {"label": "15m", "planet": "Earth ♁",   "minutes": 15}),

    ("M",   {"label": "1M",  "planet": "Neptune ♆", "minutes": 1440 * 30}),
])
EARTH_TF_OPTS = [{"label": v["label"], "value": k} for k, v in PLANETS.items()]

G_CONST = 1.0
SUN_MASS = 12.0
BLACKHOLE_MASS = 80.0
R_MIN, R_MAX = 55, 260
BLACKHOLE_BASE_DIST = 620
BLACKHOLE_DIR_DEG = 222
EARTH_WOBBLE = 70
FORECAST_STEPS = 6


def kepler_radius(period_minutes, all_periods):
    raw = {k: p ** (2.0 / 3.0) for k, p in all_periods.items()}
    lo, hi = min(raw.values()), max(raw.values())
    if hi == lo:
        return (R_MIN + R_MAX) / 2
    frac = (math.log(raw[period_minutes_key(period_minutes, all_periods)]) - math.log(lo)) / \
           (math.log(hi) - math.log(lo))
    return R_MIN + frac * (R_MAX - R_MIN)


def period_minutes_key(period_minutes, all_periods):
    for k, v in all_periods.items():
        if v == period_minutes:
            return k
    return list(all_periods.keys())[0]


def phase_angle(period_minutes, now_ms=None):
    if now_ms is None:
        now_ms = int(time.time() * 1000)
    period_ms = period_minutes * 60_000
    frac = (now_ms % period_ms) / period_ms
    return frac * 2 * math.pi


def polar_to_xy(r, theta, origin=(0.0, 0.0)):
    return origin[0] + r * math.cos(theta), origin[1] + r * math.sin(theta)


def map_price_to_radius(price, sun_value, price_scale, base_radius):
    if price_scale <= 0:
        dev = 0.0
    else:
        dev = (price - sun_value) / price_scale
    dev = max(-1.0, min(1.0, dev))
    return base_radius + dev * EARTH_WOBBLE, dev


def map_radius_to_price(radius, sun_value, price_scale, base_radius):
    dev = (radius - base_radius) / EARTH_WOBBLE
    dev = max(-1.0, min(1.0, dev))
    return sun_value + dev * price_scale

def compute_solar_system(symbol, category, earth_tf, target_time=None):
    """
    محاسبه‌ی منظومه در یک لحظه‌ی مشخص (target_time) یا لحظه‌ی فعلی (None).
    """
    all_periods = {k: v["minutes"] for k, v in PLANETS.items()}

    # --- تعیین لحظه‌ی مرجع ---
    if target_time is not None:
        if target_time.tzinfo is None:
            target_time = pytz.utc.localize(target_time)
        now_ms = int(target_time.timestamp() * 1000)
        now_ts = target_time
    else:
        now_ms = int(time.time() * 1000)
        now_ts = pd.Timestamp.utcnow()

    # --- دریافت دیتا برای همه‌ی تایم‌فریم‌ها ---
    frames = {}
    for tf in PLANETS.keys():
        frames[tf] = get_klines(symbol, tf, category, limit=250, end_time=target_time)

    monthly_for_year = frames.get("M", pd.DataFrame())
    yearly = build_yearly_proxy(monthly_for_year)

    earth_df = frames.get(earth_tf, pd.DataFrame())
    if earth_df.empty or yearly is None:
        return None

    # --- خورشید: MA200 روی تایم‌فریمِ زمین ---
    closes = earth_df["close"].to_numpy()
    ma_len = min(200, len(closes))
    sun_value = float(np.mean(closes[-ma_len:]))
    price_scale = float(np.std(closes[-ma_len:])) * 2.5 or (closes[-1] * 0.01)
    last_price = float(closes[-1])

    # --- شعاعِ پایه‌ی هر سیاره + جرم ---
    bodies = {}
    masses = {}
    for tf, meta in PLANETS.items():
        r_base = kepler_radius(meta["minutes"], all_periods)
        theta = phase_angle(meta["minutes"], now_ms)
        df_tf = frames.get(tf, pd.DataFrame())
        vol_mass = float(df_tf["volume"].tail(30).mean()) if not df_tf.empty else 1.0
        masses[tf] = vol_mass
        bodies[tf] = {
            "r_base": r_base, "theta": theta,
            "label": meta["label"], "planet": meta["planet"]
        }

    max_mass = max(masses.values()) or 1.0
    for tf in masses:
        masses[tf] = 1.5 + 8.5 * (masses[tf] / max_mass)

    # --- زمین ---
    earth_theta = bodies[earth_tf]["theta"]
    earth_r_base = bodies[earth_tf]["r_base"]
    earth_r, earth_dev = map_price_to_radius(last_price, sun_value, price_scale, earth_r_base)
    bodies[earth_tf]["r"] = earth_r
    for tf in bodies:
        if tf != earth_tf:
            bodies[tf]["r"] = bodies[tf]["r_base"]

    # --- درصد پیشرفت سال بر اساس target_time ---
    year_start = pd.Timestamp(year=now_ts.year, month=1, day=1, tz="UTC")
    year_end = pd.Timestamp(year=now_ts.year + 1, month=1, day=1, tz="UTC")
    year_progress = (now_ts - year_start) / (year_end - year_start)
    yearly_bullish = yearly["close"] >= yearly["open"]
    eased_progress = year_progress ** (0.7 if yearly_bullish else 1.3)
    blackhole_dist = BLACKHOLE_BASE_DIST * (1 - eased_progress) + 90 * eased_progress
    bh_theta = math.radians(BLACKHOLE_DIR_DEG)
    bh_x, bh_y = polar_to_xy(blackhole_dist, bh_theta)

    # --- نیروی گرانشی خالص روی زمین ---
    ex, ey = polar_to_xy(earth_r, earth_theta)
    m_earth = 1.0

    def force_from(mass, tx, ty):
        dx, dy = tx - ex, ty - ey
        d = math.hypot(dx, dy) or 1e-6
        f = G_CONST * mass * m_earth / (d ** 2)
        return f * dx / d, f * dy / d, f

    fx = fy = 0.0
    contributions = []

    sx, sy = 0.0, 0.0
    _fx, _fy, _f = force_from(SUN_MASS, sx, sy)
    fx += _fx; fy += _fy
    contributions.append({"name": "خورشید (MA200)", "force": _f, "role": "بازگشت به میانگین"})

    _fx, _fy, _f = force_from(BLACKHOLE_MASS, bh_x, bh_y)
    fx += _fx; fy += _fy
    contributions.append({"name": "سیاه‌چاله (روند سالانه)", "force": _f, "role": "پیروی از روند کلان"})

    for tf, b in bodies.items():
        if tf == earth_tf:
            continue
        px, py = polar_to_xy(b["r"], b["theta"])
        _fx, _fy, _f = force_from(masses[tf], px, py)
        fx += _fx; fy += _fy
        contributions.append({"name": f"{b['planet']} ({b['label']})", "force": _f,
                              "role": "همگرایی چندتایم‌فریمی"})

    contributions.sort(key=lambda c: -c["force"])
    total_mag = sum(c["force"] for c in contributions) or 1e-9
    align_x = fx / total_mag if total_mag else 0.0
    align_y = fy / total_mag if total_mag else 0.0
    confluence_score = min(1.0, math.hypot(align_x, align_y))

    # --- پیش‌بینی مسیر زمین ---
    period_min = PLANETS[earth_tf]["minutes"]
    omega = 2 * math.pi / period_min
    v_tangent = omega * earth_r
    vx = -math.sin(earth_theta) * v_tangent
    vy = math.cos(earth_theta) * v_tangent

    px, py = ex, ey
    ax, ay = fx / m_earth, fy / m_earth
    dt = period_min / 6.0
    path_xy = [(px, py)]
    predicted_prices = [last_price]
    for _ in range(FORECAST_STEPS):
        vx += ax * dt * 0.02
        vy += ay * dt * 0.02
        px += vx * dt * 0.02
        py += vy * dt * 0.02
        path_xy.append((px, py))
        r_pred = math.hypot(px, py)
        predicted_prices.append(map_radius_to_price(r_pred, sun_value, price_scale, earth_r_base))

    return {
        "bodies": bodies, "masses": masses, "earth_tf": earth_tf,
        "sun_value": sun_value, "price_scale": price_scale, "last_price": last_price,
        "earth_r_base": earth_r_base, "earth_r": earth_r, "earth_theta": earth_theta,
        "earth_xy": (ex, ey), "blackhole_xy": (bh_x, bh_y), "blackhole_dist": blackhole_dist,
        "year_progress": year_progress, "yearly_bullish": yearly_bullish,
        "contributions": contributions, "confluence_score": confluence_score,
        "align_xy": (align_x, align_y), "path_xy": path_xy, "predicted_prices": predicted_prices,
        "earth_df": earth_df,
        "target_time": target_time,  # برای نمایش در پنل
    }

def build_orbit_figure(state):
    fig = go.Figure()

    bh_x, bh_y = state["blackhole_xy"]
    for i, rr in enumerate([46, 34, 24]):
        fig.add_shape(type="circle", xref="x", yref="y",
                      x0=bh_x - rr, x1=bh_x + rr, y0=bh_y - rr, y1=bh_y + rr,
                      line=dict(color=ACCRETION_COLOR, width=1),
                      fillcolor=f"rgba(124,58,237,{0.08 + i * 0.05})")
    fig.add_trace(go.Scatter(x=[bh_x], y=[bh_y], mode="markers+text",
                              marker=dict(size=22, color=BLACKHOLE_COLOR,
                                          line=dict(color=ACCRETION_COLOR, width=2)),
                              text=["⚫ سیاه‌چاله (سالانه)"], textposition="bottom center",
                              textfont=dict(color=ACCRETION_COLOR, size=11, family=FONT_FA),
                              name="سیاه‌چاله"))
    fig.add_trace(go.Scatter(x=[0, bh_x], y=[0, bh_y], mode="lines",
                              line=dict(color=ACCRETION_COLOR, width=1, dash="dot"),
                              hoverinfo="skip", showlegend=False))

    fig.add_trace(go.Scatter(x=[0], y=[0], mode="markers+text",
                              marker=dict(size=26, color=SUN_COLOR,
                                          line=dict(color="#fff3c4", width=2)),
                              text=["☀ خورشید (MA200)"], textposition="top center",
                              textfont=dict(color=SUN_COLOR, size=11, family=FONT_FA),
                              name="خورشید"))

    theta_ring = np.linspace(0, 2 * math.pi, 100)
    for tf, b in state["bodies"].items():
        r_ring = b["r_base"]
        ring_x = r_ring * np.cos(theta_ring)
        ring_y = r_ring * np.sin(theta_ring)
        fig.add_trace(go.Scatter(x=ring_x, y=ring_y, mode="lines",
                                  line=dict(color=LINE, width=1),
                                  hoverinfo="skip", showlegend=False))

        is_earth = (tf == state["earth_tf"])
        px, py = polar_to_xy(b["r"], b["theta"])
        color = EARTH_COLOR if is_earth else GOLD
        size = 16 + state["masses"][tf] * 1.6
        label = f"🌍 {b['planet']} — قیمت" if is_earth else b["planet"]
        fig.add_trace(go.Scatter(x=[px], y=[py], mode="markers+text",
                                  marker=dict(size=size, color=color,
                                              line=dict(color="#ffffff",
                                                        width=1 if is_earth else 0)),
                                  text=[f"{label}<br>{b['label']}"], textposition="middle right",
                                  textfont=dict(
                                      color=TXT if not is_earth else EARTH_COLOR,
                                      size=10, family=FONT_FA),
                                  name=b["label"]))

    path = state["path_xy"]
    fig.add_trace(go.Scatter(x=[p[0] for p in path], y=[p[1] for p in path],
                              mode="lines+markers",
                              line=dict(color=EARTH_COLOR, width=2, dash="dash"),
                              marker=dict(size=5, color=EARTH_COLOR),
                              name="مسیرِ پیش‌بینی‌شده‌ی زمین"))

    lim = state["blackhole_dist"] + 60 if state["blackhole_dist"] > R_MAX else R_MAX + 80

    # عنوان بر اساس حالت (زنده / سفر در زمان)
    tt = state.get("target_time")
    if tt is not None:
        title_text = f"🌌 منظومه‌ی شمسی — سفر در زمان: {tt.strftime('%Y-%m-%d %H:%M')} UTC"
    else:
        title_text = "🌌 منظومه‌ی شمسیِ چندتایم‌فریمی — زمین = قیمت"

    fig.update_layout(
        template="plotly_dark", paper_bgcolor=BG, plot_bgcolor=BG,
        font=dict(color=TXT, family=FONT_FA),
        xaxis=dict(visible=False, range=[-lim, lim], scaleanchor="y", scaleratio=1),
        yaxis=dict(visible=False, range=[-lim, lim]),
        margin=dict(l=10, r=10, t=30, b=10), showlegend=False,
        title=dict(text=title_text, x=0.5,
                   font=dict(color=GOLD, size=14, family=FONT_FA)),
    )
    return fig

# 6) کال‌بک 1: مدیریت سفر در زمان
# ==============================================================================
@app.callback(
    Output("target-time-store", "data"),
    Output("refresh-interval", "disabled"),
    Output("time-status", "children"),
    Input("goto-time-btn", "n_clicks"),
    Input("goto-now-btn", "n_clicks"),
    State("date-input", "value"),
    State("time-input", "value"),
    prevent_initial_call=True,
)
def manage_target_time(n_goto, n_now, date_value, time_value):
    trigger = ctx.triggered_id

    if trigger == "goto-time-btn" and date_value:
        try:
            hour, minute = 0, 0
            if time_value:
                parts = time_value.split(":")
                if len(parts) >= 2:
                    hour = int(parts[0])
                    minute = int(parts[1])

            target_time = datetime.strptime(date_value, "%Y-%m-%d").replace(
                hour=hour, minute=minute, tzinfo=pytz.UTC
            )
            print(f"⏰ سفر در زمان به: {target_time.isoformat()}")
            return (
                target_time.isoformat(),
                True,   # غیرفعال کردن refresh-interval
                f"⏰ {target_time.strftime('%Y-%m-%d %H:%M')} UTC (قفل)",
            )
        except Exception as e:
            print(f"❌ خطا در پردازش زمان: {e}")
            return None, False, "❌ فرمت تاریخ/ساعت نامعتبر"

    elif trigger == "goto-now-btn":
        print("🔄 بازگشت به حالت زنده")
        return None, False, "🔴 زنده"

    return dash.no_update, dash.no_update, dash.no_update

)
def update_system(n_int, n_refresh, stored_target_time, symbol, category, earth_tf):
    symbol = (symbol or DEFAULT_SYMBOL).upper()
    category = category or DEFAULT_CATEGORY
    earth_tf = earth_tf or DEFAULT_EARTH_TF

    # بازیابی target_time از Store
    target_time = None
    if stored_target_time:
        try:
            target_time = datetime.fromisoformat(stored_target_time)
        except Exception:
            target_time = None

    state = compute_solar_system(symbol, category, earth_tf, target_time=target_time)
    if state is None:
        empty_fig = go.Figure(layout=dict(
            paper_bgcolor=BG, plot_bgcolor=BG,
            annotations=[dict(

                x=0.5, y=0.5, showarrow=False,
                font=dict(color=DN, size=14))]))
        return empty_fig, dash.no_update, "🔴 قطع"

    fig = build_orbit_figure(state)
    panel = build_side_panel(state, symbol, earth_tf)

    if target_time is not None:
        status = f"⏰ قفل | {target_time.strftime('%H:%M:%S')} UTC"
    else:
        status = f"🟢 متصل | {pd.Timestamp.now().strftime('%H:%M:%S')}"

    return fig, panel, status
