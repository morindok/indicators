REST_CANDIDATES = [
    "https://api.bybit.com",

]
SESSION = requests.Session()
SESSION.headers.update({
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
    "Accept": "application/json",
    "Referer": "https://www.bybit.com/",
})
_ACTIVE_REST_BASE = {"url": None}


def bybit_get(path, params, timeout=10):
    cands = ([_ACTIVE_REST_BASE["url"]] if _ACTIVE_REST_BASE["url"] else []) + \
            [b for b in REST_CANDIDATES if b != _ACTIVE_REST_BASE["url"]]
    for base in cands:
        try:
            r = SESSION.get(f"{base}{path}", params=params, timeout=timeout)
            if r.status_code in (403, 451):
                continue
            r.raise_for_status()
            d = r.json()
            if d.get("retCode") == 0:
                _ACTIVE_REST_BASE["url"] = base
                return d
        except Exception:
            continue
    return None


def get_klines(symbol, interval, limit=1000):
    for category in ["linear", "spot"]:
        d = bybit_get("/v5/market/kline", {
            "category": category, "symbol": symbol,
            "interval": interval, "limit": limit,
        })
        if d and "list" in (d.get("result") or {}):
            lst = d["result"]["list"]
            if lst:
                df = pd.DataFrame(lst, columns=["ts", "open", "high", "low", "close", "volume", "turnover"])
                df["ts"] = pd.to_datetime(df["ts"].astype(int), unit="ms")
                for c in ["open", "high", "low", "close", "volume"]:
                    df[c] = df[c].astype(float)
                return df.sort_values("ts").reset_index(drop=True)
    return pd.DataFrame()

# 2) محاسبه فاز (زاویه) داخل هر چرخه
# ==============================================================================
def compute_phase(dt, cycle):
    """فاز داخل چرخه را در بازه [0, 1) برمی‌گرداند."""
    if hasattr(dt, "to_pydatetime"):
        dt = dt.to_pydatetime()
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)

    if cycle == "year":
        doy = dt.timetuple().tm_yday
        days_in_year = 366 if calendar.isleap(dt.year) else 365
        return (doy - 1 + dt.hour / 24.0) / days_in_year
    elif cycle == "month":
        dim = calendar.monthrange(dt.year, dt.month)[1]
        return (dt.day - 1 + dt.hour / 24.0 + dt.minute / 1440.0) / dim
    elif cycle == "week":
        wd = dt.weekday()  # دوشنبه = 0
        return (wd + dt.hour / 24.0 + dt.minute / 1440.0) / 7.0
    elif cycle == "day":
        return (dt.hour + dt.minute / 60.0 + dt.second / 3600.0) / 24.0
    return 0.0


def interp_circular(theta_arr, r_arr, angle):
    """درون‌یابی شعاع در زاویه دلخواه با پیچش دایره‌ای."""
    th_ext = np.concatenate([theta_arr - 2 * np.pi, theta_arr, theta_arr + 2 * np.pi])
    r_ext = np.concatenate([r_arr, r_arr, r_arr])
    return float(np.interp(angle, th_ext, r_ext))


# ==============================================================================
# 3) ساخت الگوی اسپیرال برای یک تایم‌فریم
# ==============================================================================
def build_timeframe_spiral(df, cycle, current_price, n_bins=48):
    """
    برای هر کندل، فاز داخل چرخه و قیمت را محاسبه و بر اساس فاز bins‌بندی می‌کند.
    خروجی: theta (رادیان)، r_price (قیمت واقعی)، r_norm (نرمال به قیمت فعلی)، counts
    """
    phases = df["ts"].apply(lambda x: compute_phase(x.to_pydatetime(), cycle)).values
    prices = df["close"].values

    bins = np.linspace(0, 1, n_bins + 1)
    theta = (bins[:-1] + 0.5 / n_bins) * 2 * np.pi
    r_price = np.full(n_bins, np.nan)
    counts = np.zeros(n_bins)

    for b in range(n_bins):
        m = (phases >= bins[b]) & (phases < bins[b + 1])
        if m.sum() > 0:
            r_price[b] = prices[m].mean()
            counts[b] = m.sum()

    # پر کردن جاهای خالی با درون‌یابی دایره‌ای
    valid = ~np.isnan(r_price)
    if valid.sum() >= 2:
        th_ext = np.concatenate([theta[valid] - 2 * np.pi, theta[valid], theta[valid] + 2 * np.pi])
        r_ext = np.concatenate([r_price[valid], r_price[valid], r_price[valid]])
        r_price = np.interp(theta, th_ext, r_ext)
    elif valid.sum() == 1:
        r_price[:] = r_price[valid][0]
    else:
        r_price[:] = current_price

    r_norm = r_price / current_price if current_price > 0 else r_price
    return theta, r_price, r_norm, counts

# 4) محاسبه کل سیستم اسپیرال
# ==============================================================================
def compute_spiral_system(symbol):
    now = datetime.now(timezone.utc)

    # دریافت داده همه تایم‌فریم‌ها
    data = {}
    for tf in TF_CONFIG:
        df = get_klines(symbol, tf["interval"], limit=1000)
        if not df.empty:
            data[tf["interval"]] = df
        time.sleep(0.05)

    if not data:
        return None

    # قیمت فعلی از سریع‌ترین تایم‌فریم موجود
    current_price = None
    for iv in ["1", "5", "15", "30", "60", "240", "D", "W", "M"]:
        if iv in data and not data[iv].empty:
            current_price = float(data[iv]["close"].iloc[-1])
            break
    if current_price is None:
        return None

    # زاویه مرجع از اسپیرال ماهانه
    ref_cfg = next(tf for tf in TF_CONFIG if tf.get("is_ref"))
    ref_cycle = ref_cfg["cycle"]
    ref_phase = compute_phase(now, ref_cycle)
    ref_theta = ref_phase * 2 * np.pi

    spirals = []
    for tf in TF_CONFIG:
        iv = tf["interval"]
        if iv not in data:
            continue
        df = data[iv]
        theta, r_price, r_norm, counts = build_timeframe_spiral(
            df, tf["cycle"], current_price, tf["bins"]
        )
        # موقعیت لایو
        live_phase = compute_phase(now, tf["cycle"])
        live_theta = live_phase * 2 * np.pi
        live_price = float(df["close"].iloc[-1])
        live_r_norm = live_price / current_price

        # قیمت هم‌راستا در زاویه مرجع
        aligned_r_norm = interp_circular(theta, r_norm, ref_theta)
        aligned_price = aligned_r_norm * current_price
        dev_pct = (aligned_price - live_price) / live_price * 100 if live_price > 0 else 0

        spirals.append({
            "interval": iv, "label": tf["label"], "color": tf["color"],
            "is_ref": tf.get("is_ref", False), "cycle": tf["cycle"],
            "theta": theta, "r_price": r_price, "r_norm": r_norm,
            "live_phase": live_phase, "live_theta": live_theta,
            "live_price": live_price, "live_r_norm": live_r_norm,
            "aligned_r_norm": aligned_r_norm, "aligned_price": aligned_price,
            "dev_pct": dev_pct, "n_samples": int(counts.sum()),
        })

    # معیار هم‌راستایی: پراکندگی فاز لایو نسبت به فاز مرجع
    phase_devs = []
    for s in spirals:
        d = abs(s["live_phase"] - ref_phase)
        d = min(d, 1.0 - d)
        phase_devs.append(d)
    avg_phase_dev = float(np.mean(phase_devs)) if phase_devs else 0.0
    alignment_score = max(0.0, 1.0 - avg_phase_dev * 2.0) * 100.0

    return {
        "symbol": symbol,
        "current_price": current_price,
        "ref_theta": ref_theta,
        "ref_phase": ref_phase,
        "ref_label": ref_cfg["label"],
        "ref_cycle_label": CYCLE_LABELS.get(ref_cycle, ref_cycle),
        "spirals": spirals,
        "alignment_score": alignment_score,
        "avg_phase_dev": avg_phase_dev,
        "now": now,
    }

def build_polar_figure(sys):
    fig = go.Figure()
    fine_theta = np.linspace(0, 2 * np.pi, 361)
    ref_theta_deg = np.degrees(sys["ref_theta"])

    # منحنی اسپیرال هر تایم‌فریم
    for s in sys["spirals"]:
        r_fine = np.array([interp_circular(s["theta"], s["r_norm"], a) for a in fine_theta])
        fig.add_trace(go.Scatterpolar(
            r=r_fine, theta=np.degrees(fine_theta),
            mode="lines",
            line=dict(color=s["color"], width=3 if s["is_ref"] else 1.4),
            name=s["label"], opacity=0.95 if s["is_ref"] else 0.75,
            hovertemplate=f"{s['label']}<br>قیمت نرمال: %{{r:.4f}}<extra></extra>",
        ))

    max_r = max(float(np.max(s["r_norm"])) for s in sys["spirals"]) * 1.08

    # خط شعاعی زاویه مرجع
    fig.add_trace(go.Scatterpolar(
        r=[0, max_r], theta=[ref_theta_deg, ref_theta_deg],
        mode="lines", line=dict(color=GOLD, width=2.5, dash="dash"),
        name=f"زاویه مرجع ({sys['ref_label']})",
    ))

    # نقاط هم‌راستا + زنجیره شعاعی
    aligned_r = [s["aligned_r_norm"] for s in sys["spirals"]]
    fig.add_trace(go.Scatterpolar(
        r=aligned_r, theta=[ref_theta_deg] * len(aligned_r),
        mode="markers+lines",
        marker=dict(size=10, color=[s["color"] for s in sys["spirals"]],
                    line=dict(color="white", width=1.2), symbol="circle"),

        polar=dict(

        margin=dict(l=60, r=60, t=70, b=60),
        height=640,
    )
    return fig


def build_3d_figure(sys):
    fig = go.Figure()
    fine_theta = np.linspace(0, 2 * np.pi, 241)

    for i, s in enumerate(sys["spirals"]):
        r_fine = np.array([interp_circular(s["theta"], s["r_norm"], a) for a in fine_theta])
        x = r_fine * np.cos(fine_theta)
        y = r_fine * np.sin(fine_theta)
        z = np.full_like(fine_theta, float(i))
        fig.add_trace(go.Scatter3d(
            x=x, y=y, z=z, mode="lines",
            line=dict(color=s["color"], width=4 if s["is_ref"] else 2),
            name=s["label"],
        ))

    # زنجیره عمودی هم‌راستا در زاویه مرجع
    xr, yr, zr = [], [], []
    for i, s in enumerate(sys["spirals"]):
        r = s["aligned_r_norm"]
        xr.append(r * np.cos(sys["ref_theta"]))
        yr.append(r * np.sin(sys["ref_theta"]))
        zr.append(float(i))
    fig.add_trace(go.Scatter3d(
        x=xr, y=yr, z=zr, mode="lines+markers",
        marker=dict(size=7, color=[s["color"] for s in sys["spirals"]]),
        line=dict(color=GOLD, width=6),
        name="⛓️ زنجیره هم‌راستا",
    ))

    # نقاط لایو
    xl = [s["live_r_norm"] * np.cos(s["live_theta"]) for s in sys["spirals"]]
    yl = [s["live_r_norm"] * np.sin(s["live_theta"]) for s in sys["spirals"]]
    zl = [float(i) for i in range(len(sys["spirals"]))]
    fig.add_trace(go.Scatter3d(
        x=xl, y=yl, z=zl, mode="markers",
        marker=dict(size=8, color=[s["color"] for s in sys["spirals"]], symbol="diamond"),
        name="💠 لایو",
    ))

    fig.update_layout(
        scene=dict(
            xaxis=dict(title="X", gridcolor=LINE, color=MUT, showticklabels=False),
            yaxis=dict(title="Y", gridcolor=LINE, color=MUT, showticklabels=False),
            zaxis=dict(title="تایم‌فریم", gridcolor=LINE, color=MUT,
                       tickvals=list(range(len(sys["spirals"]))),
                       ticktext=[s["label"] for s in sys["spirals"]], tickfont=dict(size=9)),
            bgcolor=CARD,
            camera=dict(eye=dict(x=1.6, y=1.6, z=1.2)),
        ),
        paper_bgcolor=BG,
        font=dict(color=TXT, family=FONT_FAMILY),

                   x=0.5, font=dict(color=CYAN, size=15, family=FONT_FAMILY)),
        showlegend=True,

    Output("spiral-store", "data"),
    Output("status-text", "children"),
    Input("refresh-btn", "n_clicks"),
    Input("tick", "n_intervals"),
    State("symbol-input", "value"),

def compute_spirals(_click, _tick, symbol):
    symbol = (symbol or DEFAULT_SYMBOL).upper()
    sys_data = compute_spiral_system(symbol)
    if sys_data is None:
        return None, "🔴 خطا در دریافت داده از Bybit"

    # سریال‌سازی برای ذخیره در Store
    store = {
        "symbol": sys_data["symbol"],
        "current_price": sys_data["current_price"],
        "ref_theta": sys_data["ref_theta"],
        "ref_phase": sys_data["ref_phase"],
        "ref_label": sys_data["ref_label"],
        "ref_cycle_label": sys_data["ref_cycle_label"],
        "alignment_score": sys_data["alignment_score"],
        "avg_phase_dev": sys_data["avg_phase_dev"],
        "now": sys_data["now"].isoformat(),
        "spirals": [
            {
                "interval": s["interval"], "label": s["label"], "color": s["color"],
                "is_ref": s["is_ref"], "cycle": s["cycle"],
                "theta": s["theta"].tolist(), "r_norm": s["r_norm"].tolist(),
                "live_phase": s["live_phase"], "live_theta": s["live_theta"],
                "live_price": s["live_price"], "live_r_norm": s["live_r_norm"],
                "aligned_r_norm": s["aligned_r_norm"], "aligned_price": s["aligned_price"],
                "dev_pct": s["dev_pct"], "n_samples": s["n_samples"],
            } for s in sys_data["spirals"]
        ],
    }
    status = f"🟢 محاسبه شد | {len(sys_data['spirals'])} اسپیرال | {sys_data['now'].strftime('%H:%M:%S')}"
    return store, status


def _rebuild_sys(store):
    if not store:
        return None
    spirals = []
    for s in store["spirals"]:
        s2 = dict(s)
        s2["theta"] = np.array(s["theta"])
        s2["r_norm"] = np.array(s["r_norm"])
        spirals.append(s2)
    sys_data = dict(store)
    sys_data["spirals"] = spirals
    sys_data["now"] = datetime.fromisoformat(store["now"])
    return sys_data

def render_outputs(store):

    return kpis, polar, fig3d, table, interp
