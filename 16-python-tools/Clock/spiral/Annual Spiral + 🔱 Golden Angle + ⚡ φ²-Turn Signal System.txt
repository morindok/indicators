SYMBOL = "BTCUSDT"
CATEGORY = "linear"
YEAR_SECONDS = 365.25 * 24 * 3600

N_SPIRAL_TURNS = 6
A_SPIRAL = 0.8
B_SPIRAL = 0.07
THETA_MAX = 2 * np.pi * N_SPIRAL_TURNS

PHI = (1 + np.sqrt(5)) / 2
GA_RAD = 2 * np.pi / (PHI ** 2)
GA_DEG = GA_RAD * 180 / np.pi
GA_SUPP_RAD = 2 * np.pi - GA_RAD
GA_SUPP_DEG = GA_SUPP_RAD * 180 / np.pi
GC_RAD = np.pi + GA_RAD
GC_DEG = GC_RAD * 180 / np.pi

# ⚡ φ²-Turn: قوی‌ترین مقاومت طلایی
PHI2_RAD = (PHI ** 2) * 2 * np.pi        # 16.449 rad
PHI2_DEG = PHI2_RAD * 180 / np.pi        # 942.49°
PHI2_PROGRESS = PHI2_RAD / THETA_MAX     # ≈ 0.436 (43.6% از دوره)


def spiral_radius(theta):
    return A_SPIRAL * np.exp(B_SPIRAL * theta)


def spiral_x(theta):
    return spiral_radius(theta) * np.cos(theta)


def spiral_y(theta):
    return spiral_radius(theta) * np.sin(theta)

def format_fib_time(minutes):
    if minutes < 60:
        return f"{minutes}m"
    elif minutes < 1440:
        h = minutes // 60
        m = minutes % 60
        return f"{h}h{m:02d}m"
    elif minutes < 10080:
        d = minutes // 1440
        h = (minutes % 1440) // 60
        return f"{d}d{h}h"
    elif minutes < 43200:
        w = minutes // 10080
        d = (minutes % 10080) // 1440
        return f"{w}w{d}d"
    else:
        mo = minutes // 43200
        d = (minutes % 43200) // 1440
        return f"{mo}mo{d}d"


def build_fib_timeframes():
    tfs = []
    colors = [
        '#FFD700', '#FFC300', '#FFB000', '#FF9D00', '#FF8A00',
        '#FF7700', '#FF6400', '#FF5100', '#FF3E00', '#FF2B00',
        '#FF1800', '#FF0500', '#E000FF', '#CC00FF', '#AA00FF',
        '#8800FF', '#6600FF', '#4400FF', '#2200FF', '#0044FF',
        '#0088FF', '#00CCFF', '#00FFCC'
    ]
    for i, minutes in enumerate(FIB_MINUTES):
        tfs.append({
            'name': format_fib_time(minutes),
            'T': minutes * 60,
            'R': 0.8 + np.log10(minutes + 1) * 3.5,
            'color': colors[i % len(colors)],
            'width': max(1.0, 4.0 - i * 0.15),
            'fib_index': i + 1,
            'minutes': minutes
        })
    return tfs

def api_request(path, params, timeout=10):
    endpoints = [APP_STATE["endpoint"]] if APP_STATE["endpoint"] != "None" else []
    endpoints += [ep for ep in API_ENDPOINTS if ep != APP_STATE["endpoint"]]
    for endpoint in endpoints:
        url = f"{endpoint}{path}"
        for verify in [True, False]:
            try:
                resp = SESSION.get(url, params=params, timeout=timeout, verify=verify)
                if resp.status_code == 200:
                    data = resp.json()
                    if data.get("retCode") == 0:
                        APP_STATE["endpoint"] = endpoint
                        APP_STATE["connected"] = True
                        return data
            except Exception:
                continue
    APP_STATE["connected"] = False
    return None


def data_loop(stop_flag):
    while not stop_flag.is_set():
        try:
            data = api_request("/v5/market/tickers", {"category": CATEGORY, "symbol": SYMBOL})
            if data and data.get("result") and data["result"].get("list"):
                APP_STATE["price"] = float(data["result"]["list"][0].get("lastPrice", 0))
            for tf in TIMEFRAMES_NORMAL:
                data = api_request("/v5/market/kline", {
                    "category": CATEGORY, "symbol": SYMBOL,
                    "interval": tf['name'], "limit": 2

                    KLINE_CACHE[tf['name']] = {
                        'open': float(k[1]), 'high': float(k[2]),
                        'low': float(k[3]), 'close': float(k[4])
                    }
        except Exception:
            pass
        time.sleep(3)


threading.Thread(target=data_loop, args=(threading.Event(),), daemon=True).start()

def find_next_golden_event(t_now, T_sec):
    progress = (t_now % T_sec) / T_sec
    theta_now = progress * THETA_MAX

    events = {
        '1×GA': GA_RAD, '2×GA': 2 * GA_RAD, '3×GA': 3 * GA_RAD,
        '4×GA': 4 * GA_RAD, '5×GA': 5 * GA_RAD, '6×GA': 6 * GA_RAD,
        '7×GA': 7 * GA_RAD, '8×GA': 8 * GA_RAD, '9×GA': 9 * GA_RAD,
        '10×GA': 10 * GA_RAD,
        '1×GS': GA_SUPP_RAD, '2×GS': 2 * GA_SUPP_RAD, '3×GS': 3 * GA_SUPP_RAD,
        '4×GS': 4 * GA_SUPP_RAD, '5×GS': 5 * GA_SUPP_RAD, '6×GS': 6 * GA_SUPP_RAD,
        'GC': GC_RAD, 'GCS': np.pi + GA_SUPP_RAD,
        'φ-Turn': PHI * 2 * np.pi, 'φ²-Turn': PHI2_RAD,
        'Mid(3π)': 3 * np.pi,
    }

    candidates = []
    for name, theta_ev in events.items():
        if 0 <= theta_ev <= THETA_MAX:
            for k in range(0, 4):
                theta_target = theta_ev + k * THETA_MAX
                if theta_target > theta_now + 0.001:
                    progress_target = (theta_target % THETA_MAX) / THETA_MAX
                    t_remaining = T_sec * (progress_target - progress)
                    if t_remaining < 0:
                        t_remaining += T_sec
                    candidates.append((t_remaining, name, theta_ev))
                    break

    if not candidates:
        return None

    candidates.sort(key=lambda x: x[0])
    t_rem, event, theta_base = candidates[0]
    return {
        'time_remaining': t_rem,
        'event': event,
        'theta_base': theta_base,
        'event_time': datetime.fromtimestamp(t_now + t_rem)
    }


def get_event_strength(event):
    if event == 'φ²-Turn':
        return 5, '⚡', '#FF0055'
    elif event in ['GC', 'GCS', 'φ-Turn']:
        return 4, '🔥', '#FF6B00'
    elif event.startswith(('3×', '4×', '5×')):
        return 3, '⭐', '#FFD700'
    elif event.startswith(('1×GS', '2×GS')):
        return 3, '🟡', '#FFA500'
    elif event.startswith(('1×GA', '2×GA')):
        return 2, '🟢', '#4ECDC4'
    else:
        return 1, '➖', '#888888'


def format_time_remaining(seconds):
    if seconds < 0:
        return "—"
    if seconds < 60:
        return f"{seconds:.0f}s"
    if seconds < 3600:
        return f"{int(seconds // 60)}m {int(seconds % 60):02d}s"
    if seconds < 86400:
        return f"{int(seconds // 3600)}h {int((seconds % 3600) // 60):02d}m"
    return f"{int(seconds // 86400)}d {int((seconds % 86400) // 3600)}h"


def compute_golden_resistances(t_now, timeframes):
    results = []
    for tf in timeframes:
        progress = (t_now % tf['T']) / tf['T']
        theta_now = progress * THETA_MAX
        event = find_next_golden_event(t_now, tf['T'])
        results.append({
            'tf': tf['name'],
            'period': format_fib_time(tf['T'] // 60),
            'color': tf['color'],
            'theta_now': theta_now,
            'progress': progress,
            'event': event
        })
    return results

def find_next_phi2_event(t_now, T_sec):
    """
    فقط رویداد φ²-Turn بعدی را پیدا می‌کند.
    φ²-Turn در progress = 43.6% از دوره هر تایم‌فریم رخ می‌دهد.
    """
    progress = (t_now % T_sec) / T_sec
    
    # زمان تا φ² در دوره فعلی
    t_remaining = T_sec * (PHI2_PROGRESS - progress)
    if t_remaining < 0:
        t_remaining += T_sec  # دوره بعد
    
    return {
        'time_remaining': t_remaining,
        'event': 'φ²-Turn',
        'theta_base': PHI2_RAD,
        'event_time': datetime.fromtimestamp(t_now + t_remaining)
    }


def get_trend():
    """روند کلی از کندل‌های زنده"""
    bulls, bears = 0, 0
    for name in ['15m', '1H', '4H', '1D']:
        k = KLINE_CACHE.get(name)
        if k and k['open'] > 0:
            if k['close'] >= k['open']:
                bulls += 1
            else:
                bears += 1
    if bulls > bears:
        return 'bull', bulls, bears
    if bears > bulls:
        return 'bear', bulls, bears
    return 'neutral', bulls, bears


def get_volatility_pct():
    """نوسان زنده بازار"""
    pcts = []
    for k in KLINE_CACHE.values():
        if k['low'] > 0:
            pcts.append((k['high'] - k['low']) / k['low'] * 100)
    return float(np.mean(pcts)) if pcts else 0.5


def compute_phi2_candidates(t_now):
    """فقط رویدادهای φ²-Turn را محاسبه و رتبه‌بندی می‌کند"""
    cands = []
    for tf in ALL_TIMEFRAMES:
        ev = find_next_phi2_event(t_now, tf['T'])
        t_rem = ev['time_remaining']
        proximity = 1.0 / (1.0 + t_rem / 7200.0)
        score = 100 * proximity  # φ² همیشه حداکثر قدرت را دارد
        cands.append(dict(tf=tf, ev=ev, stars=5, icon='⚡',
                          color='#FF0055', score=score, t_rem=t_rem))
    cands.sort(key=lambda c: -c['score'])
    return cands


def build_signal(t_now):
    """ساخت سیگنال φ²-Turn با TP/SL/لوریج"""
    price = APP_STATE['price']
    cands = compute_phi2_candidates(t_now)
    if not cands:
        return None, []

    top = cands[0]
    trend, bulls, bears = get_trend()
    vol = get_volatility_pct()

    # ─── جهت: φ² همیشه معکوس روند (قوی‌ترین مقاومت/حمایت) ───
    if trend == 'bull':
        direction = 'SHORT'  # مقاومت → انتظار بازگشت به پایین
    elif trend == 'bear':
        direction = 'LONG'   # حمایت → انتظار بازگشت به بالا
    else:
        direction = 'SHORT'  # پیش‌فرض: مقاومت

    # ─── وضعیت بر اساس زمان تا رویداد ───
    if top['t_rem'] < 900:          # < 15 دقیقه
        status, status_color = '🔥 فعال (ACTIVE)', '#FF0055'
    elif top['t_rem'] < 3600:       # < 1 ساعت
        status, status_color = '⏳ آماده‌سازی (PREPARING)', '#FFA500'
    elif top['t_rem'] < 24 * 3600:  # < 24 ساعت
        status, status_color = '⏳ در انتظار (PENDING)', '#888888'
    else:
        status, status_color = '😴 دور (FAR)', '#555555'

    no_signal = top['t_rem'] > 24 * 3600

    # ─── TP/SL: φ² قوی‌ترین است → SL بزرگ‌تر و مطمئن‌تر ───
    sl_pct = max(0.6, round(vol * 1.5, 2))
    tp1_pct, tp2_pct, tp3_pct = sl_pct, sl_pct * 2, sl_pct * 3

    # ─── لوریج: φ² = قوی‌ترین رویداد = لوریج بالا ───
    base_lev = 25
    T = top['tf']['T']
    tf_factor = 1.0 if T <= 3600 else (0.7 if T <= 86400 else 0.4)
    leverage = max(3, int(base_lev * tf_factor))

    # ─── سطوح قیمت ───
    levels = {}
    if price > 0:
        if direction == 'LONG':
            levels = {
                'entry': price,
                'tp1': price * (1 + tp1_pct / 100),
                'tp2': price * (1 + tp2_pct / 100),
                'tp3': price * (1 + tp3_pct / 100),
                'sl': price * (1 - sl_pct / 100),
            }
        else:
            levels = {
                'entry': price,
                'tp1': price * (1 - tp1_pct / 100),
                'tp2': price * (1 - tp2_pct / 100),
                'tp3': price * (1 - tp3_pct / 100),
                'sl': price * (1 + sl_pct / 100),
            }

    strength = min(100, int(top['score']))

    signal = {
        'id': f"{top['tf']['name']}_PHI2_{int(top['ev']['event_time'].timestamp() // 3600)}",
        'direction': direction,
        'status': status,
        'status_color': status_color,
        'no_signal': no_signal,
        'tf': top['tf'],
        'event': top['ev'],
        'stars': 5,
        'icon': '⚡',
        'color': '#FF0055',
        'strength': strength,
        'trend': trend,
        'bulls': bulls,
        'bears': bears,
        'vol': vol,
        'sl_pct': sl_pct,
        'tp1_pct': tp1_pct,
        'tp2_pct': tp2_pct,
        'tp3_pct': tp3_pct,
        'leverage': leverage,
        'levels': levels,
        'confluence': [c for c in cands if c['t_rem'] < 3600],
        't_rem': top['t_rem'],
    }

    # ─── لاگ سیگنال ───
    if not no_signal and (not SIGNAL_LOG or SIGNAL_LOG[-1]['id'] != signal['id']):
        SIGNAL_LOG.append({
            'id': signal['id'],
            'time': datetime.fromtimestamp(t_now),
            'direction': direction,
            'tf': top['tf']['name'],
            'event': 'φ²-Turn',
            'price': price,
            'strength': strength,
        })
        del SIGNAL_LOG[:-8]

    return signal, cands

def compute_annual_spiral(t_now, timeframes):
    traces = []
    price = APP_STATE["price"]

    theta_bg = np.linspace(0, THETA_MAX, 800)
    traces.append(go.Scatter(
        x=spiral_x(theta_bg), y=spiral_y(theta_bg), mode='lines',
        line=dict(color='#2a2a3e', width=1.5, dash='dot'),
        showlegend=False, hoverinfo='skip', name='🌀 Annual Spiral'
    ))

    for turn in range(N_SPIRAL_TURNS):
        theta_c = np.linspace(turn * 2 * np.pi, (turn + 1) * 2 * np.pi, 120)
        r = A_SPIRAL * np.exp(B_SPIRAL * theta_c)
        traces.append(go.Scatter(
            x=r * np.cos(theta_c), y=r * np.sin(theta_c), mode='lines',
            line=dict(color='#1a1a2e', width=0.5),
            showlegend=False, hoverinfo='skip'
        ))

    seasons = [(0, "🌱 Jan"), (2*np.pi, "🌸 Mar"), (4*np.pi, "☀️ May"),
               (6*np.pi, "🌿 Jul"), (8*np.pi, "🍂 Sep"), (10*np.pi, "❄️ Nov")]
    for theta_s, label_s in seasons:
        if theta_s < THETA_MAX:
            traces.append(go.Scatter(
                x=[spiral_x(theta_s)], y=[spiral_y(theta_s)],
                mode='markers+text',
                marker=dict(color='#555566', size=5),
                text=[label_s], textposition='top right',
                textfont=dict(color='#777788', size=8, family='monospace'),
                showlegend=False, hoverinfo='skip'
            ))

    # ⚡ فقط φ²-Turn روی اسپیرال (بزرگ‌تر و برجسته‌تر)
    traces.append(go.Scatter(
        x=[spiral_x(PHI2_RAD)], y=[spiral_y(PHI2_RAD)],
        mode='markers+text',
        marker=dict(color='#FF0055', size=14, symbol='diamond',
                    line=dict(color='#FFD700', width=2)),
        text=["⚡φ²"], textposition='top center',
        textfont=dict(color='#FF0055', size=12, family='monospace'),
        showlegend=False,
        hovertemplate=f"<b>⚡ φ²-Turn</b><br>θ = {PHI2_RAD/np.pi:.3f}π rad<br>{PHI2_DEG:.1f}°<extra></extra>"
    ))

    # سایر نقاط طلایی (کوچک‌تر)
    ga_nodes = [(GA_RAD, 'GA'), (GA_SUPP_RAD, 'GS'), (GC_RAD, 'GC'),
                (PHI * 2 * np.pi, 'φ')]
    for theta_n, label_n in ga_nodes:
        if theta_n < THETA_MAX:
            traces.append(go.Scatter(
                x=[spiral_x(theta_n)], y=[spiral_y(theta_n)],
                mode='markers+text',
                marker=dict(color='#FF6B00', size=7, symbol='diamond',
                            line=dict(color='#FFD700', width=1)),
                text=[label_n], textposition='top center',
                textfont=dict(color='#FFD700', size=8, family='monospace'),
                showlegend=False, hoverinfo='skip'
            ))

    year_progress = (t_now % YEAR_SECONDS) / YEAR_SECONDS
    theta_year_now = year_progress * THETA_MAX
    traces.append(go.Scatter(
        x=[spiral_x(theta_year_now)], y=[spiral_y(theta_year_now)],
        mode='markers+text',
        marker=dict(color='white', size=11, symbol='star',
                    line=dict(color='#FFD700', width=2)),
        text=[f"📅 {year_progress * 100:.0f}%"],
        textposition='bottom center',
        textfont=dict(color='#FFD700', size=8, family='monospace'),
        showlegend=False,
        hovertemplate=f"<b>📅 Year</b><br>{year_progress * 100:.2f}%<extra></extra>"
    ))

    for tf in timeframes:
        progress = (t_now % tf['T']) / tf['T']
        theta_tf = progress * THETA_MAX

        theta_recent = 3 * 2 * np.pi
        theta_trail_start = max(0, theta_tf - theta_recent)
        n_pts = max(int(400 * (theta_tf - theta_trail_start) / theta_recent), 10)
        theta_trail = np.linspace(theta_trail_start, theta_tf, n_pts)

        traces.append(go.Scatter(
            x=spiral_x(theta_trail), y=spiral_y(theta_trail), mode='lines',
            line=dict(color=tf['color'], width=tf['width']),
            showlegend=True, name=tf['name'],
            hovertemplate=f"<b style='color:{tf['color']}'>{tf['name']}</b><br>📍 {progress * 100:.1f}%<extra></extra>"
        ))

        x_cur = spiral_x(theta_tf)
        y_cur = spiral_y(theta_tf)

        for g in [10, 6]:
            traces.append(go.Scatter(
                x=[x_cur], y=[y_cur], mode='markers',
                marker=dict(color=tf['color'], size=g, opacity=0.3),
                showlegend=False, hoverinfo='skip'
            ))

        hover_pt = f"<b style='color:{tf['color']}'>{tf['name']}</b><br>📊 {progress * 100:.1f}%<br>"
        if price > 0:
            hover_pt += f"💰 ${price:,.2f}"

        traces.append(go.Scatter(
            x=[x_cur], y=[y_cur], mode='markers',
            marker=dict(color='#FFF', size=6, line=dict(color=tf['color'], width=2)),
            showlegend=False,
            hovertemplate=hover_pt + "<extra></extra>"
        ))

    return traces

    fig.update_layout(

                    bgcolor="rgba(10,10,18,0.8)", font=dict(size=8, family="monospace")),
        shapes=shapes,
        xaxis=dict(visible=False, scaleanchor="y", scaleratio=1, range=[-15, 15]),
        yaxis=dict(visible=False, scaleanchor="x", scaleratio=1, range=[-15, 15]),

def render_key_nodes(t_now):
    year_progress = (t_now % YEAR_SECONDS) / YEAR_SECONDS
    theta_year_now = year_progress * THETA_MAX

    key_nodes = [
        ("⚡ φ²-Turn (2.618 turns)", PHI2_RAD, "قوی‌ترین مقاومت - فقط این سیگنال صادر می‌شود"),
        ("1× Golden Angle (GA)", GA_RAD, "مقاومت اولیه - 137.5°"),
        ("2× Golden Angle", 2 * GA_RAD, "مقاومت دوتایی قوی"),
        ("Golden Supplement (GS)", GA_SUPP_RAD, "مکمل طلایی - 222.5°"),
        ("🔥 Golden Convergence (GC)", GC_RAD, "بازگشت قوی - 317.5°"),
        ("φ-Turn (1.618 turns)", PHI * 2 * np.pi, "چرخش طلایی کامل"),
        ("2× φ-Turn", 2 * PHI * 2 * np.pi, "چرخش دوتایی"),
        ("3× Golden Angle", 3 * GA_RAD, "تشدید سه‌گانه"),
        ("5× Golden Angle", 5 * GA_RAD, "کلاستر فیبوناچی"),
        ("8× Golden Angle", 8 * GA_RAD, "کلاستر فیبوناچی"),
        ("Spiral Midpoint (3π)", 3 * np.pi, "نقطه محوری نیمه"),
        ("Spiral Terminus (6π)", 6 * np.pi, "پایان اسپیرال سالانه"),
    ]

    rows = [html.Tr([
        html.Th('نوع نقطه', style=HEADER_CELL), html.Th('θ (rad)', style=HEADER_CELL),
        html.Th('θ (deg)', style=HEADER_CELL), html.Th('θ (π)', style=HEADER_CELL),
        html.Th('% سال', style=HEADER_CELL), html.Th('بعدی', style=HEADER_CELL),
        html.Th('توضیحات', style=HEADER_CELL),
    ])]

    for name, theta, significance in key_nodes:
        if theta <= THETA_MAX:
            year_pct = (theta / THETA_MAX) * 100
            theta_deg = theta * 180 / np.pi

            if theta > theta_year_now:
                t_to = YEAR_SECONDS * ((theta - theta_year_now) / THETA_MAX)
            else:
                t_to = YEAR_SECONDS * ((THETA_MAX - theta_year_now + theta) / THETA_MAX)
            dt_next = datetime.fromtimestamp(t_now + t_to)

            if 'φ²' in name:
                color, bg = '#FF0055', 'rgba(255,0,85,0.2)'
            elif 'φ-Turn' in name or 'Convergence' in name or 'Terminus' in name:
                color, bg = '#FF6B00', 'rgba(255,107,0,0.1)'
            elif ('3×' in name or '5×' in name or '8×' in name):
                color, bg = '#FFD700', 'rgba(255,215,0,0.08)'
            else:
                color, bg = '#4ECDC4', '#0f0f1a'

            cell = {'padding': '8px', 'textAlign': 'center', 'color': '#e0e0e0'}
            rows.append(html.Tr([
                html.Td(html.Span(name, style={'color': color, 'fontWeight': 'bold'}), style=cell),
                html.Td(f"{theta:.3f}", style={**cell, 'color': '#aaa'}),
                html.Td(f"{theta_deg:.2f}°", style={**cell, 'color': '#aaa'}),
                html.Td(f"{theta/np.pi:.3f}π", style={**cell, 'color': '#4ECDC4'}),
                html.Td(f"{year_pct:.1f}%", style={**cell, 'color': '#FFD700'}),
                html.Td(dt_next.strftime('%Y-%m-%d %H:%M'),
                        style={**cell, 'color': '#888', 'fontSize': '9px'}),
                html.Td(significance, style={**cell, 'color': '#ccc', 'fontSize': '9px'}),
            ], style={'backgroundColor': bg, 'borderBottom': '1px solid #1a1a2e',
                      'fontFamily': 'monospace', 'fontSize': '10px'}))

    return html.Div([
        html.H4("🌟 نقاط کلیدی طلایی سالانه", style={
            'color': '#FFD700', 'fontFamily': 'monospace', 'marginBottom': '15px',
            'textShadow': '0 0 8px rgba(255,215,0,0.3)'}),
        html.Table(rows, style=TABLE_STYLE)
    ])

@app.callback(
    [Output("tab-content", "children"),
     Output("price-display", "children"),
     Output("status-bar", "children")],
    [Input("interval", "n_intervals"),
     Input("main-tabs", "value")]
)
def update(n, active_tab):
    t_now = time.time()
    price = APP_STATE["price"]
    connected = APP_STATE["connected"]

    price_text = f"💰 ${price:,.2f}" if price > 0 else "💰 Loading..."
    status_text = (f"{'🟢' if connected else '🔴'} Live | {SYMBOL} | "
                   f"φ={PHI:.4f} | φ²={PHI2_DEG:.1f}° | θ_max={THETA_MAX/np.pi:.1f}π")

    if active_tab == 'tab-signal':
        content = render_live_signal(t_now)
    elif active_tab == 'tab-spiral':
        traces = compute_annual_spiral(t_now, ALL_TIMEFRAMES)
        content = dcc.Graph(figure=build_spiral_figure(traces),
                            config={'displayModeBar': True, 'scrollZoom': True},
                            style={'height': '100%'})
    elif active_tab == 'tab-short':
        content = render_resistance_table(
            compute_golden_resistances(t_now, SHORT_TF),
            "مقاومت‌های کوتاه‌مدت (1m → 89m)", "🔱")
    elif active_tab == 'tab-mid':
        content = render_resistance_table(
            compute_golden_resistances(t_now, MID_TF),
            "مقاومت‌های میان‌مدت (144m → 4181m)", "🔱")
    elif active_tab == 'tab-long':
        content = render_resistance_table(
            compute_golden_resistances(t_now, LONG_TF),
            "مقاومت‌های بلندمدت (6765m → 46368m)", "🔱")
    elif active_tab == 'tab-calendar':
        content = render_golden_calendar(t_now)
    elif active_tab == 'tab-nodes':
        content = render_key_nodes(t_now)
    else:
        content = html.Div("Tab not found", style={'color': '#888'})

    return content, price_text, status_text
