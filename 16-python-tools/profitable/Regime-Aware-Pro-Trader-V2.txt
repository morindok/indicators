REST_CANDIDATES = ["https://api.bybit.com", "https://api.bytick.com", "https://api.bybit.kz"]

_ACTIVE = {"url": None}


def bybit_get(path, params, timeout=10):
    cands = ([_ACTIVE["url"]] if _ACTIVE["url"] else []) + \
            [b for b in REST_CANDIDATES if b != _ACTIVE["url"]]
    for base in cands:
        try:
            r = SESSION.get(f"{base}{path}", params=params, timeout=timeout)
            if r.status_code in (403, 451):
                continue
            r.raise_for_status()
            d = r.json()
            if d.get("retCode") == 0:
                _ACTIVE["url"] = base
                return d
        except Exception:
            continue
    return None


def get_klines(symbol, interval, category="linear", limit=1000, end_time=None):
    params = {"category": category, "symbol": symbol, "interval": interval, "limit": limit}
    if end_time is not None:
        if end_time.tzinfo is None:
            end_time = pytz.utc.localize(end_time)
        params["end"] = int(end_time.timestamp() * 1000)
    d = bybit_get("/v5/market/kline", params)
    if not d or "list" not in (d.get("result") or {}):
        return pd.DataFrame()
    lst = d["result"]["list"]
    if not lst:
        return pd.DataFrame()
    df = pd.DataFrame(lst, columns=["ts", "open", "high", "low", "close", "volume", "turnover"])
    df["ts"] = pd.to_datetime(df["ts"].astype(int), unit="ms", utc=True)
    for c in ["open", "high", "low", "close", "volume"]:
        df[c] = df[c].astype(float)
    return df.sort_values("ts").reset_index(drop=True)


def get_klines_paged(symbol, interval, category, pages=2, per=1000, end_time=None):
    """دریافت چند صفحه برای تاریخچه‌ی بلندتر (~3 ماه)"""
    frames = []
    end = end_time
    for _ in range(pages):
        df = get_klines(symbol, interval, category, limit=per, end_time=end)
        if df.empty:
            break
        frames.append(df)
        end = df["ts"].iloc[0] - pd.Timedelta(milliseconds=1)
    if not frames:
        return pd.DataFrame()
    return pd.concat(frames).drop_duplicates("ts").sort_values("ts").reset_index(drop=True)

def ema(a, s):
    return pd.Series(a).ewm(span=s, adjust=False).mean().values


def rsi(close, period=14):
    d = np.diff(close)
    g = np.where(d > 0, d, 0)
    l = np.where(d < 0, -d, 0)
    ag = pd.Series(g).ewm(span=period, adjust=False).mean().values
    al = pd.Series(l).ewm(span=period, adjust=False).mean().values
    return 100 - 100 / (1 + ag / (al + 1e-10))


def atr_arr(h, l, c, period=14):
    pc = np.concatenate([[c[0]], c[:-1]])
    tr = np.maximum(h - l, np.maximum(abs(h - pc), abs(l - pc)))
    return pd.Series(tr).ewm(span=period, adjust=False).mean().values


def adx_arr(h, l, c, period=14):
    ph = np.concatenate([[h[0]], h[:-1]])
    pl = np.concatenate([[l[0]], l[:-1]])
    up = h - ph
    dn = pl - l
    pdm = np.where((up > dn) & (up > 0), up, 0)
    mdm = np.where((dn > up) & (dn > 0), dn, 0)
    pc = np.concatenate([[c[0]], c[:-1]])
    tr = np.maximum(h - l, np.maximum(abs(h - pc), abs(l - pc)))
    atr = pd.Series(tr).ewm(span=period, adjust=False).mean().values
    pdi = 100 * pd.Series(pdm).ewm(span=period, adjust=False).mean().values / (atr + 1e-10)
    mdi = 100 * pd.Series(mdm).ewm(span=period, adjust=False).mean().values / (atr + 1e-10)
    dx = 100 * abs(pdi - mdi) / (pdi + mdi + 1e-10)
    adx = pd.Series(dx).ewm(span=period, adjust=False).mean().values
    return adx, pdi, mdi

def run_strategy(symbol, category, leverage=10):
    df4 = get_klines_paged(symbol, "240", category, pages=1)
    df1 = get_klines_paged(symbol, "60", category, pages=2)
    if df4.empty or df1.empty or len(df1) < 300:
        return None

    c = df1["close"].values
    h = df1["high"].values
    l = df1["low"].values
    ts = df1["ts"].values
    n = len(c)

    e20 = ema(c, 20)
    e50 = ema(c, 50)
    a = atr_arr(h, l, c)
    a = np.where(a <= 0, c * 0.001, a)
    r = rsi(c, 14)
    z = (c - e50) / a

    adx4, pdi4, mdi4 = adx_arr(df4["high"].values, df4["low"].values, df4["close"].values)
    ts4 = df4["ts"].astype("int64").values
    ts1 = df1["ts"].astype("int64").values
    i4 = np.clip(np.searchsorted(ts4, ts1, side="right") - 1, 0, len(ts4) - 1)
    adx1, pdi1, mdi1 = adx4[i4], pdi4[i4], mdi4[i4]

    trades = []
    equity = [INITIAL_CAPITAL]
    cap = INITIAL_CAPITAL
    in_pos = False
    pos = None
    start = 250

    for i in range(start, n):
        if in_pos:
            p = pos
            close_price = c[i]
            closed = False
            exitp = close_price

            if p["type"] == "long":
                # Trailing برای روند
                if p["mode"] == "TR":
                    if close_price >= p["entry"] + 1 * p["slp"]:
                        p["sl"] = max(p["sl"], p["entry"])
                    if close_price >= p["entry"] + 2 * p["slp"]:
                        p["sl"] = max(p["sl"], close_price - p["atr_e"])
                if close_price >= p["tp"]:
                    closed, exitp, res = True, p["tp"], "win"
                elif close_price <= p["sl"]:
                    closed, exitp = True, p["sl"]
                    res = "win" if p["sl"] > p["entry"] else ("be" if p["sl"] == p["entry"] else "loss")
                elif i - p["i"] >= p["timeout"]:
                    closed, exitp = True, close_price
                    res = "win" if close_price > p["entry"] else "loss"
            else:
                if p["mode"] == "TR":
                    if close_price <= p["entry"] - 1 * p["slp"]:
                        p["sl"] = min(p["sl"], p["entry"])
                    if close_price <= p["entry"] - 2 * p["slp"]:
                        p["sl"] = min(p["sl"], close_price + p["atr_e"])
                if close_price <= p["tp"]:
                    closed, exitp, res = True, p["tp"], "win"
                elif close_price >= p["sl"]:
                    closed, exitp = True, p["sl"]
                    res = "win" if p["sl"] < p["entry"] else ("be" if p["sl"] == p["entry"] else "loss")
                elif i - p["i"] >= p["timeout"]:
                    closed, exitp = True, close_price
                    res = "win" if close_price < p["entry"] else "loss"

            if closed:
                move = (exitp - p["entry"]) / p["entry"] if p["type"] == "long" \
                    else (p["entry"] - exitp) / p["entry"]
                net = p["f"] * (move - 2 * FEE)
                cap *= (1 + net)
                trades.append({**p, "exit_idx": i, "exit_time": ts[i],
                               "exit_price": exitp, "pnl_pct": net, "result": res})
                equity.append(cap)
                in_pos = False
            continue

        # --- ورود ---
        if a[i] / c[i] < 0.0015:
            continue

        sl_dist_mult = MR_SL if adx1[i] < ADX_RANGE else TR_SL
        slp = sl_dist_mult * a[i]
        sl_pct = slp / c[i]
        f = min(leverage, RISK / sl_pct) if sl_pct > 0 else 1.0

        setup = None
        if adx1[i] < ADX_RANGE:
            if z[i] < -Z_TH:
                setup = ("long", c[i] + MR_TP * a[i], c[i] - MR_SL * a[i], MR_TIMEOUT, "MR")
            elif z[i] > Z_TH:
                setup = ("short", c[i] - MR_TP * a[i], c[i] + MR_SL * a[i], MR_TIMEOUT, "MR")
        elif adx1[i] >= ADX_TREND:
            upt = pdi1[i] > mdi1[i]
            near = abs(c[i] - e20[i]) <= 0.5 * a[i]
            if upt and near and 40 <= r[i] <= 60:
                setup = ("long", c[i] + TR_TP * a[i], c[i] - TR_SL * a[i], TR_TIMEOUT, "TR")
            elif (not upt) and near and 40 <= r[i] <= 60:
                setup = ("short", c[i] - TR_TP * a[i], c[i] + TR_SL * a[i], TR_TIMEOUT, "TR")

        if setup:
            t_, tp_, sl_, to_, mode_ = setup
            pos = {"type": t_, "i": i, "entry_time": ts[i], "entry": c[i],
                   "tp": tp_, "sl": sl_, "slp": sl_ - c[i] if t_ == "long" else c[i] - sl_,
                   "atr_e": a[i], "timeout": to_, "mode": mode_, "f": f,
                   "lev": f}
            in_pos = True

    if len(trades) < 10:
        return None

    wins = [t for t in trades if t["result"] == "win"]
    losses = [t for t in trades if t["result"] == "loss"]
    wr = len(wins) / len(trades)
    gw = sum(t["pnl_pct"] for t in wins) if wins else 0
    gl = abs(sum(t["pnl_pct"] for t in losses)) if losses else 1e-9
    pf = gw / gl if gl > 0 else float("inf")
    eq = np.array(equity)
    peak = np.maximum.accumulate(eq)
    max_dd = float(np.min((eq - peak) / peak)) * 100
    expectancy = np.mean([t["pnl_pct"] for t in trades])

    # سیگنال فعلی + setup برای نمایش TP/SL
    sig, setup_now = "انتظار", None
    slp_now = 0
    if adx1[-1] < ADX_RANGE:
        if z[-1] < -Z_TH: sig, setup_now = "LONG (MR) 🟢", ("long", c[-1]+MR_TP*a[-1], c[-1]-MR_SL*a[-1])
        elif z[-1] > Z_TH: sig, setup_now = "SHORT (MR) 🔴", ("short", c[-1]-MR_TP*a[-1], c[-1]+MR_SL*a[-1])
    elif adx1[-1] >= ADX_TREND:
        if pdi1[-1] > mdi1[-1] and abs(c[-1]-e20[-1]) <= 0.5*a[-1]:
            sig, setup_now = "LONG (TR) 🟢", ("long", c[-1]+TR_TP*a[-1], c[-1]-TR_SL*a[-1])
        elif mdi1[-1] > pdi1[-1] and abs(c[-1]-e20[-1]) <= 0.5*a[-1]:
            sig, setup_now = "SHORT (TR) 🔴", ("short", c[-1]-TR_TP*a[-1], c[-1]+TR_SL*a[-1])

    return {"trades": trades, "df1": df1, "equity": equity, "wr": wr, "pf": pf,
            "max_dd": max_dd, "expectancy": expectancy, "final": eq[-1],
            "ret": (eq[-1]/INITIAL_CAPITAL-1)*100, "signal": sig, "setup_now": setup_now,
            "adx_now": adx1[-1], "price": float(c[-1]), "leverage": leverage}

def build_chart(st):
    fig = go.Figure()
    df = st["df1"]
    n = len(df)
    show = min(300, n)
    off = n - show
    x = np.arange(off, n)

    fig.add_trace(go.Candlestick(x=x, open=df["open"].values[off:],
        high=df["high"].values[off:], low=df["low"].values[off:],
        close=df["close"].values[off:], name="قیمت",
        increasing_line_color=UP, decreasing_line_color=DN))

    # TP/SL برای تریدهای اخیر داخل پنجره
    for t in st["trades"]:
        if t["i"] >= off:
            col = UP if t["result"] == "win" else DN
            sym = "triangle-up" if t["type"] == "long" else "triangle-down"
            fig.add_trace(go.Scatter(x=[t["i"], t["exit_idx"]], y=[t["entry"], t["entry"]],
                mode="lines", line=dict(color=TXT, width=1), showlegend=False, hoverinfo="skip"))
            fig.add_trace(go.Scatter(x=[t["i"], t["exit_idx"]], y=[t["tp"], t["tp"]],
                mode="lines", line=dict(color=UP, width=1, dash="dash"), showlegend=False, hoverinfo="skip"))
            fig.add_trace(go.Scatter(x=[t["i"], t["exit_idx"]], y=[t["sl"], t["sl"]],
                mode="lines", line=dict(color=DN, width=1, dash="dash"), showlegend=False, hoverinfo="skip"))
            fig.add_trace(go.Scatter(x=[t["i"]], y=[t["entry"]], mode="markers",
                marker=dict(color=col, size=11, symbol=sym, line=dict(color="#000", width=2)),
                showlegend=False,
                hovertemplate=f"<b>{t['mode']} {t['type']} {t['result']}</b><br>"
                              f"E:{t['entry']:,.0f} TP:{t['tp']:,.0f} SL:{t['sl']:,.0f}<br>"
                              f"PnL:{t['pnl_pct']*100:+.2f}%<extra></extra>"))

    # TP/SL سیگنال فعلی
    if st["setup_now"]:
        t_, tp_, sl_ = st["setup_now"]
        fig.add_hline(y=st["price"], line=dict(color=TXT, width=1, dash="dot"))
        fig.add_hline(y=tp_, line=dict(color=UP, width=1.5, dash="dash"),
                      annotation_text=f"TP {tp_:,.0f}")
        fig.add_hline(y=sl_, line=dict(color=DN, width=1.5, dash="dash"),
                      annotation_text=f"SL {sl_:,.0f}")

    fig.update_layout(template="plotly_dark", paper_bgcolor=BG, plot_bgcolor=BG,
        font=dict(color=TXT, family=FONT_FA), showlegend=False,
        xaxis=dict(gridcolor=LINE), yaxis=dict(gridcolor=LINE),
        margin=dict(l=10, r=10, t=40, b=10),
        title=dict(text=f"🏆 Pro V2 — اهرم {st['leverage']}x | {len(st['trades'])} ترید", x=0.5,
                   font=dict(color=GOLD, size=15, family=FONT_FA)))
    return fig

def build_trades_table(st):
    cutoff = pd.Timestamp.now(pytz.UTC) - timedelta(days=90)
    rows = []
    for t in reversed(st["trades"]):
        # ⭐ اصلاح: تبدیل به tz-aware برای مقایسه‌ی صحیح
        et = pd.Timestamp(t["entry_time"])
        if et.tzinfo is None:
            et = et.tz_localize("UTC")

        if et < cutoff:
            continue

        rc = UP if t["result"] == "win" else (MUT if t["result"] == "be" else DN)
        rl = {"win": "✅ برد", "loss": "❌ باخت", "be": "➖ سر به سر"}.get(t["result"], t["result"])

        rows.append(html.Tr([
            html.Td(et.strftime("%m-%d %H:%M"), style={"color": MUT, "fontSize": 10}),
            html.Td(t["type"].upper(), style={"color": UP if t["type"]=="long" else DN,
                                              "fontWeight": "bold", "fontSize": 11}),
            html.Td(t["mode"], style={"color": GOLD, "fontSize": 10}),
            html.Td(f"{t['lev']:.1f}x", style={"color": TXT, "fontSize": 10}),
            html.Td(f"${t['entry']:,.0f}", style={"color": TXT, "fontSize": 10}),
            html.Td(f"${t['tp']:,.0f}", style={"color": UP, "fontSize": 10}),
            html.Td(f"${t['sl']:,.0f}", style={"color": DN, "fontSize": 10}),
            html.Td(f"${t['exit_price']:,.0f}", style={"color": TXT, "fontSize": 10}),
            html.Td(f"{t['pnl_pct']*100:+.2f}%", style={"color": rc, "fontWeight": "bold", "fontSize": 11}),
            html.Td(rl, style={"color": rc, "fontSize": 11}),
        ], style={"borderBottom": f"1px solid {LINE}"}))

    if not rows:
        rows.append(html.Tr([html.Td("معامله‌ای در ۳ ماه اخیر نیست", colSpan=10,
                                     style={"color": MUT, "textAlign": "center"})]))

    return dbc.Table([
        html.Thead(html.Tr([
            html.Th("ورود", style={"color": GOLD, "fontSize": 9}),
            html.Th("نوع", style={"color": GOLD, "fontSize": 9}),
            html.Th("حالت", style={"color": GOLD, "fontSize": 9}),
            html.Th("اهرم", style={"color": GOLD, "fontSize": 9}),
            html.Th("Entry", style={"color": GOLD, "fontSize": 9}),

            html.Th("Exit", style={"color": GOLD, "fontSize": 9}),
            html.Th("PnL", style={"color": GOLD, "fontSize": 9}),
            html.Th("نتیجه", style={"color": GOLD, "fontSize": 9}),
        ], style={"borderBottom": f"2px solid {GOLD}"})),
