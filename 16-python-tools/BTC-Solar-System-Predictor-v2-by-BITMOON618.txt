def calculate_all_indicators(df):
    close = df['Close']
    high = df['High']
    low = df['Low']

    delta = close.diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    rs = gain / loss
    df['RSI'] = 100 - (100 / (1 + rs))

    ema12 = close.ewm(span=12, adjust=False).mean()
    ema26 = close.ewm(span=26, adjust=False).mean()
    df['MACD'] = ema12 - ema26
    df['MACD_Signal'] = df['MACD'].ewm(span=9, adjust=False).mean()
    df['MACD_Hist'] = df['MACD'] - df['MACD_Signal']

    df['BB_Middle'] = close.rolling(20).mean()
    bb_std = close.rolling(20).std()
    df['BB_Upper'] = df['BB_Middle'] + (bb_std * 2)
    df['BB_Lower'] = df['BB_Middle'] - (bb_std * 2)

    df['MA20'] = close.rolling(20).mean()
    df['MA50'] = close.rolling(50).mean()
    df['MA100'] = close.rolling(100).mean()
    df['MA200'] = close.rolling(200).mean()

    low_14 = low.rolling(14).min()
    high_14 = high.rolling(14).max()
    df['Stoch_K'] = 100 * (close - low_14) / (high_14 - low_14)
    df['Stoch_D'] = df['Stoch_K'].rolling(3).mean()

    tr1 = high - low
    tr2 = abs(high - close.shift())
    tr3 = abs(low - close.shift())
    df['TR'] = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
    df['ATR'] = df['TR'].rolling(14).mean()

    df['Volume_MA'] = df['Volume'].rolling(20).mean()
    df['ROC'] = ((close - close.shift(10)) / close.shift(10)) * 100

    return df

def predict_price(df, planets):
    latest = df.iloc[-1]
    current_price = latest['Close']

    weights = {
        "عطارد (RSI)": 0.15, "زهره (MACD)": 0.20,
        "زمين (MA20)": 0.10, "مريخ (MA50)": 0.15,
        "مشتري (MA200)": 0.20, "زحل (Bollinger)": 0.10,
        "اورانوس (Stoch)": 0.05, "نپتون (Momentum)": 0.05
    }

    total_score = 0
    bullish = 0
    bearish = 0

    for planet, data in planets.items():
        weight = weights[planet]
        pos = data['position']

        if "RSI" in planet or "Stoch" in planet:
            if pos > 70:
                score = -weight * (pos - 70) / 30
                bearish += 1
            elif pos < 30:
                score = weight * (30 - pos) / 30
                bullish += 1
            else:
                score = 0
        else:
            if data['bullish']:
                score = weight * min(abs(pos) / 5, 1)
                bullish += 1
            else:
                score = -weight * min(abs(pos) / 5, 1)
                bearish += 1

        total_score += score

    if total_score > 0.2:
        trend = "صعودي قوي"
    elif total_score > 0:
        trend = "صعودي ضعيف"
    elif total_score > -0.2:
        trend = "نزولي ضعيف"
    else:
        trend = "نزولي قوي"

    volatility = latest['ATR'] / current_price

    predictions = {}
    for tf, mult, conf in [("1 ساعته", 0.5, 85), ("24 ساعته", 2, 75),
                           ("7 روزه", 5, 65), ("30 روزه", 15, 50)]:
        predictions[tf] = {
            "price": current_price * (1 + total_score * volatility * mult),
            "confidence": conf,
            "range_low": current_price * (1 - volatility * mult),
            "range_high": current_price * (1 + volatility * mult)
        }

    return {
        "current_price": current_price,
        "trend": trend,
        "total_score": total_score,
        "bullish": bullish,
        "bearish": bearish,
        "predictions": predictions,
        "volatility": volatility * 100
    }

def run_backtest(df, step_size=24, forecast_hours=24):
    predictions_list = []
    start_idx = 200

    for i in range(start_idx, len(df) - forecast_hours, step_size):
        historical_df = df.iloc[:i + 1].copy()
        historical_df = calculate_all_indicators(historical_df)

        planets = planetary_analysis(historical_df)
        prediction = predict_price(historical_df, planets)

        actual_price = df['Close'].iloc[i + forecast_hours]
        predicted_price = prediction['predictions']['24 ساعته']['price']

        error = ((actual_price - predicted_price) / predicted_price) * 100
        mae = abs(error)

        predictions_list.append({
            'timestamp': df.index[i],
            'actual_price': actual_price,
            'predicted_price': predicted_price,
            'error_pct': error,
            'mae': mae,
            'trend': prediction['trend'],
            'score': prediction['total_score'],
            'volatility': prediction['volatility']
        })

    return pd.DataFrame(predictions_list)


def calculate_backtest_metrics(backtest_df):
    return {
        'total_predictions': len(backtest_df),
        'avg_error': backtest_df['error_pct'].mean(),
        'std_error': backtest_df['error_pct'].std(),
        'mae': backtest_df['mae'].mean(),
        'rmse': np.sqrt((backtest_df['error_pct'] ** 2).mean()),
        'accuracy': 100 - backtest_df['mae'].mean(),
        'positive_predictions': (backtest_df['error_pct'] > 0).sum(),
        'negative_predictions': (backtest_df['error_pct'] < 0).sum(),
        'max_profit': backtest_df['error_pct'].max(),
        'max_loss': backtest_df['error_pct'].min(),
        'win_rate': ((backtest_df['error_pct'] > -2) & (backtest_df['error_pct'] < 2)).sum() / len(backtest_df) * 100
    }

        marker_color=[GREEN, GOLD, RED, PURPLE][:len(trend_accuracy)],
        opacity=0.8
    ), row=1, col=2)

    cumulative_error = backtest_df['mae'].cumsum() / np.arange(1, len(backtest_df) + 1)
    fig.add_trace(go.Scatter(
        x=np.arange(len(backtest_df)), y=cumulative_error,
        mode='lines', name='MAE تجمعي',
        line=dict(color=GOLD, width=2)
    ), row=2, col=1)

    fig.add_trace(go.Box(
        y=backtest_df['error_pct'], name='خطا',
        marker_color=ORANGE, boxpoints='outliers'
    ), row=2, col=2)

    fig.update_layout(
        height=600, template='plotly_dark',
        paper_bgcolor='rgba(0,0,0,0)', plot_bgcolor='rgba(0,0,0,0)',
        font=dict(color=TEXT_COLOR, family='Tahoma', size=10),
        margin=dict(l=50, r=20, t=50, b=20), showlegend=False
    )

    return fig

def build_prediction_table(prediction):
    """ساخت جدول پيش‌بيني - جدا شده براي خوانايي بهتر"""
    rows = []
    current = prediction['current_price']

    for tf, data in prediction['predictions'].items():
        change_pct = ((data['price'] - current) / current) * 100
        change_str = f"+{change_pct:.2f}%" if change_pct > 0 else f"{change_pct:.2f}%"
        change_color = GREEN if change_pct > 0 else RED
        border_color = GREEN if change_pct > 0 else RED

        row = html.Div([
            html.Div([
                html.Span(tf, style={"color": TEXT_COLOR, "fontSize": "14px", "fontWeight": "bold"}),
                html.Span(f" ({data['confidence']}% اطمينان)",
                          style={"color": "rgba(255,255,255,0.4)", "fontSize": "11px"}),
            ]),
            html.Div([
                html.Span(f"${data['price']:,.0f}",
                          style={"color": GOLD, "fontSize": "18px", "fontWeight": "bold"}),
                html.Span(f" ({change_str})",
                          style={"color": change_color, "fontSize": "13px", "marginRight": "8px"}),
            ]),
            html.Div([
                html.Span(f"محدوده: ${data['range_low']:,.0f} - ${data['range_high']:,.0f}",
                          style={"color": "rgba(255,255,255,0.5)", "fontSize": "11px"}),
            ]),
        ], style={
            "background": "rgba(255,255,255,0.03)",
            "borderRadius": "8px",
            "padding": "12px 15px",
            "margin": "8px 0",
            "borderRight": f"4px solid {border_color}",
        })
        rows.append(row)

    return html.Div(rows, style={"padding": "10px 20px"})
