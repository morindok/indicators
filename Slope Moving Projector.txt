calculateAverage(s1, s2, s3, s4, s5, s6, s7, s8, s9, method) =>
    float result = na
    
    if method == "Arithmetic"
        result := (s1 + s2 + s3 + s4 + s5 + s6 + s7 + s8 + s9) / 9
    
    else if method == "Weighted"
        result := (s1*1 + s2*2 + s3*3 + s4*4 + s5*5 + s6*6 + s7*7 + s8*8 + s9*9) / 45
    
    else if method == "Exponential"
        alpha = 0.3
        ema = s1
        ema := alpha * s2 + (1 - alpha) * ema
        ema := alpha * s3 + (1 - alpha) * ema
        ema := alpha * s4 + (1 - alpha) * ema
        ema := alpha * s5 + (1 - alpha) * ema
        ema := alpha * s6 + (1 - alpha) * ema
        ema := alpha * s7 + (1 - alpha) * ema
        ema := alpha * s8 + (1 - alpha) * ema
        ema := alpha * s9 + (1 - alpha) * ema
        result := ema
    
    else if method == "Median"
        array<float> slopes = array.new_float(9)
        array.set(slopes, 0, s1)
        array.set(slopes, 1, s2)
        array.set(slopes, 2, s3)
        array.set(slopes, 3, s4)
        array.set(slopes, 4, s5)
        array.set(slopes, 5, s6)
        array.set(slopes, 6, s7)
        array.set(slopes, 7, s8)
        array.set(slopes, 8, s9)
        array.sort(slopes)
        result := array.get(slopes, 4)
    
    else if method == "Harmonic"
        sum_reciprocal = 1/s1 + 1/s2 + 1/s3 + 1/s4 + 1/s5 + 1/s6 + 1/s7 + 1/s8 + 1/s9
        result := 9 / sum_reciprocal
    
    result
