z_threshold = input.float(0.8, title="¬” «‰Â «⁄ »«— ¬„«—Ì (Z-Score)", step=0.1, minval=0.1, tooltip="„⁄Ì«— Z-Score »—«Ì  ‘ŒÌ’ ”Ìê‰«·ùÂ«Ì „⁄ »—. „ﬁ«œÌ— »«·« — ”Ìê‰«·ùÂ«Ì ò„ — «„« œﬁÌﬁù —Ì „ÌùœÂ‰œ.")

// ---  Ê«»⁄ ---
//  «»⁄ œ—Ì«›  ÅÌÊÊ ùÂ« Ê “„«‰ ¬‰ùÂ«
f_get_pivots(len) =>
    ph = ta.pivothigh(high, len, len)
    pl = ta.pivotlow(low, len, len)
    
    var float last_ph_price = na
    var int   last_ph_time  = na
    var float last_pl_price = na
    var int   last_pl_time  = na
    
    if not na(ph)
        last_ph_price := ph
        last_ph_time  := time[len]
        
    if not na(pl)
        last_pl_price := pl
        last_pl_time  := time[len]
        
    [last_ph_price, last_ph_time, last_pl_price, last_pl_time]

//  «»⁄ „Õ«”»Â —Ê‰œ »— «”«” ÅÌÊÊ ùÂ« Ê ﬁÌ„  ·«ÌÊ
f_calc_trend(php, pht, plp, plt) =>
    string trend_status = "œ— «‰ Ÿ«—..."
    color  trend_color  = color.gray
    bool   is_range     = false
    
    if not na(php) and not na(plp) and not na(pht) and not na(plt)
        ctf_ms = timeframe.in_seconds(timeframe.period) * 1000
        bars_ph = math.max(1, (time - pht) / ctf_ms)
        bars_pl = math.max(1, (time - plt) / ctf_ms)
        
        slope_pl = ((close - plp) / plp) * 100 / bars_pl
        slope_ph = ((close - php) / php) * 100 / bars_ph
        
        if slope_pl > 0 and slope_ph > 0
            trend_status := "’⁄ÊœÌ (Bullish)"
            trend_color  := color.green
        else if slope_pl < 0 and slope_ph < 0
            trend_status := "‰“Ê·Ì (Bearish)"
            trend_color  := color.red
        else
            // „‰ÿﬁ Ê÷⁄Ì  —‰Ã Ê „Õ«”»Â Œÿ „Ì«‰Ì
            is_range := true
            mid_price = (php + plp) / 2
            dist_pct = (math.abs(close - mid_price) / mid_price) * 100
            dist_str = str.tostring(dist_pct, "#.##") + "%"
            
            if close > mid_price
                trend_status := "œ— «‰ Ÿ«— ’⁄Êœ\n(›«’·Â: " + dist_str + ")"
                trend_color  := color.new(color.teal, 10)
            else if close < mid_price
                trend_status := "œ— «‰ Ÿ«— ‰“Ê·\n(›«’·Â: " + dist_str + ")"
                trend_color  := color.new(color.orange, 10)
            else
                trend_status := "Ê”ÿ —‰Ã"
                trend_color  := color.gray
            
    [trend_status, trend_color, is_range]

// --- ”Ì” „ ¬„«—Ì ÂÊ‘„‰œ (Smart Statistical System) ---
// „Õ«”»Â ‘Ì»ùÂ«Ì ﬁÌ„ Ì œ— »«“ÂùÂ«Ì „Œ ·› (Slopes)
slope_5  = (close - close[5]) / close[5] * 100
slope_10 = (close - close[10]) / close[10] * 100
slope_20 = (close - close[20]) / close[20] * 100

// „Õ«”»Â ‘Ì» „Ì«‰êÌ‰ „ Õ—ò ‰„«ÌÌ (EMA Slope)
ema10 = ta.ema(close, 10)
ema10_slope = (ema10 - ema10[5]) / ema10[5] * 100

// „Õ«”»Â ‘ «» ﬁÌ„  (Acceleration -  €ÌÌ—«  ‘Ì» òÊ «Âù„œ )
accel_5 = slope_5 - slope_5[5]

// «„ Ì«“œÂÌ  —òÌ»Ì (Composite Score) »« Ê“‰ùœÂÌ »Â ⁄Ê«„· „Œ ·›
stat_score = slope_5 * 0.4 + slope_10 * 0.3 + slope_20 * 0.2 + ema10_slope * 0.1 + accel_5 * 0.5

// „Õ«”»Â „⁄Ì«— ¬„«—Ì (Z-Score) »—«Ì ”‰Ã‘ «⁄ »«— ”Ìê‰«·
score_mean = ta.sma(stat_score, 100)
score_std  = ta.stdev(stat_score, 100)
z_score    = score_std > 0 ? (stat_score - score_mean) / score_std : 0

// „‰ÿﬁ ÅÌ‘ù»Ì‰Ì —Ê‰œ »«  Œ„Ì‰ „œ  «⁄ »«—
string pred_trend = "Œ‰ÀÌ / ‰«„‘Œ’"
color  pred_color = color.gray
float  confidence = math.abs(z_score) * 100 //  »œÌ· »Â œ—’œ
int    validity_minutes = 0 // „œ  «⁄ »«— ÅÌ‘ù»Ì‰Ì »Â œﬁÌﬁÂ

// „Õ«”»Â ﬁœ—  ”Ìê‰«· (Signal Strength) »— «”«” Z-Score Ê ‘ «»
signal_strength = math.abs(z_score) + math.abs(accel_5) * 10

if z_score > z_threshold
    pred_trend := "’⁄ÊœÌ ÅÌ‘ù»Ì‰Ì „Ìù‘Êœ ??"
    pred_color := color.green
    // „Õ«”»Â „œ  «⁄ »«— »— «”«” ﬁœ—  ”Ìê‰«·
    if signal_strength > 3.0
        validity_minutes := 120 // 2 ”«⁄ 
    else if signal_strength > 2.0
        validity_minutes := 90  // 1.5 ”«⁄ 
    else if signal_strength > 1.5
        validity_minutes := 60  // 1 ”«⁄ 
    else
        validity_minutes := 30  // 30 œﬁÌﬁÂ
        
else if z_score < -z_threshold
    pred_trend := "‰“Ê·Ì ÅÌ‘ù»Ì‰Ì „Ìù‘Êœ ??"
    pred_color := color.red
    // „Õ«”»Â „œ  «⁄ »«— »— «”«” ﬁœ—  ”Ìê‰«·
    if signal_strength > 3.0
        validity_minutes := 120 // 2 ”«⁄ 
    else if signal_strength > 2.0
        validity_minutes := 90  // 1.5 ”«⁄ 
    else if signal_strength > 1.5
        validity_minutes := 60  // 1 ”«⁄ 
    else
        validity_minutes := 30  // 30 œﬁÌﬁÂ
else
    pred_trend := "—‰Ã / «‰ Ÿ«— »—«Ì ”Ìê‰«·"
    pred_color := color.orange
    validity_minutes := 0

conf_str = str.tostring(math.min(confidence, 99.9), "#.#") + "%"
validity_str = validity_minutes > 0 ? str.tostring(validity_minutes) + " œﬁÌﬁÂ" : "‰«„‘Œ’"

// --- œ—Ì«›  «ÿ·«⁄«   «Ì„ù›—Ì„ùÂ« ---
// ?.  «Ì„ù›—Ì„ «‰ Œ«» ‘œÂ »—«Ì —”„ ŒÿÊÿ —ÊÌ ç«—

// --- „Õ«”»Â —Ê‰œ »—«Ì Â—  «Ì„ù›—Ì„ ---
