//@version=5
indicator("Golden Ichimoku{Morindok} by BITMOON618", overlay=true, max_lines_count=200, max_boxes_count=50)

// === Inputs ===
t1 = input.time(timestamp("2024-01-01 09:00 +0000"), "Time 1", confirm=true)
t2 = input.time(timestamp("2024-01-02 09:00 +0000"), "Time 2", confirm=true)

// === Find selected bars ===
var int bar1 = na
var int bar2 = na

if time >= t1 and time[1] < t1
    bar1 := bar_index

if time >= t2 and time[1] < t2
    bar2 := bar_index

barsRangeRaw = (not na(bar1) and not na(bar2)) ? math.abs(bar2 - bar1) : na
barsRange = na(barsRangeRaw) ? na : math.max(barsRangeRaw, 1)

rangeEndBar = (not na(bar1) and not na(bar2)) ? math.max(bar1, bar2) : na

// === Ichimoku calculations ===
factorConv = 9.0 / 26.0
factorSpan = 52.0 / 26.0

donchianMid(len) =>
    (ta.highest(high, len) + ta.lowest(low, len)) / 2.0

calcCloud(baseRange) =>
    convLen = na(baseRange) ? na : math.max(1, math.round(baseRange * factorConv))
    spanBLen = na(baseRange) ? na : math.max(1, math.round(baseRange * factorSpan))
    tenkan = not na(convLen) ? donchianMid(convLen) : na
    kijun = not na(baseRange) ? donchianMid(baseRange) : na
    spanA = (tenkan + kijun) / 2.0
    spanB = not na(spanBLen) ? donchianMid(spanBLen) : na
    [spanA, spanB, convLen, baseRange, spanBLen]

// === Clouds ===
[spanA1000, spanB1000, c1000, b1000, s1000] = calcCloud(barsRange)
[spanA1618, spanB1618, c1618, b1618, s1618] = calcCloud(na(barsRange) ? na : math.max(1, math.round(barsRange * 1.618)))
[spanA0618, spanB0618, c0618, b0618, s0618] = calcCloud(na(barsRange) ? na : math.max(1, math.round(barsRange * 0.618)))

// === Plotting Clouds ===
pA1000 = plot(spanA1000, color=color.new(color.white, 0), linewidth=2, offset=b1000)
pB1000 = plot(spanB1000, color=color.new(color.gray, 0), linewidth=2, offset=b1000)
fill(pA1000, pB1000, color=spanA1000 >= spanB1000 ? color.new(color.silver, 80) : color.new(color.gray, 85))

pA1618 = plot(spanA1618, color=color.new(color.blue, 0), linewidth=2, offset=b1618)
pB1618 = plot(spanB1618, color=color.new(color.navy, 0), linewidth=2, offset=b1618)
fill(pA1618, pB1618, color=spanA1618 >= spanB1618 ? color.new(color.green, 85) : color.new(color.red, 85))

pA0618 = plot(spanA0618, color=color.new(color.orange, 0), linewidth=2, offset=b0618)
pB0618 = plot(spanB0618, color=color.new(color.purple, 0), linewidth=2, offset=b0618)
fill(pA0618, pB0618, color=spanA0618 >= spanB0618 ? color.new(color.teal, 85) : color.new(color.maroon, 85))

// === Logic for Points and Boxes ===
upperWhiteNow = math.max(spanA1000[b1000], spanB1000[b1000])
lowerWhiteNow = math.min(spanA1000[b1000], spanB1000[b1000])

upperCloudTop = math.max(spanA1618[b1618], spanB1618[b1618])
upperCloudBot = math.min(spanA1618[b1618], spanB1618[b1618])
lowerCloudTop = math.max(spanA0618[b0618], spanB0618[b0618])
lowerCloudBot = math.min(spanA0618[b0618], spanB0618[b0618])

searchActive = not na(rangeEndBar) and bar_index > rangeEndBar
inUpper = searchActive and upperWhiteNow <= upperCloudTop and upperWhiteNow >= upperCloudBot
inLower = searchActive and lowerWhiteNow <= lowerCloudTop and lowerWhiteNow >= lowerCloudBot

var float p1Price = na, var int p1Bar = na
var float p2Price = na, var int p2Bar = na
var bool lineDrawn = false

var float upperEdge = na
var float lowerEdge = na

var bool trackU = false, var float uMax = na, var int uMaxB = na
var bool trackL = false, var float lMin = na, var int lMinB = na

if not lineDrawn and searchActive
    if na(p1Bar)
        if inUpper and not inUpper[1]
            trackU := true
            uMax := upperWhiteNow
            uMaxB := bar_index
        if trackU and inUpper
            if upperWhiteNow > uMax
                uMax := upperWhiteNow
                uMaxB := bar_index
        if trackU and not inUpper
            p1Price := uMax
            p1Bar := uMaxB
            trackU := false
    
    else if na(p2Bar)
        if inLower and not inLower[1]
            trackL := true
            lMin := lowerWhiteNow
            lMinB := bar_index
        if trackL and inLower
            if lowerWhiteNow < lMin
                lMin := lowerWhiteNow
                lMinB := bar_index
        if trackL and not inLower
            p2Price := lMin
            p2Bar := lMinB
            trackL := false

    if not na(p1Bar) and not na(p2Bar)
        tempMax = high
        tempMin = low
        for i = 0 to (bar_index - p1Bar)
            tempMax := math.max(tempMax, high[i])
            tempMin := math.min(tempMin, low[i])
        
        boxTop = tempMax
        boxBot = tempMin
        boxMid = (boxTop + boxBot) / 2
        boxHeight = boxTop - boxBot
        
        upperEdge := boxMid + boxHeight
        lowerEdge := boxMid - boxHeight

        box.new(left=p1Bar, top=boxTop, right=p2Bar, bottom=boxBot, border_color=color.yellow, bgcolor=color.new(color.yellow, 90), border_width=2)
        box.new(left=p1Bar, top=upperEdge, right=p2Bar, bottom=boxMid, border_color=color.red, bgcolor=color.new(color.red, 90), border_width=1, border_style=line.style_dashed)
        box.new(left=p1Bar, top=boxMid, right=p2Bar, bottom=lowerEdge, border_color=color.green, bgcolor=color.new(color.green, 90), border_width=1, border_style=line.style_dashed)

        line.new(p2Bar, upperEdge, p2Bar + 1, upperEdge, extend=extend.right, color=color.red, width=2)
        line.new(p2Bar, lowerEdge, p2Bar + 1, lowerEdge, extend=extend.right, color=color.green, width=2)
        
        lineDrawn := true

/// === SIGNAL LOGIC ===
var bool buySignalFired = false
var bool sellSignalFired = false
var bool pastChecked = false
var bool waitingForBuy = false
var bool waitingForSell = false

buyCondition = false
sellCondition = false

// »——”Ì ê–‘ Â »·«›«’·Â »⁄œ «“  ‘òÌ· »«ò”ùÂ«
if lineDrawn and not pastChecked
    pastChecked := true
    offsetMid = bar_index - p2Bar
    
    if offsetMid >= 0
        start_i = offsetMid - 1
        if start_i >= 0
            for i = 0 to start_i
                check_offset = start_i - i 
                isRed = close[check_offset] < open[check_offset]
                isGreen = close[check_offset] > open[check_offset]
                
                // À»  »—ŒÊ—œ œ— ê–‘ Â
                if low[check_offset] <= lowerEdge
                    waitingForBuy := true
                if high[check_offset] >= upperEdge
                    waitingForSell := true
                
                // ”Ìê‰«· ›—Ê‘ («Ê·Ì‰ ò‰œ· ﬁ—„“ »⁄œ «“ »—ŒÊ—œ œ— ê–‘ Â)
                if not sellSignalFired and waitingForSell and isRed
                    label.new(bar_index - check_offset, high[check_offset], "SELL", style=label.style_label_down, color=color.red, textcolor=color.white, size=size.small)
                    sellSignalFired := true
                    waitingForSell := false // —Ì”  ò—œ‰ Ê÷⁄Ì 
                
                // ”Ìê‰«· Œ—Ìœ («Ê·Ì‰ ò‰œ· ”»“ »⁄œ «“ »—ŒÊ—œ œ— ê–‘ Â)
                if not buySignalFired and waitingForBuy and isGreen
                    label.new(bar_index - check_offset, low[check_offset], "BUY", style=label.style_label_up, color=color.lime, textcolor=color.black, size=size.small)
                    buySignalFired := true
                    waitingForBuy := false // —Ì”  ò—œ‰ Ê÷⁄Ì 

// --- »——”Ì ·ÕŸÂù«Ì (¬Ì‰œÂ) ---
if lineDrawn and pastChecked
    // À»  »—ŒÊ—œ »« ·»Â Å«ÌÌ‰ (›ﬁÿ œ— ’Ê— Ì òÂ ”Ìê‰«· Œ—Ìœ Â‰Ê“ ’«œ— ‰‘œÂ »«‘œ)
    if not buySignalFired and low <= lowerEdge
        waitingForBuy := true

    // À»  »—ŒÊ—œ »« ·»Â »«·« (›ﬁÿ œ— ’Ê— Ì òÂ ”Ìê‰«· ›—Ê‘ Â‰Ê“ ’«œ— ‰‘œÂ »«‘œ)
    if not sellSignalFired and high >= upperEdge
        waitingForSell := true

    // »——”Ì ’œÊ— ”Ìê‰«· Œ—Ìœ («Ê·Ì‰ ò‰œ· ”»“ »⁄œ «“ »—ŒÊ—œ)
    if not buySignalFired and waitingForBuy and (close > open)
        buyCondition := true
        buySignalFired := true // Ã·ÊêÌ—Ì «“ ’œÊ— ”Ìê‰«· „Ãœœ
        waitingForBuy := false // —Ì”  ò—œ‰ Ê÷⁄Ì 

    // »——”Ì ’œÊ— ”Ìê‰«· ›—Ê‘ («Ê·Ì‰ ò‰œ· ﬁ—„“ »⁄œ «“ »—ŒÊ—œ)
    if not sellSignalFired and waitingForSell and (close < open)
        sellCondition := true
        sellSignalFired := true // Ã·ÊêÌ—Ì «“ ’œÊ— ”Ìê‰«· „Ãœœ
        waitingForSell := false // —Ì”  ò—œ‰ Ê÷⁄Ì 

// —”„ ”Ìê‰«·ùÂ« (»—«Ì ¬Ì‰œÂ)
plotshape(buyCondition, title="BUY", style=shape.labelup, location=location.belowbar, color=color.green, text="BUY", textcolor=color.white)
plotshape(sellCondition, title="SELL", style=shape.labeldown, location=location.abovebar, color=color.red, text="SELL", textcolor=color.white)

alertcondition(buyCondition, "Buy Alert", "Price bounced from Lower Box Edge with Green Candle")
alertcondition(sellCondition, "Sell Alert", "Price bounced from Upper Box Edge with Red Candle")

if barstate.islast
