import dash
from dash import dcc, html, Input, Output
import plotly.graph_objects as go
import numpy as np
from scipy.spatial.distance import cdist

# ÷—Ì» «”ÅÌ—«· ·ê«—Ì „Ì ÿ·«ÌÌ
PHI = 1.61803398875
b_golden = np.log(PHI) / (np.pi / 2)

app = dash.Dash(__name__)

# --- «” «Ì·ùÂ«Ì UI ---
COLORS = {
    'bg': '#0f172a',
    'card': '#1e293b',
    'text': '#f8fafc',
    'accent': '#38bdf8',
    'border': '#334155'
}

card_style = {
    'backgroundColor': COLORS['card'],
    'padding': '15px 20px',
    'borderRadius': '12px',
    'border': f'1px solid {COLORS["border"]}',
    'boxShadow': '0 4px 6px -1px rgba(0, 0, 0, 0.1)',
    'flex': '1',
    'margin': '0 10px'
}

input_style = {
    'width': '100%',
    'padding': '8px 12px',
    'marginTop': '8px',
    'borderRadius': '6px',
    'border': 'none',
    'backgroundColor': '#0f172a',
    'color': '#38bdf8',
    'fontSize': '16px',
    'outline': 'none'
}

label_style = {
    'fontSize': '14px',
    'color': '#cbd5e1',
    'fontWeight': 'bold'
}

app.layout = html.Div(style={
    'fontFamily': 'Tahoma, sans-serif', 
    'direction': 'rtl', 
    'backgroundColor': COLORS['bg'],
    'color': COLORS['text'],
    'minHeight': '100vh',
    'padding': '20px',
    'boxSizing': 'border-box'
}, children=[
    
    html.Div(
        html.H2("›—«ò «· «”ÅÌ—«· « „Ì ( Ê“Ì⁄ ﬁÌ„ )", style={'margin': '0 0 20px 0', 'textAlign': 'center', 'color': COLORS['accent']})
    ),
    
    # Å‰·  ‰ŸÌ„« 
    html.Div(style={'display': 'flex', 'justifyContent': 'space-between', 'marginBottom': '20px', 'maxWidth': '1200px', 'margin': '0 auto 20px auto'}, children=[
        html.Div(style=card_style, children=[
            html.Label("”ﬁ›  «—ÌŒÌ ﬁÌ„  (ATH):", style=label_style),
            dcc.Input(id='y_ath', type='number', value=73000.0, step='any', style=input_style)
        ]),
        
        html.Div(style=card_style, children=[
            html.Label(" ⁄œ«œ ç—Œ‘ùÂ«Ì «”ÅÌ—«· »“—ê:", style=label_style),
            dcc.Input(id='revolutions', type='number', value=4.0, step='any', style=input_style)
        ]),
        
        html.Div(style=card_style, children=[
            html.Label("ﬁÌ„  ÅÌÊÊ  ò«—»—:", style=label_style),
            dcc.Input(id='y_pivot', type='number', value=30000.0, step='any', style=input_style)
        ]),
    ]),

    # ‰„Êœ«—
    html.Div(style={'maxWidth': '1400px', 'margin': '0 auto', 'borderRadius': '16px', 'overflow': 'hidden', 'boxShadow': '0 10px 15px -3px rgba(0, 0, 0, 0.5)'}, children=[
        dcc.Graph(id='fractal-spiral-graph', style={'height': '75vh'})
    ])
])

@app.callback(
    Output('fractal-spiral-graph', 'figure'),
    [Input('y_ath', 'value'),
     Input('revolutions', 'value'),
     Input('y_pivot', 'value')]
)
def update_graph(y_ath, revs, y_pivot):
    if None in [y_ath, revs, y_pivot] or y_ath <= 0 or revs <= 0:
        return go.Figure(layout=go.Layout(paper_bgcolor=COLORS['bg'], plot_bgcolor=COLORS['bg']))

    fig = go.Figure()
    
    max_theta = revs * 2 * np.pi
    A = y_ath / (np.exp(b_golden * max_theta) - 1)
    
    # --- 1. ‘»òÂ «”ÅÌ—«·Ì Å”ù“„Ì‰Â ---
    bg_theta = np.linspace(0, max_theta, 800)
    bg_r = A * (np.exp(b_golden * bg_theta) - 1)
    
    for phase in [0, np.pi/2, np.pi, 3*np.pi/2]:
        x_bg = bg_r * np.cos(bg_theta + phase)
        y_bg = bg_r * np.sin(bg_theta + phase)
        fig.add_trace(go.Scatter(
            x=x_bg, y=y_bg,
            mode='lines', hoverinfo='skip',
            line=dict(color='rgba(255, 255, 255, 0.04)', width=1),
            showlegend=False
        ))

    # --- 2. «”ÅÌ—«· »“—ê «’·Ì ---
    theta_grand = np.linspace(0, max_theta, 3000)
    r_grand = A * (np.exp(b_golden * theta_grand) - 1)
    x_grand = r_grand * np.cos(theta_grand)
    y_grand = r_grand * np.sin(theta_grand)

    fig.add_trace(go.Scatter(
        x=x_grand, y=y_grand,
        mode='lines', name='«”ÅÌ—«· ò·«‰',
        customdata=r_grand,
        hovertemplate='<b>ﬁÌ„ :</b> %{customdata:,.2f} $ <extra>ò·«‰</extra>',
        line=dict(color='#0ea5e9', width=2.5)
    ))

    # --- 3. Ì«› ‰ ‰ﬁÿÂ ÅÌÊÊ  Ê  Ê·Ìœ «”ÅÌ—«· Œ—œ Œ«„ ---
    if 0 <= y_pivot <= y_ath:
        theta_pivot = np.log((y_pivot / A) + 1) / b_golden
        xc_p = y_pivot * np.cos(theta_pivot)
        yc_p = y_pivot * np.sin(theta_pivot)
        
        fig.add_trace(go.Scatter(
            x=[xc_p], y=[yc_p],
            mode='markers', name=f'„—ò“ ÅÌÊÊ ',
            hovertemplate=f'<b>ﬁÌ„  ÅÌÊÊ :</b> {y_pivot:,.2f} $<extra></extra>',
            marker=dict(color='#fbbf24', size=14, symbol='star', line=dict(color='#fff', width=1))
        ))

        #  Ê·Ìœ «”ÅÌ—«· Œ—œ »Â «‰œ«“Â ò«›Ì »·‰œ »—«Ì «ÿ„Ì‰«‰ «“ Œ—ÊÃ «“ „ÕœÊœÂ ò·«‰
        a_small = y_pivot * 0.05 if y_pivot > 0 else y_ath * 0.01
        theta_small_raw = np.linspace(0, 10 * np.pi, 4000)
        r_small_raw = a_small * np.exp(b_golden * theta_small_raw)
        
        x_small_raw = xc_p + r_small_raw * np.cos(theta_small_raw)
        y_small_raw = yc_p + r_small_raw * np.sin(theta_small_raw)

        # --- 4. Ì«› ‰  ﬁ«ÿ⁄ùÂ« Ê „ÕœÊœ ò—œ‰ «”ÅÌ—«· Œ—œ ---
        pts_grand = np.column_stack((x_grand, y_grand))
        pts_small = np.column_stack((x_small_raw, y_small_raw))
        
        threshold = y_ath * 0.015
        distances = cdist(pts_grand, pts_small)
        min_idx = np.where(distances < threshold)
        
        ix_x, ix_y, ix_price = [], [], []
        last_intersection_idx = 0
        
        for i, j in zip(min_idx[0], min_idx[1]):
            px, py = pts_grand[i]
            # Õ–›  ﬁ«ÿ⁄ùÂ«Ì »”Ì«— ‰“œÌò »Â ŒÊœ ‰ﬁÿÂ ÅÌÊÊ 
            if np.sqrt((px - xc_p)**2 + (py - yc_p)**2) > threshold * 2:
                # Ã·ÊêÌ—Ì «“ À»  ‰ﬁ«ÿ  ò—«—Ì ò‰«— Â„
                if not any(np.sqrt((px - hx)**2 + (py - hy)**2) < (y_ath*0.05) for hx, hy in zip(ix_x, ix_y)):
                    ix_x.append(px)
                    ix_y.append(py)
                    ix_price.append(np.sqrt(px**2 + py**2))
                    if j > last_intersection_idx:
                        last_intersection_idx = j

        # »—‘ œ«œ‰ «”ÅÌ—«· Œ—œ  « ¬Œ—Ì‰ »—ŒÊ—œ (»« ò„Ì Õ«‘ÌÂ »—«Ì ‰„«Ì‘ »Â —)
        slice_idx = last_intersection_idx + 10 if last_intersection_idx > 0 else len(theta_small_raw)
        
        x_small_final = x_small_raw[:slice_idx]
        y_small_final = y_small_raw[:slice_idx]

        # —”„ «”ÅÌ—«· Œ—œ »—‘ùŒÊ—œÂ
        fig.add_trace(go.Scatter(
            x=x_small_final, y=y_small_final,
            mode='lines', name='«”ÅÌ—«· Œ—œ',
            hoverinfo='skip',
            line=dict(color='#f59e0b', width=1.5)
        ))

        # —”„ ‰ﬁ«ÿ  ﬁ«ÿ⁄
        if ix_x:
            fig.add_trace(go.Scatter(
                x=ix_x, y=ix_y,
                mode='markers', name='‰ﬁ«ÿ  ·«ﬁÌ',
                customdata=ix_price,
                hovertemplate='<b> ·«ﬁÌ œ—:</b> %{customdata:,.2f} $ <extra>Âœ›</extra>',
                marker=dict(color='#ef4444', size=10, symbol='x', line=dict(color='#fff', width=1))
            ))

    # --- 5.  ‰ŸÌ„«  ‰„«Ì‘Ì ---
    fig.update_layout(
        paper_bgcolor=COLORS['card'],
        plot_bgcolor=COLORS['bg'],
        xaxis=dict(visible=False, scaleanchor="y", scaleratio=1),
        yaxis=dict(visible=False),
        hovermode='closest',
        margin=dict(l=10, r=10, t=10, b=10),
        legend=dict(
            font=dict(color=COLORS['text']),
            orientation="h",
            yanchor="bottom",
            y=1.02,
            xanchor="right",
            x=1
        )
    )

    return fig

if __name__ == '__main__':
    app.run_server(debug=True)
