tanhApprox(x) =>
    ex = math.exp(2 * x)
    (ex - 1) / (ex + 1)

isRejectionCandle(idx, level, isResistance) =>
    if idx > bar_index or idx < 0
        false
    else
        bodySize = math.abs(close[idx] - open[idx])
        totalSize = high[idx] - low[idx]
        
        if totalSize == 0
            false
        else
            wickRatio = isResistance ? (high[idx] - math.max(close[idx], open[idx])) / totalSize : (math.min(close[idx], open[idx]) - low[idx]) / totalSize
            bodyRatio = bodySize / totalSize
            
            isNearLevel = isResistance ? math.abs(high[idx] - level) < (upperLevel - lowerLevel) * sensitivityPercent / 100 : math.abs(low[idx] - level) < (upperLevel - lowerLevel) * sensitivityPercent / 100
            hasLongWick = wickRatio > 0.4
            hasSmallBody = bodyRatio < 0.5
            correctDirection = isResistance ? close[idx] < open[idx] : close[idx] > open[idx]
            
            isNearLevel and hasLongWick and (hasSmallBody or correctDirection)

isConsolidation(idx, level, lookback) =>
    if idx > bar_index or idx < 0
        false
    else
        touches = 0
        priceRange = upperLevel - lowerLevel
        threshold = priceRange * sensitivityPercent / 100
        
        for i = 0 to math.min(lookback - 1, 9)
            checkIdx = idx - i
            if checkIdx >= 0 and checkIdx <= bar_index
                if math.abs(high[checkIdx] - level) < threshold or math.abs(low[checkIdx] - level) < threshold or (high[checkIdx] > level and low[checkIdx] < level)
                    touches += 1
        
        touches >= 3

analyzeRatiosAdvanced(startT, endT, liveT) =>
    var float bestRatio = 0.618
    var float bestScore = 0.0
    var float[] scores = array.new_float(15, 0.0)
    var int[] touchCounts = array.new_int(15, 0)
    var float[] avgDistances = array.new_float(15, 0.0)
    
    if liveT > endT and endT > startT
        priceRange = upperLevel - lowerLevel
        maxLookback = math.min(bar_index, 5000)
        threshold = priceRange * sensitivityPercent / 100
        
        for ratioIdx = 0 to 14
            ratio = array.get(fibRatios, ratioIdx)
            totalReactionScore = 0.0
            touchCount = 0
            totalDistance = 0.0
            consecutiveTouches = 0
            maxConsecutive = 0
            
            baseUpper = adjustSlope and slopeCalculated ? currentUpperLevel : upperLevel
            baseLower = adjustSlope and slopeCalculated ? currentLowerLevel : lowerLevel
            adjustedRange = baseUpper - baseLower
            
            upperFibLevel = baseUpper + adjustedRange * (ratio - 1.0)
            lowerFibLevel = baseLower - adjustedRange * ratio
            midFibLevel = baseLower + adjustedRange * ratio
            
            var line upperLine = na
            var line lowerLine = na
            var line midLine = na
            
            for i = 0 to maxLookback - 1
                if i <= bar_index and time[i] > endT and time[i] <= liveT
                    histBaseUpper = upperLevel
                    histBaseLower = lowerLevel
                    
                    if adjustSlope and slopeCalculated and not na(startBar)
                        barsFromStartHist = (bar_index - i) - startBar
                        if barsFromStartHist >= 0
                            progress = tanhApprox(barsFromStartHist / slopeSpeed)
                            histBaseUpper := upperLevel + (upperLevelAdjusted - upperLevel) * progress
                            histBaseLower := lowerLevel + (lowerLevelAdjusted - lowerLevel) * progress
                    
                    histRange = histBaseUpper - histBaseLower
                    histUpperFib = histBaseUpper + histRange * (ratio - 1.0)
                    histLowerFib = histBaseLower - histRange * ratio
                    histMidFib = histBaseLower + histRange * ratio
                    
                    reactionStrength = 0.0
                    minDistance = priceRange * 10
                    
                    distToUpperHigh = math.abs(high[i] - histUpperFib)
                    distToUpperLow = math.abs(low[i] - histUpperFib)
                    
                    if distToUpperHigh < threshold or distToUpperLow < threshold or (high[i] > histUpperFib and low[i] < histUpperFib)
                        touchCount += 1
                        consecutiveTouches += 1
                        minDistance := math.min(minDistance, math.min(distToUpperHigh, distToUpperLow))
                        
                        reactionStrength += 5.0
                        
                        if isRejectionCandle(i, histUpperFib, true)
                            reactionStrength += 8.0 * rejectionWeight
                        
                        if isConsolidation(i, histUpperFib, 5)
                            reactionStrength += 6.0 * consolidationWeight
                        
                        accuracyBonus = (1 - minDistance / threshold) * 3.0
                        reactionStrength += accuracyBonus
                    else
                        consecutiveTouches := 0
                    
                    distToLowerHigh = math.abs(high[i] - histLowerFib)
                    distToLowerLow = math.abs(low[i] - histLowerFib)
                    
                    if distToLowerHigh < threshold or distToLowerLow < threshold or (high[i] > histLowerFib and low[i] < histLowerFib)
                        touchCount += 1
                        consecutiveTouches += 1
                        minDistance := math.min(minDistance, math.min(distToLowerHigh, distToLowerLow))
                        
                        reactionStrength += 5.0
                        
                        if isRejectionCandle(i, histLowerFib, false)
                            reactionStrength += 8.0 * rejectionWeight
                        
                        if isConsolidation(i, histLowerFib, 5)
                            reactionStrength += 6.0 * consolidationWeight
                        
                        accuracyBonus = (1 - minDistance / threshold) * 2.0
                        reactionStrength += accuracyBonus
                    
                    distToMidHigh = math.abs(high[i] - histMidFib)
                    distToMidLow = math.abs(low[i] - histMidFib)
                    
                    if distToMidHigh < threshold or distToMidLow < threshold or (high[i] > histMidFib and low[i] < histMidFib)
                        touchCount += 1
                        minDistance := math.min(minDistance, math.min(distToMidHigh, distToMidLow))
                        reactionStrength += 3.0
                        
                        accuracyBonus = (1 - minDistance / threshold) * 2.0
                        reactionStrength += accuracyBonus
                    
                    if reactionStrength > 0 and i >= 5
                        avgVolume = 0.0
                        for j = 1 to 5
                            checkIdx = i - j
                            if checkIdx >= 0 and checkIdx <= bar_index
                                avgVolume += volume[checkIdx]
                        avgVolume := avgVolume / 5
                        
                        if avgVolume > 0 and volume[i] > avgVolume * 1.3
                            volumeBonus = math.min((volume[i] / avgVolume - 1) * 5.0, 10.0) * volumeWeight
                            reactionStrength += volumeBonus
                    
                    maxConsecutive := math.max(maxConsecutive, consecutiveTouches)
                    if consecutiveTouches >= 3
                        reactionStrength += consecutiveTouches * 2.0
                    
                    if minDistance < priceRange * 10
                        totalDistance += minDistance
                    
                    totalReactionScore += reactionStrength
            
            avgScore = touchCount > 0 ? totalReactionScore / touchCount : 0.0
            avgDist = touchCount > 0 ? totalDistance / touchCount : priceRange
            
            if touchCount < minTouches
                avgScore *= (touchCount / minTouches) * 0.5
            
            if maxConsecutive >= 3
                avgScore *= (1 + maxConsecutive * 0.1)
            
            accuracyFactor = 1 + (1 - math.min(avgDist / threshold, 1.0)) * 0.5
            avgScore *= accuracyFactor
            
            array.set(scores, ratioIdx, avgScore)
            array.set(touchCounts, ratioIdx, touchCount)
            array.set(avgDistances, ratioIdx, avgDist)
            
            if avgScore > bestScore
                bestScore := avgScore
                bestRatio := ratio
            
            if showDetailedLevels and ratio == bestRatio and touchCount >= minTouches
                startBarIdx = bar_index
                endBarIdx = bar_index
                
                for i = 0 to maxLookback - 1
                    if i <= bar_index and time[i] <= endT
                        startBarIdx := bar_index - i
                    if i <= bar_index and time[i] <= liveT
                        endBarIdx := bar_index - i
                
                if not na(upperLine)
                    line.delete(upperLine)
                if not na(lowerLine)
                    line.delete(lowerLine)
                if not na(midLine)
                    line.delete(midLine)
                
                upperLine := line.new(startBarIdx, upperFibLevel, endBarIdx, upperFibLevel, color=color.new(color.yellow, 0), width=2, style=line.style_dashed)
                lowerLine := line.new(startBarIdx, lowerFibLevel, endBarIdx, lowerFibLevel, color=color.new(color.orange, 0), width=2, style=line.style_dashed)
                midLine := line.new(startBarIdx, midFibLevel, endBarIdx, midFibLevel, color=color.new(color.aqua, 0), width=1, style=line.style_dotted)
    
    [bestRatio, bestScore, scores, touchCounts, avgDistances]
