f_atan2(float y, float x) =>
    float res = 0.0
    if x > 0
        res := math.atan(y / x)
    else if x < 0 and y >= 0
        res := math.atan(y / x) + math.pi
    else if x < 0 and y < 0
        res := math.atan(y / x) - math.pi
    else if x == 0 and y > 0
        res := math.pi / 2
    else if x == 0 and y < 0
        res := -math.pi / 2
    res

f_arr_sma(float[] arr, int len) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float sum = 0.0
    if actual_len > 0
        for j = 0 to actual_len - 1
            sum := sum + array.get(arr, sz - 1 - j)
    actual_len > 0 ? sum / actual_len : na

f_arr_lowest(float[] arr, int len) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float val = na
    if actual_len > 0
        val := array.get(arr, sz - 1)
        if actual_len > 1
            for j = 1 to actual_len - 1
                val := math.min(val, array.get(arr, sz - 1 - j))
    val

f_arr_highest(float[] arr, int len) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float val = na
    if actual_len > 0
        val := array.get(arr, sz - 1)
        if actual_len > 1
            for j = 1 to actual_len - 1
                val := math.max(val, array.get(arr, sz - 1 - j))
    val

f_arr_stdev(float[] arr, int len, float mean) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float sum_sq_diff = 0.0
    if actual_len > 0 and not na(mean)
        for j = 0 to actual_len - 1
            float val = array.get(arr, sz - 1 - j)
            sum_sq_diff := sum_sq_diff + math.pow(val - mean, 2)
    actual_len > 0 ? math.sqrt(sum_sq_diff / actual_len) : na

f_arr_sum(float[] arr, int len) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float sum = 0.0
    if actual_len > 0
        for j = 0 to actual_len - 1
            sum := sum + array.get(arr, sz - 1 - j)
    sum

f_arr_wma(float[] arr, int len) =>
    int sz = array.size(arr)
    int actual_len = math.min(sz, math.max(1, len))
    float sum = 0.0
    float weight_sum = 0.0
    if actual_len > 0
        for j = 0 to actual_len - 1
            float w = actual_len - j
            sum := sum + array.get(arr, sz - 1 - j) * w
            weight_sum := weight_sum + w
    weight_sum > 0 ? sum / weight_sum : na

f_spine_vol(float[] hArr, float[] lArr, int sum_len, int wma_len) =>
    int sz = array.size(hArr)
    float wma_sum = 0.0
    float w_sum = 0.0
    int actual_wma = math.min(sz, wma_len)
    if actual_wma > 0
        for j = 0 to actual_wma - 1
            int idx = sz - 1 - j
            if idx >= 0
                float r_sum = 0.0
                int r_len = math.min(idx + 1, sum_len)
                for k = 0 to r_len - 1
                    r_sum := r_sum + (array.get(hArr, idx - k) - array.get(lArr, idx - k))
                float w = actual_wma - j
                wma_sum := wma_sum + r_sum * w
                w_sum := w_sum + w
    w_sum > 0 ? wma_sum / w_sum : 0.0

f_draw_spiral(t_base, p_base, target_len, slope_angle, is_reverse, col, dt, dp, opt_cont, mirror) =>
    float b_coeff = math.log(1.618) / (math.pi / 2)
    float a = 0.02
    chart.point[] pts = array.new<chart.point>()
    array.push(pts, chart.point.from_time(math.round(t_base), p_base))
    
    float acc_len = 0.0
    float prev_t = t_base
    float prev_p = p_base
    float end_p = p_base
    float y_dir = mirror ? -1.0 : 1.0
    
    for i = 1 to 2000
        float theta = i * 0.1
        float r = a * math.exp(b_coeff * theta)
        float t_angle = (is_reverse ? -theta : theta) + slope_angle
        
        int curr_t = math.round(t_base + r * dt * math.cos(t_angle))
        float curr_p = p_base + y_dir * (r * dp * math.sin(t_angle))
        
        float dt_norm = (curr_t - prev_t) / dt
        float seg = math.sqrt(math.pow(dt_norm * dp, 2) + math.pow(curr_p - prev_p, 2))
        
        if opt_cont or acc_len <= target_len
            array.push(pts, chart.point.from_time(curr_t, curr_p))
            acc_len += seg
            end_p := curr_p
        else
            break
        prev_t := curr_t
        prev_p := curr_p
    
    if array.size(pts) > 1 and not use_sym_ratio // <--- Added Hide Condition Here
        array.push(drawnPolys, polyline.new(pts, xloc=xloc.bar_time, line_color=col, line_width=2))
    end_p

f_get_wave(t, t_start, t_end, p_start, p_end, len_bars) =>
    float _amp = math.abs(p_end - p_start) != 0 ? math.abs(p_end - p_start) / 2 : nz(dyn_atr, 0.0)
    float _mid_time = t_start + (t_end - t_start) * 0.5
    float _bar_ms = timeframe.in_seconds(timeframe.period) * 1000
    float _rel_x = (t - _mid_time) / _bar_ms
    float _dist = math.abs(_rel_x)
    float _len = math.max(1, len_bars)
    float _cyc_idx = math.floor((_dist % (_len * 5)) / _len)
    float _fibo_amp = (math.pow(1.618, _cyc_idx) / math.pow(1.618, 4)) * _amp
    math.sin(_rel_x * 2 * math.pi / _len) * _fibo_amp

        float endOrangeVal   = array.get(orangeVals, idxInArrayEnd)
        float endPurpleVal   = array.get(purpleVals, idxInArrayEnd)
        float endRedVal      = useInvMaForRed ? endPurpleVal : endOrangeVal

            startRedTargetY := f_draw_spiral(startBarTime, startRedBaseY, finalRedDist, eff_lower_slope, true, color.red, dt, dp, opt_continue, mirror_spirals)
            
            // Draw Total Spirals (Upward by default)
            endTotalTargetY := f_draw_spiral(endBarTime, endTotalBaseY, finalTotalDist, eff_upper_slope, false, color.blue, dt, dp, opt_continue, mirror_spirals)
            startTotalTargetY := f_draw_spiral(startBarTime, startTotalBaseY, finalTotalDist, eff_upper_slope, false, color.blue, dt, dp, opt_continue, mirror_spirals)
        else
            endGreenTargetY := endGreenBaseY + y_dir * finalGreenDist
            endRedTargetY := endRedBaseY - y_dir * finalRedDist
            startGreenTargetY := startGreenBaseY + y_dir * finalGreenDist
            startRedTargetY := startRedBaseY - y_dir * finalRedDist
            
            endTotalTargetY := endTotalBaseY + y_dir * finalTotalDist
            startTotalTargetY := startTotalBaseY + y_dir * finalTotalDist
            
            if not use_sym_ratio // <--- Added Hide Condition Here
                array.push(drawnLines, line.new(x1=endBarTime, y1=endGreenBaseY, x2=endBarTime, y2=endGreenTargetY, xloc=xloc.bar_time, color=color.green, width=2))
                array.push(drawnLines, line.new(x1=endBarTime, y1=endRedBaseY, x2=endBarTime, y2=endRedTargetY, xloc=xloc.bar_time, color=color.red, width=2))
                array.push(drawnLines, line.new(x1=endBarTime, y1=endTotalBaseY, x2=endBarTime, y2=endTotalTargetY, xloc=xloc.bar_time, color=color.blue, width=2))
                
                array.push(drawnLines, line.new(x1=startBarTime, y1=startGreenBaseY, x2=startBarTime, y2=startGreenTargetY, xloc=xloc.bar_time, color=color.green, width=2, style=line.style_dashed))
                array.push(drawnLines, line.new(x1=startBarTime, y1=startRedBaseY, x2=startBarTime, y2=startRedTargetY, xloc=xloc.bar_time, color=color.red, width=2, style=line.style_dashed))
                array.push(drawnLines, line.new(x1=startBarTime, y1=startTotalBaseY, x2=startBarTime, y2=startTotalTargetY, xloc=xloc.bar_time, color=color.blue, width=2, style=line.style_dashed))

        array.push(drawnLines, line.new(x1=endBarTime, y1=endGreenTargetY, x2=endBarTime + 10000000, y2=endGreenTargetY, xloc=xloc.bar_time, color=color.green, extend=extend.right))
        array.push(drawnLines, line.new(x1=endBarTime, y1=endRedTargetY, x2=endBarTime + 10000000, y2=endRedTargetY, xloc=xloc.bar_time, color=color.red, extend=extend.right))
        array.push(drawnLines, line.new(x1=endBarTime, y1=endTotalTargetY, x2=endBarTime + 10000000, y2=endTotalTargetY, xloc=xloc.bar_time, color=color.blue, extend=extend.right))
        
        array.push(drawnLines, line.new(x1=startBarTime, y1=startGreenTargetY, x2=startBarTime + 10000000, y2=startGreenTargetY, xloc=xloc.bar_time, color=color.green, extend=extend.right, style=line.style_dashed))
        array.push(drawnLines, line.new(x1=startBarTime, y1=startRedTargetY, x2=startBarTime + 10000000, y2=startRedTargetY, xloc=xloc.bar_time, color=color.red, extend=extend.right, style=line.style_dashed))
        array.push(drawnLines, line.new(x1=startBarTime, y1=startTotalTargetY, x2=startBarTime + 10000000, y2=startTotalTargetY, xloc=xloc.bar_time, color=color.blue, extend=extend.right, style=line.style_dashed))

        // --- Calculate Global Symmetry Axes Values ---
        global_sym_green := endGreenBaseY + (endGreenTargetY - endGreenBaseY) * sym_ratio_green
        global_sym_red   := endRedBaseY   + (endRedTargetY   - endRedBaseY)   * sym_ratio_red
        global_sym_total := endTotalBaseY + (endTotalTargetY - endTotalBaseY) * sym_ratio_total

        // --- Labels ---
        if not use_sym_ratio // <--- Added Hide Condition Here
            label.new(endBarTime, window_max_p, "Green: " + str.tostring(finalGreenDist, "#.##") + "\nRed: " + str.tostring(finalRedDist, "#.##") + "\nTotal: " + str.tostring(finalTotalDist, "#.##"), xloc=xloc.bar_time, color=color.new(color.black, 100), textcolor=color.white, style=label.style_label_down)

        // --- Second Pass: Price-Time Path scaled to Interval's Price Range ---
        if show_end_path and not use_sym_ratio // <--- Added Hide Condition Here
            chart.point[] pathStartGreen = array.new<chart.point>()
            chart.point[] pathStartRed   = array.new<chart.point>()
            chart.point[] pathStartTotal = array.new<chart.point>()
            chart.point[] pathEndGreen   = array.new<chart.point>()
            chart.point[] pathEndRed     = array.new<chart.point>()
            chart.point[] pathEndTotal   = array.new<chart.point>()

            float path_vertical_range = na(window_max_p) or na(window_min_p) ? 0.0 : window_max_p - window_min_p
            if path_vertical_range <= 0
                path_vertical_range := dp

            float runSumGreenDist = 0.0
            float runSumRedDist   = 0.0
            float runSumTotalDist = 0.0
            
            for i = idxInArrayStart to idxInArrayEnd
                float c = array.get(cPrices, i)
                float o = array.get(oPrices, i)
                float h = array.get(hPrices, i)
                float l = array.get(lPrices, i)
                int t_i = array.get(tValues, i)
                
                float rMaGreen = array.get(orangeVals, i)
                float rMaRed   = useInvMaForRed ? array.get(purpleVals, i) : array.get(orangeVals, i)
                
                if c > o
                    float dist = (c > rMaGreen) ? (h - rMaGreen) : (rMaGreen - l)
                    runSumGreenDist += dist
                else if c < o
                    float dist = (c > rMaRed) ? (h - rMaRed) : (rMaRed - l)
                    runSumRedDist += dist
                    
                runSumTotalDist := runSumGreenDist + runSumRedDist
                
                float gRatio = sumGreenDist > 0 ? (runSumGreenDist / sumGreenDist) : 0.0
                float rRatio = sumRedDist > 0 ? (runSumRedDist / sumRedDist) : 0.0
                float tRatio = sumTotalDist > 0 ? (runSumTotalDist / sumTotalDist) : 0.0
                
                float green_path_offset = gRatio * path_vertical_range
                float red_path_offset   = rRatio * path_vertical_range
                float total_path_offset = tRatio * path_vertical_range
                
                float psgY = startGreenBaseY + green_path_offset * y_dir
                float psrY = startRedBaseY   - red_path_offset   * y_dir
                float pstY = startTotalBaseY + total_path_offset * y_dir
                
                float pegY = endGreenBaseY   + green_path_offset * y_dir
                float perY = endRedBaseY     - red_path_offset   * y_dir
                float petY = endTotalBaseY   + total_path_offset * y_dir
                
                array.push(pathStartGreen, chart.point.from_time(t_i, psgY))
                array.push(pathStartRed,   chart.point.from_time(t_i, psrY))
                array.push(pathStartTotal, chart.point.from_time(t_i, pstY))
                
                array.push(pathEndGreen,   chart.point.from_time(t_i, pegY))
                array.push(pathEndRed,     chart.point.from_time(t_i, perY))
                array.push(pathEndTotal,   chart.point.from_time(t_i, petY))
                
            if array.size(pathStartGreen) > 1
                array.push(drawnPolys, polyline.new(pathStartGreen, xloc=xloc.bar_time, line_color=color.green, line_width=2))
            if array.size(pathStartRed) > 1
                array.push(drawnPolys, polyline.new(pathStartRed, xloc=xloc.bar_time, line_color=color.red, line_width=2))
            if array.size(pathStartTotal) > 1
                array.push(drawnPolys, polyline.new(pathStartTotal, xloc=xloc.bar_time, line_color=color.blue, line_width=2))
                
            if array.size(pathEndGreen) > 1
                array.push(drawnPolys, polyline.new(pathEndGreen, xloc=xloc.bar_time, line_color=color.green, line_width=2))
            if array.size(pathEndRed) > 1
                array.push(drawnPolys, polyline.new(pathEndRed, xloc=xloc.bar_time, line_color=color.red, line_width=2))
            if array.size(pathEndTotal) > 1
                array.push(drawnPolys, polyline.new(pathEndTotal, xloc=xloc.bar_time, line_color=color.blue, line_width=2))


// =============================================================================
//              SYMMETRY AXES AND MIRRORED PRICE (POLYLINE)
// =============================================================================
plot(use_sym_ratio ? global_sym_green : na, "Green Symmetry Axis", color.new(color.green, 40), 1, plot.style_linebr, offset=-9999, editable=false)
plot(use_sym_ratio ? global_sym_red   : na, "Red Symmetry Axis", color.new(color.red, 40), 1, plot.style_linebr, offset=-9999, editable=false)
plot(use_sym_ratio ? global_sym_total : na, "Total Symmetry Axis", color.new(color.blue, 40), 1, plot.style_linebr, offset=-9999, editable=false)
