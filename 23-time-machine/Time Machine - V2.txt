float atr14 = ta.atr(14)
f_get_wave(t, t_start, t_end, p_start, p_end, len_bars, atr_val) =>
    float _amp = math.abs(p_end - p_start) != 0 ? math.abs(p_end - p_start) / 2 : atr_val
    float _mid_time = t_start + (t_end - t_start) * sym_center_ratio
    float _bar_ms = timeframe.in_seconds(timeframe.period) * 1000
    float _rel_x = (t - _mid_time) / _bar_ms
    float _dist = math.abs(_rel_x)
    int _max_cyc = 5
    float _len = math.max(1, len_bars)
    float _cyc_idx = reset_wave ? math.floor((_dist % (_len * _max_cyc)) / _len) : math.floor(_dist / _len)
    float _max_fib = math.pow(1.618, _max_cyc - 1)
    float _fibo_amp = (math.pow(1.618, _cyc_idx) / _max_fib) * _amp
    float _w = math.sin(_rel_x * 2 * math.pi / _len) * _fibo_amp
    _w

// --- Function to get active multiplier based on market nature ---
f_get_active_mult() =>
    float _mult = fm_multiplier
    if fm_dynamic_enabled
        if fm_dyn_method == "Vertical Boundaries Distance"
            float _progress = 0.0
            int _len = math.max(1, ma_length)
            if not na(startBarIdx) and not na(endBarIdx)
                int _total_bars_in_range = endBarIdx - startBarIdx
                int _cycle_pos = _total_bars_in_range % _len
                _progress := _cycle_pos / _len
                
            float _time_factor = math.sin(_progress * math.pi)
            _mult := fm_multiplier * (1.0 + (_time_factor * fm_dynamic_target))
            
        else if fm_dyn_method == "Volatility (ATR)"
            float atr_ratio = atr_long > 0 ? (atr_short / atr_long) : 1.0
            _mult := fm_multiplier * atr_ratio
        else if fm_dyn_method == "Volume Momentum"
            float vol_ratio = vol_long > 0 ? (vol_short / vol_long) : 1.0
            _mult := fm_multiplier * vol_ratio
        else // Green/Red Ratio
            float dyn_ratio = red_candles_passed > 0 ? (green_candles_passed / red_candles_passed) : 1.0
            _mult := dyn_ratio * fm_dynamic_target
    _mult

            sumXY += i * y_val
            sumX2 += i * i

        float valRef_end   = 0.0

            valRef_start := stepped_price[idx]

        else if calcMethod == "SMA"

                    valRef_end   := invertedMA

                    valRef_start := invertedMA
                    valRef_end   := valMA
            else
                valRef_start := valMA
                valRef_end   := valMA

            for j = 0 to len_start - 1
                sumMA_start += close[idx + j]
            float valMA_start = sumMA_start / len_start
            
            float sumMA_end = 0.0
            for j = 0 to len_end - 1
                sumMA_end += close[idx + j]

            if autoSlopeHHLL

                float endRangeClose   = close[bar_index - endBarIdx]

                    shouldInvert := true
            else
                shouldInvert := invertHHLL

                valRef_start := valMA_end
                valRef_end   := valMA_start
            else
                valRef_start := valMA_start
                valRef_end   := valMA_end

                for j = 0 to loop_limit - 1
                    float vol = nz(volume[idx + j], 1)
                    float src_val = hlc3[idx + j]
                    sumVwapDev += vol * math.pow(src_val - valVWAP, 2)
                float valVwapDev = sumVol > 0 ? math.sqrt(sumVwapDev / sumVol) : 0.0
                
                upperRef := valVWAP + bbMult * valVwapDev
                lowerRef := valVWAP - bbMult * valVwapDev

                    sumDev += math.pow(close[idx + j] - valMA, 2)
                float valDev = math.sqrt(sumDev / math.max(1, loop_limit - 1))
                
                upperRef := valMA + bbMult * valDev
                lowerRef := valMA - bbMult * valDev
                
            else if calcMethod == "ATR Bands"
                float sumTR = 0.0
                float sumClose = 0.0
                for j = 0 to loop_limit - 1
                    float prev_c = na(close[idx + j + 1]) ? open[idx + j] : close[idx + j + 1]
                    float h = high[idx + j]
                    float l = low[idx + j]
                    float tr = math.max(h - l, math.max(math.abs(h - prev_c), math.abs(l - prev_c)))
                    sumTR += tr
                    sumClose += close[idx + j]

                    shouldInvert := true
            else
                shouldInvert := invertHHLL
            
            finalInvertDecision := shouldInvert

                valRef_start := upperRef
                valRef_end   := lowerRef
            else
                valRef_start := lowerRef
                valRef_end   := upperRef
        
        float dist_start = useCandleHeight ? (high[idx] - low[idx]) : math.abs(close[idx] - valRef_start)
        float dist_end   = useCandleHeight ? (high[idx] - low[idx]) : math.abs(close[idx] - valRef_end)
        
        if useCumSlope
            dist_start := dist_start + (cumSlope * i)
            dist_end   := dist_end + (cumSlope * i)
        
        if close[idx] > open[idx]

    line.set_x2(startRedHoriz, bar_index)
    line.set_x2(endBaseHoriz, bar_index)
    line.set_x2(endRedHoriz, bar_index)

sma_val = ta.sma(close, ma_length)
val_ll  = ta.lowest(low, ma_length)
val_hh  = ta.highest(high, ma_length)

val_bb_upper = sma_val + bbMult * ta.stdev(close, ma_length)
val_bb_lower = sma_val - bbMult * ta.stdev(close, ma_length)

float safeSumVol = math.max(1e-10, math.sum(nz(volume, 1), ma_length))
vwap_val = math.sum(hlc3 * nz(volume, 1), ma_length) / safeSumVol
vwap_sq_val = math.sum(math.pow(hlc3, 2) * nz(volume, 1), ma_length) / safeSumVol
vwap_dev = math.sqrt(math.max(vwap_sq_val - math.pow(vwap_val, 2), 0))

val_vwap_upper = vwap_val + (bbMult * vwap_dev)
val_vwap_lower = vwap_val - (bbMult * vwap_dev)

dyn_tr = ta.tr(true)
dyn_atr = math.sum(dyn_tr, ma_length) / math.max(ma_length, 1)

val_atr_upper = sma_val + (atrMult * dyn_atr)
val_atr_lower = sma_val - (atrMult * dyn_atr)

int ma_len_start = math.max(1, math.round(ma_length * 0.618))
int ma_len_end   = math.max(1, math.round(ma_length * 1.618))
sma_fib_start = ta.sma(close, ma_len_start)
sma_fib_end   = ta.sma(close, ma_len_end)

float dyn_wave = f_get_wave(time, nz(actualStartTimeMs, startTime), nz(actualEndTimeMs, endTime), startPriceLevel, endPriceLevel, ma_length, atr14)

center_price = (startPriceLevel + endPriceLevel) / 2
bool globalInvert = autoSlopeHHLL ? (endPriceLevel < startPriceLevel) : invertHHLL

float purple_line = na
float orange_line = na

if calcMethod == "Flat Multiplier"
    purple_line := (2 * center_price) - (fm_symmetric_enabled ? symmetric_price : stepped_price)
    orange_line := stepped_price
else if calcMethod == "Bollinger Bands"
    purple_line := (2 * center_price) - (globalInvert ? val_bb_upper : val_bb_lower)
    orange_line := globalInvert ? val_bb_lower : val_bb_upper
else if calcMethod == "VWAP"
    purple_line := (2 * center_price) - (globalInvert ? val_vwap_upper : val_vwap_lower)
    orange_line := globalInvert ? val_vwap_lower : val_vwap_upper
else if calcMethod == "ATR Bands"
    purple_line := (2 * center_price) - (globalInvert ? val_atr_upper : val_atr_lower)
    orange_line := globalInvert ? val_atr_lower : val_atr_upper
else if calcMethod == "Fibo Wavy" 
    float base_for_wave = wave_on_ma ? sma_val : center_price
    float v_up = base_for_wave + dyn_wave
    float v_dn = base_for_wave - dyn_wave
    purple_line := (2 * center_price) - (globalInvert ? v_up : v_dn)
    orange_line := globalInvert ? v_dn : v_up
else if calcMethod == "Spiral Spine" 
    purple_line := (2 * center_price) - (globalInvert ? spine_upper : spine_lower)
    orange_line := globalInvert ? spine_lower : spine_upper
else if calcMethod == "Dual Fib SMA"
    float v_start = globalInvert ? sma_fib_end : sma_fib_start
    float v_end   = globalInvert ? sma_fib_start : sma_fib_end
    purple_line := (2 * center_price) - v_start
    orange_line := v_end
else if calcMethod == "SMA"
    float inv_sma = (2 * center_price) - sma_val
    if splitSMA
        purple_line := globalInvert ? sma_val : inv_sma
        orange_line := globalInvert ? inv_sma : sma_val
    else
        purple_line := inv_sma
        orange_line := sma_val
else // HH/LL
    purple_line := (2 * center_price) - (globalInvert ? val_hh : val_ll)
    orange_line := globalInvert ? val_ll : val_hh

float symm_price_start_base = showSymmetricPriceLine and calcComplete ? (2 * startBase) - close : na
float symm_price_start_red  = showSymmetricPriceLine and calcComplete ? (2 * fixed_L_red_start) - close : na
float symm_price_end_base   = showSymmetricPriceLine and calcComplete ? (2 * endBase) - close : na

float symm2_start_vs_end_red  = showSymmetricPriceLine and calcComplete ? (2 * fixed_L_red) - symm_price_start_red : na
float symm2_end_vs_start_base = showSymmetricPriceLine and calcComplete ? (2 * startBase) - symm_price_end_base : na
float symm2_end_vs_start_red  = showSymmetricPriceLine and calcComplete ? (2 * fixed_L_red_start) - symm_price_end_red : na
