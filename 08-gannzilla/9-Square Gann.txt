f_calc_step(float cndl, float atr_anch, float atr_now, float mult, string mode, float manual) =>
    float s = na
    if mode == "«— ›«⁄ ò‰œ·"
        s := cndl
    else if mode == "ATR  ÿ»ÌﬁÌ"
        // Ê“‰ ATR Ã«—Ì = 0.6 | Ê“‰ ATR anchor = 0.4
        // «Ì‰ ‰”»  ”ÿÊÕ —« »Â ‘—«Ìÿ ·ÕŸÂù«Ì Õ”«”ù — „Ìùò‰œ
        float blended = 0.4 * atr_anch + 0.6 * atr_now
        s := blended * mult
    else
        // Õ«·  œ” Ì
        s := manual > 0.0 ? manual : cndl
    s

// ›—„Ê·: increment = (step / 2?P) ◊ (deg/360) ◊ (1+?) ◊ ring
// Level = (?P ± increment)≤
// ??????????????????????????????????????????????????????????????
f_oct_level(float cen, float step, float ang_deg, float spin_deg, int dir, int ring) =>
    float base_sq   = math.sqrt(cen)
    float delta_one = step / (2.0 * base_sq)
    float frac      = ang_deg / 360.0
    float phi_scale = spin_deg / 360.0
    float increment = delta_one * frac * (1.0 + phi_scale) * float(ring)
    float lvl       = math.pow(base_sq + float(dir) * increment, 2.0)
    lvl
