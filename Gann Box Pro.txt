var line[] allLines = array.new_line()

rotatePoint(t, p, ct, cp, angle, tScale, pScale) =>
    tn = (t - ct) * tScale
    pn = (p - cp) / pScale
    tnr = tn * math.cos(angle) - pn * math.sin(angle)
    pnr = tn * math.sin(angle) + pn * math.cos(angle)
    [ct + tnr / tScale, cp + pnr * pScale]

var float totalHeight = 0.0
var float greenHeight = 0.0
var float redHeight = 0.0
var int[] candleTimes = array.new_int()
var float[] candleHighs = array.new_float()
var float[] candleLows = array.new_float()

if time >= time1 and time <= time2
    float candleHeight = math.abs(high - low)
    totalHeight += candleHeight
    if close >= open
        greenHeight += candleHeight
    else
        redHeight += candleHeight
    array.push(candleTimes, time)
    array.push(candleHighs, high)
    array.push(candleLows, low)

float adjustedPriceRange = 0.0
int boxTime1 = 0
int boxTime2 = 0

int centerTime = 0
float centerPrice = 0.0
float timeScale = 1.0
float priceScale = 1.0
float angleRad = 0.0

if time1 > 0 and time2 > 0
    if useCandleHeight
        if candleMode == "”»“"
            adjustedPriceRange := greenHeight
        else if candleMode == "ﬁ—„“"
            adjustedPriceRange := redHeight
        else if candleMode == "Â„Â"
            adjustedPriceRange := totalHeight
        else if candleMode == " ›«÷·"
            adjustedPriceRange := greenHeight - redHeight
    else
        adjustedPriceRange := price2 - price1

    int anchorTime = 0
    float anchorPrice = 0.0
    int timeRange = time2 - time1
    
    if useCandleHeight
        if array.size(candleTimes) > 0
            int arraySize = array.size(candleTimes)
            int selectedIndex = math.round((arraySize - 1) * centerCandlePosition)
            selectedIndex := math.max(0, math.min(selectedIndex, arraySize - 1))
            int candleTime = array.get(candleTimes, selectedIndex)
            float candleHigh = array.get(candleHighs, selectedIndex)
            float candleLow = array.get(candleLows, selectedIndex)
            int candleDuration = arraySize > 1 ? math.abs(array.get(candleTimes, 0) - array.get(candleTimes, 1)) : int(timeframe.in_seconds() * 1000)
            anchorTime := candleTime + int(candleDuration / 2)
            anchorPrice := (candleHigh + candleLow) / 2
        else
            anchorTime := int((time1 + time2) / 2)
            anchorPrice := (price1 + price2) / 2
    else
        anchorTime := int((time1 + time2) / 2)
        anchorPrice := (price1 + price2) / 2

    if useCandleHeight
        boxPrice1 := anchorPrice - adjustedPriceRange * anchorHorizontal
        boxPrice2 := boxPrice1 + adjustedPriceRange
        boxTime1 := anchorTime - int(timeRange * anchorVertical)
        boxTime2 := boxTime1 + timeRange
    else
        boxPrice1 := price1
        boxPrice2 := price2
        boxTime1 := time1
        boxTime2 := time2

    centerPrice := (boxPrice1 + boxPrice2) / 2
    centerTime := int((boxTime1 + boxTime2) / 2)

    priceScale := adjustedPriceRange != 0 and (boxTime2 - boxTime1) != 0 ? math.abs(adjustedPriceRange / (boxTime2 - boxTime1)) : 1.0

    float priceRangeNormalized = adjustedPriceRange / priceScale
    float timeRangeNormalized = (boxTime2 - boxTime1) * timeScale
    float requiredAngleDegrees = timeRangeNormalized != 0 ? math.atan(priceRangeNormalized / timeRangeNormalized) * 180 / math.pi : 0

    float angleDegrees = enableAngle ? requiredAngleDegrees : manualAngleDegrees
    angleRad := angleDegrees * math.pi / 180

//  «»⁄ „Õ«”»Â ò‰œ·ùÂ«Ì „ ﬁ«—‰
getMirroredCandle(lvl) =>
    float mOpen = na, float mHigh = na, float mLow = na, float mClose = na
    if time1 > 0 and time2 > 0 and symmetrizePrice
        float p = boxPrice1 + adjustedPriceRange * lvl
        float t1r = boxTime1
        float p1r = p
        float t2r = boxTime2
        float p2r = p
        
        if (enableAngle or manualAngleDegrees != 0)
            [rt1, rp1] = rotatePoint(boxTime1, p, centerTime, centerPrice, angleRad, timeScale, priceScale)
            [rt2, rp2] = rotatePoint(boxTime2, p, centerTime, centerPrice, angleRad, timeScale, priceScale)
            t1r := rt1
            p1r := rp1
            t2r := rt2
            p2r := rp2

        float axisSlope = 0.0
        if t2r != t1r
            axisSlope := (p2r - p1r) / (t2r - t1r)
        
        float axisPriceAtCurrentTime = p1r + axisSlope * (time - t1r)
        
        mClose := axisPriceAtCurrentTime - (close - axisPriceAtCurrentTime)
        mOpen := axisPriceAtCurrentTime - (open - axisPriceAtCurrentTime)
        mHigh := axisPriceAtCurrentTime - (high - axisPriceAtCurrentTime)
        mLow := axisPriceAtCurrentTime - (low - axisPriceAtCurrentTime)
    
    [mOpen, mHigh, mLow, mClose]
