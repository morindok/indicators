//@version=5
indicator("Quantum Spider Web & Operational Path [Non-Repaint]", overlay=true, max_lines_count=500, max_boxes_count=100)

// =====================================================================
// 1. تنظیمات ورودی
// =====================================================================
int n_lines = input.int(50, "تعداد سطوح افقی شبکه", minval=10, maxval=200, group="شبکه عنکبوتی")
float step_mult = input.float(1.618, "ضریب گام (فیبوناچی)", step=0.1, group="شبکه عنکبوتی")
int time_window = input.int(20, "عرض پنجره زمانی (کندل)", minval=5, group="شبکه عنکبوتی")
int hist_lookback = input.int(100, "عمق تحلیل تاریخی (قبل از لایو)", minval=20, group="تحلیل چندمحوره")

var float fixed_anchor = na
var float static_step = na
var bool sys_init = false
var array<line> perm_lines = array.new_line()

safe_h = high[1]
safe_l = low[1]
safe_c = close[1]

macro_center = (ta.highest(safe_h, 144) + ta.lowest(safe_l, 144)) / 2.0
base_atr = nz(ta.atr(21)[1], syminfo.mintick * 100)

// =====================================================================
// 2. رسم شبکه عنکبوتی (افقی + عمودی)
// =====================================================================
if barstate.islast and not sys_init
    fixed_anchor := macro_center
    static_step := base_atr * step_mult
    static_step := static_step == 0 ? syminfo.mintick * 10 : static_step 
    
    int half = int(n_lines / 2)
    for i = -half to (n_lines - half)
        float lvl = fixed_anchor + (i * static_step)
        color c = (i == 0) ? color.new(color.yellow, 0) : color.new(color.gray, 70)
        string st = (i % 5 == 0) ? line.style_solid : line.style_dotted
        line l = line.new(bar_index, lvl, bar_index + 200, lvl, color=c, width=(i==0?2:1), style=st, extend=extend.right)
        array.push(perm_lines, l)
    sys_init := true

// ایجاد پنجره‌های زمانی عمودی (برای ایجاد افکت شبکه شطرنجی/عنکبوتی)
is_window_start = bar_index % time_window == 0
bgcolor(is_window_start ? color.new(color.blue, 92) : na, title="پنجره زمانی شبکه")

// =====================================================================
// 3. موتور ایجنت‌ها و تحلیل چندمحوره تاریخی
// =====================================================================
v1 = safe_c - close[2]
v2 = close[2] - close[3]
a1 = math.max(math.min((v1 - v2) / nz(base_atr, 1), 3.0), -3.0)

bb_m = ta.sma(safe_c, 20)
bb_d = nz(ta.stdev(safe_c, 20), 1)
a2 = math.max(math.min((safe_c - bb_m) / bb_d, 3.0), -3.0)

mf = nz(ta.mfi(hlc3[1], 14), 50)
a3 = (mf - 50) / 25.0

r_f = ta.roc(safe_c, 5)
r_s = ta.roc(safe_c, 21)
a4 = math.max(math.min(((r_f * 0.7) + (r_s * 0.3)) / 2.0, 3.0), -3.0)

float seg_f = 0.0
int seg_i = 0
float dist_n = 0.5
if sys_init and not na(fixed_anchor) and static_step != 0
    seg_f := (safe_c - fixed_anchor) / static_step
    seg_i := int(math.round(seg_f))
    dist_n := math.abs(seg_f - seg_i)

a5 = (dist_n < 0.12) ? (safe_c > (fixed_anchor + (seg_i * static_step)) ? -2.5 : 2.5) : 0.0

// اصلاح خطای تایپی: استفاده از safe_h و safe_l به جای h_safe و l_safe
r_s2 = safe_h - safe_l == 0 ? 1 : safe_h - safe_l
a6 = (((safe_c - safe_l) / r_s2) - 0.5) * 4.0

// امتیاز لحظه‌ای (لایو)
live_score = (0.25*a1) + (0.15*a2) + (0.15*a3) + (0.20*a4) + (0.15*a5) + (0.10*a6)
live_score := nz(live_score, 0)

// تحلیل چندمحوره تاریخی (Pre-Live Analysis)
float hist_score = nz(live_score[1], 0)
for i = 1 to hist_lookback
    hist_score += nz(live_score[i], 0)
hist_baseline = hist_score / hist_lookback

// =====================================================================
// 4. یافتن بهترین مسیر عملیاتی (Quantum Operational Path)
// =====================================================================
path_deviation = (live_score - hist_baseline) * base_atr * 0.5
quantum_path = ta.ema(safe_c, 14) + path_deviation

plot(quantum_path, color=color.new(color.aqua, 0), linewidth=3, title="مسیر عملیاتی کوانتومی")
plot(safe_c, color=color.new(color.white, 50), linewidth=1, title="قیمت واقعی")

// =====================================================================
// 5. داشبورد فوق حرفه‌ای
// =====================================================================
var table dash = table.new(position.bottom_right, 3, 7, border_width=2, border_color=color.yellow, frame_width=3, frame_color=color.black)

if barstate.islast
    bool is_bull = live_score > hist_baseline
    bool strong_bull = live_score >= 1.5
    bool strong_bear = live_score <= -1.5
    
    color bg_bull = strong_bull ? color.new(color.green, 10) : color.new(color.gray, 60)
    color bg_bear = strong_bear ? color.new(color.red, 10) : color.new(color.gray, 60)
    
    color txt_bull = strong_bull ? color.white : color.green
    color txt_bear = strong_bear ? color.white : color.red
    
    string status = is_bull ? "▲ صعودی (مسیر بهینه)" : "▼ نزولی (مسیر بهینه)"
    color st_bg = is_bull ? color.new(color.green, 0) : color.new(color.red, 0)
    color st_txt = color.white

    table.cell(dash, 0, 0, "پارامتر شبکه", text_color=color.yellow, text_size=size.normal, bgcolor=color.black)
    table.cell(dash, 1, 0, "مقدار زنده", text_color=color.yellow, text_size=size.normal, text_halign=text.align_center, bgcolor=color.black)
    table.cell(dash, 2, 0, "وضعیت تاریخی", text_color=color.yellow, text_size=size.normal, text_halign=text.align_center, bgcolor=color.black)

    table.cell(dash, 0, 1, "گره فعال شبکه", text_color=color.white, text_size=size.normal, bgcolor=color.rgb(30, 30, 50))
    table.cell(dash, 1, 1, str.tostring(nz(seg_i, 0)), text_color=color.aqua, text_size=size.large, text_halign=text.align_center, bgcolor=color.rgb(30, 30, 50))
    table.cell(dash, 2, 1, "Segment", text_color=color.gray, text_size=size.small, text_halign=text.align_center, bgcolor=color.rgb(30, 30, 50))

    color sc_bg = live_score > 0 ? color.new(color.green, 70) : color.new(color.red, 70)
    color sc_txt = live_score > 0 ? color.green : color.red
    table.cell(dash, 0, 2, "امتیاز لایو (6 ایجنت)", text_color=color.white, text_size=size.normal, bgcolor=color.rgb(40, 40, 40))
    table.cell(dash, 1, 2, str.tostring(nz(live_score, 0), "#.##"), text_color=sc_txt, text_size=size.large, text_halign=text.align_center, bgcolor=sc_bg)
    table.cell(dash, 2, 2, live_score > 0 ? "مثبت" : "منفی", text_color=sc_txt, text_size=size.normal, text_halign=text.align_center, bgcolor=sc_bg)

    table.cell(dash, 0, 3, "مبنای تاریخی (Pre-Live)", text_color=color.white, text_size=size.normal, bgcolor=color.rgb(40, 40, 40))
    table.cell(dash, 1, 3, str.tostring(nz(hist_baseline, 0), "#.##"), text_color=color.orange, text_size=size.large, text_halign=text.align_center, bgcolor=color.rgb(40, 40, 40))
    table.cell(dash, 2, 3, "Baseline", text_color=color.gray, text_size=size.small, text_halign=text.align_center, bgcolor=color.rgb(40, 40, 40))

    float p_up = 100 / (1 + math.exp(-live_score * 1.8))
    table.cell(dash, 0, 4, "احتمال صعود (Up %)", text_color=color.white, text_size=size.normal, bgcolor=bg_bull)
    table.cell(dash, 1, 4, str.tostring(nz(p_up, 50), "#.#") + " %", text_color=txt_bull, text_size=size.large, text_halign=text.align_center, bgcolor=bg_bull)
    table.cell(dash, 2, 4, strong_bull ? "قوی" : "عادی", text_color=txt_bull, text_size=size.normal, text_halign=text.align_center, bgcolor=bg_bull)

    float p_dn = 100 - p_up
    table.cell(dash, 0, 5, "احتمال نزول (Down %)", text_color=color.white, text_size=size.normal, bgcolor=bg_bear)
    table.cell(dash, 1, 5, str.tostring(nz(p_dn, 50), "#.#") + " %", text_color=txt_bear, text_size=size.large, text_halign=text.align_center, bgcolor=bg_bear)
    table.cell(dash, 2, 5, strong_bear ? "قوی" : "عادی", text_color=txt_bear, text_size=size.normal, text_halign=text.align_center, bgcolor=bg_bear)

    table.cell(dash, 0, 6, "خروجی مسیر عملیاتی", text_color=color.white, text_size=size.normal, bgcolor=color.rgb(20, 20, 20))
    table.cell(dash, 1, 6, status, text_color=st_txt, text_size=size.large, text_halign=text.align_center, bgcolor=st_bg)
    table.cell(dash, 2, 6, dist_n < 0.12 ? "⚠️ گره بحرانی" : "✅ مسیر باز", text_color=dist_n < 0.12 ? color.orange : color.green, text_size=size.normal, text_halign=text.align_center, bgcolor=color.rgb(20, 20, 20))
