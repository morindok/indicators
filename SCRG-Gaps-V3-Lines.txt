slope1 = math.tan(angle1 * math.pi / 180)
slope2 = math.tan(angle2 * math.pi / 180)

barsFromStart = (time - time1) / (time2 - time1) * (price1 - price2)
adjustedPrice1 = price1 + slope1 * barsFromStart
adjustedPrice2 = price2 + slope2 * barsFromStart

redCandle = close < open
greenCandle = close >= open

mirrorOpen = greenCandle ? adjustedPrice1 - (open - adjustedPrice1) : redCandle ? adjustedPrice2 + (adjustedPrice2 - open) : na
mirrorHigh = greenCandle ? adjustedPrice1 - (high - adjustedPrice1) : redCandle ? adjustedPrice2 + (adjustedPrice2 - high) : na
mirrorLow = greenCandle ? adjustedPrice1 - (low - adjustedPrice1) : redCandle ? adjustedPrice2 + (adjustedPrice2 - low) : na
mirrorClose = greenCandle ? adjustedPrice1 - (close - adjustedPrice1) : redCandle ? adjustedPrice2 + (adjustedPrice2 - close) : na

plotcandle(mirrorOpen, mirrorHigh, mirrorLow, mirrorClose, color=greenCandle ? color.purple : color.orange, wickcolor=greenCandle ? color.purple : color.orange, bordercolor=greenCandle ? color.purple : color.orange)

origAdjOpen = redCandle ? open + (slope1 * barsFromStart) : greenCandle ? open + (slope2 * barsFromStart) : open
origAdjHigh = redCandle ? high + (slope1 * barsFromStart) : greenCandle ? high + (slope2 * barsFromStart) : high
origAdjLow = redCandle ? low + (slope1 * barsFromStart) : greenCandle ? low + (slope2 * barsFromStart) : low
origAdjClose = redCandle ? close + (slope1 * barsFromStart) : greenCandle ? close + (slope2 * barsFromStart) : close

plotcandle(origAdjOpen, origAdjHigh, origAdjLow, origAdjClose, title="Adjusted Original Candles", color=greenCandle ? color.green : color.red, wickcolor=greenCandle ? color.green : color.red, bordercolor=greenCandle ? color.green : color.red)
