//@version=5
indicator("Aligned BoS Clusters - Polyline (Live)", overlay=false, max_boxes_count=500, max_polylines_count=100, max_bars_back=5000)

// --- Inputs ---
leftBars  = input.int(10, title="Pivot Left Bars", minval=1)
rightBars = input.int(10, title="Pivot Right Bars", minval=1)

show_ma = input.bool(true, title="Show Moving Averages")
blue_offset = input.float(0.0, title="Blue Box Offset", step=0.1)
orange_offset = input.float(0.0, title="Orange Box Offset", step=0.1)

invert_blue = input.bool(false, title="Invert Blue Polyline")
invert_orange = input.bool(false, title="Invert Orange Polyline")

// --- Pivot Calculation ---
ph = ta.pivothigh(high, leftBars, rightBars)
pl = ta.pivotlow(low, leftBars, rightBars)

var float lastPH = na
var float lastPL = na

if not na(ph)
    lastPH := ph
if not na(pl)
    lastPL := pl

// --- State Logic ---
var int trendState = 0 

bullish_bos = close > lastPH
bearish_bos = close < lastPL

if bullish_bos
    trendState := 1
else if bearish_bos
    trendState := -1

stateChanged = trendState != trendState[1]

// --- Arrays for Storing Data ---
var a_H = array.new_float()
var a_L = array.new_float()
var a_C = array.new_float()
var a_idx = array.new_int()

array.push(a_H, high)
array.push(a_L, low)
array.push(a_C, close)
array.push(a_idx, bar_index)

var int prevBoxSize = 10 

// --- Variables for Live objects ---
var box liveBoxBot = na
var box liveBoxTop = na
var polyline livePolyBot = na
var polyline livePolyTop = na

// ==========================================
// 1. Drawing Logic (Previous Phase)
// ==========================================
if stateChanged and trendState[1] != 0
    curr_H = array.pop(a_H)
    curr_L = array.pop(a_L)
    curr_C = array.pop(a_C)
    curr_idx = array.pop(a_idx)
    
    size = array.size(a_C)
    
    if size > 0
        minL = array.min(a_L)
        maxH = array.max(a_H)
        
        height = maxH - minL
        shiftB = blue_offset * height
        shiftO = orange_offset * height
        
        centerB = (height / 2) - shiftB
        centerO = shiftO - (height / 2)

        pts_bottom = array.new<chart.point>()
        pts_top = array.new<chart.point>()
        pts_ma_bottom = array.new<chart.point>()
        pts_ma_top = array.new<chart.point>()
        
        for i = 0 to size - 1
            idx = array.get(a_idx, i)
            c_val = array.get(a_C, i)
            
            val_bot = (c_val - minL) - shiftB
            val_top = (c_val - maxH) + shiftO
            
            if invert_blue
                val_bot := (2 * centerB) - val_bot
            if invert_orange
                val_top := (2 * centerO) - val_top
            
            array.push(pts_bottom, chart.point.from_index(idx, val_bot))
            array.push(pts_top, chart.point.from_index(idx, val_top))
            
            if show_ma
                sum_bot = 0.0
                sum_top = 0.0
                count = math.min(i + 1, prevBoxSize)
                if count > 0
                    for j = 0 to count - 1
                        vB = ((array.get(a_C, i - j) - minL) - shiftB)
                        vT = ((array.get(a_C, i - j) - maxH) + shiftO)
                        if invert_blue
                            vB := (2 * centerB) - vB
                        if invert_orange
                            vT := (2 * centerO) - vT
                        sum_bot += vB
                        sum_top += vT
                    array.push(pts_ma_bottom, chart.point.from_index(idx, sum_bot / count))
                    array.push(pts_ma_top, chart.point.from_index(idx, sum_top / count))

        box.new(left=array.get(a_idx, 0), top=height - shiftB, right=array.get(a_idx, size-1), bottom=-shiftB, 
                 bgcolor=color.new(color.blue, 90), border_color=color.blue, border_style=line.style_dashed)
        if size > 1
            polyline.new(pts_bottom, line_color=color.blue, line_width=1)
            if show_ma
                polyline.new(pts_ma_bottom, line_color=color.white, line_width=2) 

        box.new(left=array.get(a_idx, 0), top=shiftO, right=array.get(a_idx, size-1), bottom=(minL - maxH) + shiftO, 
                 bgcolor=color.new(color.orange, 90), border_color=color.orange, border_style=line.style_dashed)
        if size > 1
            polyline.new(pts_top, line_color=color.orange, line_width=1)
            if show_ma
                polyline.new(pts_ma_top, line_color=color.white, line_width=2) 
            
        prevBoxSize := size

    array.clear(a_H)
    array.clear(a_L)
    array.clear(a_C)
    array.clear(a_idx)
    
    array.push(a_H, curr_H)
    array.push(a_L, curr_L)
    array.push(a_C, curr_C)
    array.push(a_idx, curr_idx)

// ==========================================
// 2. Live Drawing Logic (Current Phase)
// ==========================================
box.delete(liveBoxBot)
box.delete(liveBoxTop)
polyline.delete(livePolyBot)
polyline.delete(livePolyTop)

sizeLive = array.size(a_C)
if sizeLive > 0
    minL_live = array.min(a_L)
    maxH_live = array.max(a_H)
    
    heightLive = maxH_live - minL_live
    shiftB_live = blue_offset * heightLive
    shiftO_live = orange_offset * heightLive
    
    centerB_live = (heightLive / 2) - shiftB_live
    centerO_live = shiftO_live - (heightLive / 2)

    pts_bottom_live = array.new<chart.point>()
    pts_top_live = array.new<chart.point>()
    
    startIdxLive = array.get(a_idx, 0)
    endIdxLive = array.get(a_idx, sizeLive - 1)
    
    for i = 0 to sizeLive - 1
        idxLive = array.get(a_idx, i)
        cLive = array.get(a_C, i)
        
        val_bot_live = (cLive - minL_live) - shiftB_live
        val_top_live = (cLive - maxH_live) + shiftO_live
        
        if invert_blue
            val_bot_live := (2 * centerB_live) - val_bot_live
        if invert_orange
            val_top_live := (2 * centerO_live) - val_top_live
        
        array.push(pts_bottom_live, chart.point.from_index(idxLive, val_bot_live))
        array.push(pts_top_live, chart.point.from_index(idxLive, val_top_live))
        
    liveBoxBot := box.new(left=startIdxLive, top=heightLive - shiftB_live, right=endIdxLive, bottom=-shiftB_live, bgcolor=color.new(color.blue, 85), border_color=color.blue, border_style=line.style_dotted)
    if sizeLive > 1
        livePolyBot := polyline.new(pts_bottom_live, line_color=color.blue, line_width=2)
        
    liveBoxTop := box.new(left=startIdxLive, top=shiftO_live, right=endIdxLive, bottom=(minL_live - maxH_live) + shiftO_live, bgcolor=color.new(color.orange, 85), border_color=color.orange, border_style=line.style_dotted)
    if sizeLive > 1
        livePolyTop := polyline.new(pts_top_live, line_color=color.orange, line_width=2)

// ˜ÔíÏä ÎØ ÕÝÑ ãÑ˜Òí
plot(0, color=color.gray, style=plot.style_line, title="Zero Baseline")

// ==========================================
// 3. Moving Average on Main Price Chart
// ==========================================
float dynamic_ma = na
if prevBoxSize > 0 and show_ma
    sum_price = 0.0
    for i = 0 to prevBoxSize - 1
        sum_price += close[i]
    dynamic_ma := sum_price / prevBoxSize

plot(show_ma ? dynamic_ma : na, color=color.yellow, title="Main Chart MA", linewidth=2, force_overlay=true)
