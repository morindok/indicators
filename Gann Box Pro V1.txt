maLengthInput = input.int(20, "ÿÊ· MA Å«ÌÂ", minval=1, group="MA Settings")

boxColor   = input.color(color.blue, "—‰ê »«ò”")

fibWidth   = input.int(1, "÷Œ«„  ŒÿÊÿ ›Ì»Ê")
manualAngleDegrees = input.float(0, "“«ÊÌÂ œ” Ì (œ—ÃÂ)", tooltip="“«ÊÌÂ »— «”«” ‰”»  “„«‰-ﬁÌ„  Ã⁄»Â. »—«Ì ç—Œ‘ùÂ«Ì Ã“∆Ì «“ 0.01 ‘—Ê⁄ ò‰Ìœ.")

var line[] allLines = array.new_line()

var float totalHeight = 0.0
var float greenHeight = 0.0

var float[] candleHighs = array.new_float()
var float[] candleLows  = array.new_float()

if time >= time1 and time <= time2
    float ch = math.abs(high - low)
    totalHeight += ch
    if close >= open
        greenHeight += ch
    else
        redHeight   += ch
    array.push(candleTimes, time)
    array.push(candleHighs, high)
    array.push(candleLows, low)

// „ €Ì—Â«Ì Â‰œ”Ì »«ò”
float adjPriceRange = 0.0
int   bT1 = 0, bT2 = 0
float bP1 = 0.0, bP2 = 0.0
int   cT = 0
float cP = 0.0
float pScale = 1.0
float aRad   = 0.0

// „Õ«”»Â  ⁄œ«œ ò‰œ·ùÂ«Ì »«“Â »—«Ì MA ŒÊœò«— Ê Slope
int barsInRange = math.max(1, math.round(math.abs(time2 - time1) / (timeframe.in_seconds() * 1000)))

if time1 > 0 and time2 > 0
    adjPriceRange := useCandleHeight ? (candleMode=="”»“"?greenHeight : candleMode=="ﬁ—„“"?redHeight : candleMode==" ›«÷·"?greenHeight-redHeight : totalHeight) : (price2 - price1)
    int ancT = 0
    float ancP = 0.0
    int tRange = time2 - time1
    if useCandleHeight and array.size(candleTimes) > 0
        val   = math.round((array.size(candleTimes)-1) * centerCandlePosition)
        minVal = 0
        maxVal = array.size(candleTimes)-1
        idx   = math.max(minVal, math.min(val, maxVal))
        ancT := array.get(candleTimes, idx)
        ancP := (array.get(candleHighs, idx) + array.get(candleLows, idx)) / 2
    else
        ancT := int((time1 + time2) / 2)
        ancP := (price1 + price2) / 2
    if useCandleHeight
        bP1 := ancP - adjPriceRange * anchorHorizontal
        bP2 := bP1 + adjPriceRange
        bT1 := ancT - int(tRange * anchorVertical)
        bT2 := bT1 + tRange
    else
        bP1 := price1
        bP2 := price2
        bT1 := time1
        bT2 := time2
    cP := (bP1 + bP2) / 2
    cT := int((bT1 + bT2) / 2)
    pScale := adjPriceRange != 0 ? math.abs(adjPriceRange / barsInRange) : 1.0
    float reqAngle = bT2 != bT1 ? math.atan((adjPriceRange/pScale) / (bT2 - bT1)) : 0
    aRad := enableAngle ? reqAngle : manualAngleDegrees * math.pi / 180

// ==========================================

rotatePoint(t, p, ct, cp, angle, ps) =>
    tn = (t - ct)
    pn = (p - cp) / ps
    tnr = tn * math.cos(angle) - pn * math.sin(angle)
    pnr = tn * math.sin(angle) + pn * math.cos(angle)
    [ct + tnr, cp + pnr * ps]

getMALen(idx) =>
    float base = autoMALen ? barsInRange : maLengthInput
    if goldMode
        dist = math.abs(idx - 3)
        base := base * math.pow(1.618, dist)
    int len = math.max(1, math.round(base))
    math.min(len, maMaxLength)   // „ÕœÊœÌ  »—«Ì Ã·ÊêÌ—Ì «“ NA

applyMA(src, len, type) =>
    if type == "SMA"
        ta.sma(src, len)
    else
        ta.ema(src, len)

getSlopeAdj(idx) =>
    if not slopeMode
        0.0
    else
        float barsSinceStart = (time - bT1) / (timeframe.in_seconds() * 1000)
        float multiplier = (idx - 3)
        if slopeType == "Convergent"
            multiplier := -multiplier
        multiplier * (math.abs(bP2 - bP1) / barsInRange) * slopeFactor * barsSinceStart

// ==========================================

[t1r1, p1r1] = rotatePoint(bT1, basePrice1, cT, cP, aRad, pScale)
[t2r1, p2r1] = rotatePoint(bT2, basePrice1, cT, cP, aRad, pScale)
[t1r2, p1r2] = rotatePoint(bT1, basePrice2, cT, cP, aRad, pScale)
[t2r2, p2r2] = rotatePoint(bT2, basePrice2, cT, cP, aRad, pScale)
[t1r3, p1r3] = rotatePoint(bT1, basePrice3, cT, cP, aRad, pScale)
[t2r3, p2r3] = rotatePoint(bT2, basePrice3, cT, cP, aRad, pScale)
[t1r4, p1r4] = rotatePoint(bT1, basePrice4, cT, cP, aRad, pScale)
[t2r4, p2r4] = rotatePoint(bT2, basePrice4, cT, cP, aRad, pScale)
[t1r5, p1r5] = rotatePoint(bT1, basePrice5, cT, cP, aRad, pScale)
[t2r5, p2r5] = rotatePoint(bT2, basePrice5, cT, cP, aRad, pScale)
[t1r6, p1r6] = rotatePoint(bT1, basePrice6, cT, cP, aRad, pScale)
[t2r6, p2r6] = rotatePoint(bT2, basePrice6, cT, cP, aRad, pScale)

slope0 = (t2r0 != t1r0) ? (p2r0 - p1r0) / (t2r0 - t1r0) : 0.0
slope1 = (t2r1 != t1r1) ? (p2r1 - p1r1) / (t2r1 - t1r1) : 0.0
slope2 = (t2r2 != t1r2) ? (p2r2 - p1r2) / (t2r2 - t1r2) : 0.0
slope3 = (t2r3 != t1r3) ? (p2r3 - p1r3) / (t2r3 - t1r3) : 0.0
slope4 = (t2r4 != t1r4) ? (p2r4 - p1r4) / (t2r4 - t1r4) : 0.0
slope5 = (t2r5 != t1r5) ? (p2r5 - p1r5) / (t2r5 - t1r5) : 0.0
slope6 = (t2r6 != t1r6) ? (p2r6 - p1r6) / (t2r6 - t1r6) : 0.0
