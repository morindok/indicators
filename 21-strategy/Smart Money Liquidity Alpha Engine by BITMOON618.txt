heatColor(float v, float mx) =>
    float half = mx > 0 ? mx / 2.0 : 1.0
    color g = v <= half ? color.from_gradient(v, 0, half, color.blue, color.yellow) : color.from_gradient(v, half, mx, color.yellow, color.red)
    color.new(g, transp)

rowOf(float price, float lo, float cH) =>
    math.max(0, math.min(nRows - 1, int(math.floor((price - lo) / cH))))

f_bin_vols(bar_v, bar_o, bar_c, bar_h, bar_l, level_lo, level_hi) =>
    float bar_range = bar_h - bar_l
    float body_lo = math.min(bar_o, bar_c)
    float body_hi = math.max(bar_o, bar_c)
    bool is_bull = bar_c >= bar_o
    float overlap_lo = math.max(level_lo, bar_l)
    float overlap_hi = math.min(level_hi, bar_h)
    float overlap = math.max(0.0, overlap_hi - overlap_lo)
    float weight = bar_range > 0 ? overlap / bar_range : (1.0 / float(BINS))
    bool in_body = (level_hi >= body_lo) and (level_lo <= body_hi)
    bool in_uwk = level_lo >= body_hi
    bool in_lwk = level_hi <= body_lo
    float body_mult = in_body ? 1.4 : 0.75
    float bin_vol = bar_v * weight * body_mult
    float buy_ratio = is_bull ? 0.62 : 0.38
    if in_body
        buy_ratio := is_bull ? 0.70 : 0.30
    if in_uwk
        buy_ratio := is_bull ? 0.45 : 0.25
    if in_lwk
        buy_ratio := is_bull ? 0.75 : 0.55
    [bin_vol * buy_ratio, bin_vol * (1.0 - buy_ratio)]

        wins += 1
        tp2_hits += 1
