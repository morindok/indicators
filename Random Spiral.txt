//@version=5
indicator("Golden Spiral", overlay=true, max_lines_count=500)

t = input.time(0, "Candle Time", confirm=true)

var float cx = na
var float cy = na
var int dir = na
var float base_scale = na

if bar_index == 0
    math.random(42)

if time == t
    cx := (high + low) / 2
    cy := (high + low) / 2
    dir := math.random(100) > 50 ? 1 : -1
    base_scale := (high - low) * 2.5

phi = 1.6180339887
turns = 5.0
steps = 400
angle_step = (turns * 2 * math.pi) / steps

if not na(cx) and bar_index > 0
    for i = 0 to steps - 1
        a1 = i * angle_step * dir
        a2 = (i + 1) * angle_step * dir
        r1 = base_scale * math.pow(phi, a1 / (2 * math.pi))
        r2 = base_scale * math.pow(phi, a2 / (2 * math.pi))
        x1_offset = r1 * math.cos(a1)
        y1 = cy + r1 * math.sin(a1)
        x2_offset = r2 * math.cos(a2)
        y2 = cy + r2 * math.sin(a2)
        t1 = t + math.round(x1_offset * 60000)
        t2 = t + math.round(x2_offset * 60000)
        line.new(x1=t1, y1=y1, x2=t2, y2=y2, xloc=xloc.bar_time, color=color.new(color.orange, 20), width=1)
