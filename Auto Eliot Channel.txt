//@version=5
indicator("Auto Fractal Slope Projection with Elliott Ratios", overlay=true, max_lines_count=500, max_labels_count=500)

//  ‰ŸÌ„«  ›—ò «·
n = input.int(title="Fractal Periods", defval=3)
fractalsToAnalyze = input.int(10, "Number of Fractals to Analyze", minval=5, maxval=20)
startTime = input.time(timestamp("2024-01-01 00:00"), "Analysis Start Time", confirm=true)

//  ‰ŸÌ„«  „Õ«”»Â ‘Ì»
averageMethod = input.string("Arithmetic", "Average Method", options=["Arithmetic", "Weighted", "Exponential", "Median", "Harmonic"])
slopeMultiplier = input.float(1.0, "Slope Multiplier", maxval=10.0, step=0.1)

// ›Ì· —Â«Ì »Â»Êœ Ì«› Â
useVolumeFilter = input.bool(true, "Use Volume Filter", group="Filters")
volumeThreshold = input.float(1.2, "Volume Threshold", minval=0.5, maxval=3.0, step=0.1, group="Filters")
useATRFilter = input.bool(true, "Use ATR Filter", group="Filters")
atrPeriod = input.int(14, "ATR Period", minval=5, maxval=50, group="Filters")
atrMultiplier = input.float(1.5, "ATR Multiplier", minval=0.5, maxval=5.0, step=0.1, group="Filters")

// ‰”» ùÂ«Ì «·ÌÊ

showFib786 = input.bool(true, "0.786", inline="fib3", group="Elliott Wave")
showFib1000 = input.bool(true, "1.000", inline="fib3", group="Elliott Wave")
showFib1272 = input.bool(true, "1.272", inline="fib4", group="Elliott Wave")
showFib1618 = input.bool(true, "1.618", inline="fib4", group="Elliott Wave")
showFib2618 = input.bool(true, "2.618", inline="fib5", group="Elliott Wave")

// „Õ«”»Â ATR Ê ÕÃ„ „Ì«‰êÌ‰
atrValue = ta.atr(atrPeriod)
avgVolume = ta.sma(volume, 20)

//  ‘ŒÌ’ ›—ò «·ùÂ«Ì »«·«

bool upflagUpFrontier0 = true
bool upflagUpFrontier1 = true

bool upflagUpFrontier3 = true

    upflagUpFrontier1 := upflagUpFrontier1 and (high[n+1] <= high[n] and high[n+i + 1] < high[n])
    upflagUpFrontier2 := upflagUpFrontier2 and (high[n+1] <= high[n] and high[n+2] <= high[n] and high[n+i + 2] < high[n])

    downflagUpFrontier3 := downflagUpFrontier3 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+i + 3] > low[n])
    downflagUpFrontier4 := downflagUpFrontier4 and (low[n+1] >= low[n] and low[n+2] >= low[n] and low[n+3] >= low[n] and low[n+4] >= low[n] and low[n+i + 4] > low[n])

flagDownFrontier = downflagUpFrontier0 or downflagUpFrontier1 or downflagUpFrontier2 or downflagUpFrontier3 or downflagUpFrontier4
volumeConfirmDown = useVolumeFilter ? volume[n] > avgVolume[n] * volumeThreshold : true

var float[] fractalPrices_high = array.new_float(0)
var int[] fractalTimes_high = array.new_int(0)
var float[] fractalStrength_high = array.new_float(0)

var float[] fractalPrices_low = array.new_float(0)
var int[] fractalTimes_low = array.new_int(0)
var float[] fractalStrength_low = array.new_float(0)

// „Õ«”»Â ﬁœ—  ›—ò «·
calcFractalStrength(price, vol, avgVol) =>
    volStrength = vol / avgVol
    priceStrength = useATRFilter ? math.abs(price - close[n]) / atrValue[n] : 1.0
    volStrength * priceStrength

// Ã„⁄ù¬Ê—Ì ›—ò «·ùÂ« ﬁ»· «“ “„«‰ ‘—Ê⁄
if time[n] < startTime
    if upFractal

        array.push(fractalPrices_high, high[n])
        array.push(fractalTimes_high, time[n])
        array.push(fractalStrength_high, strength)
    
    if downFractal
        strength = calcFractalStrength(low[n], volume[n], avgVolume[n])
        array.push(fractalPrices_low, low[n])
        array.push(fractalTimes_low, time[n])
        array.push(fractalStrength_low, strength)

//  «»⁄ „Õ«”»Â „Ì«‰êÌ‰
calculateAverage(slopes, weights, method) =>
    float result = na
    int size = array.size(slopes)
    
    if size > 0
        if method == "Arithmetic"
            sum = 0.0
            for i = 0 to size - 1
                sum := sum + array.get(slopes, i)
            result := sum / size
        
        else if method == "Weighted"
            sumWeighted = 0.0
            sumWeights = 0.0
            for i = 0 to size - 1
                weight = array.get(weights, i)
                sumWeighted := sumWeighted + (array.get(slopes, i) * weight)
                sumWeights := sumWeights + weight
            result := sumWeights > 0 ? sumWeighted / sumWeights : na
        
        else if method == "Exponential"
            alpha = 0.3
            ema = array.get(slopes, 0)
            for i = 1 to size - 1
                ema := alpha * array.get(slopes, i) + (1 - alpha) * ema
            result := ema
        
        else if method == "Median"
            sortedSlopes = array.copy(slopes)
            array.sort(sortedSlopes)
            midIndex = math.floor(size / 2)
            result := array.get(sortedSlopes, midIndex)
        
        else if method == "Harmonic"
            sumReciprocal = 0.0
            for i = 0 to size - 1
                slope = array.get(slopes, i)
                if slope != 0
                    sumReciprocal := sumReciprocal + (1 / slope)
            result := sumReciprocal != 0 ? size / sumReciprocal : na
    
    result
